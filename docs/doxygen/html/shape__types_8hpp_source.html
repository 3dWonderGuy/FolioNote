23/**
24 * @brief Categorization of all 2D geometric vector shape primitives.
25 *
26 * Numerical values are stable for serialization. Pre-alpha: backward
27 * compatibility is NOT guaranteed. Clean removals are acceptable.
28 *
29 * Closed shapes (closed path, can be filled):
30 * Rectangle, RoundedRectangle, Ellipse, Circle,
31 * Triangle, RightTriangle, RegularPolygon, Star, Heart, Cloud
32 *
33 * NOTE: Line and LineArrow have been promoted to SmartArrowObject
34 * (connectors/smart_arrow_container.hpp). They are NOT listed here.
35 */
36enum class ShapeType : uint8_t {
37 Rectangle = 0, ///< Standard axis-aligned rectangle
38 RoundedRectangle = 1, ///< Rectangle with adjustable corner fillet radius
39 Ellipse = 2, ///< Full ellipse (two-step: major axis then minor)
40 Triangle = 3, ///< Isosceles triangle (apex at top center)
41 Diamond = 4, ///< 4-point rhombus/diamond
42 Star = 5, ///< 5-point star (inner/outer radius ratio via param2)
43 Hexagon = 6, ///< 6-sided regular polygon
44 Arrow = 7, ///< Block arrow shape pointing right
45 DoubleArrow = 8, ///< Bidirectional block arrow
46 Line = 9, ///< Linear segment between start and current points
47 Heart = 10, ///< Cubic-bezier heart shape
48 Cloud = 11, ///< Rounded cloud silhouette (add_round_rect + ellipses)
49 Circle = 12, ///< Symmetric circle dragged outward from center
50 RightTriangle = 13, ///< 90-degree orthogonal triangle (right angle bottom-left)
51 LineArrow = 14, ///< Linear connector with directional arrowheads
52 RegularPolygon = 15, ///< N-sided regular polygon (param1 = side count, 3..32)
53 SineWave = 16, ///< Continuous sine wave along horizontal axis (param1 = cycles)
54 SquareWave = 17, ///< Alternating square wave / pulse train (param1 = cycles)
55 TriangleWave = 18, ///< Symmetric triangle wave (param1 = cycles)
56 RightTriangleWave = 19, ///< Right triangle / sawtooth wave (param1 = cycles, param2 = 0: right angle right, 1: right angle left)
57};

63/**
64 * @brief Infill rendering style for closed vector shapes.
65 *
66 * None: shape is transparent (no fill drawn). Default.
67 * Solid: full opaque solid color (alpha forced to 255).
68 * SemiTransparent: soft translucent wash (alpha ~90).
69 * LinearGradient: two-stop left-to-right Blend2D gradient.
70 * RadialGradient: two-stop center-outward Blend2D gradient.
71 * Hatch*: tiled drafting texture drawn as a repeating BLPattern.
72 *
73 * Scalability: new fill types only require a new case in shape_container.cpp.
74 */
75enum class ShapeFillType : uint8_t {
76 None = 0, ///< Transparent / no fill
77 Solid = 1, ///< Fully opaque solid color
78 SemiTransparent = 2, ///< Translucent color wash (~35% opacity)
79 LinearGradient = 3, ///< Left-to-right two-stop gradient
80 RadialGradient = 4, ///< Center-outward two-stop gradient
81 HatchDiagonal = 5, ///< 45-degree parallel drafting hatch lines (///)
82 HatchCross = 6, ///< Intersecting 45-degree drafting grid (XXX)
83 HatchHorizontal = 7, ///< Horizontal parallel drafting lines (---)
84 HatchVertical = 8, ///< Vertical parallel drafting lines (|||)
85 HatchDots = 9, ///< Fine circular dot stipple (concrete/sand/soil grain)
86};

92/**
93 * @brief Border stroke pattern for vector shape outlines.
94 *
95 * Shapes with None outline and None fill are still selectable via AABB
96 * proximity test. Outlines are always rendered at alpha=255 (fully opaque).
97 */
98enum class ShapeOutlineType : uint8_t {
99 None = 0, ///< No border drawn (invisible outline)
100 Solid = 1, ///< Continuous unbroken stroke
101 Dashed = 2, ///< Evenly spaced dash segments
102 Dotted = 3, ///< Crisp circular dots along the path
103 DashDot = 4, ///< Alternating long dash and circular dot
104};
