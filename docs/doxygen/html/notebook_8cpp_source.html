67 * @brief Sets the active section, keeping activeSectionGuid and activeSectionIndex in sync.
68 *
69 * WORKING PROCESS:
70 * 1. Checks if the incoming section pointer is non-null.
71 * 2. Compares incoming section GUID against `activeSectionGuid` to detect actual switch.
72 * 3. Updates `activeSectionGuid`.

77void Notebook::SetActiveSection(const std::shared_ptr<Section>& sec) {
78 if (!sec) return;
79 activeSectionGuid = sec->guid;
80 for (size_t i = 0; i < sections.size(); ++i) {
81 if (sections[i] && sections[i]->guid == sec->guid) {
82 activeSectionIndex = i;
83 return;
84 }
85 }

88/**
89 * @brief Returns the currently active Section (from root or any SectionGroup), or nullptr.
90 *
91 * RESOLUTION PIPELINE & FALLBACK STRATEGY:
92 * 1. Primary: If `activeSectionGuid` is non-empty, looks up section across both root sections
93 * and nested SectionGroups via `FindSectionByGuid()`.
94 * 2. Secondary: If GUID lookup fails or is unset, attempts direct index access via `activeSectionIndex`.
95 * 3. Tertiary: Falls back to the first available root section (`sections.front()`).
96 * 4. Quaternary: Falls back to the first available section inside the first SectionGroup.
97 * 5. Failure: Returns nullptr if the notebook contains zero sections across all containers.
98 *
99 * @return std::shared_ptr<Section> Active Section pointer, or nullptr if none found.
100 */
101std::shared_ptr<Section> Notebook::GetActiveSection() const {
102 if (!activeSectionGuid.empty()) {
103 if (auto found = FindSectionByGuid(activeSectionGuid)) {
104 return found;
105 }
106 }
107 if (activeSectionIndex < sections.size() && sections[activeSectionIndex]) {
108 activeSectionGuid = sections[activeSectionIndex]->guid;
109 return sections[activeSectionIndex];
110 }
111 if (!sections.empty() && sections.front()) {
112 activeSectionIndex = 0;
113 activeSectionGuid = sections.front()->guid;
114 return sections.front();
115 }
116 for (const auto& grp : sectionGroups) {
117 if (grp && !grp->sections.empty() && grp->sections.front()) {
118 activeSectionGuid = grp->sections.front()->guid;
119 return grp->sections.front();
120 }
121 }
122 return nullptr;
123}

179/**
180 * @brief Moves a root section from one index to another, updating sortOrder and activeIndex.
181 *
182 * INDEX CALCULATION & ACTIVE SELECTION MATH:
183 * When an element is moved from `fromIdx` to `toIdx`:
184 * - If `activeSectionIndex == fromIdx`, the active index shifts directly to `toIdx`.
185 * - If `fromIdx < activeSectionIndex <= toIdx`, preceding elements shift left: `activeSectionIndex--`.
186 * - If `toIdx <= activeSectionIndex < fromIdx`, following elements shift right: `activeSectionIndex++`.
187 *
188 * @param fromIdx Source index in `sections` vector.
189 * @param toIdx Destination index in `sections` vector.
190 * @return bool True if indices are valid and movement succeeded; false otherwise.
191 */
192bool Notebook::MoveSection(size_t fromIdx, size_t toIdx) {
193 if (fromIdx >= sections.size() || toIdx >= sections.size() || fromIdx == toIdx) {
194 return false;
195 }
196 auto movedSec = sections[fromIdx];
197 sections.erase(sections.begin() + fromIdx);
198 sections.insert(sections.begin() + toIdx, movedSec);
199
200 // Update active section index tracking
201 if (activeSectionIndex == fromIdx) {
202 activeSectionIndex = toIdx;
203 } else if (fromIdx < activeSectionIndex && toIdx >= activeSectionIndex) {
204 activeSectionIndex--;
205 } else if (fromIdx > activeSectionIndex && toIdx <= activeSectionIndex) {
206 activeSectionIndex++;
207 }
208
209 // Re-index all sort orders
210 for (size_t i = 0; i < sections.size(); ++i) {
211 if (sections[i]) sections[i]->sortOrder = static_cast<int32_t>(i);
212 }
213 LOG_INFO(Notebook, "Reordered root section '" + (movedSec ? movedSec->name : "Unknown") + "' from index " + std::to_string(fromIdx) + " to " + std::to_string(toIdx));
214 return true;
215}

219 *
220 * @param fromIdx Source index in `sectionGroups` vector.
221 * @param toIdx Destination index in `sectionGroups` vector.
222 * @return bool True if movement succeeded; false on invalid bounds.

233 if (sectionGroups[i]) sectionGroups[i]->sortOrder = static_cast<int32_t>(i);
234 }

236 return true;

239/**
240 * @brief Moves a section (from root or another group) into a target SectionGroup.
241 *
242 * WORKING PROCESS:
243 * 1. Locates target group via `FindSectionGroupByGuid`. Returns false if not found.
244 * 2. Checks if section currently resides in root sections. If so, removes it and updates
245 * root sort orders and activeSectionIndex.
246 * 3. If not in root, searches other SectionGroups, removes it, and updates that group's orders.
247 * 4. Assigns `targetSec->groupGuid = targetGroupGuid`.
248 * 5. Inserts into target group at `targetIdx` (or appends if out of bounds).
249 * 6. Updates target group's section sort orders.
250 * 7. Maintains active section state: if the moved section was active, keeps it active and
251 * auto-expands the target group (`targetGrp->isCollapsed = false`).
252 *
253 * @param secGuid GUID of section to move.
254 * @param targetGroupGuid GUID of target SectionGroup.
255 * @param targetIdx Insertion index in target group, or (size_t)-1 to append at end.
256 * @return bool True if section was relocated successfully; false if target or section not found.
257 */
258bool Notebook::MoveSectionToGroup(const std::string& secGuid, const std::string& targetGroupGuid, size_t targetIdx) {
259 auto targetGrp = FindSectionGroupByGuid(targetGroupGuid);
260 if (!targetGrp) {
261 LOG_ERROR_CODE(Notebook, FolioErrorCode::DocSectionGroupNotFound,
262 "MoveSectionToGroup failed: Target SectionGroup not found: " + targetGroupGuid);
263 return false;
264 }
265
266 std::shared_ptr<Section> targetSec = nullptr;
267 bool wasActive = (!activeSectionGuid.empty() && activeSectionGuid == secGuid);
268
269 // 1. Try remove from root sections
270 auto rootIt = std::find_if(sections.begin(), sections.end(), [&](const std::shared_ptr<Section>& s) {
271 return s && s->guid == secGuid;
272 });
273 if (rootIt != sections.end()) {
274 targetSec = *rootIt;
275 size_t removedIdx = std::distance(sections.begin(), rootIt);
276 sections.erase(rootIt);
277
278 if (sections.empty()) {
279 activeSectionIndex = 0;
280 } else if (activeSectionIndex >= sections.size() || activeSectionIndex == removedIdx) {
281 activeSectionIndex = (sections.size() > 0) ? std::min(removedIdx, sections.size() - 1) : 0;
282 }
283
284 for (size_t i = 0; i < sections.size(); ++i) {
285 if (sections[i]) sections[i]->sortOrder = static_cast<int32_t>(i);
286 }
287 } else {
288 // 2. Try remove from other section groups
289 for (auto& grp : sectionGroups) {
290 if (!grp) continue;
291 auto it = std::find_if(grp->sections.begin(), grp->sections.end(), [&](const std::shared_ptr<Section>& s) {
292 return s && s->guid == secGuid;
293 });
294 if (it != grp->sections.end()) {
295 targetSec = *it;
296 size_t removedIdx = std::distance(grp->sections.begin(), it);
297 grp->sections.erase(it);
298
299 if (grp->sections.empty()) {
300 grp->activeSectionIndex = 0;
301 } else if (grp->activeSectionIndex >= grp->sections.size() || grp->activeSectionIndex == removedIdx) {
302 grp->activeSectionIndex = (grp->sections.size() > 0) ? std::min(removedIdx, grp->sections.size() - 1) : 0;
303 }
304
305 for (size_t i = 0; i < grp->sections.size(); ++i) {
306 if (grp->sections[i]) grp->sections[i]->sortOrder = static_cast<int32_t>(i);
307 }
308 break;
309 }
310 }
311 }
312
313 if (targetSec) {
314 targetSec->groupGuid = targetGroupGuid;
315 size_t insertPos = 0;
316 if (targetIdx < targetGrp->sections.size()) {
317 targetGrp->sections.insert(targetGrp->sections.begin() + targetIdx, targetSec);
318 insertPos = targetIdx;
319 } else {
320 insertPos = targetGrp->sections.size();
321 targetGrp->sections.push_back(targetSec);
322 }
323
324 for (size_t i = 0; i < targetGrp->sections.size(); ++i) {
325 if (targetGrp->sections[i]) targetGrp->sections[i]->sortOrder = static_cast<int32_t>(i);
326 }
327
328 // Maintain active section state
329 if (wasActive || activeSectionGuid.empty() || GetActiveSection() == nullptr) {
330 activeSectionGuid = targetSec->guid;
331 targetGrp->activeSectionIndex = insertPos;
332 targetGrp->isCollapsed = false;
333 }
334
335 LOG_INFO(Notebook, "Moved section '" + targetSec->name + "' (" + secGuid + ") into group '" + targetGrp->name + "' at index " + std::to_string(insertPos));
336 return true;
337 }
338
339 LOG_WARN_CODE(Notebook, FolioErrorCode::DocSectionNotFound,
340 "MoveSectionToGroup failed: Section not found across root or groups: " + secGuid);
341 return false;
342}

344/**
345 * @brief Moves a section from a SectionGroup back to the notebook root sections list.
346 *
347 * WORKING PROCESS:
348 * 1. Searches through all SectionGroups to locate the section by GUID.
349 * 2. Erases it from its source group and clamps the source group's activeSectionIndex.
350 * 3. Re-indexes the source group's remaining sections.
351 * 4. Clears `targetSec->groupGuid`.
352 * 5. Inserts into root `sections` at `targetIdx` (or appends if out of bounds).
353 * 6. Re-indexes all root section `sortOrder` values.
354 * 7. Maintains active selection state, pointing `activeSectionIndex` to the inserted root location.
355 *
356 * @param secGuid GUID of section to move.
357 * @param targetIdx Insertion index in root sections, or (size_t)-1 to append at end.
358 * @return bool True if relocated; false if section not found in any group.
359 */
360bool Notebook::MoveSectionToRoot(const std::string& secGuid, size_t targetIdx) {
361 std::shared_ptr<Section> targetSec = nullptr;
362 bool wasActive = (!activeSectionGuid.empty() && activeSectionGuid == secGuid);
363
364 for (auto& grp : sectionGroups) {
365 if (!grp) continue;
366 auto it = std::find_if(grp->sections.begin(), grp->sections.end(), [&](const std::shared_ptr<Section>& s) {
367 return s && s->guid == secGuid;
368 });
369 if (it != grp->sections.end()) {
370 targetSec = *it;
371 size_t removedIdx = std::distance(grp->sections.begin(), it);
372 grp->sections.erase(it);
373
374 if (grp->sections.empty()) {
375 grp->activeSectionIndex = 0;
376 } else if (grp->activeSectionIndex >= grp->sections.size() || grp->activeSectionIndex == removedIdx) {
377 grp->activeSectionIndex = (grp->sections.size() > 0) ? std::min(removedIdx, grp->sections.size() - 1) : 0;
378 }
379
380 for (size_t i = 0; i < grp->sections.size(); ++i) {
381 if (grp->sections[i]) grp->sections[i]->sortOrder = static_cast<int32_t>(i);
382 }
383 break;
384 }
385 }
386
387 if (targetSec) {
388 targetSec->groupGuid.clear();
389 size_t insertPos = 0;
390 if (targetIdx < sections.size()) {
391 sections.insert(sections.begin() + targetIdx, targetSec);
392 insertPos = targetIdx;
393 } else {
394 insertPos = sections.size();
395 sections.push_back(targetSec);
396 }
397
398 for (size_t i = 0; i < sections.size(); ++i) {
399 if (sections[i]) sections[i]->sortOrder = static_cast<int32_t>(i);
400 }
401
402 // Maintain active section state
403 if (wasActive || activeSectionGuid.empty() || GetActiveSection() == nullptr) {
404 activeSectionGuid = targetSec->guid;
405 activeSectionIndex = insertPos;
406 }
407
408 LOG_INFO(Notebook, "Moved section '" + targetSec->name + "' (" + secGuid + ") to root sections at index " + std::to_string(insertPos));
409 return true;
410 }
411
412 LOG_WARN_CODE(Notebook, FolioErrorCode::DocSectionNotFound,
413 "MoveSectionToRoot failed: Section not found in any group: " + secGuid);
414 return false;
415}

418 * @brief Removes a root section by GUID and safely clamps activeSectionIndex.
419 *
420 * SAFETY INVARIANTS:
421 * - If the removed section was the active section, updates `activeSectionGuid` to the
422 * new active section resolved via `GetActiveSection()`.
423 * - Re-indexes `sortOrder` across remaining root sections.
424 *
425 * @param secGuid GUID identifier of root section to delete.
426 * @return bool True if found and removed; false if not found.

428bool Notebook::RemoveSection(const std::string& secGuid) {

430 return s && s->guid == secGuid;
431 });
432
433 if (it != sections.end()) {
434 std::string secName = (*it)->name;
435 size_t removedIndex = std::distance(sections.begin(), it);
436 sections.erase(it);
437
438 if (sections.empty()) {
439 activeSectionIndex = 0;
440 } else if (activeSectionIndex >= sections.size() || activeSectionIndex == removedIndex) {
441 activeSectionIndex = (sections.size() > 0) ? std::min(removedIndex, sections.size() - 1) : 0;

447

449 if (sections[i]) sections[i]->sortOrder = static_cast<int32_t>(i);
450 }
451

465/**
466 * @brief Finds a section by GUID across root-level sections and all nested section groups.
467 *
468 * SEARCH ALGORITHM:
469 * 1. Checks root `sections` in O(N_root) time.
470 * 2. If not found, iterates through `sectionGroups` delegating to each group's recursive
471 * `FindSectionByGuid` in O(N_groups * N_group_sections) time.
472 *
473 * @param secGuid GUID of the section to locate.
474 * @return std::shared_ptr<Section> Matching section pointer, or nullptr if not found.
475 */
476std::shared_ptr<Section> Notebook::FindSectionByGuid(const std::string& secGuid) const {
477 for (const auto& sec : sections) {
478 if (sec && sec->guid == secGuid) return sec;
479 }
480 for (const auto& grp : sectionGroups) {
481 if (!grp) continue;
482 if (auto found = grp->FindSectionByGuid(secGuid)) {
483 return found;
484 }
485 }
486 return nullptr;
487}

489/**
490 * @brief Finds a section group by GUID across all root and nested groups.
491 *
492 * SEARCH ALGORITHM:
493 * 1. Checks if the root group GUID matches.
494 * 2. If not, delegates recursively to `FindSubGroupByGuid()` to check child sub-groups.
495 *
496 * @param grpGuid GUID of the SectionGroup to locate.
497 * @return std::shared_ptr<SectionGroup> Matching group pointer, or nullptr if not found.
498 */
499std::shared_ptr<SectionGroup> Notebook::FindSectionGroupByGuid(const std::string& grpGuid) const {
500 for (const auto& grp : sectionGroups) {
501 if (!grp) continue;
502 if (grp->guid == grpGuid) return grp;
503 if (auto found = grp->FindSubGroupByGuid(grpGuid)) {
504 return found;
505 }
506 }
507 return nullptr;
508}

510/**
511 * @brief Performs an in-memory deep copy of this notebook, including its sections and section groups.
512 *
513 * GENERAL WORKING PROCESS & INVARIANTS:
514 * 1. Generates a fresh unique persistent UUID v4 identifier via `GUIDGenerator::GenerateV4()`
515 * so the cloned notebook has distinct identity in memory and storage.
516 * 2. Determines the cloned notebook's display title: defaults to `<currentName> - Copy` if
517 * `newName` is empty.
518 * 3. Preserves visual theme accent color (`colorTag`), icon filename (`iconFile`), and expansion state (`isOpen`).
519 * 4. Deep-copies all root sections via `sec->Clone()`, ensuring page and stroke hierarchies are
520 * independently replicated with fresh GUIDs.
521 * 5. Deep-copies all section groups and their contained sections via `grp->Clone()`.
522 * 6. Guarantees at least one default section exists if source was empty.
523 * 7. Clamps and mirrors `activeSectionIndex` and resolves `activeSectionGuid`.
524 *
525 * @param newName Optional display title for the clone (defaults to source name + " - Copy").
526 * @return std::shared_ptr<Notebook> Newly allocated deep-cloned Notebook model.
527 */
528std::shared_ptr<Notebook> Notebook::Clone(const std::string& newName) const {
529 std::string safeName = newName.empty() ? (this->name + " - Copy") : newName;
530 auto copyNb = std::make_shared<Notebook>(safeName, this->colorTag, this->iconFile);
531 copyNb->guid = GUIDGenerator::GenerateV4();
532 copyNb->filePath = this->filePath;
533 copyNb->isOpen = this->isOpen;
534 copyNb->sections.clear();
535 copyNb->sectionGroups.clear();
536
537 // Deep clone root sections
538 for (const auto& sec : this->sections) {
539 if (sec) {
540 copyNb->sections.push_back(sec->Clone());
541 }
542 }
543
544 // Deep clone section groups and their child sections recursively
545 for (const auto& grp : this->sectionGroups) {
546 if (grp) {
547 copyNb->sectionGroups.push_back(grp->Clone(copyNb->guid));
548 }
549 }
550
551 // Invariant: guarantee at least one section exists
552 if (copyNb->sections.empty() && copyNb->sectionGroups.empty()) {
553 copyNb->sections.push_back(std::make_shared<Section>("New Section 1"));
554 }
555
556 // Synchronize active section tracking
557 copyNb->activeSectionIndex = (this->activeSectionIndex < copyNb->sections.size()) ? this->activeSectionIndex : 0;
558 if (!copyNb->sections.empty()) {
559 copyNb->activeSectionGuid = copyNb->sections[copyNb->activeSectionIndex]->guid;
560 }
561
562 LOG_INFO(Notebook, "Cloned notebook '" + this->name + "' -> '" + safeName + "' (" + copyNb->guid + ")");
563 return copyNb;
564}

566/**
567 * @brief Collects all CanvasPages across all sections and section groups in this notebook.
568 *
569 * TRAVERSAL & AGGREGATION ALGORITHM:
570 * 1. Iterates over all root sections. If `includeDeleted` is false, skips sections marked
571 * as soft-deleted (`sec->IsDeleted()`).
572 * 2. Collects matching canvas pages. If `includeDeleted` is false, skips pages marked
573 * as soft-deleted (`page->IsDeleted()`).
574 * 3. Recursively walks all `sectionGroups` and nested `subGroups` using a recursive lambda,
575 * applying identical soft-deletion filtering rules.
576 *
577 * @param includeDeleted If false, excludes soft-deleted items (deletedAt > 0); if true, returns all.
578 * @return std::vector<std::shared_ptr<CanvasPage>> Flat vector of matching pages in the notebook.
579 */
580std::vector<std::shared_ptr<CanvasPage>> Notebook::GetAllPages(bool includeDeleted) const {
581 std::vector<std::shared_ptr<CanvasPage>> allPages;
582
583 auto collectFromSection = [&](const std::shared_ptr<Section>& sec) {
584 if (!sec) return;
585 if (!includeDeleted && sec->IsDeleted()) return;
586 for (const auto& page : sec->pages) {
587 if (page && (includeDeleted || !page->IsDeleted())) {
588 allPages.push_back(page);
589 }
590 }
591 };
592
593 // 1. Collect from root sections
594 for (const auto& sec : sections) {
595 collectFromSection(sec);
596 }
597
598 // 2. Recursively collect from SectionGroups and nested subGroups
599 auto collectFromGroup = [&](auto& self, const std::shared_ptr<SectionGroup>& grp) -> void {
600 if (!grp) return;
601 for (const auto& sec : grp->sections) {
