35 * 1. Takes the child notebook package path (e.g. "C:/Notes/Default.foliolib/Math.notebook").

53/**
54 * @brief Writes or refreshes the self-identifying marker file (library.meta).
55 *
56 * ARCHITECTURAL PURPOSE & EXTENSION STRIPPING RESILIENCE:
57 * When users view libraries in OS File Explorers, they may rename the directory and remove
58 * the ".foliolib" extension. The `library.meta` file ensures that FolioNote can unambiguously
59 * identify the folder as a library and restore its display title.
60 *
61 * ALGORITHMIC STEPS:
62 * 1. Validate that the target path exists and is indeed a directory.
63 * 2. Construct the absolute marker filepath: `<folderPath>/library.meta`.
64 * 3. Format key-value metadata text including format identifier, schema version, name, and timestamp.
65 * 4. Write to disk using `FileManager::WriteTextAtomic` (write to temp file then rename) to prevent
66 * file corruption during sudden shutdowns or power loss.
67 *
68 * @param folderPath Absolute filesystem directory path of the library bundle.
69 * @param libraryName User-visible display label of the library.
70 */
71void LibraryManager::WriteLibraryMarker(const std::string& folderPath, const std::string& libraryName) {
72 if (!FileManager::IsDirectory(folderPath)) {
73 LOG_WARN(LibraryManager, "WriteLibraryMarker skipped: Target directory does not exist: " + folderPath);
74 return;
75 }
76
77 std::string markerPath = FileManager::JoinPath(folderPath, FOLIO_LIBRARY_MARKER_FILE);
78 std::string markerContent =
79 "# FolioNote Library Metadata\n"
80 "format=FolioNoteLibrary\n"
81 "version=1\n"
82 "name=" + (libraryName.empty() ? "Library" : libraryName) + "\n"
83 "timestamp=" + std::to_string(std::chrono::system_clock::to_time_t(std::chrono::system_clock::now())) + "\n";
84
85 if (FileManager::WriteTextAtomic(markerPath, markerContent)) {
86 LOG_INFO(LibraryManager, "Verified library marker at: " + markerPath);
87 } else {
88 LOG_ERROR(LibraryManager, "Failed to write library marker file at: " + markerPath);
89 }
90
91 // Ensure the library folder is stamped with Windows Explorer package shell identity (desktop.ini)
92 PackageMarker::MarkFolderAsPackage(folderPath, "FolioNote Library Package");
93}

95/**
96 * @brief Determines whether a given filesystem path is a recognized FolioNote library folder.
97 *
98 * DUAL-MODE DETECTION RULES & LOGIC:
99 * A directory is classified as a valid library if it meets ANY of the following criteria:
100 * 1. Extension Check: The directory path has the extension `.foliolib`.
101 * 2. Metadata Marker Check: Contains a file named `library.meta`.
102 * 3. Child Notebook Check: Contains at least one subfolder ending with `.notebook`.
103 *
104 * @param path Target filesystem path to evaluate.
105 * @return bool True if the path represents an existing directory fulfilling library criteria.
106 */
107bool LibraryManager::IsLibraryFolder(const std::string& path) {
108 if (path.empty() || !FileManager::IsDirectory(path)) {
109 return false;
110 }
111
112 // Rule 1: Packaged bundle check (.foliolib extension)
113 if (FileManager::HasExtension(path, FOLIO_LIBRARY_EXTENSION)) {
114 return true;
115 }
116
117 // Rule 2: Self-identifying marker check (library.meta)
118 std::string markerPath = FileManager::JoinPath(path, FOLIO_LIBRARY_MARKER_FILE);
119 if (FileManager::Exists(markerPath)) {
120 return true;
121 }
122
123 // Rule 3: Child .notebook packages check
124 auto entries = FileManager::ListEntries(path, FOLIO_NOTEBOOK_EXTENSION);
125 for (const auto& entry : entries) {
126 if (entry.type == FileType::Directory) {
127 return true;
128 }
129 }
130
131 return false;
132}

147/**
148 * @brief Resolves a library root path on disk, tolerating user modification of the `.foliolib` extension.
149 *
150 * HEURISTIC ADAPTATION PROCESS:
151 * - Case 1 (Direct Match): The declared path exists directly as-is on the disk.
152 * - Case 2 (Extension Stripped): The declared path ended with `.foliolib`, but the user removed
153 * the extension in the OS file explorer. Checks if `<parent>/<stem>` exists and satisfies `IsLibraryFolder`.
154 * - Case 3 (Extension Added): The declared path was plain, but the user renamed it to `.foliolib`.
155 * Checks if `<parent>/<filename>.foliolib` exists and satisfies `IsLibraryFolder`.
156 * - Fallback: Returns `declaredPath` unchanged if no alternate match is found.
157 *
158 * @param declaredPath Path currently stored in configuration or runtime memory.
159 * @return std::string Resolved existing filesystem path, or `declaredPath` if neither variant exists.
160 */
161std::string LibraryManager::ResolveActualLibraryPath(const std::string& declaredPath) {
162 if (declaredPath.empty()) return declaredPath;
163
164 // Case 1: Path exists directly as declared
165 if (FileManager::IsDirectory(declaredPath)) {
166 return declaredPath;
167 }
168
169 std::string parent = FileManager::GetParentPath(declaredPath);
170
171 // Case 2: Declared path had .foliolib, but user stripped it in Explorer/Finder
172 if (FileManager::HasExtension(declaredPath, FOLIO_LIBRARY_EXTENSION)) {
173 std::string stripped = FileManager::JoinPath(parent, FileManager::GetStem(declaredPath));
174 if (FileManager::IsDirectory(stripped) && IsLibraryFolder(stripped)) {
175 LOG_INFO(LibraryManager, "Auto-detected unpacked library folder (user removed .foliolib): " + stripped);
176 return stripped;
177 }
178 }
179
180 // Case 3: Declared path was plain, but user appended .foliolib to package it
181 std::string withExt = FileManager::JoinPath(parent, FileManager::GetFileName(declaredPath) + FOLIO_LIBRARY_EXTENSION);
182 if (FileManager::IsDirectory(withExt) && IsLibraryFolder(withExt)) {
183 LOG_INFO(LibraryManager, "Auto-detected packaged library bundle (user added .foliolib): " + withExt);
184 return withExt;
185 }
186
187 LOG_WARN(LibraryManager, "Library path unresolved or missing on disk: " + declaredPath);
188 return declaredPath;
189}

206/**
207 * @brief Cleans and canonicalizes an input path to ensure it represents a valid library directory.
208 *
209 * GENERAL WORKING PROCESS & SANITIZATION RULES:
210 * 1. Loop-strips any trailing `.notebook` package paths to prevent nested encapsulation errors.
211 * 2. If the path is empty, ".", or matches "FolioNote", routes to `<defaultParentDir>/<defaultBundleName>`.
212 * (If a legacy bundle `<defaultParentDir>/<defaultBundleName>.foliolib` exists on disk, that is favored).
213 * 3. Normalizes all path separators to platform-canonical format without forcing a `.foliolib` extension.
214 *
215 * @param inputPath User-supplied or configuration directory path.
216 * @param defaultParentDir Fallback parent directory to place bundle in.
217 * @param defaultBundleName Fallback base name (defaults to "Default").
218 * @return std::string Cleaned, canonical absolute path to a library directory.
219 */
220std::string LibraryManager::SanitizeLibraryRoot(
221 const std::string& inputPath,
222 const std::string& defaultParentDir,
223 const std::string& defaultBundleName
224) {
225 std::string p = inputPath;
226
227 // 1. Climb out of any .notebook packages to prevent recursion errors
228 while (!p.empty() && FileManager::HasExtension(p, FOLIO_NOTEBOOK_EXTENSION)) {
229 p = FileManager::GetParentPath(p);
230 }
231
232 // 2. If empty or relative dot, place default bundle in defaultParentDir
233 if (p.empty() || p == "." || p == "FolioNote") {
234 std::string candidateWithExt = FileManager::JoinPath(defaultParentDir, defaultBundleName + FOLIO_LIBRARY_EXTENSION);
235 if (FileManager::IsDirectory(candidateWithExt)) {
236 return FileManager::NormalizeSeparators(candidateWithExt);
237 }
238 return FileManager::NormalizeSeparators(FileManager::JoinPath(defaultParentDir, defaultBundleName));
239 }
240
241 // 3. For any provided paths, accept as-is and normalize path separators
242 return FileManager::NormalizeSeparators(p);
243}
