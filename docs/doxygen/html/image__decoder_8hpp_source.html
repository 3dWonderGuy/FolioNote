2/**
3 * =========================================================================================
4 * @file core/objects/media/images/image_decoder.hpp
5 * @brief High-Performance Multi-Format Image Decoding Pipeline and RAM Protection Service
6 * =========================================================================================
7 *
8 * ARCHITECTURAL DESIGN & WORKING PROCESS:
9 * ---------------------------------------
10 * ImageDecoder serves as FolioNote's unified decoding bridge, taking raw binary data
11 * (from memory buffers or companion sidecar disk files) and decoding them into
12 * hardware-accelerated Blend2D `BLImage` surfaces.
13 *
14 * Supported Decoding Pipelines:
15 * 1. Blend2D Native SIMD Pipeline:
16 * High-performance JIT-accelerated decoding for PNG, JPEG, BMP, and QOI.
17 * 2. LunaSVG Vector Rasterization Pipeline:
18 * Parses SVG XML documents and renders them to pixel-perfect `BLImage` surfaces
19 * at display resolution.
20 * 3. Extended Format Fallback Pipeline:
21 * Decodes GIF, TGA, PSD, and other formats into raw RGBA buffers and maps them
22 * to Blend2D `BL_FORMAT_PRGB32` surfaces.
23 *
24 * RAM PROTECTION & GIGABYTE FILE SAFETY:
25 * --------------------------------------
26 * A raw 16000 x 16000 uncompressed TIFF or multi-gigabyte raw scan consumes:
27 * RAM = 16000 * 16000 * 4 bytes = 1,024,000,000 bytes = 1.02 GB
28 *
29 * To prevent runaway memory consumption and Out-Of-Memory (OOM) crashes:
30 * - The original file is stored 100% untouched on disk (zero data loss).
31 * - During decoding, ImageDecoder enforces `s_maxDecodedDimension` (default: 4096 px).
32 * - If max(naturalWidth, naturalHeight) > s_maxDecodedDimension:
33 * scale = s_maxDecodedDimension / max(naturalWidth, naturalHeight)
34 * downsampledWidth = naturalWidth * scale
35 * downsampledHeight = naturalHeight * scale
36 * Capping display RAM consumption to:
37 * Max RAM = 4096 * 4096 * 4 bytes = 67.1 MB (94% RAM reduction).
38 */

52/**
53 * @struct ImageFrame
54 * @brief Encapsulates a single decoded animation frame with duration metadata.
55 *
56 * Mathematical & Timing Model:
57 * - Each frame maintains a private Blend2D BL_FORMAT_PRGB32 surface.
58 * - delayMs specifies presentation duration in milliseconds.
59 * - GIF standard defines delay in 1/100th second (10ms increments).
60 * - Web browsers standardly clamp delays <= 10ms to 100ms (10 FPS)
61 * to prevent runaway CPU usage from archaic GIF encoders.
62 */
63struct ImageFrame {
64 BLImage image; ///< Decoded Blend2D PRGB32 raster surface for this frame
65 uint32_t delayMs = 100; ///< Presentation duration in milliseconds (e.g. 100ms = 10 FPS)
66};

116 /**
117 * @brief Decodes an image from a raw binary memory buffer.
118 *
119 * Working Process:
120 * 1. Inspects filename hint or magic bytes to deduce the format.
121 * 2. If SVG: Delegates to LunaSVG vector rasterizer.
122 * 3. If PNG, JPEG, BMP, QOI: Delegates to Blend2D SIMD decoder.
123 * 4. If GIF / other: Delegates to extended fallback decoder.
124 * 5. Checks if decoded dimensions exceed s_maxDecodedDimension:
125 * If true, scales down to fit s_maxDecodedDimension to protect system RAM.
126 *
127 * @param[in] data Pointer to raw image bytes in memory.
128 * @param[in] size Size of data buffer in bytes.
129 * @param[in] filenameHint Optional path or extension hint (e.g. "asset.svg", ".webp").
130 * @return DecodedImageResult containing BLImage surface, natural dimensions, and format tag.
131 */
132 static DecodedImageResult DecodeFromMemory(
133 const uint8_t* data,
134 size_t size,
135 std::string_view filenameHint = ""
136 );

138 /**
139 * @brief Decodes an image from a physical filesystem path or notebook package asset.
140 *
141 * Working Process:
142 * 1. Resolves path using Folio::FileManager::ResolveAssetPath (handles notebook packages or absolute paths).
143 * 2. Reads binary payload into memory via FileManager::ReadBinary for safe Unicode I/O.
144 * 3. Delegates to DecodeFromMemory with filename hint.
145 *
146 * @param[in] filePath Relative package path (e.g. "imports/images/<hash>.png") or absolute disk path.
147 * @return DecodedImageResult containing BLImage surface, natural dimensions, and format tag.
148 */
149 static DecodedImageResult DecodeFromFile(const std::string& filePath);

151 /**
152 * @brief Helper to convert a raw 32-bit RGBA pixel buffer into a Blend2D PRGB32 surface.
153 *
154 * Mathematical Formula (Alpha Premultiplication):
155 * pr = (r * a + 127) / 255
156 * pg = (g * a + 127) / 255
157 * pb = (b * a + 127) / 255
158 * dstPixel = (a << 24) | (pr << 16) | (pg << 8) | pb
159 *
160 * @param[in] rgbaData Raw 32-bit RGBA pixel array (width * height * 4 bytes).
161 * @param[in] width Pixel width.
162 * @param[in] height Pixel height.
163 * @return Blend2D BLImage surface in native BL_FORMAT_PRGB32 format.
164 */
165 static BLImage ConvertRgbaToBLImage(const uint8_t* rgbaData, uint32_t width, uint32_t height);
166};
