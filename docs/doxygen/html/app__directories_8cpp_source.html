90/**
91 * @brief Resolves the dedicated, local-only application data directory deep in the system file system.
92 *
93 * GENERAL WORKING PROCESS & ARCHITECTURAL MOTIVATION:
94 * 1. Isolation from Cloud Services (OneDrive, Dropbox, iCloud):
95 * User documents (active notebooks) live in the Documents folder so they can sync to the cloud if the user desires.
96 * However, backups, ephemeral caches, and disaster recovery snapshots must NEVER be placed in a cloud-synced folder
97 * because doing so triggers continuous upload churn, eats storage quotas, causes file-locking collisions,
98 * and fails when encountering dehydrated cloud reparse points.
99 * 2. Platform-Specific Resolution:
100 * - Windows: Uses SHGetKnownFolderPath with FOLDERID_LocalAppData (%LOCALAPPDATA%/FolioNote).
101 * Guaranteed to be local machine storage excluded from OneDrive synchronization.
102 * - Android: Resolves via SDL_GetPrefPath to internal app sandboxed storage.
103 * - Linux/macOS: Uses SDL_GetPrefPath("", "FolioNote") resolving to $XDG_DATA_HOME/FolioNote/
104 * (~/.local/share/FolioNote) or ~/Library/Application Support/FolioNote/.
105 *
106 * @return Canonical normalized UTF-8 filesystem path to local-only app data.
107 */
108std::string AppDirectories::GetLocalDataDirectory() {
109 {
110 std::lock_guard<std::mutex> lock(s_localDataMutex);
111 if (!s_activeLocalDataRoot.empty()) {
112 return s_activeLocalDataRoot;
113 }
114 }
115
116#if defined(_WIN32)
117 PWSTR localAppPathW = NULL;
118 if (SUCCEEDED(SHGetKnownFolderPath(FOLDERID_LocalAppData, 0, NULL, &localAppPathW))) {
119 std::filesystem::path localRoot = std::filesystem::path(localAppPathW) / "FolioNote";
120 CoTaskMemFree(localAppPathW);
121 std::string res = PathUtils::NormalizeSeparators(PathUtils::NativePathToUtf8(localRoot));
122 std::lock_guard<std::mutex> lock(s_localDataMutex);
123 s_activeLocalDataRoot = res;
124 return res;
125 }
126#elif defined(__ANDROID__)
127 if (SDL_WasInit(0) != 0) {
128 char* pref = SDL_GetPrefPath("UniversalFramework", "FolioNote");
129 if (pref && pref[0] != '\0') {
130 std::string res = PathUtils::NormalizeSeparators(std::string(pref));
131 SDL_free(pref);
132 std::lock_guard<std::mutex> lock(s_localDataMutex);
133 s_activeLocalDataRoot = res;
134 return res;
135 }
136 if (pref) SDL_free(pref);
137 }
138#else
139 // Linux, macOS, and POSIX desktop platforms:
140 // SDL3 does not define a separate SDL_FOLDER_LOCAL_APP_DATA enum value.
141 // Instead, SDL_GetPrefPath("", "FolioNote") queries standard platform app storage:
142 // - Linux: $XDG_DATA_HOME/FolioNote/ (defaults to ~/.local/share/FolioNote/)
143 // - macOS: ~/Library/Application Support/FolioNote/
144 // This directory is strictly machine-local and immune to cloud folder sync churn.
145 char* pref = SDL_GetPrefPath("", "FolioNote");
146 if (pref && pref[0] != '\0') {
147 std::string res = PathUtils::NormalizeSeparators(std::string(pref));
148 SDL_free(pref);
149 std::lock_guard<std::mutex> lock(s_localDataMutex);
150 s_activeLocalDataRoot = res;
151 return res;
152 }
153 if (pref) {
154 SDL_free(pref);
155 }
156#endif
157
158 std::string fallback = PathUtils::JoinPath(GetAppRootDirectory(), "local");
159 std::lock_guard<std::mutex> lock(s_localDataMutex);
160 s_activeLocalDataRoot = fallback;
161 return fallback;
162}

199std::string AppDirectories::ResolveAssetPath(const std::string& path) {
200 if (path.empty()) {
201 return "";
202 }
203
204 std::error_code ec;
205
206 // 1. If path is already absolute and exists on disk, normalize and return
207 if (PathUtils::IsAbsolutePath(path) && std::filesystem::exists(PathUtils::Utf8ToNativePath(path), ec)) {
208 return PathUtils::NormalizeSeparators(path);
209 }
210
211 // 2. Check against the registered active notebook package root
212 std::string root = GetActivePackageRoot();
213 if (!root.empty()) {
214 std::string candidate = PathUtils::JoinPath(root, path);
215 if (std::filesystem::exists(PathUtils::Utf8ToNativePath(candidate), ec)) {
216 return candidate;
217 }
218 }
219
220 // 3. Fallback: check if relative to application current working directory
221 if (std::filesystem::exists(PathUtils::Utf8ToNativePath(path), ec)) {
222 return PathUtils::NormalizeSeparators(path);
223 }
224
225 // 4. If not physically on disk yet, return candidate joined with active package root if available
226 if (!root.empty() && !PathUtils::IsAbsolutePath(path)) {
227 return PathUtils::JoinPath(root, path);
228 }
229
230 return PathUtils::NormalizeSeparators(path);
231}
