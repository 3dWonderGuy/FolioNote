83 /**
84 * @brief Linearly interpolates two 32-bit packed ImU32 RGBA colors.
85 * @param a Starting color.
86 * @param b Ending color.
87 * @param t Interpolation parameter in [0.0, 1.0].
88 * @return Interpolated packed ImU32 color.
89 */
90 static inline ImU32 Lerp(ImU32 a, ImU32 b, float t) noexcept {
91 if (t <= 0.0f) return a;
92 if (t >= 1.0f) return b;
93
94 uint32_t aR = (a >> IM_COL32_R_SHIFT) & 0xFF;
95 uint32_t aG = (a >> IM_COL32_G_SHIFT) & 0xFF;
96 uint32_t aB = (a >> IM_COL32_B_SHIFT) & 0xFF;
97 uint32_t aA = (a >> IM_COL32_A_SHIFT) & 0xFF;
98
99 uint32_t bR = (b >> IM_COL32_R_SHIFT) & 0xFF;
100 uint32_t bG = (b >> IM_COL32_G_SHIFT) & 0xFF;
101 uint32_t bB = (b >> IM_COL32_B_SHIFT) & 0xFF;
102 uint32_t bA = (b >> IM_COL32_A_SHIFT) & 0xFF;
103
104 uint32_t r = static_cast<uint32_t>(aR + t * (static_cast<float>(bR) - aR));
105 uint32_t g = static_cast<uint32_t>(aG + t * (static_cast<float>(bG) - aG));
106 uint32_t bVal = static_cast<uint32_t>(aB + t * (static_cast<float>(bB) - aB));
107 uint32_t aVal = static_cast<uint32_t>(aA + t * (static_cast<float>(bA) - aA));
108
109 return IM_COL32(r, g, bVal, aVal);
110 }
111
112 /**
113 * @brief Adjusts the alpha channel of a packed ImU32 color.
114 * @param col Base color.
115 * @param alpha Multiplier in [0.0, 1.0].
116 * @return Color with adjusted alpha.
117 */
118 static inline ImU32 WithAlpha(ImU32 col, float alpha) noexcept {
119 uint32_t r = (col >> IM_COL32_R_SHIFT) & 0xFF;
120 uint32_t g = (col >> IM_COL32_G_SHIFT) & 0xFF;
121 uint32_t b = (col >> IM_COL32_B_SHIFT) & 0xFF;
122 uint32_t a = (col >> IM_COL32_A_SHIFT) & 0xFF;
123 uint32_t newA = static_cast<uint32_t>(static_cast<float>(a) * std::clamp(alpha, 0.0f, 1.0f));
124 return IM_COL32(r, g, b, newA);
125 }
