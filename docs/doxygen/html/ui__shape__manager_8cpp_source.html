14void UIShapeManager::DrawShadow(
15 ImDrawList* drawList,
16 ImVec2 pMin,
17 ImVec2 pMax,
18 float elevation,
19 float rounding
20) {
21 if (!drawList || elevation <= 0.0f) return;
22
23 // Multi-pass layered ambient occlusion shadow:
24 // Pass 1: Direct contact shadow (tight, slightly darker)
25 float off1 = elevation * 0.35f;
26 float spread1 = 1.5f;
27 ImVec2 min1(pMin.x - spread1, pMin.y + off1 - spread1);
28 ImVec2 max1(pMax.x + spread1, pMax.y + off1 + spread1);
29 drawList->AddRectFilled(min1, max1, IM_COL32(0, 0, 0, static_cast<int>(std::min(45.0f, elevation * 3.5f))), rounding + 1.0f);
30
31 // Pass 2: Soft ambient field shadow (diffuse, soft falloff)
32 float off2 = elevation * 0.75f;
33 float spread2 = elevation * 0.85f;
34 ImVec2 min2(pMin.x - spread2, pMin.y + off2 - spread2);
35 ImVec2 max2(pMax.x + spread2, pMax.y + off2 + spread2);
36 drawList->AddRectFilled(min2, max2, IM_COL32(0, 0, 0, static_cast<int>(std::min(30.0f, elevation * 2.2f))), rounding + spread2 * 0.6f);
37}

39void UIShapeManager::DrawAcrylicPanel(
40 ImDrawList* drawList,
41 ImVec2 pMin,
42 ImVec2 pMax,
43 ImU32 baseFill,
44 ImU32 borderCol,
45 float rounding,
46 float elevation
47) {
48 if (!drawList) return;
49
50 // 1. Cast shadow if elevated
51 if (elevation > 0.0f) {
52 DrawShadow(drawList, pMin, pMax, elevation, rounding);
53 }
54
55 // 2. Base acrylic fill with subtle top-to-bottom luminance gradient
56 // Top is slightly highlighted (+10% luminance), bottom is base
57 uint32_t r = (baseFill >> IM_COL32_R_SHIFT) & 0xFF;
58 uint32_t g = (baseFill >> IM_COL32_G_SHIFT) & 0xFF;
59 uint32_t b = (baseFill >> IM_COL32_B_SHIFT) & 0xFF;
60 uint32_t a = (baseFill >> IM_COL32_A_SHIFT) & 0xFF;
61
62 uint32_t topR = std::min(255u, r + 15u);
63 uint32_t topG = std::min(255u, g + 15u);
64 uint32_t topB = std::min(255u, b + 15u);
65 ImU32 topCol = IM_COL32(topR, topG, topB, a);
66
67 drawList->AddRectFilledMultiColor(pMin, pMax, topCol, topCol, baseFill, baseFill);
68
69 // 3. Perimeter border
70 if ((borderCol & IM_COL32_A_MASK) != 0) {
71 drawList->AddRect(pMin, pMax, borderCol, rounding, 0, 1.0f);
72 }
73
74 // 4. Specular top-edge highlight (1px inset highlight simulating direct light)
75 if (rounding > 0.0f) {
76 float inset = std::min(rounding, 8.0f);
77 ImVec2 specStart(pMin.x + inset, pMin.y + 1.0f);
78 ImVec2 specEnd(pMax.x - inset, pMin.y + 1.0f);
79 drawList->AddLine(specStart, specEnd, IM_COL32(255, 255, 255, 38), 1.0f);
80 }
81}

107void UIShapeManager::DrawPillBadge(
108 ImDrawList* drawList,
109 ImVec2 center,
110 const char* text,
111 ImU32 bgCol,
112 ImU32 textCol,
113 ImFont* font
114) {
115 if (!drawList || !text || text[0] == '\0') return;
116
117 if (font) ImGui::PushFont(font);
118 ImVec2 textSize = ImGui::CalcTextSize(text);
119
120 float padX = 7.0f;
121 float padY = 2.5f;
122 float halfW = (textSize.x + padX * 2.0f) * 0.5f;
123 float halfH = (textSize.y + padY * 2.0f) * 0.5f;
124
125 ImVec2 bMin(center.x - halfW, center.y - halfH);
126 ImVec2 bMax(center.x + halfW, center.y + halfH);
127
128 drawList->AddRectFilled(bMin, bMax, bgCol, halfH);
129 drawList->AddText(ImVec2(center.x - textSize.x * 0.5f, center.y - textSize.y * 0.5f), textCol, text);
130
131 if (font) ImGui::PopFont();
132}

134void UIShapeManager::DrawGlow(
135 ImDrawList* drawList,
136 ImVec2 center,
137 float radius,
138 ImU32 glowColor
139) {
140 if (!drawList || radius <= 0.0f) return;
141
142 uint32_t r = (glowColor >> IM_COL32_R_SHIFT) & 0xFF;
143 uint32_t g = (glowColor >> IM_COL32_G_SHIFT) & 0xFF;
144 uint32_t b = (glowColor >> IM_COL32_B_SHIFT) & 0xFF;
145 uint32_t baseA = (glowColor >> IM_COL32_A_SHIFT) & 0xFF;
146
147 // 3 concentric alpha rings to simulate radial falloff
148 const int rings = 3;
149 for (int i = 0; i < rings; ++i) {
150 float factor = static_cast<float>(i + 1) / static_cast<float>(rings);
151 float currentRadius = radius * factor;
152 float alphaFraction = (1.0f - static_cast<float>(i) / rings);
153 int a = static_cast<int>(static_cast<float>(baseA) * 0.35f * alphaFraction);
154 drawList->AddCircleFilled(center, currentRadius, IM_COL32(r, g, b, a), 24);
155 }
156}
