375 * @brief Hit-tests the gizmo at the given screen pixel coordinates.
376 */
377 GizmoHitResult HitTest(float screenX, float screenY, const CanvasTransform& transform) const {
378 GizmoHitResult res;
379 if (!hasSelection || selectedObjects.empty()) return res;
380
381 // Locked gizmo handling for single selection
382 if (selectedObjects.size() == 1) {
383 GizmoStyle style = selectedObjects[0]->GetGizmoStyle();
384 if (style == GizmoStyle::None) return res;
385
386 if (style == GizmoStyle::TwoPoint) {
387 auto arrow = std::static_pointer_cast<Folio::SmartArrowObject>(selectedObjects[0]);
388 Point2D s0 = transform.WorldToScreen(arrow->x1, arrow->y1);
389 float dx0 = screenX - static_cast<float>(s0.x);
390 float dy0 = screenY - static_cast<float>(s0.y);
391 if ((dx0 * dx0 + dy0 * dy0) <= (HIT_RADIUS * HIT_RADIUS)) {
392 res.hit = true;
393 res.role = HandleRole::Custom;
394 res.customId = 0;
395 return res;
396 }
397
398 Point2D s1 = transform.WorldToScreen(arrow->x2, arrow->y2);
399 float dx1 = screenX - static_cast<float>(s1.x);
400 float dy1 = screenY - static_cast<float>(s1.y);
401 if ((dx1 * dx1 + dy1 * dy1) <= (HIT_RADIUS * HIT_RADIUS)) {
402 res.hit = true;
403 res.role = HandleRole::Custom;
404 res.customId = 1;
405 return res;
406 }
407
408 // If not hitting endpoint handles, check if clicking the connector body
409 Point2D worldPt = transform.ScreenToWorld(screenX, screenY);
410 if (arrow->HitTest(worldPt.x, worldPt.y)) {
411 res.hit = true;
412 res.role = HandleRole::Body;
413 return res;
414 }
415 return res;
416 }
417 }
418
419 // Standard 8-point Bounding Box Hit-Testing
420 Point2D sMin = transform.WorldToScreen(bounds.minX, bounds.minY);
421 Point2D sMax = transform.WorldToScreen(bounds.maxX, bounds.maxY);
422
423 float left = static_cast<float>(std::min(sMin.x, sMax.x));
424 float top = static_cast<float>(std::min(sMin.y, sMax.y));
425 float right = static_cast<float>(std::max(sMin.x, sMax.x));
426 float bottom = static_cast<float>(std::max(sMin.y, sMax.y));
427 float midX = (left + right) * 0.5f;
428 float midY = (top + bottom) * 0.5f;
429
430 auto HitCircle = [&](float cx, float cy) -> bool {
431 float dx = screenX - cx;
432 float dy = screenY - cy;
433 return (dx * dx + dy * dy) <= (HIT_RADIUS * HIT_RADIUS);
434 };
435
436 // For MoveOnly objects, skip rotation knob and resize handles entirely!
437 bool isMoveOnly = (selectedObjects.size() == 1 && selectedObjects[0]->GetGizmoStyle() == GizmoStyle::MoveOnly);
438
439 if (!isMoveOnly) {
440 // 1. Rotation knob
441 if (HitCircle(midX, top - ROTATION_ARM_LENGTH)) {
442 res.hit = true;
443 res.role = HandleRole::Rotation;
444 return res;
445 }
446
447 // 2. Corner and edge handles
448 if (HitCircle(left, top)) { res.hit = true; res.role = HandleRole::TopLeft; return res; }
449 if (HitCircle(midX, top)) { res.hit = true; res.role = HandleRole::TopCenter; return res; }
450 if (HitCircle(right, top)) { res.hit = true; res.role = HandleRole::TopRight; return res; }
451 if (HitCircle(right, midY)) { res.hit = true; res.role = HandleRole::RightCenter; return res; }
452 if (HitCircle(right, bottom)) { res.hit = true; res.role = HandleRole::BottomRight; return res; }
453 if (HitCircle(midX, bottom)) { res.hit = true; res.role = HandleRole::BottomCenter; return res; }
454 if (HitCircle(left, bottom)) { res.hit = true; res.role = HandleRole::BottomLeft; return res; }
455 if (HitCircle(left, midY)) { res.hit = true; res.role = HandleRole::LeftCenter; return res; }
456 }
457
458 // 3. Interior body (move)
459 const float PADDING = 4.0f;
460 if (screenX >= (left - PADDING) && screenX <= (right + PADDING) &&
461 screenY >= (top - PADDING) && screenY <= (bottom + PADDING)) {
462 // If only a single object is selected, verify that the pointer actually hits the object's body
463 // or is near the gizmo border. This ensures that clicking inside an unfilled shape container
464 // doesn't block selection of items inside it!
465 if (selectedObjects.size() == 1 && selectedObjects[0]) {
466 Point2D worldPt = transform.ScreenToWorld(screenX, screenY);
467 bool nearBorder = (screenX <= (left + PADDING + 4.0f) || screenX >= (right - PADDING - 4.0f) ||
468 screenY <= (top + PADDING + 4.0f) || screenY >= (bottom - PADDING - 4.0f));
469 if (nearBorder || selectedObjects[0]->HitTest(worldPt.x, worldPt.y)) {
470 res.hit = true;
471 res.role = HandleRole::Body;
472 return res;
473 }
474 } else {
475 res.hit = true;
476 res.role = HandleRole::Body;
477 return res;
478 }
479 }
480
481 return res;
482 }

545 /**
546 * @brief Updates the transformation during an active drag with optional grid locking.
547 *
548 * Mathematical Process:
549 * - When ShouldSnapToGrid() is true:
550 * * Move: Target bounds top-left snaps to nearest grid multiple:
551 * snapped = round((initMin + delta) / G) * G.
552 * Translation applied from initialStates snapshots, avoiding numeric drift.
553 * * Resize: Active handle world coordinate snaps to grid line:
554 * snappedHandle = round((initHandle + delta) / G) * G.
555 * Dimensions scale relative to fixed opposing anchor, guaranteeing integer cell widths.
556 * * Custom handles: Target endpoint coordinate snaps to nearest grid vertex.
557 * - When ShouldSnapToGrid() is false (or ink stroke selected):
558 * * Free floating subpixel continuous manipulation with aspect-ratio corner preservation.
559 */
560 bool OnPointerMove(float screenX, float screenY, const CanvasTransform& transform,
561 bool inLockToGrid = false, double inGridSpacingMm = 5.0) {
562 if (!isDragging || selectedObjects.empty()) return false;
563
564 lockToGrid = inLockToGrid;
565 gridSpacingMm = inGridSpacingMm;
566
567 Point2D currentWorld = transform.ScreenToWorld(screenX, screenY);
568
569 // 1. Move (Translation)
570 if (activeRole == HandleRole::Body) {
571 if (ShouldSnapToGrid()) {
572 // Snap top-left of the bounding box to the nearest paper grid intersection
573 double targetX = initialBounds.minX + (currentWorld.x - dragStartWorld.x);
574 double targetY = initialBounds.minY + (currentWorld.y - dragStartWorld.y);
575 double snappedX = std::round(targetX / gridSpacingMm) * gridSpacingMm;
576 double snappedY = std::round(targetY / gridSpacingMm) * gridSpacingMm;
577 double totalDx = snappedX - initialBounds.minX;
578 double totalDy = snappedY - initialBounds.minY;
579
580 BLMatrix2D trans = BLMatrix2D::make_translation(totalDx, totalDy);
581 for (size_t i = 0; i < selectedObjects.size(); ++i) {
582 if (i < initialStates.size() && selectedObjects[i]) {
583 selectedObjects[i]->transform = initialStates[i].initialTransform;
584 selectedObjects[i]->ApplyTransform(trans);
585 }
586 }
587 lastDragWorld = currentWorld;
588 RecalculateBounds();
589 } else {
590 double dx = currentWorld.x - lastDragWorld.x;
591 double dy = currentWorld.y - lastDragWorld.y;
592 if (dx != 0.0 || dy != 0.0) {
593 BLMatrix2D trans = BLMatrix2D::make_translation(dx, dy);
594 for (auto& obj : selectedObjects) {
595 if (obj) obj->ApplyTransform(trans);
596 }
597 lastDragWorld = currentWorld;
598 RecalculateBounds();
599 }
600 }
601 return true;
602 }
603
604 // 2. Rotation (Dual-Mode: 45° Snapping when close, continuous when far)
605 if (activeRole == HandleRole::Rotation) {
606 currentPointerScreen = { screenX, screenY };
607
608 Point2D centerScreen = transform.WorldToScreen(initialCenterWorld.x, initialCenterWorld.y);
609 float mouseDistScreen = std::hypot(screenX - static_cast<float>(centerScreen.x),
610 screenY - static_cast<float>(centerScreen.y));
611
612 // Dynamic snap radius based on bounding box size
613 Point2D sMin = transform.WorldToScreen(initialBounds.minX, initialBounds.minY);
614 Point2D sMax = transform.WorldToScreen(initialBounds.maxX, initialBounds.maxY);
615 float boxHalfDiag = 0.5f * std::hypot(static_cast<float>(sMax.x - sMin.x), static_cast<float>(sMax.y - sMin.y));
616 currentSnapRadius = std::max(boxHalfDiag + 45.0f, 100.0f);
617
618 double startAngle = std::atan2(dragStartWorld.y - initialCenterWorld.y, dragStartWorld.x - initialCenterWorld.x);
619 double currAngle = std::atan2(currentWorld.y - initialCenterWorld.y, currentWorld.x - initialCenterWorld.x);
620 double rawDeltaAngle = currAngle - startAngle;
621
622 // Handle branch cut [-PI, PI]
623 while (rawDeltaAngle > M_PI) rawDeltaAngle -= 2.0 * M_PI;
624 while (rawDeltaAngle < -M_PI) rawDeltaAngle += 2.0 * M_PI;
625
626 // Close to box: locks rotation to 45 degree increments
627 // Moving mouse a bit out: full continuous precision control
628 if (mouseDistScreen < currentSnapRadius) {
629 isAngleSnapped = true;
630 double rawDeg = rawDeltaAngle * (180.0 / M_PI);
631 double snappedDeg = std::round(rawDeg / 45.0) * 45.0;
632 currentRotationAngle = snappedDeg * (M_PI / 180.0);
633 displayAngleDeg = snappedDeg;
634 } else {
635 isAngleSnapped = false;
636 currentRotationAngle = rawDeltaAngle;
637 displayAngleDeg = rawDeltaAngle * (180.0 / M_PI);
638 }
639
640 while (displayAngleDeg > 180.0) displayAngleDeg -= 360.0;
641 while (displayAngleDeg <= -180.0) displayAngleDeg += 360.0;
642
643 BLMatrix2D rotMatrix = BLMatrix2D::make_identity();
644 rotMatrix.post_translate(-initialCenterWorld.x, -initialCenterWorld.y);
645 rotMatrix.post_rotate(currentRotationAngle);
646 rotMatrix.post_translate(initialCenterWorld.x, initialCenterWorld.y);
647
648 // Apply to each object from its initial state snapshot
649 for (size_t i = 0; i < selectedObjects.size(); ++i) {
650 if (i < initialStates.size() && selectedObjects[i]) {
651 selectedObjects[i]->transform = initialStates[i].initialTransform;
652 selectedObjects[i]->ApplyTransform(rotMatrix);
653 }
654 }
655 // Do NOT recalculate AABB bounds during active rotation; box rotates along with contents!
656 return true;
657 }
658
659 // 3. Scaling / Stretching (8 Bounding Box Grips)
660 if (activeRole >= HandleRole::TopLeft && activeRole <= HandleRole::LeftCenter) {
661 double initW = initialBounds.maxX - initialBounds.minX;
662 double initH = initialBounds.maxY - initialBounds.minY;
663 if (initW <= 0.001) initW = 1.0;
664 if (initH <= 0.001) initH = 1.0;
665
666 // Determine fixed anchor point opposite to the active handle
667 Point2D anchor;
668 Point2D handleInit;
669 bool scaleX = true;
670 bool scaleY = true;
671
672 switch (activeRole) {
673 case HandleRole::TopLeft:
674 anchor = { initialBounds.maxX, initialBounds.maxY };
675 handleInit = { initialBounds.minX, initialBounds.minY };
676 break;
677 case HandleRole::TopCenter:
678 anchor = { (initialBounds.minX + initialBounds.maxX) * 0.5, initialBounds.maxY };
679 handleInit = { anchor.x, initialBounds.minY };
680 scaleX = false;
681 break;
682 case HandleRole::TopRight:
683 anchor = { initialBounds.minX, initialBounds.maxY };
684 handleInit = { initialBounds.maxX, initialBounds.minY };
685 break;
686 case HandleRole::RightCenter:
687 anchor = { initialBounds.minX, (initialBounds.minY + initialBounds.maxY) * 0.5 };
688 handleInit = { initialBounds.maxX, anchor.y };
689 scaleY = false;
690 break;
691 case HandleRole::BottomRight:
692 anchor = { initialBounds.minX, initialBounds.minY };
693 handleInit = { initialBounds.maxX, initialBounds.maxY };
694 break;
695 case HandleRole::BottomCenter:
696 anchor = { (initialBounds.minX + initialBounds.maxX) * 0.5, initialBounds.minY };
697 handleInit = { anchor.x, initialBounds.maxY };
698 scaleX = false;
699 break;
700 case HandleRole::BottomLeft:
701 anchor = { initialBounds.maxX, initialBounds.minY };
702 handleInit = { initialBounds.minX, initialBounds.maxY };
703 break;
704 case HandleRole::LeftCenter:
705 anchor = { initialBounds.maxX, (initialBounds.minY + initialBounds.maxY) * 0.5 };
706 handleInit = { initialBounds.minX, anchor.y };
707 scaleY = false;
708 break;
709 default:
710 return false;
711 }
712
713 double sx = 1.0;
714 double sy = 1.0;
715
716 if (ShouldSnapToGrid()) {
717 // Snap moving handle coordinates directly to paper grid intervals
718 double rawHandleX = handleInit.x + (currentWorld.x - dragStartWorld.x);
719 double rawHandleY = handleInit.y + (currentWorld.y - dragStartWorld.y);
720 double snappedHandleX = std::round(rawHandleX / gridSpacingMm) * gridSpacingMm;
721 double snappedHandleY = std::round(rawHandleY / gridSpacingMm) * gridSpacingMm;
722
723 if (scaleX) {
724 double signedDistX = snappedHandleX - anchor.x;
725 if (handleInit.x < anchor.x) signedDistX = -signedDistX;
726 signedDistX = (std::max)(gridSpacingMm, signedDistX); // Clamped to minimum 1 grid unit
727 sx = std::clamp(signedDistX / initW, 0.05, 50.0);
728 }
729
730 if (scaleY) {
731 double signedDistY = snappedHandleY - anchor.y;
732 if (handleInit.y < anchor.y) signedDistY = -signedDistY;
733 signedDistY = (std::max)(gridSpacingMm, signedDistY); // Clamped to minimum 1 grid unit
734 sy = std::clamp(signedDistY / initH, 0.05, 50.0);
735 }
736 } else {
737 if (scaleX) {
738 // Vector along X from anchor to current cursor vs anchor to initial handle
739 double signedDistX = currentWorld.x - anchor.x;
740 if (dragStartWorld.x < anchor.x) signedDistX = -signedDistX;
741 sx = std::clamp(signedDistX / initW, 0.05, 50.0);
742 }
743
744 if (scaleY) {
745 // Vector along Y from anchor to current cursor vs anchor to initial handle
746 double signedDistY = currentWorld.y - anchor.y;
747 if (dragStartWorld.y < anchor.y) signedDistY = -signedDistY;
748 sy = std::clamp(signedDistY / initH, 0.05, 50.0);
749 }
750
751 // Aspect-Ratio Lock:
752 // - Corner grips (TopLeft, TopRight, BottomRight, BottomLeft) where both scaleX and scaleY are true:
753 // Enforce proportional uniform scaling preserving the object's native aspect ratio.
754 // - Edge/side grips (Left, Right, Top, Bottom) where only one axis is scaled:
755 // Allow 1D squish / stretch along the targeted axis without locking, giving the user full creative control.
756 if (scaleX && scaleY) {
757 // Uniform scale factor preserving aspect ratio without axis flipping or runaway jumps
758 double s = 0.5 * (sx + sy);
759 s = std::clamp(s, 0.05, 50.0);
760 sx = s;
761 sy = s;
762 }
763 }
764
765 BLMatrix2D scaleMatrix = BLMatrix2D::make_identity();
766 scaleMatrix.post_translate(-anchor.x, -anchor.y);
767 scaleMatrix.post_scale(sx, sy);
768 scaleMatrix.post_translate(anchor.x, anchor.y);
769
770 // Apply to each object from initial snapshot
771 for (size_t i = 0; i < selectedObjects.size(); ++i) {
772 if (i < initialStates.size() && selectedObjects[i]) {
773 selectedObjects[i]->transform = initialStates[i].initialTransform;
774 selectedObjects[i]->ApplyTransform(scaleMatrix);
775 }
776 }
777 RecalculateBounds();
778 return true;
779 }
780
781 // 4. TwoPoint Handle Drag (SmartArrow endpoints)
782 if (activeRole == HandleRole::Custom && selectedObjects.size() == 1) {
783 Point2D effectiveWorld = currentWorld;
784 if (ShouldSnapToGrid()) {
785 effectiveWorld.x = std::round(effectiveWorld.x / gridSpacingMm) * gridSpacingMm;
786 effectiveWorld.y = std::round(effectiveWorld.y / gridSpacingMm) * gridSpacingMm;
787 }
788 Point2D worldDelta = { effectiveWorld.x - lastDragWorld.x, effectiveWorld.y - lastDragWorld.y };
789 auto arrow = std::static_pointer_cast<Folio::SmartArrowObject>(selectedObjects[0]);
790 if (activeCustomId == 0) {
791 arrow->x1 += worldDelta.x;
792 arrow->y1 += worldDelta.y;
793 arrow->UpdateBounds();
794 lastDragWorld = effectiveWorld;
795 RecalculateBounds();
796 return true;
797 } else if (activeCustomId == 1) {
798 arrow->x2 += worldDelta.x;
799 arrow->y2 += worldDelta.y;
800 arrow->UpdateBounds();
801 lastDragWorld = effectiveWorld;
802 RecalculateBounds();
803 return true;
804 }
805 }
806
807 return false;
808 }

810 /**
811 * @brief Finalizes the active drag operation. Snaps rotated box back to axis-aligned AABB.
812 * @param session Optional DocumentSession pointer to record TransformObjectsCommand for undo/redo.
813 */
814 void OnPointerUp(DocumentSession* session = nullptr) {
815 if (!isDragging) return;
816 const char* roleStr = (activeRole == HandleRole::Body) ? "Body Move" :
817 (activeRole == HandleRole::Rotation) ? "Rotation" : "Resize Handle";
818 LOG_INFO(CanvasEngine, "Selection Gizmo drag completed (operation=" + std::string(roleStr) +
819 ", finalBounds=[" + std::to_string(bounds.minX) + ", " + std::to_string(bounds.minY) +
820 " to " + std::to_string(bounds.maxX) + ", " + std::to_string(bounds.maxY) + "])");
821
822 // Bake transforms into intrinsic coordinates for objects that support it (e.g. ShapeObject)
823 for (auto& obj : selectedObjects) {
824 if (obj) {
825 obj->BakeTransform();
826 }
827 }
828
829 // Record transformation command into page history for undo/redo
830 if (session && !initialClones.empty()) {
831 auto activePage = session->GetActivePage();
832 if (activePage) {
833 std::vector<Folio::TransformObjectsCommand::Entry> entries;
834 for (const auto& obj : selectedObjects) {
835 if (!obj) continue;
836 auto it = initialClones.find(obj->uid);
837 if (it != initialClones.end()) {
838 entries.push_back({
839 obj->uid,
840 it->second,
841 std::shared_ptr<CanvasObject>(obj->Clone().release())
842 });
843 }
844 }
845 if (!entries.empty()) {
846 activePage->history.RecordCommand(std::make_unique<Folio::TransformObjectsCommand>(std::move(entries)));
847 activePage->isModified = true;
848 }
849 }
850 }
851 initialClones.clear();
852
853 isDragging = false;
854 activeRole = HandleRole::None;
855 currentRotationAngle = 0.0;
856 displayAngleDeg = 0.0;
857 isAngleSnapped = false;
858 initialStates.clear();
859 RecalculateBounds(); // Snaps selection box back to axis-aligned bounding box!
860 }

863 /**
864 * @brief Renders the visual cues during rotation: center pivot, snap ring, 45° ticks, sweep arc, and degree HUD badge.
865 */
866 void RenderRotationVisualCues(BLContext& ctx, const CanvasTransform& transform, float knobX, float knobY) const {
867 Point2D centerScreen = transform.WorldToScreen(initialCenterWorld.x, initialCenterWorld.y);
868 float cx = static_cast<float>(centerScreen.x);
869 float cy = static_cast<float>(centerScreen.y);
870 float snapR = currentSnapRadius;
871
872 // 1. Center of Rotation Pivot (Crosshair + concentric circles)
873 ctx.set_stroke_style(BLRgba32(0x00, 0x78, 0xD4, 0xAA));
874 ctx.set_stroke_width(1.2);
875 ctx.stroke_line(cx - 7.0f, cy, cx + 7.0f, cy);
876 ctx.stroke_line(cx, cy - 7.0f, cx, cy + 7.0f);
877 ctx.fill_circle(cx, cy, 3.5, BLRgba32(0x00, 0x78, 0xD4, 0xFF));
878 ctx.set_stroke_style(BLRgba32(0xFF, 0xFF, 0xFF, 0xFF));
879 ctx.set_stroke_width(1.0);
880 ctx.stroke_circle(cx, cy, 3.5);
881
882 // 2. Snap Circle Guide
883 // Soft tint inside snap region
884 ctx.fill_circle(cx, cy, snapR, BLRgba32(0x00, 0x78, 0xD4, isAngleSnapped ? 0x0E : 0x05));
885
886 // Ring boundary
887 ctx.set_stroke_style(isAngleSnapped ? BLRgba32(0x00, 0xA4, 0xEF, 0x90) : BLRgba32(0x00, 0x78, 0xD4, 0x40));
888 ctx.set_stroke_width(isAngleSnapped ? 1.5 : 1.0);
889 ctx.stroke_circle(cx, cy, snapR);
890
891 // 3. 45-degree Radial Ticks on the Snap Ring
892 Point2D unrotMidWorld = { (initialBounds.minX + initialBounds.maxX) * 0.5, initialBounds.minY };
893 Point2D unrotMidScreen = transform.WorldToScreen(unrotMidWorld.x, unrotMidWorld.y);
894 double baseAngle = std::atan2(unrotMidScreen.y - cy, unrotMidScreen.x - cx);
895
896 for (int i = 0; i < 8; ++i) {
897 double tickAngle = baseAngle + i * (M_PI * 0.25);
898 float cosT = static_cast<float>(std::cos(tickAngle));
899 float sinT = static_cast<float>(std::sin(tickAngle));
900
901 bool isMajor = (i % 2 == 0); // 0, 90, 180, 270
902 float tickInner = isMajor ? (snapR - 9.0f) : (snapR - 5.0f);
903 float tickOuter = isMajor ? (snapR + 9.0f) : (snapR + 5.0f);
904
905 // Check if current rotation matches this tick
906 double angleDiff = std::remainder(currentRotationAngle - i * (M_PI * 0.25), 2.0 * M_PI);
907 bool isActiveTick = isAngleSnapped && (std::abs(angleDiff) < 0.08);
908
909 if (isActiveTick) {
910 // Ray from center to snapped tick
911 ctx.set_stroke_style(BLRgba32(0x00, 0xD2, 0xFF, 0x70));
912 ctx.set_stroke_width(1.5);
913 ctx.stroke_line(cx, cy, cx + cosT * snapR, cy + sinT * snapR);
914
915 // Snapped tick mark highlight
916 ctx.set_stroke_style(BLRgba32(0x00, 0xD2, 0xFF, 0xFF));
917 ctx.set_stroke_width(2.5);
918 ctx.stroke_line(cx + cosT * tickInner, cy + sinT * tickInner,
919 cx + cosT * tickOuter, cy + sinT * tickOuter);
920 ctx.fill_circle(cx + cosT * snapR, cy + sinT * snapR, 3.5, BLRgba32(0x00, 0xD2, 0xFF, 0xFF));
921 } else {
922 ctx.set_stroke_style(isMajor ? BLRgba32(0xFF, 0xFF, 0xFF, 0x80) : BLRgba32(0xFF, 0xFF, 0xFF, 0x38));
923 ctx.set_stroke_width(isMajor ? 1.5 : 1.0);
924 ctx.stroke_line(cx + cosT * tickInner, cy + sinT * tickInner,
925 cx + cosT * tickOuter, cy + sinT * tickOuter);
926 }
927 }
928
929 // 4. Angle Arc Sweep (from 0 to currentRotationAngle)
930 float arcRadius = std::min(snapR * 0.45f, 44.0f);
931 if (std::abs(currentRotationAngle) > 0.01) {
932 ctx.set_stroke_style(isAngleSnapped ? BLRgba32(0x00, 0xD2, 0xFF, 0xDD) : BLRgba32(0x00, 0x78, 0xD4, 0xBB));
933 ctx.set_stroke_width(2.0);
934 ctx.stroke_arc(cx, cy, arcRadius, baseAngle, currentRotationAngle);
935 }
936
937 // 5. Connecting line from center to active rotation knob
938 ctx.set_stroke_style(BLRgba32(0x00, 0x78, 0xD4, 0x50));
939 ctx.set_stroke_width(1.0);
940 ctx.stroke_line(cx, cy, knobX, knobY);
941
942 // 6. Degree HUD Badge
943 EnsureFontLoaded();
944
945 char textBuf[64];
946 if (isAngleSnapped) {
947 snprintf(textBuf, sizeof(textBuf), "%d°", static_cast<int>(std::round(displayAngleDeg)));
948 } else {
949 snprintf(textBuf, sizeof(textBuf), "%.1f°", displayAngleDeg);
950 }
951
952 float badgeW = isAngleSnapped ? 46.0f : 56.0f;
953 float badgeH = 22.0f;
954 float badgeX = knobX + 16.0f;
955 float badgeY = knobY - 26.0f;
956
957 // Background pill
958 ctx.fill_round_rect(badgeX, badgeY, badgeW, badgeH, 5.0, 5.0, BLRgba32(0x18, 0x1B, 0x22, 0xF4));
959 ctx.set_stroke_style(isAngleSnapped ? BLRgba32(0x00, 0xD2, 0xFF, 0xFF) : BLRgba32(0x45, 0x55, 0x68, 0xCC));
960 ctx.set_stroke_width(isAngleSnapped ? 1.5 : 1.0);
961 ctx.stroke_round_rect(badgeX, badgeY, badgeW, badgeH, 5.0, 5.0);
962
963 if (hudFont.is_valid()) {
964 float textX = badgeX + (isAngleSnapped ? 8.0f : 6.0f);
965 float textY = badgeY + 15.5f;
966 ctx.fill_utf8_text(BLPoint(textX, textY), hudFont, textBuf, SIZE_MAX, BLRgba32(0xFF, 0xFF, 0xFF, 0xFF));
967 if (isAngleSnapped) {
968 // Snap indicator dot
969 ctx.fill_circle(badgeX + badgeW - 9.0f, badgeY + 11.0f, 2.5, BLRgba32(0x00, 0xD2, 0xFF, 0xFF));
970 }
971 } else {
972 DrawFallbackText(ctx, badgeX + 8.0f, badgeY + 6.0f, textBuf);
973 }
974 }

1010 /**
1011 * @brief Robust vector stroke text fallback for numerals, letters, and symbols.
1012 */
1013 static void DrawFallbackText(BLContext& ctx, float x, float y, const char* str) {
1014 ctx.set_stroke_style(BLRgba32(0xFF, 0xFF, 0xFF, 0xFF));
1015 ctx.set_stroke_width(1.2);
1016 float curX = x;
1017 for (const char* p = str; *p; ++p) {
1018 char c = *p;
1019 if (c >= '0' && c <= '9') {
1020 int d = c - '0';
1021 static const uint8_t segs[10] = {
1022 0b00111111, // 0
1023 0b00000110, // 1
1024 0b01011011, // 2
1025 0b01001111, // 3
1026 0b01100110, // 4
1027 0b01101101, // 5
1028 0b01111101, // 6
1029 0b00000111, // 7
1030 0b01111111, // 8
1031 0b01101111 // 9
1032 };
1033 uint8_t s = segs[d];
1034 float w = 5.0f, h = 9.0f, mh = 4.5f;
1035 if (s & (1 << 0)) ctx.stroke_line(curX, y, curX + w, y);
1036 if (s & (1 << 1)) ctx.stroke_line(curX + w, y, curX + w, y + mh);
1037 if (s & (1 << 2)) ctx.stroke_line(curX + w, y + mh, curX + w, y + h);
1038 if (s & (1 << 3)) ctx.stroke_line(curX, y + h, curX + w, y + h);
1039 if (s & (1 << 4)) ctx.stroke_line(curX, y + mh, curX, y + h);
1040 if (s & (1 << 5)) ctx.stroke_line(curX, y, curX, y + mh);
1041 if (s & (1 << 6)) ctx.stroke_line(curX, y + mh, curX + w, y + mh);
1042 curX += w + 2.5f;
1043 } else if (c == '-') {
1044 ctx.stroke_line(curX, y + 4.5f, curX + 4.0f, y + 4.5f);
1045 curX += 6.0f;
1046 } else if (c == '.') {
1047 ctx.fill_circle(curX + 1.0f, y + 8.5f, 0.8, BLRgba32(0xFF, 0xFF, 0xFF, 0xFF));
1048 curX += 3.5f;
1049 } else if (c == ':') {
1050 ctx.fill_circle(curX + 1.0f, y + 2.5f, 0.8, BLRgba32(0xFF, 0xFF, 0xFF, 0xFF));
1051 ctx.fill_circle(curX + 1.0f, y + 6.5f, 0.8, BLRgba32(0xFF, 0xFF, 0xFF, 0xFF));
1052 curX += 3.5f;
1053 } else if (c == '#') {
1054 ctx.stroke_line(curX + 1.5f, y, curX + 1.5f, y + 9);
1055 ctx.stroke_line(curX + 4.0f, y, curX + 4.0f, y + 9);
1056 ctx.stroke_line(curX, y + 3.0f, curX + 5.5f, y + 3.0f);
1057 ctx.stroke_line(curX, y + 6.0f, curX + 5.5f, y + 6.0f);
1058 curX += 7.0f;
1059 } else if (c == '[') {
1060 ctx.stroke_line(curX + 2.5f, y, curX, y);
1061 ctx.stroke_line(curX, y, curX, y + 9);
1062 ctx.stroke_line(curX, y + 9, curX + 2.5f, y + 9);
1063 curX += 4.5f;
1064 } else if (c == ']') {
1065 ctx.stroke_line(curX, y, curX + 2.5f, y);
1066 ctx.stroke_line(curX + 2.5f, y, curX + 2.5f, y + 9);
1067 ctx.stroke_line(curX + 2.5f, y + 9, curX, y + 9);
1068 curX += 4.5f;
