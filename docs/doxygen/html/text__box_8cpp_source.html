1/**
2 * @file text_box.cpp
3 * @brief Implementation of TextBoxObject with 3-layer styling and Blend2D vector typography.
4 *
5 * GENERAL WORKING PROCESS:
6 * ------------------------
7 * 1. Layer 1 (Infill): Renders rounded container background with solid or semi-transparent
8 * pastel washes (Sticky Note presets) or transparent canvas notes.
9 * 2. Layer 2 (Border & Grip Chrome): When hovered or editing, renders a subtle top grab handle
10 * bar (OneNote style) allowing the user to reposition notes effortlessly on canvas.
11 * 3. Layer 3 (Vector Typography): Resolves crisp typography via FontManager and draws text
12 * runs using native sub-pixel glyph rasterization. When actively editing, draws selection
13 * highlight rectangles and the synchronized blinking caret.
14 *
15 * MATHEMATICAL FOUNDATIONS:
16 * -------------------------
17 * - Typographical Baseline Positioning:
18 * Baseline = worldY + localLineY + ascent
19 * - Stroke Scaling Invariance:
20 * strokeW = baseStrokeW / sqrt(det(transform))
21 * Prevents line fattening or thinning during active canvas zooms.
22 */

192std::vector<TextBoxObject::FormattedSpan> TextBoxObject::GetSpansForRange(size_t rangeStart, size_t rangeEnd) const {
193 std::vector<FormattedSpan> spans;
194 if (rangeStart >= rangeEnd) return spans;
195
196 std::string full = PlainText();
197 if (full.empty()) return spans;
198 rangeStart = (std::min)(rangeStart, full.size());
199 rangeEnd = (std::min)(rangeEnd, full.size());
200 if (rangeStart >= rangeEnd) return spans;
201
202 if (runs.empty()) {
203 FormattedSpan s;
204 s.text = full.substr(rangeStart, rangeEnd - rangeStart);
205 s.run = nullptr;
206 BLFont f = FontManager::Instance().GetFont(fontFamily, fontSize, isBold, isItalic);
207 s.width = FontManager::Instance().MeasureTextWidth(f, s.text);
208 spans.push_back(s);
209 return spans;
210 }
211
212 size_t curRunStart = 0;
213 for (const auto& r : runs) {
214 size_t curRunEnd = curRunStart + r.text.size();
215 if (rangeEnd <= curRunStart) break;
216 if (rangeStart < curRunEnd) {
217 size_t s = (std::max)(rangeStart, curRunStart);
218 size_t e = (std::min)(rangeEnd, curRunEnd);
219 FormattedSpan span;
220 span.text = r.text.substr(s - curRunStart, e - s);
221 span.run = &r;
222 if (r.elementType == InlineElementType::TextRun) {
223 BLFont rf = FontManager::Instance().GetFont(r.fontFamily, r.fontSize, r.bold, r.italic);
224 span.width = FontManager::Instance().MeasureTextWidth(rf, span.text);
225 } else {
226 span.width = r.inlineWidth;
227 }
228 spans.push_back(span);
229 }
230 curRunStart = curRunEnd;
231 }
232
233 if (spans.empty()) {
234 FormattedSpan s;
235 s.text = full.substr(rangeStart, rangeEnd - rangeStart);
236 s.run = nullptr;
237 BLFont f = FontManager::Instance().GetFont(fontFamily, fontSize, isBold, isItalic);
238 s.width = FontManager::Instance().MeasureTextWidth(f, s.text);
239 spans.push_back(s);
240 }
241 return spans;
242}

253void TextBoxObject::GetLineMetricsForRange(size_t rangeStart, size_t rangeEnd, double& outAscent, double& outLineHeight) const {
254 auto spans = GetSpansForRange(rangeStart, rangeEnd);
255 double maxAscent = 0.0;
256 double maxHeight = 0.0;
257
258 for (const auto& s : spans) {
259 if (s.run && s.run->elementType == InlineElementType::TextRun) {
260 BLFont f = FontManager::Instance().GetFont(s.run->fontFamily, s.run->fontSize, s.run->bold, s.run->italic);
261 BLFontMetrics fm = FontManager::Instance().GetMetrics(f);
262 double h = fm.ascent + fm.descent + fm.line_gap;
263 if (h <= 0.001) h = s.run->fontSize * 1.25;
264 double a = (fm.ascent > 0.001) ? fm.ascent : (s.run->fontSize * 0.9);
265 maxAscent = (std::max)(maxAscent, a);
266 maxHeight = (std::max)(maxHeight, h);
267 } else if (s.run) {
268 maxAscent = (std::max)(maxAscent, s.run->inlineHeight);
269 maxHeight = (std::max)(maxHeight, s.run->inlineHeight);
270 }
271 }
272
273 if (maxHeight <= 0.001) {
274 BLFont f = FontManager::Instance().GetFont(fontFamily, fontSize, isBold, isItalic);
275 BLFontMetrics fm = FontManager::Instance().GetMetrics(f);
276 maxHeight = fm.ascent + fm.descent + fm.line_gap;
277 if (maxHeight <= 0.001) maxHeight = fontSize * 1.25;
278 maxAscent = (fm.ascent > 0.001) ? fm.ascent : (fontSize * 0.9);
279 }
280
281 outAscent = maxAscent;
282 outLineHeight = maxHeight;
283}

307/**
308 * @brief Evaluates whether a world-space point intersects the text box or its top grab bar.
309 *
310 * Working Process:
311 * 1. Check visibility & selectability: Rejects hidden or non-selectable text boxes.
312 * 2. Expand bounding box vertically by 4.0mm at the top (hitBox.minY -= 4.0) to account
313 * for the interactive text drag handle / grab bar.
314 * 3. Containment test: If (wx, wy) falls inside the expanded hitBox:
315 * - If already selected (isSelected == true), returns true immediately to support
316 * frictionless double-click text editing and drag operations.
317 * - Otherwise returns true as an interior hit.
318 *
319 * @param wx World X coordinate in mm.
320 * @param wy World Y coordinate in mm.
321 * @return True if (wx, wy) hits the text box or grab handle.
322 */
323bool TextBoxObject::HitTest(double wx, double wy) const {
324 if (!isVisible || !isSelectable) return false;
325
326 // Include 4mm top grab bar area when testing for hit
327 AABB hitBox = bounds;
328 hitBox.minY -= 4.0;
329 if (!hitBox.Contains(wx, wy)) return false;
330
331 // Fast-path: already selected text box allows immediate interaction
332 if (isSelected) return true;
333
334 return true;
335}

350void TextBoxObject::Render(BLContext& ctx, const Viewport& viewport) const {
351 if (!isVisible) return;
352
353 ctx.save();
354 ctx.apply_transform(transform);
355
356 // =========================================================================
357 // LAYER 1: CONTAINER INFILL (Sticky Note wash or Transparent)
358 // =========================================================================
359 if (fillType != ShapeFillType::None) {
360 BLRoundRect rr(worldX, worldY, worldWidth, worldHeight, cornerRadius, cornerRadius);
361 ctx.set_fill_style(fillColor);
362 ctx.fill_round_rect(rr);
363 }
364
365 // =========================================================================
366 // LAYER 2: BORDER & ONENOTE CONTAINER HEADER
367 // =========================================================================
368 double det = std::abs(transform.m00 * transform.m11 - transform.m01 * transform.m10);
369 double scale = (det > 1e-6) ? std::sqrt(det) : 1.0;
370 double invScale = (scale > 1e-4) ? (1.0 / scale) : 1.0;
371
372 if (outlineType == ShapeOutlineType::Solid) {
373 ctx.set_stroke_style(strokeColor);
374 ctx.set_stroke_width(strokeWidth * invScale);
375 BLRoundRect rr(worldX, worldY, worldWidth, worldHeight, cornerRadius, cornerRadius);
376 ctx.stroke_round_rect(rr);
377 }
378
379 // OneNote-style Container Header Handle (shown when hovered, selected, or actively editing)
380 if (isHovered || isSelected || isEditing) {
381 double headerH = 3.5;
382 BLRoundRect topBar(worldX, worldY - headerH - 0.5, worldWidth, headerH, 2.0, 2.0);
383 BLRgba32 barColor = (isEditing) ? BLRgba32(0x3B, 0x82, 0xF6, 0xD0) : BLRgba32(0x94, 0xA3, 0xB8, 0x90);
384 ctx.set_fill_style(barColor);
385 ctx.fill_round_rect(topBar);
386
387 // Subtle frame around container
388 ctx.set_stroke_style(BLRgba32(0x3B, 0x82, 0xF6, 0x80));
389 ctx.set_stroke_width(0.4 * invScale);
390 BLRoundRect frame(worldX - 1.0, worldY - 1.0, worldWidth + 2.0, worldHeight + 2.0, cornerRadius, cornerRadius);
391 ctx.stroke_round_rect(frame);
392 }
393
394 // =========================================================================
395 // LAYER 3: BLEND2D VECTOR TYPOGRAPHY PASS (Rich Inline & Styled Runs)
396 // =========================================================================
397 std::string fullText = PlainText();
398 if (!fullText.empty()) {
399 double curY = worldY + 2.0;
400 const double maxWrapWidth = (std::max)(10.0, worldWidth - 4.0);
401
402 /**
403 * @brief Renders a single measured line slice with rich styled spans and inline elements.
404 *
405 * Mathematical Layout:
406 * - Span start X: lineX + \sum_{prev} span.width
407 * - Baseline: yTop + maxLineAscent
408 * - Text baseline render: ctx.fill_utf8_text(..., spanFont, ...)
409 */
410 auto drawFormattedLine = [&](size_t lineStart, size_t lineEnd, double yTop, double lineAscent, double lineH) {
411 double totalLineWidth = MeasureRange(lineStart, lineEnd);
412 double lineX = worldX + 2.0;
413 if (alignment == 1 && maxWrapWidth > totalLineWidth) {
414 lineX += (maxWrapWidth - totalLineWidth) * 0.5; // Center alignment
415 } else if (alignment == 2 && maxWrapWidth > totalLineWidth) {
416 lineX += (maxWrapWidth - totalLineWidth); // Right alignment
417 }
418
419 auto spans = GetSpansForRange(lineStart, lineEnd);
420 double spanX = lineX;
421
422 for (const auto& span : spans) {
423 if (span.width <= 0.0001 && span.text.empty()) continue;
424
425 // Resolve styles for this span
426 BLRgba32 spanTextColor = textColor;
427 BLRgba32 spanHighlightColor = highlightColor;
428 bool spanBold = isBold;
429 bool spanItalic = isItalic;
430 bool spanUnderline = isUnderline;
431 bool spanStrikethrough = isStrikethrough;
432 std::string spanFontFamily = fontFamily;
433 double spanFontSize = fontSize;
434
435 if (span.run) {
436 spanTextColor = span.run->color;
437 spanHighlightColor = span.run->highlightColor;
438 spanBold = span.run->bold;
439 spanItalic = span.run->italic;
440 spanUnderline = span.run->underline;
441 spanStrikethrough = span.run->strikethrough;
442 spanFontFamily = span.run->fontFamily;
443 spanFontSize = span.run->fontSize;
444 }
445
446 // Check for inline rich elements (InlineImage, InlineMath, InlineTable)
447 if (span.run && span.run->elementType != InlineElementType::TextRun) {
448 if (!span.run->renderedImage.is_empty()) {
449 // Center vertically within the line height
450 double imgY = yTop + (lineH - span.run->inlineHeight) * 0.5;
451 ctx.blit_image(BLPoint(spanX, imgY), span.run->renderedImage);
452 } else if (span.run->elementType == InlineElementType::InlineMath) {
453 // Fallback: draw LaTeX source text in italic font
454 BLFont mathFont = FontManager::Instance().GetFont(spanFontFamily, spanFontSize, false, true);
455 double baselineY = yTop + lineAscent;
456 ctx.fill_utf8_text(BLPoint(spanX, baselineY), mathFont, span.run->latexSource.data(), span.run->latexSource.size(), spanTextColor);
457 }
458 spanX += span.width;
459 continue;
460 }
461
462 // Standard TextRun formatting
463 BLFont spanFont = FontManager::Instance().GetFont(spanFontFamily, spanFontSize, spanBold, spanItalic);
464 if (!spanFont.is_valid()) {
465 spanFont = FontManager::Instance().GetFont(fontFamily, fontSize, isBold, isItalic);
466 }
467
468 // Background highlight if present
469 if (spanHighlightColor.a() > 0) {
470 ctx.set_fill_style(spanHighlightColor);
471 ctx.fill_rect(BLRect(spanX - 0.5, yTop, span.width + 1.0, lineH));
472 }
473
474 // Text glyph vector rasterization
475 double baselineY = yTop + lineAscent;
476 ctx.fill_utf8_text(BLPoint(spanX, baselineY), spanFont, span.text.data(), span.text.size(), spanTextColor);
477
478 // Underline decoration
479 if (spanUnderline) {
480 ctx.set_stroke_style(spanTextColor);
481 ctx.set_stroke_width(0.5);
482 ctx.stroke_line(spanX, baselineY + 1.2, spanX + span.width, baselineY + 1.2);
483 }
484
485 // Strikethrough decoration
486 if (spanStrikethrough) {
487 ctx.set_stroke_style(spanTextColor);
488 ctx.set_stroke_width(0.5);
489 double strikeY = baselineY - (lineAscent * 0.35);
490 ctx.stroke_line(spanX, strikeY, spanX + span.width, strikeY);
491 }
492
493 spanX += span.width;
494 }
495 };
496
497 size_t i = 0;
498 while (i < fullText.size()) {
499 size_t newlinePos = fullText.find('\n', i);
500 size_t paraEnd = (newlinePos != std::string::npos) ? newlinePos : fullText.size();
501 size_t paraLen = paraEnd - i;
502
503 if (!isWrap || maxWrapWidth <= 10.0) {
504 // Entire paragraph rendered on single line
505 double lineAscent = 0.0, lineH = 0.0;
506 GetLineMetricsForRange(i, paraEnd, lineAscent, lineH);
507 drawFormattedLine(i, paraEnd, curY, lineAscent, lineH);
508 curY += lineH;
509 } else {
510 // Word wrapping using per-span width measurements
511 size_t lineStartIdx = i;
512 size_t scanIdx = i;
513
514 while (scanIdx < paraEnd) {
515 // Find next word boundary
516 size_t nextSpace = fullText.find(' ', scanIdx);
517 if (nextSpace > paraEnd) nextSpace = std::string::npos;
518 size_t wordEnd = (nextSpace != std::string::npos) ? nextSpace : paraEnd;
519 size_t candidateEnd = (nextSpace != std::string::npos) ? nextSpace + 1 : paraEnd;
520
521 double testW = MeasureRange(lineStartIdx, wordEnd);
522 if (testW <= maxWrapWidth || scanIdx == lineStartIdx) {
523 // Word fits, advance scan
524 scanIdx = candidateEnd;
525 } else {
526 // Word overflows, commit current line slice [lineStartIdx, scanIdx]
527 size_t actualLineEnd = scanIdx;
528 if (actualLineEnd > lineStartIdx && fullText[actualLineEnd - 1] == ' ') {
529 actualLineEnd--; // trim trailing space from measure
530 }
531 double lineAscent = 0.0, lineH = 0.0;
532 GetLineMetricsForRange(lineStartIdx, actualLineEnd, lineAscent, lineH);
533 drawFormattedLine(lineStartIdx, actualLineEnd, curY, lineAscent, lineH);
534 curY += lineH;
535 lineStartIdx = scanIdx;
536 scanIdx = candidateEnd;
537 }
538 }
539
540 if (lineStartIdx < paraEnd) {
541 double lineAscent = 0.0, lineH = 0.0;
542 GetLineMetricsForRange(lineStartIdx, paraEnd, lineAscent, lineH);
543 drawFormattedLine(lineStartIdx, paraEnd, curY, lineAscent, lineH);
544 curY += lineH;
545 }
546 }
547
548 if (newlinePos != std::string::npos) {
549 i = newlinePos + 1;
550 } else {
551 break;
552 }
553 }
554 }
555
556 ctx.restore();
557}

559void TextBoxObject::RenderWithEditor(BLContext& ctx, const Viewport& viewport, const TextEditorState& editor) const {
560 // 1. Draw base container and vector text
561 Render(ctx, viewport);
562
563 ctx.save();
564 ctx.apply_transform(transform);
565
566 // 2. Draw selection highlight boxes
567 std::vector<AABB> selBoxes = editor.GetSelectionBoxes();
568 if (!selBoxes.empty()) {
569 ctx.set_fill_style(BLRgba32(0x3B, 0x82, 0xF6, 0x66)); // Translucent accent selection
570 for (const auto& box : selBoxes) {
571 ctx.fill_rect(box.minX, box.minY, box.Width(), box.Height());
572 }
573 }
574
575 // 3. Draw blinking caret line
576 if (editor.IsCaretVisible()) {
577 Point2D caretPt = editor.GetCursorWorldPos();
578 double caretH = editor.GetCaretHeight();
579
580 ctx.set_stroke_style(textColor);
581 ctx.set_stroke_width(0.7);
582 ctx.stroke_line(caretPt.x, caretPt.y, caretPt.x, caretPt.y + caretH);
583 }
584
585 ctx.restore();
586}
