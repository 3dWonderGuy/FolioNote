49/**
50 * @brief Deduces the ImageFormat from a file extension or filename string.
51 *
52 * Working Process:
53 * 1. Finds the last dot '.' separator in the input path or extension.
54 * 2. Extracts the substring and transforms characters to lowercase.
55 * 3. Matches against known extension aliases (.jpg, .jpeg, .tif, .tiff, .svg, etc.).
56 * 4. Returns ImageFormat::Unknown if no match is found.
57 *
58 * @param[in] pathOrExt Filename (e.g. "photo.jpeg") or extension (e.g. ".png" or "webp").
59 * @return Deducible ImageFormat enumeration, or ImageFormat::Unknown if unrecognized.
60 */
61inline ImageFormat ImageFormatFromExtension(std::string_view pathOrExt) {
62 if (pathOrExt.empty()) {
63 return ImageFormat::Unknown;
64 }
65
66 // Locate trailing extension delimiter
67 const auto lastDot = pathOrExt.rfind('.');
68 std::string_view ext = (lastDot != std::string_view::npos)
69 ? pathOrExt.substr(lastDot + 1)
70 : pathOrExt;
71
72 // Convert to lowercase for case-insensitive matching
73 std::string lowerExt;
74 lowerExt.reserve(ext.size());
75 for (char c : ext) {
76 lowerExt.push_back(static_cast<char>(std::tolower(static_cast<unsigned char>(c))));
77 }
78
79 if (lowerExt == "png") return ImageFormat::PNG;
80 if (lowerExt == "jpg" || lowerExt == "jpeg") return ImageFormat::JPEG;
81 if (lowerExt == "webp") return ImageFormat::WebP;
82 if (lowerExt == "bmp") return ImageFormat::BMP;
83 if (lowerExt == "gif") return ImageFormat::GIF;
84 if (lowerExt == "tif" || lowerExt == "tiff") return ImageFormat::TIFF;
85 if (lowerExt == "qoi") return ImageFormat::QOI;
86 if (lowerExt == "svg") return ImageFormat::SVG;
87
88 return ImageFormat::Unknown;
89}

154/**
155 * @brief Resolves an ImageFormat enum from a standard MIME type string.
156 *
157 * @param[in] mime The MIME type string to parse (e.g. "image/png", "image/svg+xml").
158 * @return Corresponding ImageFormat enumeration, or ImageFormat::Unknown if unmatched.
159 */
160inline ImageFormat ImageFormatFromMimeType(std::string_view mime) {
161 if (mime == "image/png") return ImageFormat::PNG;
162 if (mime == "image/jpeg") return ImageFormat::JPEG;
163 if (mime == "image/jpg") return ImageFormat::JPEG;
164 if (mime == "image/webp") return ImageFormat::WebP;
165 if (mime == "image/bmp") return ImageFormat::BMP;
166 if (mime == "image/gif") return ImageFormat::GIF;
167 if (mime == "image/tiff") return ImageFormat::TIFF;
168 if (mime == "image/qoi") return ImageFormat::QOI;
169 if (mime == "image/svg+xml" || mime == "image/svg") return ImageFormat::SVG;
170 return ImageFormat::Unknown;
171}
