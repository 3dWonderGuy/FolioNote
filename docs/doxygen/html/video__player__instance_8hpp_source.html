2/**
3 * =========================================================================================
4 * @file core/objects/media/videos/video_player_instance.hpp
5 * @brief Threaded libVLC Video Player Engine — Frame Buffer Bridge to Blend2D Canvas
6 * =========================================================================================
7 *
8 * ARCHITECTURAL DESIGN & WORKING PROCESS:
9 * ----------------------------------------
10 * VideoPlayerInstance wraps a libVLC media player and decodes video frames on a
11 * DEDICATED BACKGROUND THREAD (the "3rd layer" in FolioNote's rendering pipeline).
12 *
13 * The 3-Layer Rendering Pipeline:
14 * Layer 1 — staticCanvasLayer (BLImage): Grid + committed objects. Rebuilt on needsFullRebake.
15 * Layer 2 — liveInkingLayer (BLImage): In-flight pen stroke. Wiped each gesture.
16 * Layer 3 — VIDEO DECODE THREAD : libVLC decodes into pixelBuffer[] at native FPS.
17 * ↓ (atomic swap on main thread during Render)
18 * compositeSurface (BLImage): Compositor reads pixelBuffer → blit_image()
19 *
20 * Zero-Copy Frame Hand-off:
21 * 1. libVLC decode thread calls lock_cb → receives pointer to pixelBuffer (BGRA32 / PRGB32).
22 * 2. libVLC decodes frame directly into our buffer (hardware-accelerated: D3D11VA on Windows).
23 * 3. unlock_cb sets hasNewFrame = true (atomic, memory_order_release).
24 * 4. Main render thread checks hasNewFrame → wraps pixelBuffer in BLImage via
25 * BLImage::createFromData() (zero-copy external data reference) → blit_image() into scene.
26 *
27 * Thread Safety Model:
28 * - pixelBuffer is protected by frameMutex (std::mutex).
29 * - hasNewFrame is std::atomic<bool> (lock-free fast-path check).
30 * - All libVLC instance/player pointers are owned exclusively by the VideoPlayerInstance.
31 * - Public API (Play/Pause/Seek/SetVolume) is safe to call from main thread.
32 *
33 * Graceful Degradation (no VLC installed or SDK missing):
34 * - FOLIO_HAS_LIBVLC compile-time guard disables all VLC code paths.
35 * - VideoPlayerInstance still compiles as a stub that reports ErrorState.
36 * - Render() falls back to the placeholder player card.
37 *
38 * Input/Output:
39 * @input sourceUrl — Filesystem path or HTTP/HLS URL string
40 * @output pixelBuffer — BGRA32 raw pixel data at (videoW * videoH * 4) bytes
41 * @output hasNewFrame — Atomic flag; main thread polls each frame
42 * @output playbackState — Current playback state enum
43 * @output timeSec — Current playback position in seconds (from libVLC)
44 * @output durationSec — Total media duration in seconds
45 */

89/**
90 * @class VideoPlayerInstance
91 * @brief Owns the libVLC instance, media player, and pixel ring buffer for one VideoObject.
92 *
93 * Lifetime Model:
94 * 1. Construct with a source URL.
95 * 2. Call Open() to begin asynchronous media loading.
96 * 3. Main render loop calls HasNewFrame() → if true, acquires LockFrame() and
97 * blits the raw BGRA32 pixels using BLImage::createFromData().
98 * 4. Call Stop() before destruction (destructor also handles teardown).

134 * Memory layout:

230 /**
231 * @brief Returns true when a new video frame is decoded and ready to blit.
232 *
233 * Usage in VideoObject::Render():
234 * @code
235 * if (player && player->HasNewFrame()) {
236 * int fw, fh;
237 * const uint8_t* px = player->LockFrame(fw, fh);
238 * if (px) {
239 * BLImage frame;
240 * frame.createFromData(fw, fh, BL_FORMAT_PRGB32,
241 * const_cast<uint8_t*>(px), fw * 4);
242 * ctx.blit_image(BLPointI(screenX, screenY), frame);
243 * player->UnlockFrame();
244 * }
245 * }
246 * @endcode
247 */
248 [[nodiscard]] bool HasNewFrame() const noexcept {
249 auto s = state.load(std::memory_order_relaxed);
250 return hasNewFrame.load(std::memory_order_acquire) &&
251 (s == VideoPlayerState::Playing ||
252 s == VideoPlayerState::Paused ||
253 s == VideoPlayerState::Loading);
254 }
255
256 /**
257 * @brief Gets the current lifecycle state of the player engine.
258 */
259 [[nodiscard]] VideoPlayerState GetState() const noexcept {
260 return state.load(std::memory_order_relaxed);
261 }
262
263 /**
264 * @brief Checks if a valid, decoded video frame is present in the pixel buffer.
265 *
266 * Unlike HasNewFrame() (which is reset to false once consumed), HasValidFrame() remains
267 * true as long as at least one decoded frame has been received. This allows the canvas
268 * compositor to keep displaying the frozen frame while the video is paused or seeking.
269 *
270 * @return true only after at least one real frame has been decoded and pixelBuffer is populated.
271 */
272 [[nodiscard]] bool HasValidFrame() const noexcept {
273 return hasValidDecodedFrame.load(std::memory_order_acquire) &&
274 videoW > 0 && videoH > 0 && !pixelBuffer.empty();
275 }
276
277 /**
278 * @brief Locks the frame mutex and returns a raw pointer to the BGRA32 pixel data.
279 *
280 * @param[out] outWidth Frame width in pixels.
281 * @param[out] outHeight Frame height in pixels.
282 * @return Pointer to BGRA32 pixel data, or nullptr if no valid frame.
283 *
284 * IMPORTANT: Must be followed by UnlockFrame() to release the mutex.
285 * Never hold this lock across a full render frame.
286 */
287 const uint8_t* LockFrame(int& outWidth, int& outHeight) {
288 frameMutex.lock();
289 if (pixelBuffer.empty() || videoW <= 0 || videoH <= 0) {
290 frameMutex.unlock();
291 outWidth = outHeight = 0;
292 return nullptr;
293 }
294 hasNewFrame.store(false, std::memory_order_release);
295 outWidth = videoW;
296 outHeight = videoH;
297 return pixelBuffer.data();
298 }
299
300 /**
301 * @brief Releases the frame mutex acquired by LockFrame().
302 */
303 void UnlockFrame() {
304 frameMutex.unlock();
305 }

352 /**
