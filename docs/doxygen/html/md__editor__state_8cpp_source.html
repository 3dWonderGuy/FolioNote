65void MdEditorState::Backspace() {
66 if (selection.hasSelection) {
67 PushUndoSnapshot();
68 DeleteSelection();
69 return;
70 }
71
72 if (selection.cursorOffset == 0 || markdownText.empty()) return;
73 PushUndoSnapshot();
74
75 size_t pos = selection.cursorOffset;
76 // Basic UTF-8 multibyte backward skip
77 size_t eraseCount = 1;
78 if (pos >= 1 && (static_cast<uint8_t>(markdownText[pos - 1]) & 0xC0) == 0x80) {
79 while (pos >= eraseCount && (static_cast<uint8_t>(markdownText[pos - eraseCount]) & 0xC0) == 0x80) {
80 ++eraseCount;
81 }
82 }
83
84 size_t delStart = (pos >= eraseCount) ? (pos - eraseCount) : 0;
85 markdownText.erase(delStart, pos - delStart);
86 selection.cursorOffset = delStart;
87 selection.anchorOffset = delStart;
88 selection.hasSelection = false;
89 caret.offset = delStart;
90 caret.visible = true;
91}

93void MdEditorState::DeleteForward() {
94 if (selection.hasSelection) {
95 PushUndoSnapshot();
96 DeleteSelection();
97 return;
98 }
99
100 if (selection.cursorOffset >= markdownText.size()) return;
101 PushUndoSnapshot();
102
103 size_t pos = selection.cursorOffset;
104 size_t eraseCount = 1;
105 uint8_t lead = static_cast<uint8_t>(markdownText[pos]);
106 if ((lead & 0xE0) == 0xC0) eraseCount = 2;
107 else if ((lead & 0xF0) == 0xE0) eraseCount = 3;
108 else if ((lead & 0xF8) == 0xF0) eraseCount = 4;
109
110 markdownText.erase(pos, eraseCount);
111 caret.offset = selection.cursorOffset;
112 caret.visible = true;
113}

207void MdEditorState::WrapSelectionWith(const std::string& prefix, const std::string& suffix) {
208 PushUndoSnapshot();
209 if (selection.hasSelection) {
210 size_t minO = selection.MinOffset();
211 size_t maxO = selection.MaxOffset();

213 std::string replacement = prefix + selected + suffix;
214 markdownText.replace(minO, maxO - minO, replacement);
215 selection.anchorOffset = minO;
216 selection.cursorOffset = minO + replacement.size();

221 selection.cursorOffset = pos + prefix.size();
222 selection.anchorOffset = selection.cursorOffset;
223 selection.hasSelection = false;
224 }
