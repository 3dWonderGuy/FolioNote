1/**
2 * =========================================================================================
3 * @file core/objects/media/videos/video_player_instance.cpp
4 * @brief libVLC Memory Callback Implementation — Decode Thread to Blend2D Frame Bridge
5 * =========================================================================================
6 *
7 * GENERAL WORKING PROCESS:
8 * -------------------------
9 * This file implements the VideoPlayerInstance class defined in video_player_instance.hpp.
10 * It contains all libVLC C API calls and the three core memory callbacks that form the
11 * "video sink": lock_cb, unlock_cb, display_cb.
12 *
13 * The decode pipeline for each frame (all on VLC's internal decode thread):
14 * 1. lock_cb() → give VLC our pixelBuffer pointer as decode target
15 * 2. libVLC decodes H.264/HEVC/VP9 frame into pixelBuffer (D3D11VA hw accel on Win)
16 * 3. unlock_cb() → set hasNewFrame = true (atomic release)
17 * 4. display_cb() → call onFrameReady() so the canvas marks isDirty = true
18 * 5. Main thread render loop sees isDirty, calls Render(), checks HasNewFrame()
19 * 6. LockFrame() → BLImage::createFromData(pixelBuffer) → ctx.blit_image()
20 * 7. UnlockFrame() releases mutex
21 */

35// OPEN — Initialize libVLC instance, load media, wire callbacks, start decode
36// =============================================================================
37
38bool VideoPlayerInstance::Open(const std::string& url, bool startPaused) {
39 if (url.empty()) {
40 errorMessage = "Empty URL";
41 state.store(VideoPlayerState::Error);
42 return false;
43 }
44
45 LOG_INFO(CanvasObject, "VideoPlayerInstance::Open: target='" + url + "' (startPaused=" + (startPaused ? "true" : "false") + ")");
46
47 // 1. Stop and release any previously loaded media
48 Stop();
49 Release();
50 hasValidDecodedFrame.store(false);
51
52 // -------------------------------------------------------------------------
53 // 2. Create libVLC instance with minimal plugin output
54 // "--no-video-title" : don't show title overlay on decode start
55 // "--avcodec-hw=any" : enable hardware decoding (D3D11VA on Windows)
56 // -------------------------------------------------------------------------
57 const char* vlcArgs[] = {
58 "--no-video-title-show",
59 "--avcodec-hw=any",
60 "--aout=directsound", // Use DirectSound audio on Windows
61 };
62 constexpr int vlcArgCount = static_cast<int>(sizeof(vlcArgs) / sizeof(vlcArgs[0]));
63
64 vlcInstance = libvlc_new(vlcArgCount, vlcArgs);
65 if (!vlcInstance) {
66 errorMessage = "libvlc_new() failed: VLC plugins directory not found";
67 state.store(VideoPlayerState::Error);
68 LOG_ERROR(CanvasObject, "Failed to create libVLC instance: " + errorMessage);
69 return false;
70 }
71
72 // Attach logger to pipe libVLC internal logs into FolioNote logger
73 libvlc_log_set(vlcInstance, log_cb, this);
74 LOG_INFO(CanvasObject, "libVLC instance created successfully (version: " + std::string(libvlc_get_version()) + ")");
75
76 // -------------------------------------------------------------------------
77 // 3. Open media from URL or file path
78 // libvlc_media_new_path: file:// URL from local filesystem path
79 // libvlc_media_new_location: http/https/rtsp/hls streams
80 // -------------------------------------------------------------------------
81 bool isNetworkUrl = (url.find("http://") == 0 || url.find("https://") == 0 ||
82 url.find("rtsp://") == 0 || url.find("rtmp://") == 0 ||
83 url.find("file://") == 0);
84
85 if (isNetworkUrl) {
86 LOG_INFO(CanvasObject, "Opening media location MRL: " + url);
87 vlcMedia = libvlc_media_new_location(vlcInstance, url.c_str());
88 } else {
89 std::filesystem::path p(url);
90 std::string nativePath = p.make_preferred().string();
91 LOG_INFO(CanvasObject, "Opening media filesystem path: " + nativePath);
92 vlcMedia = libvlc_media_new_path(vlcInstance, nativePath.c_str());
93 }
94
95 if (!vlcMedia) {
96 errorMessage = "libvlc_media_new_*() failed: bad URL or path";
97 state.store(VideoPlayerState::Error);
98 LOG_ERROR(CanvasObject, "Failed to open media: " + url);
99 return false;
100 }
101
102 if (isNetworkUrl) {
103 // Smooth out network jitter, buffer ahead 2 seconds, and prevent TLS disconnects
104 libvlc_media_add_option(vlcMedia, ":network-caching=2000");
105 libvlc_media_add_option(vlcMedia, ":live-caching=2000");
106 libvlc_media_add_option(vlcMedia, ":clock-jitter=1000");
107 libvlc_media_add_option(vlcMedia, ":clock-synchro=0");
108 }
109
110 // -------------------------------------------------------------------------
111 // 4. Create the media player
112 // -------------------------------------------------------------------------
113 vlcPlayer = libvlc_media_player_new_from_media(vlcMedia);
114 if (!vlcPlayer) {
115 errorMessage = "libvlc_media_player_new_from_media() failed";
116 state.store(VideoPlayerState::Error);
117 return false;
118 }
119
120 // -------------------------------------------------------------------------
121 // 5. Wire memory video callbacks (our "video sink")
122 // This tells VLC to decode frames into our pixelBuffer instead of a window.
123 //
124 // libvlc_video_set_format_callbacks: called once when VLC knows frame dimensions
125 // libvlc_video_set_callbacks: called every frame (lock/unlock/display)
126 // -------------------------------------------------------------------------
127 libvlc_video_set_format_callbacks(vlcPlayer, video_format_cb, video_cleanup_cb);
128 libvlc_video_set_callbacks(vlcPlayer, lock_cb, unlock_cb, display_cb, this);
129
130 // -------------------------------------------------------------------------
131 // 6. Subscribe to playback lifecycle events
132 // -------------------------------------------------------------------------
133 libvlc_event_manager_t* evtMgr = libvlc_media_player_event_manager(vlcPlayer);
134 if (evtMgr) {
135 libvlc_event_attach(evtMgr, libvlc_MediaPlayerPlaying, event_cb, this);
136 libvlc_event_attach(evtMgr, libvlc_MediaPlayerPaused, event_cb, this);
137 libvlc_event_attach(evtMgr, libvlc_MediaPlayerStopped, event_cb, this);
138 libvlc_event_attach(evtMgr, libvlc_MediaPlayerEndReached, event_cb, this);
139 libvlc_event_attach(evtMgr, libvlc_MediaPlayerEncounteredError, event_cb, this);
140 libvlc_event_attach(evtMgr, libvlc_MediaPlayerTimeChanged, event_cb, this);
141 libvlc_event_attach(evtMgr, libvlc_MediaPlayerLengthChanged, event_cb, this);
142 }
143
144 // -------------------------------------------------------------------------
145 // 7. Start playback (or stay paused if startPaused)
146 // -------------------------------------------------------------------------
147 state.store(VideoPlayerState::Loading);
148
149 if (libvlc_media_player_play(vlcPlayer) != 0) {
150 errorMessage = "libvlc_media_player_play() failed to start";
151 state.store(VideoPlayerState::Error);
152 return false;
153 }
154
155 if (startPaused) {
156 // Give VLC a moment to decode the first frame, then pause
157 libvlc_media_player_set_pause(vlcPlayer, 1);
158 }
159
160 LOG_INFO(CanvasObject, "Opened media: " + url);
161 return true;
162}

192void VideoPlayerInstance::SeekTo(float positionSec) {
193 if (!vlcPlayer) return;
194 float dur = durationSec.load(std::memory_order_relaxed);
195 if (dur <= 0.0f) return;
196
197 // Normalized position: p ∈ [0.0, 1.0]
198 float normalizedPos = positionSec / dur;
199 normalizedPos = std::max(0.0f, std::min(1.0f, normalizedPos));
200
201 libvlc_media_player_set_position(vlcPlayer, normalizedPos);
202}

245// =============================================================================
246// VIDEO FORMAT CALLBACK — Called once when VLC resolves frame dimensions
247// =============================================================================
248
249unsigned int VideoPlayerInstance::video_format_cb(void** data, char* chroma,
250 unsigned int* width,
251 unsigned int* height,
252 unsigned int* pitches,
253 unsigned int* lines)
254{
255 auto* self = static_cast<VideoPlayerInstance*>(*data);
256
257 // Force BGRA output format — matches Blend2D BL_FORMAT_PRGB32 native layout
258 // chroma is a 4-character string (not null-terminated): "BGRA"
259 std::memcpy(chroma, "BGRA", 4);
260
261 // Store decoded frame dimensions
262 self->videoW = static_cast<int>(*width);
263 self->videoH = static_cast<int>(*height);
264
265 // Tightly-packed rows: stride = width * 4 bytes (no padding)
266 pitches[0] = (*width) * 4;
267 lines[0] = *height;
268
269 // Allocate pixel buffer: width * height * 4 bytes (BGRA, 4 bytes/pixel)
270 {
271 std::lock_guard<std::mutex> lock(self->frameMutex);
272 self->pixelBuffer.assign(static_cast<size_t>(*width) * (*height) * 4, 0);
273 }
274
275 LOG_INFO(CanvasObject,
276 "Video format set: " + std::to_string(*width) + "x" + std::to_string(*height) + " BGRA");
277
278 // Return 1 plane (BGRA is a packed single-plane format)
279 return 1u;
280}

286// =============================================================================
287// LOCK CALLBACK — Called before each frame decode; hands VLC our pixel buffer
288// =============================================================================
289
290void* VideoPlayerInstance::lock_cb(void* data, void** planes) {
291 auto* self = static_cast<VideoPlayerInstance*>(data);
292
293 // Lock the frame mutex so the render thread cannot read while VLC is writing
294 self->frameMutex.lock();
295
296 // Give VLC our pixel buffer as the decode destination
297 planes[0] = self->pixelBuffer.empty() ? nullptr : self->pixelBuffer.data();
298
299 // Return nullptr as the 'picture' token (passed to unlock_cb and display_cb)
300 return nullptr;
301}
302
303// =============================================================================
304// UNLOCK CALLBACK — Called after a frame has been fully decoded into pixelBuffer
305// =============================================================================
306
307void VideoPlayerInstance::unlock_cb(void* data, void* /*picture*/, void* const* /*planes*/) {
308 auto* self = static_cast<VideoPlayerInstance*>(data);
309
310 // Mark that at least ONE valid decoded frame is present in pixelBuffer
311 self->hasValidDecodedFrame.store(true, std::memory_order_release);
312
313 // Signal the main render thread that a new frame is ready (atomic release barrier)
314 self->hasNewFrame.store(true, std::memory_order_release);
315
316 // Release the frame mutex so the main render thread can blit the frame
317 self->frameMutex.unlock();
318}

320// =============================================================================
321// DISPLAY CALLBACK — VLC "presents" the frame; we trigger canvas repaint
322// =============================================================================
323
324void VideoPlayerInstance::display_cb(void* data, void* /*picture*/) {
325 auto* self = static_cast<VideoPlayerInstance*>(data);
326
327 // Invoke the canvas dirty callback to schedule a Blend2D repaint.
328 // This is safe to call from VLC's decode thread as long as onFrameReady
329 // only does an atomic store (e.g. isDirty = true).
330 if (self->onFrameReady) {
331 self->onFrameReady();
332 }
333}

336 char buf[1024];

349// =============================================================================
350// EVENT CALLBACK — Handles VLC playback state transitions and time updates
351// =============================================================================
352
353void VideoPlayerInstance::event_cb(const libvlc_event_t* event, void* data) {
354 auto* self = static_cast<VideoPlayerInstance*>(data);
355
356 switch (event->type) {
357 case libvlc_MediaPlayerPlaying:
358 self->state.store(VideoPlayerState::Playing, std::memory_order_relaxed);
359 LOG_INFO(CanvasObject, "libVLC state: PLAYING");
360 break;
361
362 case libvlc_MediaPlayerPaused:
363 self->state.store(VideoPlayerState::Paused, std::memory_order_relaxed);
364 LOG_INFO(CanvasObject, "libVLC state: PAUSED");
365 break;
366
367 case libvlc_MediaPlayerStopped:
368 self->state.store(VideoPlayerState::Ended, std::memory_order_relaxed);
369 self->hasNewFrame.store(false, std::memory_order_relaxed);
370 LOG_INFO(CanvasObject, "libVLC state: STOPPED");
371 break;
372
373 case libvlc_MediaPlayerEndReached:
374 self->state.store(VideoPlayerState::Ended, std::memory_order_relaxed);
375 self->hasNewFrame.store(false, std::memory_order_relaxed);
376 LOG_INFO(CanvasObject, "libVLC state: END REACHED");
377 break;
378
379 case libvlc_MediaPlayerEncounteredError:
380 self->state.store(VideoPlayerState::Error, std::memory_order_relaxed);
381 self->errorMessage = "libVLC encountered a playback error";
382 LOG_ERROR(CanvasObject, self->errorMessage);
383 break;
384
385 case libvlc_MediaPlayerTimeChanged: {
386 // event->u.media_player_time_changed.new_time is in milliseconds
387 float newTimeSec = static_cast<float>(event->u.media_player_time_changed.new_time) / 1000.0f;
388 self->timeSec.store(newTimeSec, std::memory_order_relaxed);
389 break;
390 }
391
392 case libvlc_MediaPlayerLengthChanged: {
393 // event->u.media_player_length_changed.new_length is in milliseconds
394 float newDurSec = static_cast<float>(event->u.media_player_length_changed.new_length) / 1000.0f;
395 self->durationSec.store(newDurSec, std::memory_order_relaxed);
396 LOG_INFO(CanvasObject, "libVLC media duration: " + std::to_string(newDurSec) + "s");
397 break;
398 }
399
400 default:
401 break;
402 }
403}
