53 entry.heightMm = res.heightMm;

63 // Generate OpenGL texture for fast blitting
64 glGenTextures(1, &entry.glTexture);
65 glBindTexture(GL_TEXTURE_2D, entry.glTexture);
66 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR);
67 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);
68 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_CLAMP_TO_EDGE);
69 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_CLAMP_TO_EDGE);
70
71 BLImageData imgData;
72 if (entry.image.get_data(&imgData) == BL_SUCCESS) {
73 glPixelStorei(GL_UNPACK_ROW_LENGTH, static_cast<GLint>(imgData.stride / 4));
74#if defined(__ANDROID__)
75 // OpenGL ES 3.0 does not expose GL_BGRA or GL_UNSIGNED_INT_8_8_8_8_REV in core headers.
76 // Blend2D BL_FORMAT_PRGB32 layout on little-endian is [B, G, R, A] in memory.
77 // We upload as GL_RGBA / GL_UNSIGNED_BYTE and configure hardware texture swizzling
78 // (mapping texture B to Red and texture R to Blue) to eliminate any CPU-side channel swap overhead.
79 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_SWIZZLE_R, GL_BLUE);
80 glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_SWIZZLE_B, GL_RED);
81 glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA8, entry.pixelW, entry.pixelH, 0, GL_RGBA, GL_UNSIGNED_BYTE, imgData.pixel_data);
82#else
83 glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA8, entry.pixelW, entry.pixelH, 0, GL_BGRA, GL_UNSIGNED_INT_8_8_8_8_REV, imgData.pixel_data);
84#endif
85 glPixelStorei(GL_UNPACK_ROW_LENGTH, 0);
86 }
87
88 // Load text layer for interaction and selection
89 PdfRenderer::LoadTextLayer(pdfPath, pageIdx, entry.textLayer);
90 entry.isLoaded = true;
91
92 pageCache[pageIdx] = std::move(entry);
93 return &pageCache[pageIdx];
94}
