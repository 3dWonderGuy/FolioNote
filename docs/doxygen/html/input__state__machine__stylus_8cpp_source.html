1/**
2 * @file input_state_machine_stylus.cpp
3 * @brief Implementation of Stylus input dispatching: inking, selecting, erasing, shape creation,
4 * and stylus gesture / double-tap handling.
5 *
6 * COORDINATE SPACE MAPPING:
7 * -------------------------
8 * Raw stylus coordinates reported by SDL3 are in window pixel coordinates: $(x_{pen}, y_{pen})$.
9 * CanvasEngine maintains an origin offset $(canvasOriginX, canvasOriginY)$ corresponding to
10 * top/side UI chrome (ribbon, navigation drawer).
11 *
12 * Local canvas projection:
13 * $x_{local} = x_{pen} - canvasOriginX$
14 * $y_{local} = y_{pen} - canvasOriginY$
15 *
16 * When hit-testing or creating objects, $x_{local}, y_{local}$ is projected into continuous
17 * world millimeter space via the camera transform matrix:
18 * $P_{world} = \text{ScreenToWorld}(x_{local}, y_{local})$
19 */

26void InputStateMachine::DispatchStylus(CanvasEngine& canvas, DocumentSession& session, bool imguiWantsInput) {
27 const bool justDown = (currentStylusState == StylusState::Engaged && oldStylusState != StylusState::Engaged);
28 const bool isMoving = (currentStylusState == StylusState::Engaged && oldStylusState == StylusState::Engaged);
29 const bool justUp = (currentStylusState != StylusState::Engaged && oldStylusState == StylusState::Engaged);
30
31 // -------------------------------------------------------------------------
32 // 1. UI CAPTURE EVALUATION
33 // -------------------------------------------------------------------------
34 // If stylus contact initiated over UI chrome (e.g. toolbar button), lock capture
35 // to UI until the stylus lifts off the screen.
36 if (justDown) {
37 uiCapturedStylus = imguiWantsInput;
38
39 // Check for double-tap special action (e.g. quick toggle to eraser or undo)
40 if (!uiCapturedStylus) {
41 uint64_t now = SDL_GetTicks();
42 if (specialActions.EvaluateDoubleClick(pen.x, pen.y, now, config)) {
43 LOG_INFO(InputStateMachine, "Stylus Double-Tap detected! Toggling eraser/pen tool.");
44 // If currently inking, toggle to eraser; if eraser, toggle back to inking
45 if (currentAction == InteractionState::Inking) {
46 currentAction = InteractionState::Eraser;
47 SetToolForDevice(DeviceType::Stylus, InteractionState::Eraser);
48 } else if (currentAction == InteractionState::Eraser) {
49 currentAction = InteractionState::Inking;
50 SetToolForDevice(DeviceType::Stylus, InteractionState::Inking);
51 }
52 }
53 }
54 }
55
56 if (uiCapturedStylus) {
57 if (justUp) uiCapturedStylus = false;
58 return;
59 }
60
61 // Transform from window pixel space to canvas-local coordinates
62 const float canvasLocalX = pen.x - canvasOriginX;
63 const float canvasLocalY = pen.y - canvasOriginY;
64
65 if (currentAction != InteractionState::DrawingShape && canvas.shapeCreation.lockDrawingMode) {
66 canvas.shapeCreation.lockDrawingMode = false;
67 canvas.shapeCreation.isActive = false;
68 }
69
70 // -------------------------------------------------------------------------
71 // 2. SEMANTIC ACTION DISPATCH
72 // -------------------------------------------------------------------------
73 switch (currentAction) {
74 // =====================================================================
75 // INKING (Pressure & Tilt Sensitive Digital Ink)
76 // =====================================================================
77 case InteractionState::Inking: {
78 if (justDown) {
79 canvas.OnPointerDown(canvasLocalX, canvasLocalY, pen.pressure, latestEventTimeSec, palette.GetActivePen(), pen.tiltX, pen.tiltY);
80 } else if (isMoving) {
81 canvas.OnPointerMove(canvasLocalX, canvasLocalY, pen.pressure, latestEventTimeSec, pen.tiltX, pen.tiltY);
82 } else if (justUp) {
83 canvas.OnPointerUp(session, palette.GetActivePen());
84 }
85 break;
86 }
87
88 // =====================================================================
89 // SELECTING (Object Hit-Testing, Gizmo Manipulation, Lasso / Box Selection)
90 // =====================================================================
91 case InteractionState::Selecting: {
92 if (justDown) {
93 // First test if pointer clicked an active selection gizmo handle
94 if (canvas.selectionGizmo.OnPointerDown(canvasLocalX, canvasLocalY, canvas.transform,
95 canvas.shapeCreation.lockToGrid, canvas.gridSpacingMm)) {
96 canvas.isDirty = true;
97 } else {
98 // Convert local coordinates to world millimeters for spatial hit-testing
99 Point2D worldMm = canvas.transform.ScreenToWorld(canvasLocalX, canvasLocalY);
100 auto activePage = session.GetActivePage();
101 std::shared_ptr<CanvasObject> clickedObj = nullptr;
102
103 if (activePage) {
104 clickedObj = activePage->HitTestSingleClick(worldMm.x, worldMm.y, config.objectHitTestRadiusMm);
105 }
106
107 if (clickedObj) {
108 canvas.ClearSelection(&session);
109 clickedObj->isSelected = 1;
110 canvas.selectionGizmo.SetSelectedObjects(activePage->objects);
111 canvas.selectionGizmo.OnPointerDown(canvasLocalX, canvasLocalY, canvas.transform,
112 canvas.shapeCreation.lockToGrid, canvas.gridSpacingMm);
113 canvas.needsFullRebake = true;
114 canvas.isDirty = true;
115 LOG_INFO(InputStateMachine, "Stylus direct click selected object uid=" + std::to_string(clickedObj->uid));
116 } else {
117 // Clicked empty canvas: clear selection and begin lasso or box select
118 canvas.ClearSelection(&session);
119 if (canvas.selectionMode == CanvasEngine::SelectionMode::Lasso) {
120 canvas.OnLassoDown(canvasLocalX, canvasLocalY);
121 } else {
122 canvas.OnBoxSelectDown(canvasLocalX, canvasLocalY);
123 }
124 }
125 }
126 }
127 else if (isMoving) {
128 if (canvas.selectionGizmo.isDragging) {
129 if (canvas.selectionGizmo.OnPointerMove(canvasLocalX, canvasLocalY, canvas.transform,
130 canvas.shapeCreation.lockToGrid, canvas.gridSpacingMm)) {
131 canvas.InvalidateLayer();
132 }
133 } else if (canvas.selectionMode == CanvasEngine::SelectionMode::Lasso) {
134 canvas.OnLassoMove(canvasLocalX, canvasLocalY);
135 } else if (canvas.marqueeBox.isActive) {
136 canvas.OnBoxSelectMove(canvasLocalX, canvasLocalY);
137 }
138 }
139 else if (justUp) {
140 if (canvas.selectionGizmo.isDragging) {
141 canvas.selectionGizmo.OnPointerUp(&session);
142 canvas.SyncSelectionToSpatialIndex(&session);
143 canvas.InvalidateLayer();
144 } else if (canvas.selectionMode == CanvasEngine::SelectionMode::Lasso) {
145 canvas.OnLassoUp(&session);
146 } else if (canvas.marqueeBox.isActive) {
147 canvas.OnBoxSelectUp(&session);
148 }
149 }
150 break;
151 }
152
153 // =====================================================================
154 // ERASER (Segment and Stroke Erasing)
155 // =====================================================================
156 case InteractionState::Eraser: {
157 double currentTimeSec = (latestEventTimeSec > 0.0) ? latestEventTimeSec : (SDL_GetTicks() * 0.001);
158 float effectiveRadius = UpdateDynamicEraserRadius(canvasLocalX, canvasLocalY, currentTimeSec, isMoving, isEraserActive || justDown);
159
160 if (justDown) {
161 lastEraserX = canvasLocalX;
162 lastEraserY = canvasLocalY;
163 isEraserActive = true;
164 session.BeginEraseTransaction();
165 canvas.EraseSegment(canvasLocalX, canvasLocalY, canvasLocalX, canvasLocalY, effectiveRadius, session, isStrokeEraser);
166 } else if (isMoving && isEraserActive) {
167 canvas.EraseSegment(lastEraserX, lastEraserY, canvasLocalX, canvasLocalY, effectiveRadius, session, isStrokeEraser);
168 lastEraserX = canvasLocalX;
169 lastEraserY = canvasLocalY;
170 } else if (justUp) {
171 isEraserActive = false;
172 session.EndEraseTransaction(&canvas);
173 }
174 canvas.SetEraserCursor(canvasLocalX, canvasLocalY, effectiveRadius, isEraserActive, isStrokeEraser);
175 break;
176 }
177
178 // =====================================================================
179 // DRAWING SHAPE (Geometric Vector Shape Drag-Creation)
180 // =====================================================================
181 case InteractionState::DrawingShape: {
182 if (ImGui::IsKeyPressed(ImGuiKey_Escape)) {
183 canvas.CancelShapeCreation();
184 canvas.ClearSelection(&session);
185 currentAction = InteractionState::Selecting;
186 SetToolForDevice(DeviceType::Stylus, InteractionState::Selecting);
187 SetToolForDevice(DeviceType::Mouse, InteractionState::Selecting);
188 } else if (justDown) {
189 // Clean state isolation: suppress previous gizmo interactions when actively drawing
190 if (canvas.selectionGizmo.HasSelection()) {
191 canvas.ClearSelection(&session);
192 canvas.selectionGizmo.ClearSelection();
193 }
194 canvas.OnShapeDrawDown(canvasLocalX, canvasLocalY, &session);
195 canvas.isDirty = true;
196 } else if (isMoving) {
197 if (canvas.shapeCreation.isDragging || (canvas.shapeCreation.shapeType == Folio::ShapeType::Ellipse && canvas.shapeCreation.ellipseStep == 1)) {
198 canvas.OnShapeDrawMove(canvasLocalX, canvasLocalY, &session);
199 canvas.isDirty = true;
200 }
201 } else if (justUp) {
202 if (canvas.shapeCreation.isDragging || (canvas.shapeCreation.shapeType == Folio::ShapeType::Ellipse && canvas.shapeCreation.ellipseStep == 1)) {
203 canvas.OnShapeDrawUp(&session);
204 canvas.needsFullRebake = true;
205 canvas.isDirty = true;
206 if (canvas.shapeCreation.ellipseStep == 0) {
207 if (!canvas.shapeCreation.lockDrawingMode) {
208 currentAction = InteractionState::Selecting;
209 SetToolForDevice(DeviceType::Stylus, InteractionState::Selecting);
210 SetToolForDevice(DeviceType::Mouse, InteractionState::Selecting);
211 }
212 }
213 }
214 }
215 break;
216 }
217
218 default: break;
219 }
