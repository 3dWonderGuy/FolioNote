41bool ObjectRegistry::Register(
42 ObjectType type,
43 const std::string& typeName,
44 ObjectFactoryFn factory,
45 const std::string& defaultIcon,
46 bool isSerializable
47) {
48 if (!factory) {
49 LOG_WARN(ObjectRegistry, "Registration failed: Null factory provided for type=" + typeName);
50 return false;
51 }
52
53 auto& storage = GetStorage();
54 std::lock_guard<std::mutex> lock(storage.mutex);
55
56 ObjectTypeMetadata meta;
57 meta.type = type;
58 meta.typeName = typeName;
59 meta.defaultIcon = defaultIcon;
60 meta.isSerializable = isSerializable;
61 meta.factory = std::move(factory);
62
63 storage.registry[type] = std::move(meta);
64 LOG_INFO(ObjectRegistry, "Registered CanvasObject type: [" + typeName +
65 "] (ID=" + std::to_string(static_cast<uint32_t>(type)) + ")");
66 return true;
67}

69std::shared_ptr<CanvasObject> ObjectRegistry::Create(ObjectType type) {
70 auto& storage = GetStorage();
71 std::lock_guard<std::mutex> lock(storage.mutex);
72
73 auto it = storage.registry.find(type);
74 if (it != storage.registry.end() && it->second.factory) {
75 auto obj = it->second.factory();
76 if (obj) {
77 obj->type = type; // Ensure type discriminator is synchronized
78 }
79 return obj;
80 }
81
82 LOG_WARN(ObjectRegistry, "Create() failed: No factory registered for ObjectType=" +
83 std::to_string(static_cast<uint32_t>(type)));
84 return nullptr;
85}
