1/**
2 * =========================================================================================
3 * @file notebook_cloner.cpp
4 * @brief High-Fidelity Package Container Initializer and Pure Filesystem Replicator
5 * =========================================================================================
6 *
7 * ARCHITECTURAL PURPOSE & DESIGN RATIONALE:
8 * In FolioNote, a .notebook package is a composite bundle comprising:
9 * 1. An SQLite database (structure.db) housing notebook metadata, section
10 * hierarchies, canvas page geometries, and search indexes.
11 * 2. A `pages/` subdirectory housing individual `.ink` binary vector stroke files.
12 * 3. An `imports/` subdirectory housing referenced PDFs and image assets.
13 *
14 * PURE FILESYSTEM CLONING:
15 * To maintain 100% vector fidelity and zero-cost duplication without loading megabytes of
16 * stroke points into RAM, `NotebookCloner::CloneNotebook` executes a pure recursive filesystem
17 * copy using `Folio::FileManager::CopyDirectoryRecursive`.
18 *
19 * ZERO DATABASE COUPLING:
20 * This subsystem operates purely on filesystem containers and paths. It does NOT include
21 * SQLite or `PageRepository`. Package database schema creation is deferred to the active
22 * `Workspace` session (`workspace.repository.OpenNotebookPackage`).
23 */

36/**
37 * @brief Synchronously initializes a physical .notebook package directory container on disk.
38 *
39 * GENERAL WORKING PROCESS & INVARIANTS:
40 * 1. Validates that the notebook has a non-empty `filePath`.
41 * 2. Creates the parent package directory and the child `pages/` folder via `FileManager`.
42 * 3. Leaves database creation and active document management to the `Workspace` layer.
43 *
44 * @param notebook Shared pointer to the in-memory Notebook model.
45 * @return bool True if the package directories were successfully created on disk; false otherwise.
46 */
47bool NotebookCloner::InitializePackage(const std::shared_ptr<Notebook>& notebook) {
48 if (!notebook || notebook->filePath.empty()) {
49 LOG_ERROR(LibraryManager, "NotebookCloner::InitializePackage rejected: Null notebook or empty filePath.");
50 return false;
51 }
52
53 // Create package folder and pages subdirectory
54 if (!FileManager::CreateDirectories(notebook->filePath)) {
55 LOG_ERROR(LibraryManager, "NotebookCloner failed to create package directory: " + notebook->filePath);
56 return false;
57 }
58
59 std::string pagesDir = FileManager::JoinPath(notebook->filePath, "pages");
60 if (!FileManager::CreateDirectories(pagesDir)) {
61 LOG_ERROR(LibraryManager, "NotebookCloner failed to create pages directory: " + pagesDir);
62 return false;
63 }
64
65 LOG_INFO(LibraryManager, "NotebookCloner: Initialized package container at: " + notebook->filePath);
66
67 // Apply Windows Shell package identity (desktop.ini)
68 PackageMarker::MarkFolderAsPackage(notebook->filePath, "FolioNote Notebook Package");
69 return true;
70}

72/**
73 * @brief Duplicates an existing notebook package, ensuring fresh unique GUIDs across
74 * the entire document hierarchy while preserving 100% binary vector stroke fidelity.
75 *
76 * GENERAL WORKING PROCESS & UNIQUE UID REMAPPING ALGORITHM:
77 * 1. Resolves destination library root directory (falling back to source parent or default).
78 * 2. Formats a collision-free package path inside destination using `FileManager::DisambiguatePath`.
79 * 3. Creates the physical package folder (`<pkgPath>/`) and `<pkgPath>/pages/`.
80 * 4. Copies imported assets (`<pkgPath>/imports/`) recursively if present.
81 * 5. Deep-clones the in-memory Notebook model via `sourceNb->Clone(safeName)`:
82 * - New UUID v4 assigned to Notebook.
83 * - New UUID v4 assigned to each SectionGroup (and recursively to sub-groups).
84 * - New UUID v4 assigned to each Section.
85 * - New UUID v4 assigned to each CanvasPage.
86 * 6. Collects all pages from both source and cloned models via `GetAllPages(true)`:
87 * Because the clone traverses sections and groups in identical deterministic order,
88 * `srcPages[i]` maps 1:1 to `dstPages[i]`.
89 * 7. For each page pair, copies `<srcPkg>/pages/{srcPages[i]->guid}.ink` directly to
90 * `<dstPkg>/pages/{dstPages[i]->guid}.ink`.
91 * - 100% vector fidelity (raw binary preserved byte-for-byte).
92 * - Zero RAM overhead (no deserializing or re-compressing strokes).
93 * - Crucially: On disk, every `.ink` file is indexed by its BRAND NEW UNIQUE GUID!
94 * 8. SQLite schema and metadata persistence is deferred to `Workspace::OpenAndActivateNotebook`
95 * which writes clean tables populated with the new unique GUIDs.
96 *
97 * @param sourceNb Shared pointer to the source notebook being duplicated.
98 * @param newName Proposed title for the duplicate notebook (e.g. "Calculus - Final").
99 * @param targetLibraryPath Path to the target library folder (empty for default or source parent).
100 * @param fallbackLibraryPath Fallback path if targetLibraryPath is invalid.
101 * @return Shared pointer to the newly cloned Notebook object, or nullptr on failure.
102 */
103std::shared_ptr<Notebook> NotebookCloner::CloneNotebook(
104 const std::shared_ptr<Notebook>& sourceNb,
105 const std::string& newName,
106 const std::string& targetLibraryPath,
107 const std::string& fallbackLibraryPath
108) {
109 if (!sourceNb || sourceNb->filePath.empty()) {
110 LOG_ERROR(LibraryManager, "NotebookCloner::CloneNotebook rejected: Source notebook pointer is null or has empty filePath.");
111 return nullptr;
112 }
113
114 // Step 1: Resolve destination library directory
115 std::string destLib = targetLibraryPath;
116 if (destLib.empty() || !LibraryManager::IsLibraryFolder(destLib)) {
117 std::string srcParent = FileManager::GetParentPath(sourceNb->filePath);
118 if (LibraryManager::IsLibraryFolder(srcParent)) {
119 destLib = srcParent;
120 } else {
121 destLib = fallbackLibraryPath;
122 }
123 }
124
125 FileManager::CreateDirectories(destLib);
126
127 // Step 2: Determine collision-free destination package path
128 std::string safeName = newName.empty() ? (sourceNb->name + " - Copy") : newName;
129 std::string newPkgPath = FileManager::DisambiguatePath(destLib, safeName, FOLIO_NOTEBOOK_EXTENSION);
130
131 // Step 3: Initialize physical package directory structure
132 if (!FileManager::CreateDirectories(newPkgPath)) {
133 LOG_ERROR(LibraryManager, "NotebookCloner failed to create package directory: " + newPkgPath);
134 return nullptr;
135 }
136
137 std::string dstPagesDir = FileManager::JoinPath(newPkgPath, "pages");
138 if (!FileManager::CreateDirectories(dstPagesDir)) {
139 LOG_ERROR(LibraryManager, "NotebookCloner failed to create pages directory: " + dstPagesDir);
140 return nullptr;
141 }
142
143 // Step 4: Copy imported assets (PDFs, images) if present
144 std::string srcImportsDir = FileManager::JoinPath(sourceNb->filePath, "imports");
145 if (FileManager::IsDirectory(srcImportsDir)) {
146 std::string dstImportsDir = FileManager::JoinPath(newPkgPath, "imports");
147 FileManager::CopyDirectoryRecursive(srcImportsDir, dstImportsDir);
148 }
149
150 // Step 5: Deep-clone in-memory model generating fresh unique UUIDs across all tiers
151 auto copyNb = sourceNb->Clone(safeName);
152 copyNb->filePath = newPkgPath;
153
154 // Step 6: Map source pages to cloned pages and copy raw .ink files under the new unique GUIDs
155 std::string srcPagesDir = FileManager::JoinPath(sourceNb->filePath, "pages");
156 if (FileManager::IsDirectory(srcPagesDir)) {
157 auto srcPages = sourceNb->GetAllPages(true);
158 auto dstPages = copyNb->GetAllPages(true);
159
160 if (srcPages.size() == dstPages.size()) {
161 size_t copiedInkCount = 0;
162 for (size_t i = 0; i < srcPages.size(); ++i) {
163 if (!srcPages[i] || !dstPages[i]) continue;
164 std::string srcInk = FileManager::JoinPath(srcPagesDir, srcPages[i]->guid + ".ink");
165 std::string dstInk = FileManager::JoinPath(dstPagesDir, dstPages[i]->guid + ".ink");
166 if (FileManager::IsRegularFile(srcInk)) {
167 if (FileManager::CopySingleFile(srcInk, dstInk, true)) {
168 copiedInkCount++;
169 }
170 }
171 }
172 LOG_INFO(LibraryManager, "NotebookCloner: Re-mapped and copied " + std::to_string(copiedInkCount) +
173 " .ink files with unique GUIDs to: " + dstPagesDir);
174 } else {
175 LOG_WARN(LibraryManager, "NotebookCloner: Page count mismatch between source (" +
176 std::to_string(srcPages.size()) + ") and clone (" +
177 std::to_string(dstPages.size()) + ")");
178 }
179 }
180
181 LOG_INFO(LibraryManager, "NotebookCloner: Successfully cloned notebook package with unique GUIDs to: " + newPkgPath);
182
183 // Apply Windows Shell package identity (desktop.ini)
184 PackageMarker::MarkFolderAsPackage(newPkgPath, "FolioNote Notebook Package");
185 return copyNb;
186}

188/**
189 * @brief Directly copies raw `.ink` stroke binary files between package `pages/` directories.
190 *
191 * @param srcPackagePath Absolute filesystem path to source .notebook package directory.
192 * @param dstPackagePath Absolute filesystem path to destination .notebook package directory.
193 * @return size_t Count of `.ink` page stroke files copied.
194 */
195size_t NotebookCloner::CopyPageInkFiles(
196 const std::string& srcPackagePath,
197 const std::string& dstPackagePath
198) {
199 size_t count = 0;
200 std::string srcPagesDir = FileManager::JoinPath(srcPackagePath, "pages");
201 std::string dstPagesDir = FileManager::JoinPath(dstPackagePath, "pages");
202
203 if (FileManager::IsDirectory(srcPagesDir)) {
204 FileManager::CreateDirectories(dstPagesDir);
205 auto pageEntries = FileManager::ListEntries(srcPagesDir, ".ink");
206 for (const auto& entry : pageEntries) {
207 std::string dstInk = FileManager::JoinPath(dstPagesDir, entry.fileName);
208 if (FileManager::CopySingleFile(entry.fullPath, dstInk, true)) {
209 count++;
210 }
211 }
212 }
213 return count;
214}
