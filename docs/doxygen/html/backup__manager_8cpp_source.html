7 * ARCHITECTURAL IMPLEMENTATION DETAILS:
8 * 1. Multi-Ring Grandfather-Father-Son Rotation:
9 * - Tier 1 (Daily): Evaluates interval threshold $\Delta t \ge 24\text{ hours}$ ($86.4 \times 10^6\text{ ms}$).
10 * Retains $N_{\text{daily}} = 7$ rolling snapshots.
11 * - Tier 2 (Weekly): Evaluates interval threshold $\Delta t \ge 7\text{ days}$ ($604.8 \times 10^6\text{ ms}$).
12 * Retains $N_{\text{weekly}} = 4$ rolling snapshots.
13 * - Tier 3 (Monthly): Evaluates interval threshold $\Delta t \ge 30\text{ days}$ ($2.592 \times 10^9\text{ ms}$).
14 * Retains $N_{\text{monthly}} = 12$ rolling snapshots.
15 * 2. Deep Isolated Filesystem Paths:
16 * Organized cleanly under `<AppRoot>/backups/tiers/{daily,weekly,monthly}/` or `<AppRoot>/backups/manual/`.
17 * 3. Safe SQLite WAL Checkpoint Integration:
18 * Scans the source directory tree for any `.db` files and executes `sqlite3_wal_checkpoint_v2`
19 * with `SQLITE_CHECKPOINT_TRUNCATE`. This forces SQLite to merge all uncommitted WAL pages
20 * into the primary database file and reset the journal, eliminating Windows file sharing
21 * violations (`ERROR_SHARING_VIOLATION`) during `std::filesystem::copy`.
22 * 4. Non-Blocking Background ThreadPool Execution:
23 * All intensive recursive directory copying and integrity verification run asynchronously.
24 */

47 * Format: "YYYYMMDD_HHMMSS" (e.g., "20260920_013000").
48 */
49std::string GetTimestampString() {
50 auto now = std::chrono::system_clock::now();
51 std::time_t tt = std::chrono::system_clock::to_time_t(now);
52 std::tm tmVal{};
53#if defined(_WIN32)
54 localtime_s(&tmVal, &tt);
55#else
56 localtime_r(&tt, &tmVal);
57#endif
58 std::ostringstream oss;
59 oss << std::put_time(&tmVal, "%Y%m%d_%H%M%S");
60 return oss.str();
61}

81/**
82 * @brief Calculates the recursive byte size of all files inside a directory.
83 *
84 * MATHEMATICAL PROCESS:
85 * Accumulates $\sum \text{file\_size}(f)$ for all directory regular files.
86 *
87 * @param dirPath Path to evaluate.
88 * @return Total accumulated size in bytes.
89 */
90uint64_t CalculateDirectorySize(const std::string& dirPath) {
91 uint64_t total = 0;
92 std::error_code ec;
93 for (const auto& entry : std::filesystem::recursive_directory_iterator(dirPath, ec)) {
94 if (ec) break;
95 if (std::filesystem::is_regular_file(entry.status())) {
96 total += entry.file_size(ec);
97 }
98 }
99 return total;
100}

135/**
136 * @brief Parses a backup.meta file into a BackupInfo structure.
137 */
138bool ParseBackupMetadata(const std::string& snapshotDir, BackupInfo& outInfo) {
139 std::string metaPath = FileManager::JoinPath(snapshotDir, "backup.meta");
140 std::string content;
141 if (!FileManager::ReadText(metaPath, content)) {
142 return false;
143 }
144
145 outInfo.backupPath = FileManager::NormalizeSeparators(snapshotDir);
146 outInfo.totalBytes = CalculateDirectorySize(snapshotDir);
147
148 std::istringstream stream(content);
149 std::string line;
150 while (std::getline(stream, line)) {
151 if (line.empty() || line[0] == '#') continue;
152 auto eq = line.find('=');
153 if (eq == std::string::npos) continue;
154
155 std::string key = line.substr(0, eq);
156 std::string val = line.substr(eq + 1);
157
158 if (key == "id") outInfo.backupId = val;
159 else if (key == "sourceName") outInfo.sourceName = val;
160 else if (key == "sourcePath") outInfo.sourcePath = val;
161 else if (key == "isLibrary") outInfo.isLibrary = (val == "1" || val == "true");
162 else if (key == "tier") outInfo.tier = BackupManager::StringToTier(val);
163 else if (key == "timestamp") {
164 try { outInfo.timestamp = std::stoull(val); } catch (...) {}
165 }
166 else if (key == "timestampStr") outInfo.timestampStr = val;
167 else if (key == "notebookCount") {
168 try { outInfo.notebookCount = std::stoul(val); } catch (...) {}
169 }
170 }
171
172 if (outInfo.backupId.empty()) {
173 outInfo.backupId = FileManager::GetFileName(snapshotDir);
174 }
175 return true;
176}

178/**
179 * @brief Recursively copies a directory tree while gracefully handling OneDrive cloud placeholders,
180 * broken reparse points, and files locked by background processes.
181 *
182 * GENERAL WORKING PROCESS & ERROR RESILIENCE:
183 * 1. Directory Tree Traversal:
184 * Iterates through sourceDir using std::filesystem::recursive_directory_iterator with
185 * skip_permission_denied enabled to prevent OS permission traps from crashing the pass.
186 * 2. Reparse Point & Symlink Verification:
187 * Cloud providers (like Microsoft OneDrive or iCloud) create placeholder reparse points for
188 * dehydrated files. If a placeholder has no local backing or cannot be resolved, a naive
189 * std::filesystem::copy fails immediately with ERROR_PATH_NOT_FOUND (The system cannot find the path specified).
190 * RobustCopyDirectory inspects each entry individually: if it is a broken reparse point/symlink,
191 * it emits a diagnostic warning and skips it rather than aborting the backup of valid notebooks and data.
192 * 3. Atomic Per-File Replication:
193 * Directories are created on the destination as encountered. Regular files are copied with
194 * overwrite_existing. Any single file read error (e.g. temporary sharing violation) is logged as a
195 * warning and does not invalidate the entire snapshot.
196 *
197 * @param sourceDir Absolute source directory path.
198 * @param destDir Target snapshot directory path.
199 * @param ec Output error code, set only if destination directory cannot be created at all.
200 * @return true if destination directory was prepared and valid content was copied; false on fatal initialization failure.
201 */
202bool RobustCopyDirectory(const std::filesystem::path& sourceDir, const std::filesystem::path& destDir, std::error_code& ec) {
203 ec.clear();
204 std::filesystem::create_directories(destDir, ec);
205 if (ec) {
206 return false;
207 }
208
209 auto options = std::filesystem::directory_options::skip_permission_denied;
210 for (std::filesystem::recursive_directory_iterator it(sourceDir, options, ec), end; it != end; it.increment(ec)) {
211 if (ec) {
212 LOG_WARN(FileManager, "RobustCopyDirectory: Skipping unreadable traversal node: " + ec.message());
213 ec.clear();
214 continue;
215 }
216
217 try {
218 const auto& entry = *it;
219 std::error_code relEc;
220 auto relPath = std::filesystem::relative(entry.path(), sourceDir, relEc);
221 if (relEc) {
222 continue;
223 }
224 auto targetPath = destDir / relPath;
225
226 std::error_code statusEc;
227 auto symStatus = entry.symlink_status(statusEc);
228 if (statusEc) {
229 continue;
230 }
231
232 // Check for broken symlinks and cloud placeholders
233 if (std::filesystem::is_symlink(symStatus)) {
234 if (!std::filesystem::exists(entry.path(), statusEc)) {
235 LOG_WARN(FileManager, "RobustCopyDirectory: Skipping broken cloud reparse point / symlink: " + entry.path().string());
236 continue;
237 }
238 }
239
240 if (entry.is_directory(statusEc)) {
241 std::filesystem::create_directories(targetPath, statusEc);
242 } else if (entry.is_regular_file(statusEc)) {
243 std::filesystem::create_directories(targetPath.parent_path(), statusEc);
244 std::filesystem::copy_file(entry.path(), targetPath, std::filesystem::copy_options::overwrite_existing, statusEc);
245 if (statusEc) {
246 LOG_WARN(FileManager, "RobustCopyDirectory: Skipping copy for file (" + statusEc.message() + "): " + entry.path().string());
247 }
248 }
249 } catch (const std::exception& ex) {
250 LOG_WARN(FileManager, std::string("RobustCopyDirectory exception on entry: ") + ex.what());
251 }
252 }
253 return true;
254}

297bool BackupManager::FlushSqliteWal(const std::string& rootDirectory) {
298 if (rootDirectory.empty() || !FileManager::IsDirectory(rootDirectory)) {
299 return false;
300 }
301
302 std::error_code ec;
303 bool allSuccess = true;
304
305 for (const auto& entry : std::filesystem::recursive_directory_iterator(rootDirectory, ec)) {
306 if (ec) break;
307 if (std::filesystem::is_regular_file(entry.status()) && entry.path().extension() == ".db") {
308 std::string dbPath = entry.path().string();
309 sqlite3* db = nullptr;
310 int rc = sqlite3_open(dbPath.c_str(), &db);
311 if (rc == SQLITE_OK && db != nullptr) {
312 int chkRc = sqlite3_wal_checkpoint_v2(db, nullptr, SQLITE_CHECKPOINT_TRUNCATE, nullptr, nullptr);
313 if (chkRc != SQLITE_OK) {
314 LOG_WARN(FileManager, "FlushSqliteWal: WAL checkpoint returned status " +
315 std::to_string(chkRc) + " for " + dbPath);
316 allSuccess = false;
317 }
318 sqlite3_close(db);
319 } else {
320 if (db) sqlite3_close(db);
321 allSuccess = false;
322 }
323 }
324 }
325
326 return allSuccess;
327}

337bool BackupManager::BackupLibraryTiered(const std::string& libraryPath, BackupTier tier, const std::string& customRoot) {
338 if (libraryPath.empty() || !FileManager::IsDirectory(libraryPath)) {
339 LOG_ERROR(FileManager, "BackupLibraryTiered: Invalid or missing library path: " + libraryPath);
340 return false;
341 }
342
343 std::string libName = FileManager::GetStem(libraryPath);
344 if (libName.empty()) libName = "Library";
345
346 std::string targetDir = GetTierDirectory(tier, customRoot);
347 if (!FileManager::CreateDirectories(targetDir)) {
348 LOG_ERROR(FileManager, "BackupLibraryTiered: Cannot create target directory: " + targetDir);
349 return false;
350 }
351
352 // Flush SQLite WAL logs to prevent Windows sharing violations and dirty copies
353 FlushSqliteWal(libraryPath);
354
355 std::string timeSuffix = GetTimestampString();
356 std::string backupFolderName = libName + "_Backup_" + timeSuffix;
357 std::string snapshotPath = FileManager::JoinPath(targetDir, backupFolderName);
358
359 LOG_INFO(FileManager, "Starting " + TierToString(tier) + " backup of library '" + libName + "' to: " + snapshotPath);
360
361 std::error_code ec;
362 std::filesystem::create_directories(snapshotPath, ec);
363 if (ec) {
364 LOG_ERROR(FileManager, "Failed to create snapshot directory: " + ec.message());
365 return false;
366 }
367
368 size_t nbCount = 0;
369 for (const auto& entry : std::filesystem::directory_iterator(libraryPath, ec)) {
370 if (std::filesystem::is_directory(entry.status()) && entry.path().extension() == ".notebook") {
371 nbCount++;
372 }
373 }
374
375 std::error_code copyEc;
376 if (!RobustCopyDirectory(libraryPath, snapshotPath, copyEc)) {
377 LOG_ERROR(FileManager, "BackupLibraryTiered recursive copy failed: " + copyEc.message());
378 std::filesystem::remove_all(snapshotPath, copyEc);
379 return false;
380 }
381
382 std::string backupId = "backup_lib_" + TierToString(tier) + "_" + timeSuffix;
383 WriteBackupMetadata(snapshotPath, backupId, libName, libraryPath, true, nbCount, tier);
384
385 // If a secondary backup mirror destination is configured, mirror the snapshot to it
386 const auto& secDir = SettingsManager::Instance().secondaryBackupDirectory;
387 if (!secDir.empty() && customRoot.empty()) {
388 std::string secTargetDir = GetTierDirectory(tier, secDir);
389 if (FileManager::CreateDirectories(secTargetDir)) {
390 std::string secSnapshotPath = FileManager::JoinPath(secTargetDir, backupFolderName);
391 std::error_code secEc;
392 if (RobustCopyDirectory(snapshotPath, secSnapshotPath, secEc)) {
393 WriteBackupMetadata(secSnapshotPath, backupId, libName, libraryPath, true, nbCount, tier);
394 LOG_INFO(FileManager, "Mirrored backup snapshot to secondary destination: " + secSnapshotPath);
395 }
396 }
397 }
398
399 uint64_t totalBytes = CalculateDirectorySize(snapshotPath);
400 LOG_INFO(FileManager, "Completed " + TierToString(tier) + " library backup. Snapshot: '" + snapshotPath +
401 "' | Size: " + std::to_string(totalBytes / 1024) + " KB | Notebooks: " + std::to_string(nbCount));
402 return true;
403}

486std::vector<BackupInfo> BackupManager::ListBackups(const std::string& backupRootDir) {
487 std::string root = backupRootDir.empty() ? FileManager::GetBackupsDirectory() : backupRootDir;
488 std::vector<BackupInfo> backups;
489
490 std::error_code ec;
491 if (!std::filesystem::exists(root, ec) || !std::filesystem::is_directory(root, ec)) {
492 return backups;
493 }
494
495 for (const auto& entry : std::filesystem::recursive_directory_iterator(root, ec)) {
496 if (ec) break;
497 if (std::filesystem::is_regular_file(entry.status()) && entry.path().filename() == "backup.meta") {
498 std::string snapshotDir = entry.path().parent_path().string();
499 BackupInfo info;
500 if (ParseBackupMetadata(snapshotDir, info)) {
501 backups.push_back(info);
502 }
503 }
504 }
505
506 std::sort(backups.begin(), backups.end(), [](const BackupInfo& a, const BackupInfo& b) {
507 return a.timestamp > b.timestamp;
508 });
509
510 return backups;
511}

518bool BackupManager::RestoreBackup(const std::string& backupSnapshotPath, const std::string& targetDestinationDir) {
519 if (backupSnapshotPath.empty() || !FileManager::IsDirectory(backupSnapshotPath)) {
520 LOG_ERROR(FileManager, "RestoreBackup rejected: Invalid backup snapshot path: " + backupSnapshotPath);
521 return false;
522 }
523 if (targetDestinationDir.empty()) {
524 LOG_ERROR(FileManager, "RestoreBackup rejected: Empty target destination directory.");
525 return false;
526 }
527
528 BackupInfo info;
529 if (!ParseBackupMetadata(backupSnapshotPath, info)) {
530 LOG_WARN(FileManager, "RestoreBackup: Snapshot lacks valid backup.meta; proceeding with raw file copy.");
531 }
532
533 std::error_code ec;
534 FileManager::CreateDirectories(targetDestinationDir);
535
536 LOG_INFO(FileManager, "Restoring backup snapshot '" + backupSnapshotPath + "' into '" + targetDestinationDir + "'");
537
538 std::filesystem::copy(
539 backupSnapshotPath,
540 targetDestinationDir,
541 std::filesystem::copy_options::recursive | std::filesystem::copy_options::overwrite_existing,
542 ec
543 );
544
545 if (ec) {
546 LOG_ERROR(FileManager, "RestoreBackup copy operation failed: " + ec.message());
547 return false;
548 }
549
550 std::string metaInDest = FileManager::JoinPath(targetDestinationDir, "backup.meta");
551 if (FileManager::Exists(metaInDest)) {
552 std::filesystem::remove(metaInDest, ec);
553 }
554
555 LOG_INFO(FileManager, "RestoreBackup successfully restored to: " + targetDestinationDir);
556 return true;
557}

579size_t BackupManager::PruneOldBackups(const std::string& sourceName, size_t maxToKeep, const std::string& backupRootDir) {
580 auto allBackups = ListBackups(backupRootDir);
581 if (allBackups.empty()) return 0;
582
583 std::vector<BackupInfo> matching;
584 for (const auto& b : allBackups) {
585 if (sourceName.empty() || b.sourceName == sourceName) {
586 matching.push_back(b);
587 }
588 }
589
590 if (matching.size() <= maxToKeep) {
591 return 0;
592 }
593
594 size_t prunedCount = 0;
595 for (size_t i = maxToKeep; i < matching.size(); ++i) {
596 if (DeleteBackup(matching[i].backupPath)) {
597 prunedCount++;
598 }
599 }
600
601 LOG_INFO(FileManager, "PruneOldBackups: Purged " + std::to_string(prunedCount) +
602 " obsolete snapshots for '" + (sourceName.empty() ? "All" : sourceName) + "'.");
603 return prunedCount;
604}

606size_t BackupManager::PruneTierBackups(BackupTier tier, const std::string& sourceName, size_t maxToKeep, const std::string& customRoot) {
607 std::string tierDir = GetTierDirectory(tier, customRoot);
608 if (maxToKeep == 0) {
609 // Resolve default retention from settings
610 const auto& settings = SettingsManager::Instance();
611 switch (tier) {
612 case BackupTier::Daily: maxToKeep = static_cast<size_t>(settings.dailyBackupsRetention); break;
613 case BackupTier::Weekly: maxToKeep = static_cast<size_t>(settings.weeklyBackupsRetention); break;
614 case BackupTier::Monthly: maxToKeep = static_cast<size_t>(settings.monthlyBackupsRetention); break;
615 default: maxToKeep = 5; break;
616 }
617 }
618
619 return PruneOldBackups(sourceName, maxToKeep, tierDir);
620}

622void BackupManager::CheckAndRunScheduledBackups(const std::string& libraryPath, const std::string& customRoot) {
623 if (libraryPath.empty() || !FileManager::IsDirectory(libraryPath)) {
624 return;
625 }
626
627 const auto& settings = SettingsManager::Instance();
628 auto now = std::chrono::system_clock::now();
629 uint64_t nowEpochMs = std::chrono::duration_cast<std::chrono::milliseconds>(now.time_since_epoch()).count();
630
631 constexpr uint64_t DAY_MS = 86400000ULL; // 24 * 3600 * 1000
632 constexpr uint64_t WEEK_MS = 7ULL * DAY_MS; // 7 days
633 constexpr uint64_t MONTH_MS = 30ULL * DAY_MS; // 30 days
634
635 // 1. Daily Backup Check
636 if (settings.autoBackupDailyEnabled) {
637 auto dailyBackups = ListTierBackups(BackupTier::Daily, customRoot);
638 bool shouldRunDaily = dailyBackups.empty() || (nowEpochMs - dailyBackups.front().timestamp >= DAY_MS);
639 if (shouldRunDaily) {
640 LOG_INFO(FileManager, "CheckAndRunScheduledBackups: Running scheduled Daily backup.");
641 if (BackupLibraryTiered(libraryPath, BackupTier::Daily, customRoot)) {
642 PruneTierBackups(BackupTier::Daily, "", static_cast<size_t>(settings.dailyBackupsRetention), customRoot);
643 }
644 }
645 }
646
647 // 2. Weekly Backup Check
648 if (settings.autoBackupWeeklyEnabled) {
649 auto weeklyBackups = ListTierBackups(BackupTier::Weekly, customRoot);
650 bool shouldRunWeekly = weeklyBackups.empty() || (nowEpochMs - weeklyBackups.front().timestamp >= WEEK_MS);
651 if (shouldRunWeekly) {
652 LOG_INFO(FileManager, "CheckAndRunScheduledBackups: Running scheduled Weekly backup.");
653 if (BackupLibraryTiered(libraryPath, BackupTier::Weekly, customRoot)) {
654 PruneTierBackups(BackupTier::Weekly, "", static_cast<size_t>(settings.weeklyBackupsRetention), customRoot);
655 }
656 }
657 }
658
659 // 3. Monthly Backup Check
660 if (settings.autoBackupMonthlyEnabled) {
661 auto monthlyBackups = ListTierBackups(BackupTier::Monthly, customRoot);
662 bool shouldRunMonthly = monthlyBackups.empty() || (nowEpochMs - monthlyBackups.front().timestamp >= MONTH_MS);
663 if (shouldRunMonthly) {
664 LOG_INFO(FileManager, "CheckAndRunScheduledBackups: Running scheduled Monthly backup.");
665 if (BackupLibraryTiered(libraryPath, BackupTier::Monthly, customRoot)) {
666 PruneTierBackups(BackupTier::Monthly, "", static_cast<size_t>(settings.monthlyBackupsRetention), customRoot);
667 }
668 }
669 }
670}
