24void CanvasEngine::GetCalculatedPageBoundsMm(double& outW, double& outH) const {
25 double stdW, stdH;

34 double calcH = usedW * aspect;
35 if (contentMaxYMm > calcH) {
36 usedW = contentMaxYMm / aspect;

39 outW = usedW;
