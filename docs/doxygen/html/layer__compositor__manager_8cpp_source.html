2 * =========================================================================================
3 * @file core/layers/layer_compositor_manager.cpp
4 * @brief Implementation of LayerCompositorManager (3-Layer Master Compositor)
5 * =========================================================================================
6 *
7 * GENERAL ARCHITECTURAL FLOW:
8 * ---------------------------
9 * This translation unit orchestrates the frame composition pipeline:
10 * 1. Surface Allocation: Propagates window dimensions to `BakedCanvasLayer` to ensure backing
11 * PRGB32 memory buffers match screen metrics.
12 * 2. Invalidation Routing: Directs full or partial dirty bounding boxes to Layer 1.
13 * 3. Deterministic 3-Pass Frame Assembly:
14 * - Pass 1: BakedCanvasLayer::Render() updates static cache if dirty, then blits cached
15 * PRGB32 surface at (0, 0).
16 * - Pass 2: LiveInteractionLayer::Render() overlays active ephemeral vector graphics
17 * (pen ink, gizmo, marquee box, eraser reticle).
18 * - Pass 3: EmbeddedAppLayer::UpdateAndRender() updates active runtime app frames and controls.
19 */

72/**
73 * @brief Executes multi-tier frame composition into destination context.
74 *
75 * @param[in,out] finalContext Target context for blitting and composite output.
76 * @param[in] activePage The current active canvas page.
77 * @param[in] viewport Camera viewport matrices and frustum bounds.
78 * @param[in] activeAppObjects Running interactive Layer 3 objects.
79 * @param[in] deltaTime Frame delta time in seconds.
80 */
81void LayerCompositorManager::RenderFrame(BLContext& finalContext,
82 CanvasPage* activePage,
83 const Viewport& viewport,
84 const std::unordered_map<uint32_t, const CanvasObject*>& activeAppObjects,
85 double deltaTime) {
86 if (!activePage || m_widthPx <= 0 || m_heightPx <= 0) {
87 return;
88 }
89
90 // --- PASS 1: Layer 1 Baked Canvas (Committed Entities Cache) ---
91 // BakedCanvasLayer acts as a passive raster cache: Render() updates its internal
92 // backing surface (handling full or partial dirty regions), and we composite the
93 // resulting raster directly to the destination frame context at (0, 0).
94 m_bakedLayer.Render(activePage, viewport);
95 const BLImage& bakedSurface = m_bakedLayer.GetSurface();
96 if (!bakedSurface.is_empty()) {
97 finalContext.blit_image(BLPointI(0, 0), bakedSurface);
98 }
99
100 // --- PASS 2: Layer 2 Live Interaction (Ink / Marquee / Gizmo / Text / Eraser) ---
101 if (m_liveLayer.HasActiveInteraction()) {
102 m_liveLayer.Render(finalContext, viewport);
103 }
104
105 // --- PASS 3: Layer 3 Embedded Running Apps ---
106 m_appLayer.UpdateAndRender(viewport, activeAppObjects, deltaTime);
107}
