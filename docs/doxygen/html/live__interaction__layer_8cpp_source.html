1/**
2 * =========================================================================================
3 * @file core/layers/live_interaction_layer.cpp
4 * @brief Implementation of Layer 2 Real-Time Ephemeral Interaction Overlay
5 * =========================================================================================
6 *
7 * ARCHITECTURAL DESIGN & COORDINATE DISCIPLINE:
8 * 1. Zero-Copy Borrowing:
9 * - Uses C++20 `std::span` for zero-allocation access to active tool telemetry.
10 * - Never clones geometry or allocates heap storage during per-frame render ticks.
11 * 2. Coordinate Transformations:
12 * - World Space (mm): In-flight strokes, laser paths, and text editing boundaries are
13 * transformed using `viewport.worldToScreenMatrix` to scale and pan with camera motion.
14 * - Screen Space (px): Gizmo frames, scale handles, and rotation stems are rendered in
15 * fixed-pixel metrics via `viewport.WorldToScreenRect` so that touch/mouse hit targets
16 * remain sharp and constant regardless of zoom level.
17 * 3. State Isolation:
18 * - Every rendering pass wraps its transformation and raster state inside `ctx.save()`
19 * and `ctx.restore()`.
20 */

51void LiveInteractionLayer::Render(BLContext& ctx, const Viewport& viewport) {
52 if (!HasActiveInteraction()) {
53 return;
54 }
55
56 // Pass 1: In-Flight Vector Ink Stroke (World Coordinates)
57 RenderInFlightInk(ctx, viewport);
58
59 // Pass 2: Laser Pointer Glowing Trail (World Coordinates with chronological alpha fade)
60 RenderLaserPointer(ctx, viewport);
61 if (!m_inkEngine.GetEphemeralStrokes().empty()) {
62 ctx.save();
63 ctx.set_transform(viewport.worldToScreenMatrix);
64 uint64_t nowMs = static_cast<uint64_t>(std::chrono::duration_cast<std::chrono::milliseconds>(
65 std::chrono::steady_clock::now().time_since_epoch()).count());
66 m_inkEngine.RenderEphemeralStrokes(ctx, nowMs);
67 ctx.restore();
68 }
69
70 // Pass 3: Selection / Transform Gizmo & Handles (Screen Coordinates for crisp fixed-size handles)
71 RenderGizmo(ctx, viewport);
72
73 // Pass 4: Active Text Editor Caret & Selection Highlights (World Coordinates)
74 RenderTextEditor(ctx, viewport);
75
76 // Pass 5: Marquee Selection Box / Lasso Polygon
77 RenderMarquee(ctx, viewport);
78
79 // Pass 6: Live Eraser Circular Cursor Reticle (Screen Coordinates)
80 RenderEraser(ctx);
81
82 // Pass 7: Custom Tool Callbacks (Alignment snapping lines, custom preview gizmos)
83 if (m_customRenderer) {
84 ctx.save();
85 m_customRenderer(ctx, viewport);
86 ctx.restore();
87 }
88}

90/**
91 * @brief Renders the active in-flight pen stroke in world space.
92 *
93 * MATHEMATICAL PROCESS:
94 * - Priority 1: High-Performance Modeled Ribbon Polygon (Google Ink spring-damper physics).
95 * Fills the 2D extruded polygon contour using non-zero winding rules.
96 * - Priority 2: Fallback for borrowed spans / raw points with round caps & round joins.
97 */
98void LiveInteractionLayer::RenderInFlightInk(BLContext& ctx, const Viewport& viewport) {
99 // 1. Primary Vector Inking Engine: Continuous Modeled Ribbon Polygon (Google Ink & Spring Physics)
100 if (m_inkEngine.IsStrokeActive()) {
101 ctx.save();
102 ctx.set_transform(viewport.worldToScreenMatrix);
103 m_inkEngine.RenderLiveStroke(ctx);
104 ctx.restore();
105 return;
106 }
107
108 // 2. Fallback for Borrowed Tool Spans / Direct LivePoint Ingestion
109 std::span<const LivePoint> pts;
110 BLRgba32 strokeColor;
111 double strokeWidthMm = 0.5;
112
113 if (m_hasBorrowedStroke && m_borrowedStroke.points.size() >= 2) {
114 pts = m_borrowedStroke.points;
115 strokeColor = m_borrowedStroke.color;
116 strokeWidthMm = m_borrowedStroke.baseWidthMm;
117 } else if (m_isInking && m_internalPoints.size() >= 2) {
118 pts = m_internalPoints;
119 strokeColor = BLRgba32(m_penColor);
120 strokeWidthMm = m_penWidthMm;
121 } else {
122 return;
123 }
124
125 ctx.save();
126 ctx.set_transform(viewport.worldToScreenMatrix);
127
128 BLPath path;
129 path.move_to(pts[0].worldX, pts[0].worldY);
130 for (size_t i = 1; i < pts.size(); ++i) {
131 path.line_to(pts[i].worldX, pts[i].worldY);
132 }
133
134 ctx.set_stroke_style(strokeColor);
135 ctx.set_stroke_width(strokeWidthMm);
136 ctx.set_stroke_caps(BL_STROKE_CAP_ROUND);
137 ctx.set_stroke_join(BL_STROKE_JOIN_ROUND);
138 ctx.stroke_path(path);
139
140 ctx.restore();
141}

143/**
144 * @brief Renders transient laser pointer trail with chronological time-decay opacity.
145 *
146 * MATHEMATICAL FADE MODEL:
147 * - For sample at timestamp t_i and current clock T:
148 * age = T - t_i
149 * decayFactor = clamp(1.0 - (age / trailDecayMs), 0.0, 1.0)
150 * alpha = originalAlpha * decayFactor
151 * - Segments older than trailDecayMs are skipped.
152 */
153void LiveInteractionLayer::RenderLaserPointer(BLContext& ctx, const Viewport& viewport) {
154 if (!m_hasActiveLaser || m_activeLaser.points.size() < 2) {
155 return;
156 }
157
158 const auto& pts = m_activeLaser.points;
159 const uint64_t now = m_activeLaser.currentTimeMs;
160 const uint64_t decayMs = m_activeLaser.trailDecayMs > 0 ? m_activeLaser.trailDecayMs : 600;
161 const BLRgba32 baseCol = m_activeLaser.color;
162
163 ctx.save();
164 ctx.set_transform(viewport.worldToScreenMatrix);
165 ctx.set_stroke_caps(BL_STROKE_CAP_ROUND);
166 ctx.set_stroke_join(BL_STROKE_JOIN_ROUND);
167
168 for (size_t i = 0; i + 1 < pts.size(); ++i) {
169 const auto& p0 = pts[i];
170 const auto& p1 = pts[i + 1];
171
172 uint64_t sampleTime = (p0.timestamp + p1.timestamp) / 2;
173 if (now < sampleTime) sampleTime = now;
174 uint64_t age = now - sampleTime;
175 if (age >= decayMs) continue;
176
177 double factor = 1.0 - (static_cast<double>(age) / static_cast<double>(decayMs));
178 factor = (std::clamp)(factor, 0.0, 1.0);
179
180 uint8_t alpha = static_cast<uint8_t>(baseCol.a() * factor);
181 if (alpha == 0) continue;
182
183 BLRgba32 segColor(baseCol.r(), baseCol.g(), baseCol.b(), alpha);
184 ctx.set_stroke_style(segColor);
185 ctx.set_stroke_width(m_activeLaser.radiusMm * 2.0 * factor);
186 ctx.stroke_line(p0.worldX, p0.worldY, p1.worldX, p1.worldY);
187 }
188
189 // Laser tip glow spot at head
190 const auto& head = pts.back();
191 ctx.set_fill_style(baseCol);
192 ctx.fill_circle(head.worldX, head.worldY, m_activeLaser.radiusMm);
193
194 ctx.restore();
195}

197/**
198 * @brief Renders the selection and transform gizmo in crisp SCREEN coordinates.
199 *
200 * GEOMETRIC LAYOUT (Screen Pixels):
201 * - Bounding box frame: `screenBox = viewport.WorldToScreenRect(targetBounds)`.
202 * - 8 Scale handles positioned at the 4 corners and 4 edge midpoints:
203 * Top-Left, Top-Mid, Top-Right, Right-Mid, Bottom-Right, Bottom-Mid, Bottom-Left, Left-Mid.
204 * - Rotation handle: Positioned above Top-Mid by `rotationHandleOffsetPx` with connecting stem.
205 */
206void LiveInteractionLayer::RenderGizmo(BLContext& ctx, const Viewport& viewport) {
207 // 1. Direct Bound Application Gizmo Pass (takes precedence)
208 if (m_boundGizmo && m_boundTransform && m_boundGizmo->HasSelection()) {
209 m_boundGizmo->Render(ctx, *m_boundTransform);
210 return;
211 }
212
213 // 2. Fallback Lightweight Gizmo Pass
214 if (!m_hasActiveGizmo || m_activeGizmo.targetBounds.IsEmpty()) {
215 return;
216 }
217
218 BLRect box = viewport.WorldToScreenRect(m_activeGizmo.targetBounds);
219 if (box.w <= 0.0 || box.h <= 0.0) {
220 return;
221 }
222
223 ctx.save();
224 // Context remains in identity screen space (pixels)
225
226 // 1. Draw outer bounding rectangle
227 ctx.set_stroke_style(m_activeGizmo.outlineColor);
228 ctx.set_stroke_width(m_activeGizmo.borderWidthPx);
229 ctx.stroke_rect(box);
230
231 const double r = m_activeGizmo.handleRadiusPx;
232
233 // 2. Draw 8 scale / resize handles
234 if (m_activeGizmo.showScaleHandles) {
235 const double x0 = box.x;
236 const double x1 = box.x + box.w * 0.5;
237 const double x2 = box.x + box.w;
238 const double y0 = box.y;
239 const double y1 = box.y + box.h * 0.5;
240 const double y2 = box.y + box.h;
241
242 const BLPoint handles[8] = {
243 { x0, y0 }, { x1, y0 }, { x2, y0 }, // Top
244 { x2, y1 }, // Right
245 { x2, y2 }, { x1, y2 }, { x0, y2 }, // Bottom
246 { x0, y1 } // Left
247 };
248
249 for (const auto& pt : handles) {
250 ctx.set_fill_style(m_activeGizmo.handleFillColor);
251 ctx.fill_circle(pt.x, pt.y, r);
252 ctx.set_stroke_style(m_activeGizmo.handleBorderColor);
253 ctx.set_stroke_width(1.2);
254 ctx.stroke_circle(pt.x, pt.y, r);
255 }
256 }
257
258 // 3. Draw rotation handle & connecting stem
259 if (m_activeGizmo.showRotateHandle) {
260 const double midX = box.x + box.w * 0.5;
261 const double topY = box.y;
262 const double rotY = topY - m_activeGizmo.rotationHandleOffsetPx;
263
264 // Stem line
265 ctx.set_stroke_style(m_activeGizmo.outlineColor);
266 ctx.set_stroke_width(1.2);
267 ctx.stroke_line(midX, topY, midX, rotY);
268
269 // Rotation circle handle
270 ctx.set_fill_style(m_activeGizmo.handleFillColor);
271 ctx.fill_circle(midX, rotY, r * 1.1);
272 ctx.set_stroke_style(m_activeGizmo.handleBorderColor);
273 ctx.set_stroke_width(1.5);
274 ctx.stroke_circle(midX, rotY, r * 1.1);
275 }
276
277 ctx.restore();
278}

318void LiveInteractionLayer::RenderMarquee(BLContext& ctx, const Viewport& viewport) {
319 // Check borrowed marquee first
320 if (m_hasActiveMarquee) {
321 // Direct screen-space rectangular marquee (pixel coordinates)
322 if (m_activeMarquee.isScreenRect) {
323 float bx = (std::min)(m_activeMarquee.screenMinX, m_activeMarquee.screenMaxX);
324 float by = (std::min)(m_activeMarquee.screenMinY, m_activeMarquee.screenMaxY);
325 float bw = std::abs(m_activeMarquee.screenMaxX - m_activeMarquee.screenMinX);
326 float bh = std::abs(m_activeMarquee.screenMaxY - m_activeMarquee.screenMinY);
327
328 ctx.save();
329 ctx.set_fill_style(m_activeMarquee.fillColor);
330 ctx.fill_rect(bx, by, bw, bh);
331 ctx.set_stroke_style(m_activeMarquee.strokeColor);
332 ctx.set_stroke_width(m_activeMarquee.strokeWidthPx);
333 ctx.stroke_rect(bx, by, bw, bh);
334 ctx.restore();
335 return;
336 }
337
338 if (m_activeMarquee.isLasso && m_activeMarquee.polygonPoints.size() >= 2) {
339 ctx.save();
340 ctx.set_transform(viewport.worldToScreenMatrix);
341
342 BLPath path;
343 const auto& pts = m_activeMarquee.polygonPoints;
344 path.move_to(pts[0].x, pts[0].y);
345 for (size_t i = 1; i < pts.size(); ++i) {
346 path.line_to(pts[i].x, pts[i].y);
347 }
348
349 ctx.set_stroke_style(m_activeMarquee.strokeColor);
350 ctx.set_stroke_width(0.5);
351 ctx.stroke_path(path);
352 ctx.restore();
353 return;
354 }
355
356 if (!m_activeMarquee.isLasso && !m_activeMarquee.rectBounds.IsEmpty()) {
357 BLRect sRect = viewport.WorldToScreenRect(m_activeMarquee.rectBounds);
358 ctx.save();
359 ctx.set_fill_style(m_activeMarquee.fillColor);
360 ctx.fill_rect(sRect);
361 ctx.set_stroke_style(m_activeMarquee.strokeColor);
362 ctx.set_stroke_width(m_activeMarquee.strokeWidthPx);
363 ctx.stroke_rect(sRect);
364 ctx.restore();
365 return;
366 }
367 }
368
369 // Fallback: internal lasso buffer from direct ingestion
370 if (m_isLassoing && m_internalLassoPoints.size() >= 2) {
371 ctx.save();
372 ctx.set_transform(viewport.worldToScreenMatrix);
373
374 BLPath path;
375 path.move_to(m_internalLassoPoints[0].x, m_internalLassoPoints[0].y);
376 for (size_t i = 1; i < m_internalLassoPoints.size(); ++i) {
377 path.line_to(m_internalLassoPoints[i].x, m_internalLassoPoints[i].y);
378 }
379
380 ctx.set_stroke_style(BLRgba32(0x3B, 0x82, 0xF6, 0xD0)); // Accent blue
381 ctx.set_stroke_width(0.5);
382 ctx.stroke_path(path);
383
384 ctx.restore();
385 }

479std::vector<Segment1D> LiveInteractionLayer::FinalizeStrokeAsSegments(float baseWidthMm) {
480 m_isInking = false;
481 std::vector<LivePoint> pts = std::move(m_internalPoints);
482 m_internalPoints.clear();
483 m_internalPoints.reserve(1024);
484
485 std::vector<Segment1D> segments;
486 if (pts.empty()) return segments;
487
488 float w = baseWidthMm > 0.0f ? baseWidthMm : static_cast<float>(m_penWidthMm);
489 if (w <= 0.0f) w = 0.5f;
490
491 if (pts.size() == 1) {
492 Segment1D seg;
493 seg.p0 = Point2D{ pts[0].worldX, pts[0].worldY };
494 seg.p1 = Point2D{ pts[0].worldX + 0.01, pts[0].worldY };
495 seg.width = w * (std::max)(0.1f, pts[0].pressure);
496 segments.push_back(seg);
497 return segments;
498 }
499
500 segments.reserve(pts.size() - 1);
501 for (size_t i = 0; i + 1 < pts.size(); ++i) {
502 Segment1D seg;
503 seg.p0 = Point2D{ pts[i].worldX, pts[i].worldY };
504 seg.p1 = Point2D{ pts[i + 1].worldX, pts[i + 1].worldY };
505 seg.width = w * (std::max)(0.1f, pts[i + 1].pressure);
506 segments.push_back(seg);
507 }
508 return segments;
509}
