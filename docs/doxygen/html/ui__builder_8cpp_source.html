61bool ButtonBuilder::Render() {
62 ImGuiWindow* window = ImGui::GetCurrentWindow();
63 if (window->SkipItems) return false;
64
65 ImGuiContext& g = *GImGui;
66 const ImGuiStyle& style = g.Style;
67 const ImGuiID id = window->GetID(label_.c_str());
68
69 // 1. Height and padding based on Size tier
70 float height = 34.0f;
71 float padX = Spacing::md;
72 float fontSize = ImGui::GetFontSize();
73 float rounding = Radii::sm;
74
75 switch (size_) {
76 case UIButtonSize::Small:
77 height = 26.0f;
78 padX = Spacing::sm;
79 rounding = Radii::xs;
80 break;
81 case UIButtonSize::Large:
82 height = 44.0f;
83 padX = Spacing::lg;
84 rounding = Radii::md;
85 break;
86 default:
87 height = 34.0f;
88 padX = Spacing::md;
89 rounding = Radii::sm;
90 break;
91 }
92
93 // 2. Measure content width
94 ImVec2 labelSize = ImGui::CalcTextSize(label_.c_str(), nullptr, true);
95 float contentW = labelSize.x;
96 if (!iconText_.empty()) {
97 contentW += ImGui::CalcTextSize(iconText_.c_str()).x + Spacing::sm;
98 }
99 if (!badgeText_.empty()) {
100 contentW += ImGui::CalcTextSize(badgeText_.c_str()).x + Spacing::md;
101 }
102
103 float totalW = (customWidth_ > 0.0f) ? customWidth_ : (contentW + padX * 2.0f);
104 ImVec2 size(totalW, height);
105
106 ImVec2 pos = window->DC.CursorPos;
107 ImRect bb(pos, ImVec2(pos.x + size.x, pos.y + size.y));
108 ImGui::ItemSize(size, style.FramePadding.y);
109 if (!ImGui::ItemAdd(bb, id)) {
110 return false;
111 }
112
113 // 3. Interaction evaluation
114 bool hovered = false;
115 bool held = false;
116 bool pressed = false;
117 if (!isDisabled_) {
118 pressed = ImGui::ButtonBehavior(bb, id, &hovered, &held);
119 }
120
121 // 4. Smooth animation transition alphas
122 float hoverAlpha = 0.0f;
123 float activeAlpha = 0.0f;
124 UIAnimationManager::Instance().GetInteractionAlphas(id, hovered || isSelected_, held, hoverAlpha, activeAlpha);
125
126 // 5. Theme color resolution
127 const auto& theme = ThemeManager::Instance();
128 bool isDark = (theme.colorBg.x < 0.5f);
129
130 ImU32 baseFill = 0;
131 ImU32 hoverFill = 0;
132 ImU32 baseBorder = 0;
133 ImU32 hoverBorder = 0;
134 ImU32 textCol = isDark ? IM_COL32(235, 238, 245, 255) : IM_COL32(25, 28, 35, 255);
135
136 // Signature warm accent
137 ImU32 accentCol = IM_COL32(242, 115, 38, 255);
138 ImU32 accentHover = IM_COL32(250, 135, 60, 255);
139
140 switch (variant_) {
141 case UIButtonVariant::Primary:
142 baseFill = accentCol;
143 hoverFill = accentHover;
144 baseBorder = IM_COL32(0, 0, 0, 0);
145 hoverBorder = IM_COL32(255, 255, 255, 60);
146 textCol = IM_COL32(255, 255, 255, 255);
147 break;
148 case UIButtonVariant::Ghost:
149 baseFill = IM_COL32(0, 0, 0, 0);
150 hoverFill = isDark ? IM_COL32(255, 255, 255, 22) : IM_COL32(0, 0, 0, 16);
151 baseBorder = IM_COL32(0, 0, 0, 0);
152 hoverBorder = isDark ? IM_COL32(255, 255, 255, 35) : IM_COL32(0, 0, 0, 25);
153 break;
154 case UIButtonVariant::Danger:
155 baseFill = isDark ? IM_COL32(180, 45, 45, 180) : IM_COL32(220, 60, 60, 200);
156 hoverFill = isDark ? IM_COL32(210, 55, 55, 240) : IM_COL32(240, 70, 70, 255);
157 baseBorder = IM_COL32(255, 80, 80, 80);
158 hoverBorder = IM_COL32(255, 100, 100, 160);
159 textCol = IM_COL32(255, 255, 255, 255);
160 break;
161 case UIButtonVariant::Glassmorphic:
162 baseFill = isDark ? IM_COL32(32, 36, 46, 175) : IM_COL32(255, 255, 255, 190);
163 hoverFill = isDark ? IM_COL32(48, 54, 68, 215) : IM_COL32(255, 255, 255, 240);
164 baseBorder = isDark ? IM_COL32(80, 88, 105, 140) : IM_COL32(200, 205, 218, 180);
165 hoverBorder = isDark ? IM_COL32(120, 132, 155, 200) : IM_COL32(160, 170, 190, 220);
166 break;
167 case UIButtonVariant::Secondary:
168 default:
169 baseFill = isDark ? IM_COL32(38, 42, 53, 230) : IM_COL32(245, 247, 250, 240);
170 hoverFill = isDark ? IM_COL32(50, 56, 70, 255) : IM_COL32(236, 240, 246, 255);
171 baseBorder = isDark ? IM_COL32(65, 72, 88, 180) : IM_COL32(215, 220, 230, 200);
172 hoverBorder = isDark ? IM_COL32(95, 105, 128, 220) : IM_COL32(175, 185, 200, 240);
173 break;
174 }
175
176 if (isSelected_ && variant_ != UIButtonVariant::Primary) {
177 baseBorder = accentCol;
178 hoverBorder = accentHover;
179 }
180
181 if (isDisabled_) {
182 textCol = ColorUtils::WithAlpha(textCol, 0.45f);
183 baseFill = ColorUtils::WithAlpha(baseFill, 0.5f);
184 hoverFill = baseFill;
185 }
186
187 // 6. Draw surface with subtle scale press effect
188 ImDrawList* drawList = window->DrawList;
189 ImVec2 bMin = bb.Min;
190 ImVec2 bMax = bb.Max;
191
192 if (activeAlpha > 0.0f) {
193 float scaleInset = 1.0f * activeAlpha;
194 bMin.x += scaleInset;
195 bMin.y += scaleInset;
196 bMax.x -= scaleInset;
197 bMax.y -= scaleInset;
198 }
199
200 if (variant_ == UIButtonVariant::Glassmorphic) {
201 UIShapeManager::DrawAcrylicPanel(drawList, bMin, bMax, ColorUtils::Lerp(baseFill, hoverFill, hoverAlpha),
202 ColorUtils::Lerp(baseBorder, hoverBorder, hoverAlpha), rounding, Elevation::low);
203 } else {
204 UIShapeManager::DrawInteractiveCard(drawList, bMin, bMax, baseFill, hoverFill, baseBorder, hoverBorder, hoverAlpha, rounding);
205 }
206
207 // 7. Render centered text and icons
208 float startX = bMin.x + (bMax.x - bMin.x - contentW) * 0.5f;
209 float startY = bMin.y + (bMax.y - bMin.y - fontSize) * 0.5f;
210
211 if (!iconText_.empty()) {
212 drawList->AddText(ImVec2(startX, startY), textCol, iconText_.c_str());
213 startX += ImGui::CalcTextSize(iconText_.c_str()).x + Spacing::sm;
214 }
215
216 drawList->AddText(ImVec2(startX, startY), textCol, label_.c_str());
217 startX += labelSize.x;
218
219 if (!badgeText_.empty()) {
220 startX += Spacing::sm;
221 ImU32 badgeBg = isDark ? IM_COL32(255, 255, 255, 28) : IM_COL32(0, 0, 0, 20);
222 ImU32 badgeTextCol = isDark ? IM_COL32(200, 205, 215, 220) : IM_COL32(100, 105, 118, 220);
223 UIShapeManager::DrawPillBadge(drawList, ImVec2(startX + 18.0f, bMin.y + height * 0.5f), badgeText_.c_str(), badgeBg, badgeTextCol);
224 }
225
226 // 8. Tooltip
227 if (!tooltip_.empty() && hovered) {
228 ImGui::SetTooltip("%s", tooltip_.c_str());
229 }
230
231 return pressed;
232}

251bool SwitchBuilder::Render() {
252 ImGuiWindow* window = ImGui::GetCurrentWindow();
253 if (window->SkipItems || !value_) return false;
254
255 ImGuiContext& g = *GImGui;
256 const ImGuiStyle& style = g.Style;
257 const ImGuiID id = window->GetID(label_.c_str());
258
259 float trackW = 42.0f;
260 float trackH = 22.0f;
261 float knobRadius = 8.5f;
262
263 ImVec2 labelSize = ImGui::CalcTextSize(label_.c_str());
264 float rowW = ImGui::GetContentRegionAvail().x;
265 float rowH = std::max(trackH, description_.empty() ? labelSize.y : (labelSize.y + 18.0f));
266
267 ImVec2 pos = window->DC.CursorPos;
268 ImRect bb(pos, ImVec2(pos.x + rowW, pos.y + rowH));
269 ImGui::ItemSize(bb, style.FramePadding.y);
270 if (!ImGui::ItemAdd(bb, id)) return false;
271
272 bool hovered = false;
273 bool held = false;
274 bool clicked = false;
275 if (!isDisabled_) {
276 clicked = ImGui::ButtonBehavior(bb, id, &hovered, &held);
277 if (clicked) {
278 *value_ = !(*value_);
279 }
280 }
281
282 // Animated knob sliding parameter [0.0 = OFF, 1.0 = ON]
283 std::string animKey = "switch_" + std::to_string(id);
284 float targetT = (*value_) ? 1.0f : 0.0f;
285 float t = UIAnimationManager::Instance().GetFloat(animKey, targetT, 16.0f);
286
287 const auto& theme = ThemeManager::Instance();
288 bool isDark = (theme.colorBg.x < 0.5f);
289
290 // Track colors
291 ImU32 offTrack = isDark ? IM_COL32(55, 60, 75, 220) : IM_COL32(205, 212, 225, 230);
292 ImU32 onTrack = IM_COL32(242, 115, 38, 255); // Folio accent orange
293 ImU32 currentTrack = ColorUtils::Lerp(offTrack, onTrack, t);
294
295 // Track bounds (anchored to the right)
296 ImVec2 trackMin(bb.Max.x - trackW - Spacing::sm, bb.Min.y + (rowH - trackH) * 0.5f);
297 ImVec2 trackMax(trackMin.x + trackW, trackMin.y + trackH);
298
299 ImDrawList* drawList = window->DrawList;
300 drawList->AddRectFilled(trackMin, trackMax, currentTrack, trackH * 0.5f);
301
302 // Sliding knob position
303 float knobMinX = trackMin.x + knobRadius + 2.5f;
304 float knobMaxX = trackMax.x - knobRadius - 2.5f;
305 float knobX = knobMinX + (knobMaxX - knobMinX) * t;
306 float knobY = trackMin.y + trackH * 0.5f;
307
308 // Knob drop shadow & circle
309 drawList->AddCircleFilled(ImVec2(knobX, knobY + 1.0f), knobRadius, IM_COL32(0, 0, 0, 45), 18);
310 drawList->AddCircleFilled(ImVec2(knobX, knobY), knobRadius, IM_COL32(255, 255, 255, 255), 18);
311
312 // Text labels (on the left)
313 ImU32 titleCol = isDark ? IM_COL32(235, 238, 245, 255) : IM_COL32(25, 28, 35, 255);
314 drawList->AddText(ImVec2(bb.Min.x, bb.Min.y + (description_.empty() ? (rowH - labelSize.y) * 0.5f : 0.0f)), titleCol, label_.c_str());
315
316 if (!description_.empty()) {
317 ImU32 descCol = isDark ? IM_COL32(150, 158, 175, 220) : IM_COL32(115, 122, 138, 220);
318 drawList->AddText(ImVec2(bb.Min.x, bb.Min.y + labelSize.y + 2.0f), descCol, description_.c_str());
319 }
320
321 return clicked;
322}

339bool SegmentedSwitchBuilder::Render() {
340 ImGuiWindow* window = ImGui::GetCurrentWindow();
341 if (window->SkipItems || options_.empty() || !selectedIndex_) return false;
342
343 ImGuiContext& g = *GImGui;
344 const ImGuiStyle& style = g.Style;
345 const ImGuiID widgetId = window->GetID(id_.c_str());
346
347 float height = 32.0f;
348 float totalW = (customWidth_ > 0.0f) ? customWidth_ : ImGui::GetContentRegionAvail().x;
349 float segW = totalW / static_cast<float>(options_.size());
350
351 ImVec2 pos = window->DC.CursorPos;
352 ImRect bb(pos, ImVec2(pos.x + totalW, pos.y + height));
353 ImGui::ItemSize(bb, style.FramePadding.y);
354 if (!ImGui::ItemAdd(bb, widgetId)) return false;
355
356 const auto& theme = ThemeManager::Instance();
357 bool isDark = (theme.colorBg.x < 0.5f);
358
359 // Background container
360 ImDrawList* drawList = window->DrawList;
361 ImU32 bgFill = isDark ? IM_COL32(28, 32, 42, 230) : IM_COL32(235, 238, 245, 240);
362 drawList->AddRectFilled(bb.Min, bb.Max, bgFill, Radii::sm);
363
364 // Smooth sliding highlight pill
365 std::string pillKey = "seg_pill_" + id_;
366 float targetPillX = bb.Min.x + static_cast<float>(*selectedIndex_) * segW;
367 float pillX = UIAnimationManager::Instance().GetFloat(pillKey, targetPillX, 18.0f);
368
369 float pad = 2.0f;
370 ImVec2 pillMin(pillX + pad, bb.Min.y + pad);
371 ImVec2 pillMax(pillX + segW - pad, bb.Max.y - pad);
372
373 ImU32 pillFill = isDark ? IM_COL32(50, 56, 72, 255) : IM_COL32(255, 255, 255, 255);
374 drawList->AddRectFilled(pillMin, pillMax, pillFill, Radii::xs);
375 drawList->AddRect(pillMin, pillMax, IM_COL32(255, 255, 255, 20), Radii::xs);
376
377 // Item evaluation and rendering
378 bool changed = false;
379 for (size_t i = 0; i < options_.size(); ++i) {
380 ImVec2 segMin(bb.Min.x + static_cast<float>(i) * segW, bb.Min.y);
381 ImVec2 segMax(segMin.x + segW, bb.Max.y);
382 ImGuiID segId = window->GetID((id_ + "_" + std::to_string(i)).c_str());
383
384 bool hovered = false;
385 bool held = false;
386 if (ImGui::ButtonBehavior(ImRect(segMin, segMax), segId, &hovered, &held)) {
387 if (*selectedIndex_ != static_cast<int>(i)) {
388 *selectedIndex_ = static_cast<int>(i);
389 changed = true;
390 }
391 }
392
393 // Text rendering
394 ImVec2 optSize = ImGui::CalcTextSize(options_[i].c_str());
395 ImVec2 textPos(segMin.x + (segW - optSize.x) * 0.5f, segMin.y + (height - optSize.y) * 0.5f);
396
397 bool isSelected = (*selectedIndex_ == static_cast<int>(i));
398 ImU32 textCol = isSelected
399 ? (isDark ? IM_COL32(255, 255, 255, 255) : IM_COL32(15, 18, 24, 255))
400 : (isDark ? IM_COL32(160, 168, 185, 210) : IM_COL32(110, 118, 132, 210));
401
402 drawList->AddText(textPos, textCol, options_[i].c_str());
403 }
404
405 return changed;
406}
