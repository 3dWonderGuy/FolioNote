17void UIAnimationManager::Update(float deltaTime) {
18 // Clamp delta time to avoid physics explosion during debugging pauses or window freezes
19 currentDeltaTime_ = std::clamp(deltaTime, 0.001f, 0.1f);
20 currentFrameCount_++;
21
22 // Prune stale widget interaction states older than 300 frames (~2.5 seconds @ 120Hz)
23 if ((currentFrameCount_ % 120) == 0) {
24 for (auto it = widgetStates_.begin(); it != widgetStates_.end(); ) {
25 if (currentFrameCount_ - it->second.lastFrameActive > 300) {
26 it = widgetStates_.erase(it);
27 } else {
28 ++it;
29 }
30 }
31 }
32}

34float UIAnimationManager::GetFloat(const std::string& key, float target, float speed) {
35 auto& tw = tweens_[key];
36 if (!tw.initialized) {
37 tw.current = target;
38 tw.target = target;
39 tw.initialized = true;
40 return target;
41 }
42
43 tw.target = target;
44
45 // Frame-rate independent exponential decay:
46 // x(t + dt) = target + (current - target) * exp(-speed * dt)
47 float decayFactor = std::exp(-speed * currentDeltaTime_);
48 tw.current = target + (tw.current - target) * decayFactor;
49
50 // Snap if extremely close to prevent continuous micro-updates
51 if (std::abs(tw.current - target) < 0.0005f) {
52 tw.current = target;
53 }
54
55 return tw.current;
56}

58float UIAnimationManager::GetSpring(const std::string& key, float target, float stiffness, float damping) {
59 auto& sp = springs_[key];
60 if (!sp.initialized) {
61 sp.position = target;
62 sp.velocity = 0.0f;
63 sp.target = target;
64 sp.initialized = true;
65 return target;
66 }
67
68 sp.target = target;
69
70 // Sub-stepped semi-implicit Euler integration for numerical stability
71 // 2 sub-steps ensure stability even if frame rate drops to 30-60 FPS
72 const int subSteps = 2;
73 float dt = currentDeltaTime_ / static_cast<float>(subSteps);
74
75 for (int i = 0; i < subSteps; ++i) {
76 float displacement = sp.position - sp.target;
77 // Hooke's Law with damping: F = -k * x - c * v
78 float springForce = -stiffness * displacement;
79 float dampingForce = -damping * sp.velocity;
80 float acceleration = springForce + dampingForce; // Assuming mass m = 1.0 kg
81
82 sp.velocity += acceleration * dt;
83 sp.position += sp.velocity * dt;
84 }
85
86 // Snap to rest when displacement and velocity are below perceptual threshold
87 if (std::abs(sp.position - sp.target) < 0.001f && std::abs(sp.velocity) < 0.01f) {
88 sp.position = sp.target;
89 sp.velocity = 0.0f;
90 }
91
92 return sp.position;
93}

95void UIAnimationManager::GetInteractionAlphas(ImGuiID id, bool isHovered, bool isActive, float& outHoverAlpha, float& outActiveAlpha) {
96 auto& state = widgetStates_[id];
97 state.lastFrameActive = currentFrameCount_;
98
99 // Asymmetric hover speed: fade in quickly (14/s), fade out slightly slower (9/s)
100 float hoverSpeed = isHovered ? 14.0f : 9.0f;
101 float targetHover = isHovered ? 1.0f : 0.0f;
102 float hoverDecay = std::exp(-hoverSpeed * currentDeltaTime_);
103 state.hoverAlpha = targetHover + (state.hoverAlpha - targetHover) * hoverDecay;
104
105 // Asymmetric active speed: press in instantly (26/s), release smoothly (12/s)
106 float activeSpeed = isActive ? 26.0f : 12.0f;
107 float targetActive = isActive ? 1.0f : 0.0f;
108 float activeDecay = std::exp(-activeSpeed * currentDeltaTime_);
109 state.activeAlpha = targetActive + (state.activeAlpha - targetActive) * activeDecay;
110
111 // Clamping to guaranteed [0.0, 1.0] range
112 outHoverAlpha = std::clamp(state.hoverAlpha, 0.0f, 1.0f);
113 outActiveAlpha = std::clamp(state.activeAlpha, 0.0f, 1.0f);
114}

128float UIAnimationManager::ApplyEasing(float t, EasingType type) noexcept {
129 t = std::clamp(t, 0.0f, 1.0f);
130 switch (type) {
131 case EasingType::Linear:
132 return t;
133 case EasingType::EaseInQuad:
134 return t * t;
135 case EasingType::EaseOutQuad:
136 return 1.0f - (1.0f - t) * (1.0f - t);
137 case EasingType::EaseInOutCubic:
138 return (t < 0.5f) ? (4.0f * t * t * t) : (1.0f - std::pow(-2.0f * t + 2.0f, 3.0f) * 0.5f);
139 case EasingType::EaseOutBack: {
140 constexpr float c1 = 1.70158f;
141 constexpr float c3 = c1 + 1.0f;
142 return 1.0f + c3 * std::pow(t - 1.0f, 3.0f) + c1 * std::pow(t - 1.0f, 2.0f);
143 }
144 default:
145 return t;
146 }
147}
