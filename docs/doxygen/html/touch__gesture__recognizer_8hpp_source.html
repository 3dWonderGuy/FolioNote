1/**
2 * @file touch_gesture_recognizer.hpp
3 * @brief Translates raw multi-touch contact telemetry into high-level UI gestures.
4 *
5 * GESTURE RECOGNITION ARCHITECTURE:
6 * ---------------------------------
7 * Glass touchscreens report discrete contact points across consecutive frames.
8 * The recognizer classifies these contacts into distinct semantic gestures:
9 * 1. Tap: Single finger contact released within HOLD_TIME_MS without moving beyond DRAG_THRESHOLD_PX.
10 * 2. Press & Hold: Single finger dwelling in place >= HOLD_TIME_MS within DRAG_THRESHOLD_PX.
11 * 3. Single Finger Scroll / Pan: Single finger moving beyond DRAG_THRESHOLD_PX.
12 * 4. Two-Finger Pinch-Pan: Two fingers moving; centroid translation drives pan, Euclidean distance ratio drives zoom.
13 * 5. Two-Finger Tap: Quick two-finger tap released within MULTI_TAP_MAX_MS (Digital ink shortcut for Undo).
14 * 6. Three-Finger Tap: Quick three-finger tap released within MULTI_TAP_MAX_MS (Digital ink shortcut for Redo).
15 * 7. Gizmo Transform: Route single-finger drag directly to selection handles when active.
16 */
17
18#pragma once
19#include <array>
20#include <cmath>
21#include <cstdint>
22#include <algorithm>

25// The different types of gestures recognized from capacitive multi-touch contacts.
26enum class TouchGestureType : uint8_t {
27 None = 0,
28 Tap,
29 PressAndHold,
30 SingleFingerScroll,
31 GizmoTransform,
32 TwoFingerPinchPan,
33 TwoFingerTap, ///< Quick 2-finger tap (Undo)
34 ThreeFingerTap ///< Quick 3-finger tap (Redo)
35};
36
37// Payload sent out whenever a touch gesture is triggered or updated.
38struct TouchGestureEvent {
39 TouchGestureType type = TouchGestureType::None;
40 float focusX = 0.0f; ///< Focal point X coordinate (tap spot or centroid between fingers)
41 float focusY = 0.0f; ///< Focal point Y coordinate
42 float deltaX = 0.0f; ///< Horizontal displacement delta (for panning/scrolling)
43 float deltaY = 0.0f; ///< Vertical displacement delta
44 float zoomDelta = 1.0f; ///< Scale factor for pinch-to-zoom (1.0 = no change, >1 = zoom in, <1 = zoom out)
45};

47// Translates raw finger contacts on glass into clean UI gestures.
48class TouchGestureRecognizer {
49private:
50 // Tracking variables for single-finger touches
51 float startX = 0.0f;
52 float startY = 0.0f;
53 float prevX = 0.0f;
54 float prevY = 0.0f;
55 uint64_t touchStartTimeMs = 0;
56
57 // Status flags to track what the user is currently performing
58 bool isScrolling = false;
59 bool isHolding = false;
60 bool isPinching = false;
61
62 // Tracking variables for two-finger gestures (pinch & pan)
63 float prevPinchDist = 0.0f;
64 float prevMidX = 0.0f;
65 float prevMidY = 0.0f;
66 uint64_t twoFingerStartTimeMs = 0;
67 bool twoFingerMoved = false;
68
69 // Tracking variables for three-finger gestures
70 uint64_t threeFingerStartTimeMs = 0;
71
72 // Remembers how many fingers were on screen during the previous frame
73 size_t lastActiveCount = 0;
74
75 // Spatial slop threshold in pixels
76 static constexpr float DRAG_THRESHOLD_PX = 8.0f;
77
78 // Minimum dwell duration in ms for Press and Hold
79 static constexpr uint64_t HOLD_TIME_MS = 450;
80
81 // Maximum duration in ms for multi-finger tap gestures (Undo/Redo)
82 static constexpr uint64_t MULTI_TAP_MAX_MS = 250;
83
84public:
85 /**
86 * @brief Evaluates the current state of active touch contacts and yields a gesture event.
87 *
88 * @param fingers Array of tracked touch slots.
89 * @param activeCount Number of currently engaged touch slots (fingerID != -1).
90 * @param nowMs Current timestamp in milliseconds.
91 * @param isInteractingWithGizmo True if user is dragging an active selection gizmo handle.
92 * @return TouchGestureEvent containing the resolved gesture type and transform deltas.
93 */
94 TouchGestureEvent Evaluate(const std::array<TouchSlot, 10>& fingers, size_t activeCount, uint64_t nowMs, bool isInteractingWithGizmo) {
95 TouchGestureEvent gesture;
96
97 // ---------------------------------------------------------------------
98 // 0 FINGERS: ALL TOUCHES RELEASED
99 // ---------------------------------------------------------------------
100 // Evaluate completion of tap gestures (Single Tap, Two-Finger Tap, Three-Finger Tap)
101 if (activeCount == 0) {
102 // Case A: Two-Finger Tap (Undo)
103 if (lastActiveCount == 2 && (nowMs - twoFingerStartTimeMs <= MULTI_TAP_MAX_MS) && !twoFingerMoved && twoFingerStartTimeMs > 0) {
104 gesture.type = TouchGestureType::TwoFingerTap;
105 gesture.focusX = prevMidX;
106 gesture.focusY = prevMidY;
107 }
108 // Case B: Three-Finger Tap (Redo)
109 else if (lastActiveCount == 3 && (nowMs - threeFingerStartTimeMs <= MULTI_TAP_MAX_MS) && threeFingerStartTimeMs > 0) {
110 gesture.type = TouchGestureType::ThreeFingerTap;
111 gesture.focusX = prevMidX;
112 gesture.focusY = prevMidY;
113 }
114 // Case C: Single Finger Tap
115 else if (!isScrolling && !isHolding && !isPinching && (nowMs - touchStartTimeMs < HOLD_TIME_MS) && touchStartTimeMs > 0) {
116 gesture.type = TouchGestureType::Tap;
117 gesture.focusX = startX;
118 gesture.focusY = startY;
119 }
120
121 // Wipe clean internal tracking state
122 isScrolling = false;
123 isHolding = false;
124 isPinching = false;
125 prevPinchDist = 0.0f;
126 prevMidX = 0.0f;
127 prevMidY = 0.0f;
128 touchStartTimeMs = 0;
129 twoFingerStartTimeMs = 0;
130 threeFingerStartTimeMs = 0;
131 twoFingerMoved = false;
132 lastActiveCount = 0;
133 return gesture;
134 }
135
136 // ---------------------------------------------------------------------
137 // 1 FINGER: SCROLLING, GIZMO DRAGGING, TAP PREPARATION, OR PRESS & HOLD
138 // ---------------------------------------------------------------------
139 if (activeCount == 1) {
140 size_t idx = 0;
141 while (idx < fingers.size() && fingers[idx].fingerID == -1) idx++;
142 if (idx >= fingers.size()) {
143 lastActiveCount = 0;
144 return gesture;
145 }
146
147 const auto& pt = fingers[idx].point;
148
149 // Handle transition from multi-touch down to single touch
150 if (lastActiveCount > 1 || touchStartTimeMs == 0) {
151 touchStartTimeMs = nowMs;
152 startX = pt.x;
153 startY = pt.y;
154 prevX = pt.x;
155 prevY = pt.y;
156 isPinching = false;
157 isHolding = false;
158 lastActiveCount = 1;
159 return gesture;
160 }
161
162 lastActiveCount = 1;
163
164 // Total Euclidean distance moved from where the finger first touched down:
165 // D = sqrt((x - x0)^2 + (y - y0)^2)
166 float totalDist = std::hypot(pt.x - startX, pt.y - startY);
167
168 // Movement delta since the previous frame
169 float dx = pt.x - prevX;
170 float dy = pt.y - prevY;
171 prevX = pt.x;
172 prevY = pt.y;
173
174 // Option 1: Gizmo Dragging
175 if (isInteractingWithGizmo) {
176 gesture.type = TouchGestureType::GizmoTransform;
177 gesture.focusX = pt.x;
178 gesture.focusY = pt.y;
179 gesture.deltaX = dx;
180 gesture.deltaY = dy;
181 return gesture;
182 }
183
184 // Option 2: Press and Hold Trigger
185 // If contact remains stationary within DRAG_THRESHOLD_PX for >= HOLD_TIME_MS:
186 if (!isScrolling && totalDist < DRAG_THRESHOLD_PX) {
187 if ((nowMs - touchStartTimeMs >= HOLD_TIME_MS) && !isHolding) {
188 isHolding = true; // Prevents firing repeatedly every frame
189 gesture.type = TouchGestureType::PressAndHold;
190 gesture.focusX = pt.x;
191 gesture.focusY = pt.y;
192 return gesture;
193 }
194 }
195
196 // Option 3: Single Finger Scroll / Pan
197 // Moving beyond the slop threshold initiates scrolling
198 if (totalDist >= DRAG_THRESHOLD_PX || isScrolling) {
199 isScrolling = true;
200 gesture.type = TouchGestureType::SingleFingerScroll;
201 gesture.focusX = pt.x;
202 gesture.focusY = pt.y;
203 gesture.deltaX = dx;
204 gesture.deltaY = dy;
205 return gesture;
206 }
207
208 return gesture;
209 }
210
211 // ---------------------------------------------------------------------
212 // 2 FINGERS: PINCH-TO-ZOOM & TWO-FINGER PANNING
213 // ---------------------------------------------------------------------
214 if (activeCount == 2) {
215 int idx1 = -1, idx2 = -1;
216 for (size_t i = 0; i < fingers.size(); ++i) {
217 if (fingers[i].fingerID != -1) {
218 if (idx1 == -1) idx1 = static_cast<int>(i);
219 else if (idx2 == -1) { idx2 = static_cast<int>(i); break; }
220 }
221 }
222
223 if (idx1 != -1 && idx2 != -1) {
224 const auto& p1 = fingers[idx1].point;
225 const auto& p2 = fingers[idx2].point;
226
227 // Midpoint (centroid) between both fingers:
228 // M = ((p1.x + p2.x) / 2, (p1.y + p2.y) / 2)
229 float midX = (p1.x + p2.x) * 0.5f;
230 float midY = (p1.y + p2.y) * 0.5f;
231
232 // Straight-line Euclidean distance between both fingers:
233 // dist = sqrt((p2.x - p1.x)^2 + (p2.y - p1.y)^2)
234 float dist = std::hypot(p2.x - p1.x, p2.y - p1.y);
235
236 // Initial frame for 2-finger contact
237 if (!isPinching || lastActiveCount != 2) {
238 isPinching = true;
239 prevPinchDist = dist;
240 prevMidX = midX;
241 prevMidY = midY;
242 twoFingerStartTimeMs = nowMs;
243 twoFingerMoved = false;
244 } else {
245 // Check if movement exceeded tap slop
246 if (std::abs(dist - prevPinchDist) > DRAG_THRESHOLD_PX ||
247 std::hypot(midX - prevMidX, midY - prevMidY) > DRAG_THRESHOLD_PX) {
248 twoFingerMoved = true;
249 }
250
251 // Continuous 2-finger pan & zoom updates
252 gesture.type = TouchGestureType::TwoFingerPinchPan;
253 gesture.focusX = midX;
254 gesture.focusY = midY;
255
256 // Centroid translation for canvas panning
257 gesture.deltaX = midX - prevMidX;
258 gesture.deltaY = midY - prevMidY;
259
260 // Distance ratio for zoom factor:
261 // zoomDelta = dist_current / dist_previous
262 gesture.zoomDelta = (prevPinchDist > 0.001f) ? (dist / prevPinchDist) : 1.0f;
263
264 prevPinchDist = dist;
265 prevMidX = midX;
266 prevMidY = midY;
267 }
268 }
269
270 lastActiveCount = 2;
271 return gesture;
272 }
273
274 // ---------------------------------------------------------------------
275 // 3 FINGERS: THREE-FINGER GESTURES (e.g. Redo tap tracking)
276 // ---------------------------------------------------------------------
277 if (activeCount == 3) {
278 if (lastActiveCount != 3) {
279 threeFingerStartTimeMs = nowMs;
280 }
281 lastActiveCount = 3;
282 return gesture;
283 }
284
285 lastActiveCount = activeCount;
286 return gesture;
287 }
288};
