60 virtual void OnEnd(ActionEndReason reason, ActionContext& ctx) { (void)reason; (void)ctx; }

135 * @class SequentialAction
136 * @brief Executes a series of child actions sequentially, advancing to the next when the current finishes.
137 */
138class SequentialAction : public ICanvasAction {
139private:
140 std::vector<std::unique_ptr<ICanvasAction>> actions;
141 size_t currentIndex{0};
142 bool startedCurrent{false};
143 uint32_t aggregatedResources{0};
144
145public:
146 SequentialAction() = default;
147
148 explicit SequentialAction(std::vector<std::unique_ptr<ICanvasAction>> sequence) {
149 for (auto& act : sequence) {
150 AddAction(std::move(act));
151 }
152 }
153
154 void AddAction(std::unique_ptr<ICanvasAction> action) {
155 if (action) {
156 aggregatedResources |= action->GetRequiredResources();
157 actions.push_back(std::move(action));
158 }
159 }
160
161 [[nodiscard]] std::string_view GetName() const noexcept override { return "SequentialAction"; }
162 [[nodiscard]] uint32_t GetRequiredResources() const noexcept override { return aggregatedResources; }
163
164 void OnStart(ActionContext& ctx) override {
165 currentIndex = 0;
166 startedCurrent = false;
167 if (!actions.empty()) {
168 actions[currentIndex]->OnStart(ctx);
169 startedCurrent = true;
170 }
171 }
172
173 void OnUpdate(double dt, ActionContext& ctx) override {
174 while (currentIndex < actions.size()) {
175 auto& current = actions[currentIndex];
176 if (!startedCurrent) {
177 current->OnStart(ctx);
178 startedCurrent = true;
179 }
180
181 current->OnUpdate(dt, ctx);
182
183 if (current->IsFinished(ctx)) {
184 current->OnEnd(ActionEndReason::Finished, ctx);
185 ++currentIndex;
186 startedCurrent = false;
187 } else {
188 break; // Still executing current action
189 }
190 }
191 }
192
193 [[nodiscard]] bool IsFinished(ActionContext& ctx) const override {
194 (void)ctx;
195 return currentIndex >= actions.size();
196 }
197
198 void OnEnd(ActionEndReason reason, ActionContext& ctx) override {
199 if (reason != ActionEndReason::Finished && currentIndex < actions.size()) {
200 if (startedCurrent) {
201 actions[currentIndex]->OnEnd(reason, ctx);
202 }
203 }
204 }
205};

207/**
208 * @class ParallelAction
209 * @brief Executes multiple child actions concurrently until all have completed.
210 */
211class ParallelAction : public ICanvasAction {
212private:
213 struct ChildState {
214 std::unique_ptr<ICanvasAction> action;
215 bool finished{false};
216 };
217
218 std::vector<ChildState> children;
219 uint32_t aggregatedResources{0};
220
221public:
222 ParallelAction() = default;
223
224 void AddAction(std::unique_ptr<ICanvasAction> action) {
225 if (action) {
226 aggregatedResources |= action->GetRequiredResources();
227 children.push_back(ChildState{std::move(action), false});
228 }
229 }
230
231 [[nodiscard]] std::string_view GetName() const noexcept override { return "ParallelAction"; }
232 [[nodiscard]] uint32_t GetRequiredResources() const noexcept override { return aggregatedResources; }
233
234 void OnStart(ActionContext& ctx) override {
235 for (auto& child : children) {
236 child.finished = false;
237 child.action->OnStart(ctx);
238 }
239 }
240
241 void OnUpdate(double dt, ActionContext& ctx) override {
242 for (auto& child : children) {
243 if (!child.finished) {
244 child.action->OnUpdate(dt, ctx);
245 if (child.action->IsFinished(ctx)) {
246 child.action->OnEnd(ActionEndReason::Finished, ctx);
247 child.finished = true;
248 }
249 }
250 }
251 }
252
253 [[nodiscard]] bool IsFinished(ActionContext& ctx) const override {
254 (void)ctx;
255 return std::all_of(children.begin(), children.end(), [](const ChildState& c) {
256 return c.finished;
257 });
258 }
259
260 void OnEnd(ActionEndReason reason, ActionContext& ctx) override {
261 for (auto& child : children) {
262 if (!child.finished) {
263 child.action->OnEnd(reason, ctx);
264 child.finished = true;
265 }
266 }
267 }
