1/**
2 * @file input_state_machine_mouse.cpp
3 * @brief Implementation of Mouse input dispatching: navigation, transient pan overrides,
4 * gizmo cursors, selection, and scroll-wheel zooming.
5 *
6 * NAVIGATION OVERVIEW:
7 * --------------------
8 * Unlike a stylus, which defaults to inking, the mouse primarily functions as a navigation
9 * and selection device. Transient navigation modes are supported:
10 * 1. Middle-Click Drag: Immediately pans the canvas (`canvas.Pan(dx, dy)`) without modifying tool state.
11 * 2. Spacebar + Left Drag: Standard graphics editor hand tool navigation.
12 * 3. Mouse Wheel: Continuous zoom centered at current cursor position (`canvas.ZoomAt(x, y, factor)`).
13 */

28void InputStateMachine::DispatchMouse(CanvasEngine& canvas, DocumentSession& session, bool imguiWantsInput) {
29 InteractionState oldAction = currentAction;
30
31 // -------------------------------------------------------------------------
32 // 1. TRANSIENT TOOL EVALUATION
33 // -------------------------------------------------------------------------
34 // Space bar or middle mouse button transiently overrides the current tool with Panning
35 if (mouse.middleButton || keyboard.space) {
36 currentAction = InteractionState::Panning;
37 } else {
38 currentAction = mouseTool.savedTool;
39 }
40
41 if (currentAction != InteractionState::DrawingShape && canvas.shapeCreation.lockDrawingMode) {
42 canvas.shapeCreation.lockDrawingMode = false;
43 canvas.shapeCreation.isActive = false;
44 }
45
46 if (oldAction != currentAction) {
47 std::string actionName = (currentAction == InteractionState::Inking) ? "Inking" :
48 (currentAction == InteractionState::Eraser) ? "Eraser" :
49 (currentAction == InteractionState::Selecting) ? "Selecting" :
50 (currentAction == InteractionState::Panning) ? "Panning" :
51 (currentAction == InteractionState::DrawingShape) ? "DrawingShape" : "Idle";
52 LOG_INFO(InputStateMachine, "Mouse interaction state changed to: " + actionName);
53 }
54
55 // Edge transition booleans
56 const bool middleJustDown = mouse.middleButton && !wasMiddleDown;
57 const bool middleIsMoving = mouse.middleButton && wasMiddleDown;
58 const bool middleJustUp = !mouse.middleButton && wasMiddleDown;
59
60 const bool justDown = mouse.leftButton && !wasMouseDown;
61 const bool isMoving = mouse.leftButton && wasMouseDown;
62 const bool justUp = !mouse.leftButton && wasMouseDown;
63
64 // -------------------------------------------------------------------------
65 // 2. DIRECT MIDDLE & SPACE PAN (Fires before UI capture)
66 // -------------------------------------------------------------------------
67 if (currentAction == InteractionState::Panning && middleIsMoving) {
68 pointerIcons.Lock(FolioInput::PointerCursorShape::Grabbing, FolioInput::CursorLockReason::Panning);
69 canvas.Pan(mouse.dx, mouse.dy);
70 }
71
72 if (keyboard.space && (isMoving || middleIsMoving)) {
73 pointerIcons.Lock(FolioInput::PointerCursorShape::Grabbing, FolioInput::CursorLockReason::Panning);
74 canvas.Pan(mouse.dx, mouse.dy);
75 }
76
77 if (middleJustUp || (keyboard.space && justUp)) {
78 pointerIcons.Unlock(FolioInput::CursorLockReason::Panning);
79 mouse.dx = 0.0f;
80 mouse.dy = 0.0f;
81 }
82
83 // -------------------------------------------------------------------------
84 // 3. UI CAPTURE CHECK
85 // -------------------------------------------------------------------------
86 if (justDown) {
87 uiCapturedMouse = imguiWantsInput;
88
89 // Double click detection on canvas
90 if (!uiCapturedMouse) {
91 uint64_t now = SDL_GetTicks();
92 isLastClickDouble = specialActions.EvaluateDoubleClick(mouse.x, mouse.y, now, config);
93 if (isLastClickDouble) {
94 LOG_INFO(InputStateMachine, "Mouse Double-Click detected!");
95 }
96 } else {
97 isLastClickDouble = false;
98 }
99 }
100
101 if (uiCapturedMouse) {
102 if (justUp) uiCapturedMouse = false;
103 return;
104 }
105
106 const float canvasLocalX = mouse.x - canvasOriginX;
107 const float canvasLocalY = mouse.y - canvasOriginY;
108
109 if (currentAction != InteractionState::Eraser) {
110 canvas.HideEraserCursor();
111 }
112
113 // -------------------------------------------------------------------------
114 // 4. MOUSE ACTION DISPATCH
115 // -------------------------------------------------------------------------
116 if (currentAction == InteractionState::Inking) {
117 pointerIcons.SetHoverShape(FolioInput::PointerCursorShape::Crosshair);
118 if (justDown) {
119 pointerIcons.Lock(FolioInput::PointerCursorShape::Crosshair, FolioInput::CursorLockReason::Inking);
120 canvas.OnPointerDown(canvasLocalX, canvasLocalY, 1.0f, latestEventTimeSec, palette.GetActivePen(), 0.0f, 0.0f);
121 } else if (isMoving) {
122 canvas.OnPointerMove(canvasLocalX, canvasLocalY, 1.0f, latestEventTimeSec, 0.0f, 0.0f);
123 } else if (justUp) {
124 pointerIcons.Unlock(FolioInput::CursorLockReason::Inking);
125 canvas.OnPointerUp(session, palette.GetActivePen());
126 }
127 }
128 else if (currentAction == InteractionState::Eraser) {
129 pointerIcons.SetHoverShape(FolioInput::PointerCursorShape::Hidden);
130 double currentTimeSec = (latestEventTimeSec > 0.0) ? latestEventTimeSec : (SDL_GetTicks() * 0.001);
131 bool isPointerMoving = isMoving || (mouse.dx != 0.0f || mouse.dy != 0.0f);
132 float effectiveRadius = UpdateDynamicEraserRadius(canvasLocalX, canvasLocalY, currentTimeSec, isPointerMoving, isEraserActive || justDown);
133
134 if (justDown) {
135 pointerIcons.Lock(FolioInput::PointerCursorShape::Hidden, FolioInput::CursorLockReason::Eraser);
136 lastEraserX = canvasLocalX;
137 lastEraserY = canvasLocalY;
138 isEraserActive = true;
139 session.BeginEraseTransaction();
140 canvas.EraseSegment(canvasLocalX, canvasLocalY, canvasLocalX, canvasLocalY, effectiveRadius, session, isStrokeEraser);
141 } else if (isMoving && isEraserActive) {
142 canvas.EraseSegment(lastEraserX, lastEraserY, canvasLocalX, canvasLocalY, effectiveRadius, session, isStrokeEraser);
143 lastEraserX = canvasLocalX;
144 lastEraserY = canvasLocalY;
145 } else if (justUp) {
146 pointerIcons.Unlock(FolioInput::CursorLockReason::Eraser);
147 isEraserActive = false;
148 session.EndEraseTransaction(&canvas);
149 }
150
151 canvas.SetEraserCursor(canvasLocalX, canvasLocalY, effectiveRadius, isEraserActive, isStrokeEraser);
152 }
153 else if (currentAction == InteractionState::Selecting) {
154 if (justDown) {
155 clickDownScreenX = mouse.x;
156 clickDownScreenY = mouse.y;
157
158 if (canvas.selectionGizmo.OnPointerDown(canvasLocalX, canvasLocalY, canvas.transform,
159 canvas.shapeCreation.lockToGrid, canvas.gridSpacingMm)) {
160 canvas.isDirty = true;
161 auto role = canvas.selectionGizmo.activeRole;
162 auto shape = pointerIcons.GetShapeForHandleRole(role);
163 auto lockReason = (role == HandleRole::Body) ? FolioInput::CursorLockReason::GizmoMove :
164 (role == HandleRole::Rotation) ? FolioInput::CursorLockReason::GizmoRotate :
165 FolioInput::CursorLockReason::GizmoResize;
166 pointerIcons.Lock(shape, lockReason);
167 } else {
168 Point2D worldMm = canvas.transform.ScreenToWorld(canvasLocalX, canvasLocalY);
169 lastCanvasClickWorldMm = worldMm;
170
171 auto activePage = session.GetActivePage();
172 std::shared_ptr<CanvasObject> clickedObj = nullptr;
173
174 if (activePage) {
175 clickedObj = activePage->HitTestSingleClick(worldMm.x, worldMm.y, config.objectHitTestRadiusMm);
176 }
177
178 if (clickedObj) {
179 // Detach any inactive/empty text editor
180 if (canvas.textEditor.IsActive()) {
181 auto prevTarget = canvas.textEditor.GetTarget();
182 canvas.textEditor.Detach(&session);
183 if (prevTarget && prevTarget->PlainText().empty()) {
184 if (activePage) activePage->RemoveObjectByUid(prevTarget->uid);
185 }
186 }
187 hasPendingEmptyTextBox = false;
188 pendingTextBoxUid = 0;
189
190 // ---------------------------------------------------------
191 // DOMAIN CLICK DISPATCH BY OBJECT TYPE
192 // ---------------------------------------------------------
193 // 1. Audio: handles play/pause toggle and scrubber clicks
194 if (clickedObj->type == ObjectType::Audio) {
195 auto* audio = static_cast<Folio::AudioObject*>(clickedObj.get());
196 if (audio->HandleCanvasClick(worldMm.x, worldMm.y, [&canvas]() {
197 canvas.needsFullRebake = true;
198 canvas.isDirty = true;
199 })) {
200 return;
201 }
202 }
203
204 // 2. Double-click actions (Text editing, Attachment launching)
205 if (isLastClickDouble) {
206 if (clickedObj->type == ObjectType::Text) {
207 auto* textBox = static_cast<Folio::TextBoxObject*>(clickedObj.get());
208 if (canvas.textEditor.IsActive() && canvas.textEditor.GetTarget() &&
209 canvas.textEditor.GetTarget() != textBox &&
210 canvas.textEditor.GetTarget()->PlainText().empty()) {
211 auto page = session.GetActivePage();
212 if (page) page->RemoveObjectByUid(canvas.textEditor.GetTarget()->uid);
213 }
214 canvas.ClearSelection(&session);
215 canvas.textEditor.Detach(&session);
216 canvas.textEditor.Attach(textBox, &session);
217 canvas.textEditor.OnMouseDown(worldMm.x, worldMm.y, keyboard.shift);
218 canvas.needsFullRebake = true;
219 canvas.isDirty = true;
220 LOG_INFO(InputStateMachine, "Double-click activated text editor on text box uid=" + std::to_string(textBox->uid));
221 return;
222 } else if (clickedObj->type == ObjectType::AttachmentFile) {
223 auto* attachment = static_cast<Folio::AttachmentObject*>(clickedObj.get());
224 LOG_INFO(InputStateMachine, "Double-clicked attachment: opening file '" + attachment->filePath + "'");
225 attachment->OpenFile();
226 return;
227 }
228 }
229
230 // Check for click on object with live overlay (WebOverlay, YouTube player, live widgets)
231 if (clickedObj->HasLiveOverlay()) {
232 canvas.interactiveOverlayHost.SetFocusedObject(clickedObj.get());
233 }
234
235 canvas.ClearSelection(&session);
236 clickedObj->isSelected = 1;
237 canvas.selectionGizmo.SetSelectedObjects(activePage->objects);
238 canvas.selectionGizmo.OnPointerDown(canvasLocalX, canvasLocalY, canvas.transform,
239 canvas.shapeCreation.lockToGrid, canvas.gridSpacingMm);
240 auto role = canvas.selectionGizmo.activeRole;
241 auto shape = pointerIcons.GetShapeForHandleRole(role);
242 auto lockReason = (role == HandleRole::Body) ? FolioInput::CursorLockReason::GizmoMove :
243 (role == HandleRole::Rotation) ? FolioInput::CursorLockReason::GizmoRotate :
244 FolioInput::CursorLockReason::GizmoResize;
245 pointerIcons.Lock(shape, lockReason);
246 canvas.needsFullRebake = true;
247 canvas.isDirty = true;
248 LOG_INFO(InputStateMachine, "Mouse direct click selected object uid=" + std::to_string(clickedObj->uid));
249 return;
250 } else {
251 // Clicked on empty canvas:
252 // Commits pending edits, detaches the editor, deselects any active text box or objects
253 canvas.interactiveOverlayHost.ClearFocusedObject();
254 if (canvas.textEditor.IsActive()) {
255 auto prevTarget = canvas.textEditor.GetTarget();
256 canvas.textEditor.Detach(&session);
257 if (prevTarget && prevTarget->PlainText().empty()) {
258 if (activePage) activePage->RemoveObjectByUid(prevTarget->uid);
259 }
260 canvas.needsFullRebake = true;
261 }
262 hasPendingEmptyTextBox = false;
263 pendingTextBoxUid = 0;
264 canvas.ClearSelection(&session);
265
266 if (canvas.selectionMode == CanvasEngine::SelectionMode::Lasso) {
267 pointerIcons.Lock(FolioInput::PointerCursorShape::Crosshair, FolioInput::CursorLockReason::BoxSelecting);
268 canvas.OnLassoDown(canvasLocalX, canvasLocalY);
269 } else {
270 pointerIcons.Lock(FolioInput::PointerCursorShape::Crosshair, FolioInput::CursorLockReason::BoxSelecting);
271 canvas.OnBoxSelectDown(canvasLocalX, canvasLocalY);
272 }
273 }
274 }
275 }
276 else if (isMoving) {
277 // If user begins dragging while a pending empty text box was placed,
278 // clean up the empty box so the user can marquee/lasso select smoothly!
279 if (hasPendingEmptyTextBox && pendingTextBoxUid != 0) {
280 float distSq = (mouse.x - clickDownScreenX) * (mouse.x - clickDownScreenX) +
281 (mouse.y - clickDownScreenY) * (mouse.y - clickDownScreenY);
282 if (distSq > 16.0f) { // Moved > 4 pixels
283 if (canvas.textEditor.IsActive() && canvas.textEditor.GetTarget() &&
284 canvas.textEditor.GetTarget()->uid == pendingTextBoxUid &&
285 canvas.textEditor.GetTarget()->PlainText().empty()) {
286 auto activePage = session.GetActivePage();
287 if (activePage) activePage->RemoveObjectByUid(pendingTextBoxUid);
288 canvas.textEditor.Detach();
289 canvas.needsFullRebake = true;
290 canvas.isDirty = true;
291 }
292 hasPendingEmptyTextBox = false;
293 pendingTextBoxUid = 0;
294 }
295 }
296
297 if (canvas.selectionGizmo.isDragging) {
298 if (canvas.selectionGizmo.OnPointerMove(canvasLocalX, canvasLocalY, canvas.transform,
299 canvas.shapeCreation.lockToGrid, canvas.gridSpacingMm)) {
300 canvas.InvalidateLayer();
301 }
302 } else if (canvas.selectionMode == CanvasEngine::SelectionMode::Lasso) {
303 canvas.OnLassoMove(canvasLocalX, canvasLocalY);
304 } else if (canvas.marqueeBox.isActive) {
305 canvas.OnBoxSelectMove(canvasLocalX, canvasLocalY);
306 }
307 }
308 else if (justUp) {
309 hasPendingEmptyTextBox = false;
310 pendingTextBoxUid = 0;
311
312 pointerIcons.Unlock(FolioInput::CursorLockReason::GizmoResize);
313 pointerIcons.Unlock(FolioInput::CursorLockReason::GizmoMove);
314 pointerIcons.Unlock(FolioInput::CursorLockReason::GizmoRotate);
315 pointerIcons.Unlock(FolioInput::CursorLockReason::BoxSelecting);
316
317 if (canvas.selectionGizmo.isDragging) {
318 canvas.selectionGizmo.OnPointerUp(&session);
319 canvas.SyncSelectionToSpatialIndex(&session);
320 canvas.InvalidateLayer();
321 } else if (canvas.selectionMode == CanvasEngine::SelectionMode::Lasso) {
322 canvas.OnLassoUp(&session);
323 } else if (canvas.marqueeBox.isActive) {
324 canvas.OnBoxSelectUp(&session);
325 }
326 }
327
328 // Dynamically update mouse cursor based on hovered gizmo handle when not dragging
329 if (!canvas.selectionGizmo.isDragging && canvas.selectionGizmo.HasSelection()) {
330 auto hit = canvas.selectionGizmo.HitTest(canvasLocalX, canvasLocalY, canvas.transform);
331 if (hit.hit) {
332 pointerIcons.SetHoverForHandleRole(hit.role);
333 } else {
334 pointerIcons.SetHoverShape(FolioInput::PointerCursorShape::Arrow);
335 }
336 } else if (!canvas.selectionGizmo.isDragging) {
337 pointerIcons.SetHoverShape(FolioInput::PointerCursorShape::Arrow);
338 }
339 }
340 else if (currentAction == InteractionState::DrawingShape) {
341 pointerIcons.SetHoverShape(FolioInput::PointerCursorShape::Crosshair);
342 if (ImGui::IsKeyPressed(ImGuiKey_Escape)) {
343 pointerIcons.Unlock(FolioInput::CursorLockReason::ShapeDrawing);
344 canvas.CancelShapeCreation();
345 canvas.ClearSelection(&session);
346 currentAction = InteractionState::Selecting;
347 SetToolForDevice(DeviceType::Mouse, InteractionState::Selecting);
348 SetToolForDevice(DeviceType::Stylus, InteractionState::Selecting);
349 } else if (justDown) {
350 pointerIcons.Lock(FolioInput::PointerCursorShape::Crosshair, FolioInput::CursorLockReason::ShapeDrawing);
351 if (canvas.selectionGizmo.HasSelection()) {
352 canvas.ClearSelection(&session);
353 canvas.selectionGizmo.ClearSelection();
354 }
355 canvas.OnShapeDrawDown(canvasLocalX, canvasLocalY, &session);
356 canvas.isDirty = true;
357 } else if (isMoving) {
358 if (canvas.shapeCreation.isDragging || (canvas.shapeCreation.shapeType == Folio::ShapeType::Ellipse && canvas.shapeCreation.ellipseStep == 1)) {
359 canvas.OnShapeDrawMove(canvasLocalX, canvasLocalY, &session);
360 canvas.isDirty = true;
361 }
362 } else if (justUp) {
363 pointerIcons.Unlock(FolioInput::CursorLockReason::ShapeDrawing);
364 if (canvas.shapeCreation.isDragging || (canvas.shapeCreation.shapeType == Folio::ShapeType::Ellipse && canvas.shapeCreation.ellipseStep == 1)) {
365 canvas.OnShapeDrawUp(&session);
366 canvas.needsFullRebake = true;
367 canvas.isDirty = true;
368 if (canvas.shapeCreation.ellipseStep == 0) {
369 if (!canvas.shapeCreation.lockDrawingMode) {
370 currentAction = InteractionState::Selecting;
371 SetToolForDevice(DeviceType::Mouse, InteractionState::Selecting);
372 SetToolForDevice(DeviceType::Stylus, InteractionState::Selecting);
373 }
374 }
375 }
376 }
377 }
378 else if (currentAction == InteractionState::Text) {
379 pointerIcons.SetHoverShape(FolioInput::PointerCursorShape::TextInput);
380 Point2D worldMm = canvas.transform.ScreenToWorld(canvasLocalX, canvasLocalY);
381 if (justDown) {
382 auto activePage = session.GetActivePage();
383 std::shared_ptr<Folio::TextBoxObject> clickedBox = nullptr;
384 if (activePage) {
385 for (auto it = activePage->objects.rbegin(); it != activePage->objects.rend(); ++it) {
386 auto& obj = *it;
387 if (obj && obj->isVisible && obj->type == ObjectType::Text) {
388 if (obj->HitTest(worldMm.x, worldMm.y)) {
389 clickedBox = std::dynamic_pointer_cast<Folio::TextBoxObject>(obj);
390 break;
391 }
392 }
393 }
394 }
395
396 if (clickedBox) {
397 if (canvas.textEditor.GetTarget() != clickedBox.get()) {
398 canvas.textEditor.Detach();
399 canvas.textEditor.Attach(clickedBox.get());
400 }
401 canvas.textEditor.OnMouseDown(worldMm.x, worldMm.y, keyboard.shift);
402 canvas.isDirty = true;
403 } else {
404 canvas.textEditor.Detach();
405 auto newBox = std::make_shared<Folio::TextBoxObject>(worldMm.x, worldMm.y, 70.0, 20.0);
406 session.AddTextBox(newBox);
407 canvas.textEditor.Attach(newBox.get());
408 canvas.needsFullRebake = true;
409 canvas.isDirty = true;
410 LOG_INFO(InputStateMachine, "OneNote hybrid text click created text box at (" +
411 std::to_string(worldMm.x) + ", " + std::to_string(worldMm.y) + ")");
412 }
413 } else if (isMoving && canvas.textEditor.IsActive()) {
414 canvas.textEditor.OnMouseDrag(worldMm.x, worldMm.y);
415 canvas.isDirty = true;
416 }
417 }
418 else if (currentAction == InteractionState::Panning) {
419 pointerIcons.SetHoverShape(FolioInput::PointerCursorShape::Hand);
420 if (justDown) {
421 pointerIcons.Lock(FolioInput::PointerCursorShape::Grabbing, FolioInput::CursorLockReason::Panning);
422 } else if (isMoving) {
423 canvas.Pan(mouse.dx, mouse.dy);
424 } else if (justUp) {
425 pointerIcons.Unlock(FolioInput::CursorLockReason::Panning);
426 }
427 }
428
429 // -------------------------------------------------------------------------
430 // 5. SCROLL WHEEL ZOOM
431 // -------------------------------------------------------------------------
432 if (mouse.wheelY != 0.0f) {
433 if (isCanvasHovered && !imguiWantsInput) {
434 float factor = (mouse.wheelY > 0.0f) ? config.mouseWheelZoomInFactor : config.mouseWheelZoomOutFactor;
435 canvas.ZoomAt(canvasLocalX, canvasLocalY, factor);
436 }
437 mouse.wheelX = 0.0f;
438 mouse.wheelY = 0.0f;
439 }
440
441 // -------------------------------------------------------------------------
442 // 6. SYNCHRONIZE HARDWARE POINTER CURSOR
443 // -------------------------------------------------------------------------
444 pointerIcons.Apply(isCanvasHovered, imguiWantsInput);
445}
