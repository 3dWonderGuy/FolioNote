22/**
23 * @brief Advances to the next sequential CanvasPage within the active section.
24 * Touches the newly activated page, lazy-loads it if evicted, and notifies observers.
25 *
26 * @return true if navigation succeeded, false if already at the last page or no section is active.
27 */
28bool DocumentSession::NextPage() {
29 auto sec = GetActiveSection();
30 if (!sec || sec->pages.empty()) return false;
31
32 if (sec->activePageIndex + 1 < sec->pages.size()) {
33 auto oldPage = GetActivePage();
34 sec->activePageIndex++;
35 auto newPage = GetActivePage();
36 if (newPage) {
37 newPage->Touch();
38 }
39 NotifyActivePageChanged(newPage, oldPage);
40 NotifyHistoryChanged();
41 LOG_INFO(DocumentSession, "Navigated forward to page: " + (newPage ? newPage->title : "null"));
42 return true;
43 }
44 return false;
45}

76/**
77 * @brief Navigates directly to any CanvasPage by its persistent GUID across the active notebook
78 * or resident workspace notebooks.
79 * Traverses root sections, section groups, and sub-groups to resolve the target page,
80 * switches the active section if necessary, lazy-loads page content into RAM, and notifies observers.
81 *
82 * @param pageGuid Persistent UUID v4 of the destination page.
83 * @return true if page was located and activated; false if not found.
84 */
85/**
86 * @brief Navigates directly to a CanvasPage by persistent GUID across the active notebook or workspace.
87 *
88 * GENERAL WORKING PROCESS & INVARIANT ASSURANCES:
89 * 1. Recursion Guard:
90 * A thread-local recursion counter (`s_navDepth`) prevents cyclic or infinite navigation loops.
91 * If re-entrant navigation reaches depth >= 3, it is automatically intercepted and aborted with a warning.
92 * 2. Active Notebook Resolution:
93 * First attempts rapid lookup within the currently resident active notebook (`nb`):
94 * - Scans root-level sections.
95 * - Recursively scans section groups and nested sub-groups.
96 * - When located, switches the active section tab (if needed) and points `activePageIndex` directly to the target.
97 * - Emits observer events (`NotifyActiveSectionChanged`, `NotifyActivePageChanged`, `NotifyHistoryChanged`).
98 * 3. Cross-Notebook Direct Transition (Zero Recursion):
99 * If the page is not in the active notebook, the workspace is scanned for other resident notebooks:
100 * - When located inside `otherNb`, captures the target section and page index.
101 * - Flushes unwritten modifications on the outgoing notebook asynchronously.
102 * - Directly re-indexes `workspace.activeNotebookIndex` to `nbIdx`, bypassing ambiguous GUID lookups.
103 * - Sets the active section and active page directly on `otherNb` in a single atomic operation.
104 * - Avoids blind recursive re-entry to guarantee 100% loop-free transitions.
105 *
106 * @param pageGuid Persistent UUID v4 of the destination page.
107 * @return true if page was located and activated; false if not found.
108 */
109bool DocumentSession::NavigateToPage(const std::string& pageGuid) {
110 if (pageGuid.empty()) return false;
111
112 // Guard against cyclic re-entrant calls
113 static thread_local int s_navDepth = 0;
114 if (s_navDepth >= 3) {
115 LOG_WARN(DocumentSession, "NavigateToPage: Cyclic recursion prevented for page GUID: " + pageGuid);
116 return false;
117 }
118 struct RecursionGuard {
119 int& depth;
120 explicit RecursionGuard(int& d) : depth(d) { ++depth; }
121 ~RecursionGuard() { --depth; }
122 } guard(s_navDepth);
123
124 auto nb = GetActiveNotebook();
125 if (!nb) return false;
126
127 auto oldPage = GetActivePage();
128 auto oldSec = GetActiveSection();
129
130 auto searchSection = [&](const std::shared_ptr<Section>& sec) -> bool {
131 if (!sec) return false;
132 for (size_t i = 0; i < sec->pages.size(); ++i) {
133 if (sec->pages[i] && sec->pages[i]->guid == pageGuid) {
134 if (sec != oldSec) {
135 nb->SetActiveSection(sec);
136 NotifyActiveSectionChanged(sec, oldSec);
137 }
138 sec->activePageIndex = i;
139 auto newPage = GetActivePage();
140 if (newPage) {
141 newPage->Touch();
142 }
143 NotifyActivePageChanged(newPage, oldPage);
144 NotifyHistoryChanged();
145 LOG_INFO(DocumentSession, "NavigateToPage: Activated page '" + (newPage ? newPage->title : "") + "' [" + pageGuid + "]");
146 return true;
147 }
148 }
149 return false;
150 };
151
152 // 1. Scan root sections in active notebook
153 for (const auto& sec : nb->sections) {
154 if (searchSection(sec)) return true;
155 }
156
157 // 2. Scan section groups and nested sub-groups in active notebook
158 for (const auto& grp : nb->sectionGroups) {
159 if (!grp) continue;
160 for (const auto& sec : grp->sections) {
161 if (searchSection(sec)) return true;
162 }
163 for (const auto& sub : grp->subGroups) {
164 if (!sub) continue;
165 for (const auto& sec : sub->sections) {
166 if (searchSection(sec)) return true;
167 }
168 }
169 }
170
171 // 3. Fallback: Search all other resident notebooks in the workspace for cross-notebook navigation
172 for (size_t nbIdx = 0; nbIdx < workspace.notebooks.size(); ++nbIdx) {
173 const auto& otherNb = workspace.notebooks[nbIdx];
174 if (!otherNb || otherNb == nb) continue;
175
176 std::shared_ptr<Section> foundSection = nullptr;
177 size_t foundPageIndex = 0;
178 bool foundInOther = false;
179
180 auto checkSection = [&](const std::shared_ptr<Section>& sec) -> bool {
181 if (!sec) return false;
182 for (size_t pIdx = 0; pIdx < sec->pages.size(); ++pIdx) {
183 if (sec->pages[pIdx] && sec->pages[pIdx]->guid == pageGuid) {
184 foundSection = sec;
185 foundPageIndex = pIdx;
186 return true;
187 }
188 }
189 return false;
190 };
191
192 for (const auto& sec : otherNb->sections) {
193 if (checkSection(sec)) { foundInOther = true; break; }
194 }
195 if (!foundInOther) {
196 for (const auto& grp : otherNb->sectionGroups) {
197 if (!grp) continue;
198 for (const auto& sec : grp->sections) {
199 if (checkSection(sec)) { foundInOther = true; break; }
200 }
201 if (foundInOther) break;
202 for (const auto& sub : grp->subGroups) {
203 if (!sub) continue;
204 for (const auto& sec : sub->sections) {
205 if (checkSection(sec)) { foundInOther = true; break; }
206 }
207 if (foundInOther) break;
208 }
209 }
210 }
211
212 if (foundInOther && foundSection) {
213 LOG_INFO(DocumentSession, "NavigateToPage: Found page [" + pageGuid + "] in notebook '" + otherNb->name + "'. Directly activating destination notebook.");
214
215 // Flush outgoing notebook changes
216 if (nb) {
217 workspace.FlushActiveNotebookAsync();
218 }
219
220 auto oldNb = nb;
221 workspace.activeNotebookIndex = nbIdx;
222 NotifyActiveNotebookChanged(otherNb, oldNb);
223
224 otherNb->SetActiveSection(foundSection);
225 NotifyActiveSectionChanged(foundSection, oldSec);
226
227 foundSection->activePageIndex = foundPageIndex;
228 auto newPage = GetActivePage();
229 if (newPage) {
230 newPage->Touch();
231 }
232 NotifyActivePageChanged(newPage, oldPage);
233 NotifyHistoryChanged();
234 LOG_INFO(DocumentSession, "NavigateToPage: Activated page '" + (newPage ? newPage->title : "") + "' [" + pageGuid + "] in notebook '" + otherNb->name + "'");
235 return true;
236 }
237 }
238
239 LOG_WARN(DocumentSession, "NavigateToPage failed: Page GUID not found in workspace: " + pageGuid);
240 return false;
241}

283/**
284 * @brief Factory method: Instantiates and appends a new CanvasPage to the active section.
285 * Automatically sets the newly created page as active, marks it modified, registers it
286 * for persistence, and broadcasts creation and active page events to observers.
287 *
288 * @param title Display title for the new page (defaults to "New Untitled").
289 * @param parentGuid Optional parent page GUID for hierarchical subpage nesting.
290 * @param level Nesting hierarchy level (0 = top-level page, 1 = subpage, 2 = sub-subpage).
291 * @return std::shared_ptr<CanvasPage> Pointer to newly created and activated CanvasPage.
292 */
293std::shared_ptr<CanvasPage> DocumentSession::CreateNewPage(std::string title,
294 std::string parentGuid,
295 int32_t level) {
296 auto sec = GetActiveSection();
297 if (!sec) {
298 LOG_ERROR(DocumentSession, "CreateNewPage rejected: No active section present.");
299 return nullptr;
300 }
301
302 auto oldPage = GetActivePage();
303 auto newPage = sec->CreatePage(std::move(title), std::move(parentGuid), level);
304 if (!newPage) return nullptr;
305
306 sec->activePageIndex = sec->pages.size() - 1;
307 newPage->Touch();
308 newPage->isModified = true;
309
310 // Schedule async persistence for initial page record
311 workspace.repository.SavePageAsync(newPage, sec->guid, newPage->sortOrder);
312
313 NotifyPageCreated(newPage);
314 NotifyActivePageChanged(newPage, oldPage);
315 NotifyHistoryChanged();
316 LOG_INFO(DocumentSession, "CreateNewPage: Created and activated '" + newPage->title + "' [" + newPage->guid + "]");
317 return newPage;
318}

320/**
321 * @brief Deletes or soft-deletes the currently active CanvasPage.
322 * Guarantees section safety: if the active page is the only page in the section,
323 * a clean blank "Untitled page" is automatically created first so the section is never empty.
324 *
325 * @param moveToTrash If true, moves to recycle bin in SQLite (30-day retention window).
326 * @return true if active page was successfully removed.
327 */
328bool DocumentSession::DeleteActivePage(bool moveToTrash) {
329 auto sec = GetActiveSection();
330 if (!sec || sec->pages.empty()) return false;
331
332 auto oldPage = GetActivePage();
333 if (!oldPage) return false;
334
335 std::string deletedGuid = oldPage->guid;
336
337 // Invariant: Section must always contain at least one valid page.
338 // If deleting the only remaining page, generate a replacement blank page first.
339 if (sec->pages.size() <= 1) {
340 sec->CreatePage("Untitled page");
341 }
342
343 if (moveToTrash) {
344 workspace.repository.SoftDeletePageAsync(deletedGuid);
345 LOG_INFO(DocumentSession, "DeleteActivePage: Soft-deleted page [" + deletedGuid + "] to recycle bin");
346 }
347
348 sec->RemovePage(deletedGuid);
349 auto newPage = GetActivePage();
350 if (newPage) {
351 newPage->Touch();
352 }
353
354 NotifyPageDeleted(deletedGuid);
355 NotifyActivePageChanged(newPage, oldPage);
356 NotifyHistoryChanged();
357 return true;
358}

360/**
361 * @brief Creates a deep duplicate of the active CanvasPage with cloned objects and fresh UUIDs.
362 * Inserts the clone immediately after the active page, sets it active, and persists it.
363 *
364 * @return std::shared_ptr<CanvasPage> Cloned and activated page pointer.
365 */
366std::shared_ptr<CanvasPage> DocumentSession::DuplicateActivePage() {
367 auto sec = GetActiveSection();
368 if (!sec) return nullptr;
369

371 if (!oldPage) return nullptr;
372

374 if (!clone) return nullptr;
375

377 sec->InsertPage(insertIdx, clone);

382

384 NotifyActivePageChanged(clone, oldPage);
385 NotifyHistoryChanged();

387 return clone;
388}

390/**
391 * @brief Moves the active page forward (+1) or backward (-1) in display sequence.
392 * @param delta Direction offset (-1 for up/earlier, +1 for down/later).
393 * @return true if page moved successfully; false if already at boundary.
394 */
395bool DocumentSession::MoveActivePage(int delta) {
396 auto sec = GetActiveSection();
397 if (!sec || sec->pages.size() <= 1) return false;
398
399 int newIdx = static_cast<int>(sec->activePageIndex) + delta;
400 if (newIdx < 0 || newIdx >= static_cast<int>(sec->pages.size())) {
401 return false;
402 }
403
404 bool ok = sec->MovePage(sec->activePageIndex, static_cast<size_t>(newIdx));
405 if (ok) {
406 auto page = GetActivePage();
407 NotifyActivePageChanged(page, page);
408 }
409 return ok;
410}

416/**
417 * @brief Caches the camera pan and zoom viewport state for a specific page in RAM.
418 *
419 * MATHEMATICAL PROJECTION:
420 * P_screen = (P_world + panMm) * (DPI / 25.4 * zoom)
421 * P_world = (P_screen / (DPI / 25.4 * zoom)) - panMm
422 *
423 * Persists in RAM throughout the active application run without time-based eviction,
424 * ensuring that users return to their exact viewport position when switching pages.
425 * Starts clean (homed at 0, 0, 1.0x) upon application reboot.
426 *
427 * @param pageGuid Target CanvasPage persistent GUID.
428 * @param panX World X pan coordinate in millimeters.
429 * @param panY World Y pan coordinate in millimeters.
430 * @param zoom Camera magnification scale factor (e.g. 1.0 = 100%).
431 */
432void DocumentSession::SavePageViewport(const std::string& pageGuid, double panX, double panY, double zoom) {
433 auto page = workspace.FindPageByGuid(pageGuid);
434 if (page) {
435 page->inMemoryViewport.panXMm = panX;
436 page->inMemoryViewport.panYMm = panY;
437 page->inMemoryViewport.zoom = zoom;
438 page->inMemoryViewport.hasCustomViewport = true;
439 page->inMemoryViewport.lastViewportAccessMs = SDL_GetTicks();
440 }
441}

448 * @param outZoom Output camera magnification factor.

477/**
478 * @brief Navigates to a specific canvas coordinate on any page, centering the viewport on that point.
479 *
480 * MATHEMATICAL CENTERING DERIVATION:
481 * The world-to-screen camera projection is:
482 * P_screen = (P_world + panMm) * (DPI / 25.4 * zoom)
483 * Setting P_screen to the center of the viewport (W_screen / 2, H_screen / 2):
484 * (targetWorldXMm + panXMm) * scale = W_screen * 0.5
485 * (targetWorldYMm + panYMm) * scale = H_screen * 0.5
486 *
487 * Solving for camera pan offsets:
488 * panXMm = (W_screen * 0.5 / scale) - targetWorldXMm
489 * panYMm = (H_screen * 0.5 / scale) - targetWorldYMm
490 * where scale = pixelsPerMm * zoom.
491 *
492 * @param pageGuid Target CanvasPage GUID.
493 * @param targetWorldXMm Destination X coordinate in world millimeters.
494 * @param targetWorldYMm Destination Y coordinate in world millimeters.
495 * @param screenWidthPx Current viewport rendering width in pixels.
496 * @param screenHeightPx Current viewport rendering height in pixels.
497 * @param pixelsPerMm Physical display density (e.g. 96.0 / 25.4).
498 * @param zoom Target camera zoom scale (defaults to 1.0x).
499 * @return true if navigation succeeded and viewport was homed to target point.
500 */
501bool DocumentSession::NavigateToCanvasLocation(const std::string& pageGuid,
502 double targetWorldXMm, double targetWorldYMm,
503 double screenWidthPx, double screenHeightPx,
504 double pixelsPerMm, double zoom) {
505 if (pageGuid.empty()) return false;
506 double scale = pixelsPerMm * zoom;
507 if (scale <= 1e-4) scale = 1.0;
508
509 double panX = (screenWidthPx * 0.5 / scale) - targetWorldXMm;
510 double panY = (screenHeightPx * 0.5 / scale) - targetWorldYMm;
511
512 // Cache pre-computed centered viewport coordinates
513 SavePageViewport(pageGuid, panX, panY, zoom);
514
515 // Switch to target page
516 return NavigateToPage(pageGuid);
517}

519/**
520 * @brief Navigates to a specific CanvasObject by UID on any page, centering the viewport on its
521 * bounding box centroid and optionally selecting the object.
522 *
523 * MATHEMATICAL DERIVATION:
524 * Evaluates the Axis-Aligned Bounding Box (AABB) of the target entity:
525 * Center_X = (bounds.minX + bounds.maxX) * 0.5
526 * Center_Y = (bounds.minY + bounds.maxY) * 0.5
527 * Then delegates to NavigateToCanvasLocation(pageGuid, Center_X, Center_Y, ...).
528 *
529 * @param pageGuid Target CanvasPage persistent GUID.
530 * @param objectUid Unique runtime ID of the target object.
531 * @param screenWidthPx Current screen width in pixels.
532 * @param screenHeightPx Current screen height in pixels.
533 * @param pixelsPerMm Physical display density.
534 * @param selectObject If true, clears active selection and selects the target entity.
535 * @param zoom Target camera magnification scale.
536 * @return true if object was found and centered in viewport.
537 */
538bool DocumentSession::NavigateToObject(const std::string& pageGuid, uint32_t objectUid,
539 double screenWidthPx, double screenHeightPx,
540 double pixelsPerMm, bool selectObject,
541 double zoom) {
542 if (pageGuid.empty() || objectUid == 0) return false;
543
544 // Ensure target page is navigated/loaded
545 if (!NavigateToPage(pageGuid)) return false;
546
547 auto activePage = GetActivePage();
548 if (!activePage) return false;
549
550 std::shared_ptr<CanvasObject> targetObj = nullptr;
551 for (const auto& obj : activePage->objects) {
552 if (obj && obj->uid == objectUid) {
553 targetObj = obj;
554 break;
555 }
556 }
557
558 if (!targetObj) {
559 LOG_WARN(DocumentSession, "NavigateToObject: Object UID " + std::to_string(objectUid) +
560 " not found on page " + pageGuid);
561 return false;
562 }
563
564 const auto& bounds = targetObj->bounds;
565 double cx = (bounds.minX + bounds.maxX) * 0.5;
566 double cy = (bounds.minY + bounds.maxY) * 0.5;
567
568 double scale = pixelsPerMm * zoom;
569 if (scale <= 1e-4) scale = 1.0;
570
571 double panX = (screenWidthPx * 0.5 / scale) - cx;
572 double panY = (screenHeightPx * 0.5 / scale) - cy;
573
574 SavePageViewport(pageGuid, panX, panY, zoom);
575
576 if (selectObject) {
577 DeselectAll();
578 targetObj->isSelected = 1;
579 NotifyPageModified(activePage);
580 }
581
582 LOG_INFO(DocumentSession, "NavigateToObject: Centered on object UID " + std::to_string(objectUid) +
583 " at world (" + std::to_string(cx) + ", " + std::to_string(cy) + ") mm");
584 return true;
585}

587/**
588 * @brief Parses a dynamic deep-link URI into a structured CanvasDeepLink target.
589 *
590 * Supported URI Formats:
591 * 1. folionote://page/<guid>
592 * 2. folionote://page/<guid>?x=120.5&y=450.2&zoom=1.5
593 * 3. folionote://page/<guid>#obj=<uid> or folionote://page/<guid>?obj=<uid>
594 * 4. folionote://page/<guid>?pos=120.5,450.2,1.5
595 * 5. <guid> (direct GUID string)
596 *
597 * @param uri Raw URI string.
598 * @return std::optional<CanvasDeepLink> containing parsed target fields if valid.
599 */
600std::optional<DocumentSession::CanvasDeepLink> DocumentSession::ParseDeepLinkUri(const std::string& uri) {
601 if (uri.empty()) return std::nullopt;
602
603 CanvasDeepLink link;
604 std::string clean = uri;
605
606 // Strip scheme if present
607 const std::string scheme = "folionote://";
608 if (clean.rfind(scheme, 0) == 0) {
609 clean = clean.substr(scheme.length());
610 }
611 const std::string pagePrefix = "page/";
612 if (clean.rfind(pagePrefix, 0) == 0) {
613 clean = clean.substr(pagePrefix.length());
614 }
615
616 // Split query and fragment
617 std::string queryStr;
618 std::string fragmentStr;
619
620 size_t hashPos = clean.find('#');
621 if (hashPos != std::string::npos) {
622 fragmentStr = clean.substr(hashPos + 1);
623 clean = clean.substr(0, hashPos);
624 }
625 size_t qPos = clean.find('?');
626 if (qPos != std::string::npos) {
627 queryStr = clean.substr(qPos + 1);
628 clean = clean.substr(0, qPos);
629 }
630
631 link.pageGuid = clean;
632 if (link.pageGuid.empty()) return std::nullopt;
633
634 // Parse query params (e.g. x=10&y=20&zoom=1.5 or obj=42)
635 auto parseParams = [&](const std::string& s) {
636 if (s.empty()) return;
637 size_t start = 0;
638 while (start < s.length()) {
639 size_t end = s.find('&', start);
640 if (end == std::string::npos) end = s.length();
641 std::string pair = s.substr(start, end - start);
642 size_t eq = pair.find('=');
643 if (eq != std::string::npos) {
644 std::string key = pair.substr(0, eq);
645 std::string val = pair.substr(eq + 1);
646 try {
647 if (key == "x") {
648 link.worldXMm = std::stod(val);
649 link.hasTargetCoords = true;
650 } else if (key == "y") {
651 link.worldYMm = std::stod(val);
652 link.hasTargetCoords = true;
653 } else if (key == "zoom") {
654 link.zoom = std::stod(val);
655 } else if (key == "obj" || key == "object") {
656 link.objectUid = static_cast<uint32_t>(std::stoul(val));
657 link.hasTargetObject = true;
658 } else if (key == "pos") {
659 // Comma separated x,y or x,y,zoom
660 size_t c1 = val.find(',');
661 if (c1 != std::string::npos) {
662 link.worldXMm = std::stod(val.substr(0, c1));
663 size_t c2 = val.find(',', c1 + 1);
664 if (c2 != std::string::npos) {
665 link.worldYMm = std::stod(val.substr(c1 + 1, c2 - c1 - 1));
666 link.zoom = std::stod(val.substr(c2 + 1));
667 } else {
668 link.worldYMm = std::stod(val.substr(c1 + 1));
669 }
670 link.hasTargetCoords = true;
671 }
672 }
673 } catch (...) {
674 // Ignore malformed numeric values
675 }
676 } else if (pair.rfind("obj_", 0) == 0) {
677 // Format: #obj_123
678 try {
679 link.objectUid = static_cast<uint32_t>(std::stoul(pair.substr(4)));
680 link.hasTargetObject = true;
681 } catch (...) {}
682 }
683 start = end + 1;
684 }
685 };
686
687 parseParams(queryStr);
688 parseParams(fragmentStr);
689
690 return link;
691}

695 */
696bool DocumentSession::NavigateToDeepLink(const CanvasDeepLink& link,
697 double screenWidthPx, double screenHeightPx,
698 double pixelsPerMm, bool selectObject) {
699 if (link.pageGuid.empty()) return false;
700
701 if (link.hasTargetObject && link.objectUid != 0) {
702 return NavigateToObject(link.pageGuid, link.objectUid,
703 screenWidthPx, screenHeightPx,
704 pixelsPerMm, selectObject, link.zoom);
705 }

707 if (link.hasTargetCoords) {
708 return NavigateToCanvasLocation(link.pageGuid, link.worldXMm, link.worldYMm,
709 screenWidthPx, screenHeightPx,
710 pixelsPerMm, link.zoom);
711 }
712
713 return NavigateToPage(link.pageGuid);
714}
