52 /**
53 * @brief Advances animation timers and calculates instantaneous FPS.
54 * Guaranteed zero heap allocation.
55 */
56 void OnUpdate(uint64_t nowMs, double deltaTime) override {
57 (void)nowMs;
58 if (deltaTime > 0.00001) {
59 // Exponential moving average filter for smooth FPS presentation
60 // alpha = 0.1 gives 90% history weight, 10% new sample
61 const double instantFps = 1.0 / deltaTime;
62 currentFps = (currentFps * 0.9) + (instantFps * 0.1);
63 }
64
65 // Advance sinusoidal phase (speed: 3 radians per second)
66 oscillationPhase += static_cast<float>(deltaTime * 3.0);
67 if (oscillationPhase > 6.2831853f) {
68 oscillationPhase -= 6.2831853f;
69 }
70 }

76 /**
77 * @brief Renders the mock overlay on the screen composite context.
78 * Guaranteed zero heap allocation: uses stack-allocated char buffers for text formatting.
79 */
80 void OnRenderOverlay(BLContext& ctx, const OverlayRect& screenRect) override {
81 if (screenRect.IsEmpty()) return;
82
83 ctx.save();
84
85 // 1. Clip to screen bounds with rounded corners
86 constexpr double cornerRadius = 6.0;
87 ctx.clip_to_round_rect(BLRoundRect(screenRect.x, screenRect.y,
88 screenRect.width, screenRect.height,
89 cornerRadius, cornerRadius));
90
91 // 2. Toggled background tint based on click state
92 static constexpr std::array<BLRgba32, 4> bgPalette = {
93 BLRgba32(0x1E, 0x22, 0x2D, 0xF5), // Slate dark
94 BLRgba32(0x1B, 0x2E, 0x3D, 0xF5), // Navy blue
95 BLRgba32(0x19, 0x36, 0x2D, 0xF5), // Deep emerald
96 BLRgba32(0x33, 0x1E, 0x36, 0xF5) // Royal amethyst
97 };
98 ctx.set_fill_style(bgPalette[paletteIndex % bgPalette.size()]);
99 ctx.fill_round_rect(BLRoundRect(screenRect.x, screenRect.y,
100 screenRect.width, screenRect.height,
101 cornerRadius, cornerRadius));
102
103 // 3. Subsystem Header Bar
104 ctx.set_fill_style(BLRgba32(0x14, 0x17, 0x1F, 0xE0));
105 ctx.fill_rect(BLRect(screenRect.x, screenRect.y, screenRect.width, 24.0));
106
107 // 4. Running 60 FPS counter (formatted into stack buffer)
108 char fpsText[48];
109 std::snprintf(fpsText, sizeof(fpsText), "Live Overlay | FPS: %5.1f", currentFps);
110 ctx.set_fill_style(BLRgba32(0x00, 0xE5, 0xFF, 0xFF)); // Neon Cyan
111 ctx.fill_utf8_text(BLPoint(screenRect.x + 8.0, screenRect.y + 16.0), cachedFont, fpsText);
112
113 // 5. Sinusoidal Oscillating Colored Bar
114 // Math: t in [0.0, 1.0] from sin(phase)
115 const double t = 0.5 + 0.5 * std::sin(static_cast<double>(oscillationPhase));
116 const double barMargin = 10.0;
117 const double availableWidth = std::max(10.0, static_cast<double>(screenRect.width) - (barMargin * 2.0));
118 const double barWidth = availableWidth * 0.6;
119 const double travelRange = availableWidth - barWidth;
120 const double barX = screenRect.x + barMargin + (t * travelRange);
121 const double barY = screenRect.y + 32.0;
122
123 // Dynamic color shifting from Electric Blue to Vivid Violet
124 const uint8_t r = static_cast<uint8_t>(0x3B + t * (0x93 - 0x3B));
125 const uint8_t g = static_cast<uint8_t>(0x82 + (1.0 - t) * (0x3B - 0x82));
126 const uint8_t b = static_cast<uint8_t>(0xF6);
127 ctx.set_fill_style(BLRgba32(r, g, b, 0xFF));
128 ctx.fill_round_rect(BLRoundRect(barX, barY, barWidth, 12.0, 3.0, 3.0));
129
130 // 6. Interactive Click Me Button
131 ComputeButtonRect(screenRect);
132 const bool pressed = isButtonPressed;
133 const bool hovered = isButtonHovered;
134
135 // Button background
136 if (pressed) {
137 ctx.set_fill_style(BLRgba32(0x25, 0x63, 0xEB, 0xFF)); // Active blue
138 } else if (hovered) {
139 ctx.set_fill_style(BLRgba32(0x3B, 0x82, 0xF6, 0xEE)); // Hover light blue
140 } else {
141 ctx.set_fill_style(BLRgba32(0x27, 0x2D, 0x3B, 0xDD)); // Resting dark slate
142 }
143 ctx.fill_round_rect(BLRoundRect(cachedButtonRect.x, cachedButtonRect.y,
144 cachedButtonRect.width, cachedButtonRect.height,
145 4.0, 4.0));
146
147 // Button outline
148 ctx.set_stroke_style(hovered ? BLRgba32(0x60, 0xA5, 0xFA, 0xFF) : BLRgba32(0x4B, 0x55, 0x63, 0xFF));
149 ctx.set_stroke_width(1.0);
150 ctx.stroke_round_rect(BLRoundRect(cachedButtonRect.x, cachedButtonRect.y,
151 cachedButtonRect.width, cachedButtonRect.height,
152 4.0, 4.0));
153
154 // Button text (stack buffer)
155 char btnText[64];
156 std::snprintf(btnText, sizeof(btnText), "Click Me: %u clicks", clickCount);
157 ctx.set_fill_style(BLRgba32(0xFF, 0xFF, 0xFF, 0xFF));
158 ctx.fill_utf8_text(BLPoint(cachedButtonRect.x + 12.0, cachedButtonRect.y + 18.0),
159 buttonFont, btnText);
160
161 ctx.restore();
162 }

168 /**
169 * @brief Consumes mouse events within overlay bounds.
170 *
171 * Input Isolation Test:
172 * Clicking the button increments the click counter and cycles the background
173 * tint. Returning true ensures the event never reaches CanvasEngine, proving
174 * that active overlays isolate input without disturbing canvas selection or gizmos.
175 */
176 [[nodiscard]] bool OnInputEvent(const SDL_Event& event, const OverlayRect& screenRect) override {
177 ComputeButtonRect(screenRect);
178
179 switch (event.type) {
180 case SDL_EVENT_MOUSE_MOTION: {
181 const int mx = static_cast<int>(event.motion.x);
182 const int my = static_cast<int>(event.motion.y);
183 isButtonHovered = cachedButtonRect.Contains(mx, my);
184 // Consume motion if cursor is inside this overlay so canvas hover doesn't trigger
185 return screenRect.Contains(mx, my);
186 }
187
188 case SDL_EVENT_MOUSE_BUTTON_DOWN: {
189 if (event.button.button == SDL_BUTTON_LEFT) {
190 const int mx = static_cast<int>(event.button.x);
191 const int my = static_cast<int>(event.button.y);
192 if (cachedButtonRect.Contains(mx, my)) {
193 isButtonPressed = true;
194 clickCount++;
195 paletteIndex = (paletteIndex + 1) % 4;
196 return true; // Consumed by button
197 }
198 if (screenRect.Contains(mx, my)) {
199 return true; // Consumed by overlay card
200 }
201 }
202 break;
203 }
204
205 case SDL_EVENT_MOUSE_BUTTON_UP: {
206 if (event.button.button == SDL_BUTTON_LEFT) {
207 const bool wasPressed = isButtonPressed;
208 isButtonPressed = false;
209 const int mx = static_cast<int>(event.button.x);
210 const int my = static_cast<int>(event.button.y);
211 if (wasPressed || screenRect.Contains(mx, my)) {
212 return true; // Consumed
213 }
214 }
215 break;
216 }
217
218 default:
219 break;
220 }
221
222 return false;
223 }
