1/**
2 * =========================================================================================
3 * @file core/layers/embedded_app_layer.cpp
4 * @brief Implementation of Layer 3 EmbeddedAppLayer Runtime Coordinator
5 * =========================================================================================
6 *
7 * GENERAL ARCHITECTURAL FLOW:
8 * ---------------------------
9 * This translation unit coordinates active interactive widgets on the canvas:
10 * 1. Lifecycle Management:
11 * - Owns `std::unique_ptr<IEmbeddedApp>` instances mapped to object UIDs.
12 * - Guarantees clean resource release through `OnTeardown()` when widgets are unmounted
13 * or the layer is destroyed.
14 * 2. Real-Time Spatial Projection:
15 * - Maps world-space physical millimeters to screen-space device pixels using:
16 * p_screen = M_worldToScreen * p_world
17 * - Maps top-left (worldX, worldY) and bottom-right (worldX + worldWidth, worldY + worldHeight)
18 * to calculate the exact bounding box `AppScreenRegion`.
19 * 3. Execution Pass:
20 * - Dispatches per-frame simulation via `app->Tick(deltaTime)`.
21 * - Dispatches graphics rendering via `app->RenderApp(region)`.
22 */

33/**
34 * @brief Mounts an active runtime widget session onto a specific canvas object UID.
35 *
36 * If an app was previously mounted under this UID, it is safely dismounted and torn down first.
37 *
38 * @param[in] objectUid Unique identifier of the host CanvasObject.
39 * @param[in] app Unique pointer to the running widget implementation.
40 */
41void EmbeddedAppLayer::MountApp(uint32_t objectUid, std::unique_ptr<IEmbeddedApp> app) {
42 if (!app) return;
43 DismountApp(objectUid);
44 m_activeApps[objectUid] = std::move(app);
45}

47/**
48 * @brief Dismounts and terminates an app session associated with a CanvasObject UID.
49 *
50 * @param[in] objectUid Unique identifier of the CanvasObject to dismount.
51 */
52void EmbeddedAppLayer::DismountApp(uint32_t objectUid) {
53 auto it = m_activeApps.find(objectUid);
54 if (it != m_activeApps.end()) {
55 if (it->second) {
56 it->second->OnTeardown();
57 }
58 m_activeApps.erase(it);
59 }
60}

62/**
63 * @brief Dismounts all active applications and releases all native handles.

65void EmbeddedAppLayer::DismountAll() {
66 for (auto& [uid, app] : m_activeApps) {

84/**
85 * @brief Updates and renders all mounted apps matching visible canvas objects.
86 *
87 * MATHEMATICAL PROJECTION:
88 * Let M = viewport.worldToScreenMatrix:
89 * topLeftScreen = M * (worldX, worldY)
90 * bottomRightScreen = M * (worldX + worldWidth, worldY + worldHeight)
91 * region.x = topLeftScreen.x
92 * region.y = topLeftScreen.y
93 * region.width = bottomRightScreen.x - topLeftScreen.x
94 * region.height = bottomRightScreen.y - topLeftScreen.y
95 *
96 * @param[in] viewport Current camera viewport with transformation matrices.
97 * @param[in] activeObjects Active candidate objects on the current page.
98 * @param[in] deltaTime Elapsed frame time in seconds.
99 */
100void EmbeddedAppLayer::UpdateAndRender(const Viewport& viewport,
101 const std::unordered_map<uint32_t, const CanvasObject*>& activeObjects,
102 double deltaTime) {
103 for (auto& [uid, app] : m_activeApps) {
104 if (!app) continue;
105
106 auto objIt = activeObjects.find(uid);
107 if (objIt == activeObjects.end() || !objIt->second) {
108 continue;
109 }
110
111 const CanvasObject* obj = objIt->second;
112
113 // Project world mm coordinates to screen pixel region
114 BLPoint topLeft = viewport.worldToScreenMatrix.map_point(BLPoint(obj->worldX, obj->worldY));
115 BLPoint bottomRight = viewport.worldToScreenMatrix.map_point(
116 BLPoint(obj->worldX + obj->worldWidth, obj->worldY + obj->worldHeight));
117
118 AppScreenRegion region;
119 region.x = static_cast<float>(topLeft.x);
120 region.y = static_cast<float>(topLeft.y);
121 region.width = static_cast<float>(bottomRight.x - topLeft.x);
122 region.height = static_cast<float>(bottomRight.y - topLeft.y);
123
124 app->Tick(deltaTime);
125 app->RenderApp(region);
126 }
127}
