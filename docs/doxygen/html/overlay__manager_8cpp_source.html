18void OverlayManager::RenderCanvasFloatingHUDs(
19 CanvasEngine& canvas,
20 DocumentSession& session,
21 const ImVec2& canvasOrigin
22) {
23 // When an AudioObject is selected on the canvas, render a floating modern
24 // glassmorphic capsule HUD directly below the badge tracking canvas transform.
25 for (const auto& obj : session.GetSelectedObjects()) {
26 if (obj && obj->type == ObjectType::Audio) {
27 if (auto audioObj = std::dynamic_pointer_cast<Folio::AudioObject>(obj)) {
28 Folio::AudioOverlayUI::Render(*audioObj, canvas.transform, canvasOrigin, [&]() {
29 canvas.isDirty = true;
30 });
31 }
32 }
33 }
34}

36void OverlayManager::RenderModalsAndOverlays(
37 CanvasEngine& canvas,
38 InputStateMachine& stateMachine,
39 WindowStateManager& windowSM,
40 DocumentSession& session,
41 const ThemeManager& themeManager
42) {
43 // =========================================================================
44 // 1. ATTACHMENT FLOW MODAL
45 // =========================================================================
46 if (canvas.m_attachModalOpen) {
47 ImGui::OpenPopup("Attach File##Modal");
48 }
49
50 ImGui::SetNextWindowSize(ImVec2(380, 0), ImGuiCond_Appearing);
51 if (ImGui::BeginPopupModal("Attach File##Modal", nullptr, ImGuiWindowFlags_NoSavedSettings | ImGuiWindowFlags_NoResize)) {
52 ImGui::Text("File: %s", canvas.m_pendingAttachName.c_str());
53 ImGui::Separator();
54 ImGui::Spacing();
55
56 ImGui::TextWrapped("How would you like to attach this file?");
57 ImGui::Spacing();
58
59 if (ImGui::Button("Embed in Notebook", ImVec2(-1, 30))) {
60 canvas.CommitAttachment(true, canvas.m_pendingAttachSession);
61 ImGui::CloseCurrentPopup();
62 }
63 ImGui::TextColored(themeManager.colorTextMuted, "Copies the file into the notebook sidecar. Portable and safe.");
64 ImGui::Spacing();
65
66 if (ImGui::Button("Attach as Link", ImVec2(-1, 30))) {
67 canvas.CommitAttachment(false, canvas.m_pendingAttachSession);
68 ImGui::CloseCurrentPopup();
69 }
70 ImGui::TextColored(themeManager.colorTextMuted, "Stores the original absolute path. Breaks if the file is moved.");
71
72 ImGui::Spacing();
73 ImGui::Separator();
74 if (ImGui::Button("Cancel", ImVec2(100, 0))) {
75 canvas.m_attachModalOpen = false;
76 canvas.m_pendingAttachPath.clear();
77 canvas.m_pendingAttachName.clear();
78 canvas.m_pendingAttachSession = nullptr;
79 ImGui::CloseCurrentPopup();
80 }
81 ImGui::EndPopup();
82 }
83
84 // =========================================================================
85 // 2. DIAGNOSTIC TOOLS & DEV STUDIOS
86 // =========================================================================
87 devTelemetry.Render(canvas, stateMachine, windowSM, session,
88 stateMachine.canvasOriginX, stateMachine.canvasOriginY, themeManager);
89 tuningStudio.Render(themeManager);
90 toolbarDemo.Render(themeManager);
91
92 // =========================================================================
93 // 3. DOCUMENT IMPORT MODAL
94 // =========================================================================
95 pdfImportModal.Render(session, canvas, themeManager);
96
97 // =========================================================================
98 // 4. SPOTLIGHT ACTION COMMAND PALETTE (Ctrl+K / Search)
99 // =========================================================================
100 commandPalette.Render(themeManager);
101}
