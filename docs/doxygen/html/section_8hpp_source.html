12/**
13 * =========================================================================================
14 * @file section.hpp
15 * @brief Represents a Section tab within a Notebook or Section Group.
16 * =========================================================================================
17 *
18 * ARCHITECTURAL ROLE:
19 * - A Section represents an individual thematic tab (e.g., "Lectures", "Homework", "Quick Notes").
20 * - In the document hierarchy:
21 * Notebook -> [Section Group ->] Section -> CanvasPage
22 * - A Section can either reside directly at the root of a Notebook or be nested inside a SectionGroup.
23 * - When created, a Section automatically instantiates a default blank CanvasPage so that the canvas
24 * is never in an uninitialized empty state.
25 *
26 * COLOR & ICON UNIFICATION:
27 * - Employs a single template vector icon (`FOLIO_SECTION_DEFAULT_ICON` = "orange-section-simple.svg").
28 * - Dynamic color styling is governed by `colorTag` (ImVec4 RGBA), matching the design pattern
29 * established for Library and Notebook entities.
30 * - Tab icons are rendered as monochrome white masks and dynamically tinted at runtime, avoiding
31 * redundant disk assets for color variations.
32 *
33 * KEY ATTRIBUTES:
34 * - `guid`: Persistent unique UUID v4 identifier.
35 * - `groupGuid`: If non-empty, points to the parent SectionGroup UUID.
36 * - `name`: Display name of this section.
37 * - `colorTag`: Accent color vector (RGBA) used to tint the tab icon, headers, and UI indicators.
38 * - `iconFile`: Base template SVG icon filename.
39 * - `sortOrder`: 0-indexed integer defining the display order in the UI navigation bar.
40 * - `pages`: Ordered vector of CanvasPages.
41 * - `activePageIndex`: Tracks the currently viewed page within this section.
42 *
43 * POTENTIAL FUTURE ENHANCEMENTS:
44 * - Password Protection: AES-256 encrypted sections (like OneNote's locked sections).
45 * - Section Archiving / Export: Export a section as a standalone `.section` package or PDF bundle.
46 * - Read-Only / Locking: Prevent accidental edits on finalized sections.
47 */

65/**
66 * @brief Selects a random accent color from the curated preset color palette.
67 *
68 * MATHEMATICAL WORKING PROCESS:
69 * - Entropy Source: Hardware entropy via `std::random_device rd` seeds a 32-bit Mersenne
70 * Twister engine (`std::mt19937 gen`).
71 * - Discrete Uniform Distribution:
72 * P(X = k) = 1 / N, where N = SECTION_PRESET_COLORS.size()
73 * Guarantees equal probabilistic likelihood across all curated accent tones.
74 * - Invariant: Thread-safe local static PRNG generator ensures efficient O(1) sampling
75 * without heap allocations or re-initialization overhead.
76 *
77 * @return ImVec4 Random RGBA color vector from SECTION_PRESET_COLORS.
78 */
79inline ImVec4 GetRandomSectionColor() {
80 thread_local std::mt19937 gen([]() {
81 std::random_device rd;
82 return std::mt19937(rd());
83 }());
84 std::uniform_int_distribution<size_t> dis(0, SECTION_PRESET_COLORS.size() - 1);
85 return SECTION_PRESET_COLORS[dis(gen)];
86}

225 /**
226 * @brief Inserts a CanvasPage at a specific index, maintains contiguous sort orders,
227 * marks the page dirty for persistence, and logs the operation.
228 *
229 * MATHEMATICAL WORKING PROCESS & SORT ORDER SEQUENCING:
230 * - Clamps insertion index to [0, pages.size()] to prevent vector iterator out-of-bounds.
231 * - Re-indexes all subsequent page positions monotonically:
232 * \forall i \in [0, N-1], \quad pages[i]->sortOrder = i
233 * - Sets `page->isModified = true` ensuring the inserted page is scheduled for SQLite upsert.
234 *
235 * @param index 0-indexed position where the page should be inserted.
236 * @param page Shared pointer to the CanvasPage to insert.
237 * @return bool True if successfully inserted; false if page was nullptr.
238 */
239 bool InsertPage(size_t index, std::shared_ptr<CanvasPage> page) {
240 if (!page) return false;
241 if (index > pages.size()) {
242 index = pages.size();
243 }
244 page->isModified = true;
245 std::string pageTitle = page->title;
246 std::string pageGuid = page->guid;
247 pages.insert(pages.begin() + index, std::move(page));
248 for (size_t i = 0; i < pages.size(); ++i) {
249 if (pages[i]) pages[i]->sortOrder = static_cast<int32_t>(i);
250 }
251 LOG_INFO(Section, "Inserted page '" + pageTitle + "' (" + pageGuid + ") at index " + std::to_string(index) + " in section '" + name + "'. Total pages: " + std::to_string(pages.size()));
252 return true;
253 }

283 /**
284 * @brief Removes a page by its persistent GUID, adjusts activePageIndex,
285 * maintains contiguous sortOrder numbering, and logs the operation.
286 *
287 * @param pageGuid GUID identifier of page to delete.
288 * @return bool True if a page was found and removed; false otherwise.
289 */
290 bool RemovePage(const std::string& pageGuid) {
291 auto it = std::find_if(pages.begin(), pages.end(), [&](const std::shared_ptr<CanvasPage>& p) {
292 return p && p->guid == pageGuid;
293 });
294
295 if (it != pages.end()) {
296 std::string pageTitle = (*it)->title;
297 size_t removedIndex = std::distance(pages.begin(), it);
298 pages.erase(it);
299
300 // Re-anchor active page index to a valid position
301 if (pages.empty()) {
302 activePageIndex = 0;
303 } else if (activePageIndex >= pages.size() || activePageIndex == removedIndex) {
304 activePageIndex = (pages.size() > 0) ? std::min(removedIndex, pages.size() - 1) : 0;
305 }
306
307 // Maintain contiguous 0-indexed sortOrder sequence
308 for (size_t i = 0; i < pages.size(); ++i) {
309 if (pages[i]) pages[i]->sortOrder = static_cast<int32_t>(i);
310 }
311 LOG_INFO(Section, "Removed page '" + pageTitle + "' (" + pageGuid + ") from section '" + name + "'. Remaining pages: " + std::to_string(pages.size()));
312 return true;
313 }
314 LOG_WARN_CODE(Section, FolioErrorCode::DocPageNotFound,
315 "RemovePage failed: Page not found in section '" + name + "': " + pageGuid);
316 return false;
317 }

336 /**
337 * @brief Moves a page from one index to another, keeping activePageIndex accurate and logging movement.
338 *
339 * INDEX CALCULATION & ACTIVE SELECTION MATH:
340 * When an element is moved from `fromIdx` to `toIdx`:
341 * - If `activePageIndex == fromIdx`, the active index shifts directly to `toIdx`.
342 * - If `fromIdx < activePageIndex <= toIdx`, preceding elements shift left: `activePageIndex--`.
343 * - If `toIdx <= activePageIndex < fromIdx`, following elements shift right: `activePageIndex++`.
344 *
345 * @param fromIdx Source index in `pages` vector.
346 * @param toIdx Destination index in `pages` vector.
347 * @return bool True if indices are valid and movement succeeded; false otherwise.
348 */
349 bool MovePage(size_t fromIdx, size_t toIdx) {
350 if (fromIdx >= pages.size() || toIdx >= pages.size() || fromIdx == toIdx) {
351 return false;
352 }
353 auto movedPage = pages[fromIdx];
354 pages.erase(pages.begin() + fromIdx);
355 pages.insert(pages.begin() + toIdx, movedPage);
356
357 // Update active page index if affected
358 if (activePageIndex == fromIdx) {
359 activePageIndex = toIdx;
360 } else if (fromIdx < activePageIndex && toIdx >= activePageIndex) {
361 activePageIndex--;
362 } else if (fromIdx > activePageIndex && toIdx <= activePageIndex) {
363 activePageIndex++;
364 }
365
366 // Re-index sortOrder
367 for (size_t i = 0; i < pages.size(); ++i) {
368 if (pages[i]) pages[i]->sortOrder = static_cast<int32_t>(i);
369 }
370 LOG_INFO(Section, "Reordered page '" + (movedPage ? movedPage->title : "Unknown") + "' in section '" + name + "' from " + std::to_string(fromIdx) + " to " + std::to_string(toIdx));
371 return true;
372 }
