46void PdfEngine::LoadDocument(const std::string& filePath) {
47 if (filePath.empty()) return;
48 if (document.filePath == filePath && (document.totalPages > 0 || isLoading)) return;
49
50 Clear();
51 document.filePath = filePath;
52 virtualizer.ResetViewport();
53 selectedPageIndex = -1;
54 currentSelection.Clear();
55
56 loadingDocPath = filePath;
57 loadingDocName = FileManager::GetFileName(filePath);
58 isLoading = true;
59 loadingFinished.store(false);
60 loadingFailed.store(false);
61 loadingCurrentPage.store(0);
62 loadingTotalPages.store(0);
63
64 LOG_INFO(PdfStorage, "PdfEngine starting background structure load: " + filePath);
65
66 GetGlobalThreadPool().Enqueue([this, filePath]() {
67 PdfDocSummary summary;
68 bool ok = PdfRenderer::InspectAndLoadDocStructure(filePath, summary, [this](int cur, int tot) {
69 loadingCurrentPage.store(cur);
70 loadingTotalPages.store(tot);
71 });
72
73 if (ok && summary.pageCount > 0) {
74 std::lock_guard<std::mutex> lock(pendingSummaryMutex);
75 pendingSummary = std::move(summary);
76 loadingFinished.store(true);
77 } else {
78 loadingFailed.store(true);
79 LOG_ERROR(PdfStorage, "PdfEngine background structure load failed for '" + filePath +
80 "': " + (ok ? "document reported 0 pages" : "InspectAndLoadDocStructure failed (PDFium missing or document unreadable)"));
81 }
82 });
83}

85void PdfEngine::Update(DocumentSession* session) {
86 // 1. Check if background loading thread finished
87 if (isLoading && loadingFinished.load()) {
88 isLoading = false;
89 std::lock_guard<std::mutex> lock(pendingSummaryMutex);
90 document.SetStructure(std::move(pendingSummary));
91
92 if (session) {
93 auto activePage = session->GetActivePage();
94 if (activePage) {
95 document.LoadBookmarksFromString(activePage->dedicatedPdfBookmarks);
96 document.LoadHighlightsFromString(activePage->dedicatedPdfHighlights);
97 }
98 }
99
100 LOG_INFO(PdfStorage, "PdfEngine background loading complete: " + document.filePath + " (" + std::to_string(document.totalPages) + " pages)");
101 }
102
103 // 2. Working-set maintenance: evict textures and home viewports of documents absent for >60,000ms
104 uint64_t nowMs = SDL_GetTicks();
105 constexpr uint64_t WORKING_SET_TIMEOUT_MS = 60000;
106 for (auto& [path, docState] : documentWorkingSet) {
107 if (path != document.filePath && (nowMs - docState.lastAccessTimeMs > WORKING_SET_TIMEOUT_MS)) {
108 docState.EvictTextures();
109 docState.HomeViewport();
110 docState.isLoaded = false;
111 }
112 }
113}

197bool PdfEngine::RenderPageToBlend2D(BLContext& ctx, int pageIndex, double xMm, double yMm, double scale, bool invert) {
198 if (document.filePath.empty() || pageIndex < 0 || pageIndex >= document.totalPages) {
199 return false;
200 }
201
202 CachedPdfViewerPage* cached = tileCache.GetOrLoadPage(document.filePath, pageIndex, document.docHighlights, 150.0, invert);
203 if (!cached || cached->image.is_empty()) {
204 return false;
205 }
206
207 double drawW_mm = cached->widthMm * scale;
208 double drawH_mm = cached->heightMm * scale;
209
210 ctx.save();
211 BLRect dstRect(xMm, yMm, drawW_mm, drawH_mm);
212 ctx.blit_image(dstRect, cached->image);
213 ctx.restore();
214 return true;
215}
