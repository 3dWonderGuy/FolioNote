1/**
2 * =========================================================================================
3 * @file core/objects/ink_container/ink_container.cpp
4 * @brief Implementation of Persistent Vector Ink Manager, Spatial Engine, and Renderer
5 * =========================================================================================
6 *
7 * MATHEMATICAL PRINCIPLES & WORKING PROCESS:
8 * ------------------------------------------
9 * 1. Bounding Box Calculation (UpdateBounds):
10 * - Local Stroke Envelope: Each stroke boundary is derived either directly from its
11 * pre-baked closed 2D polygon outline (BLPath get_bounding_box) or by expanding
12 * centerline points and segments:
13 * r = width / 2
14 * x_min = min(p0.x - r, p1.x - r), x_max = max(p0.x + r, p1.x + r)
15 * y_min = min(p0.y - r, p1.y - r), y_max = max(p0.y + r, p1.y + r)
16 * - Affine Coordinate Mapping:
17 * Corners of the local bounding rectangle are mapped through the 2D affine matrix M:
18 * p' = [m00*x + m10*y + m20, m01*x + m11*y + m21]^T
19 * The enclosing world AABB is constructed from extrema [min(x'_i), min(y'_i)] to
20 * [max(x'_i), max(y'_i)] with a 1.0 mm antialiasing safety padding.
21 *
22 * 2. Two-Tier Spatial Hit Testing (HitTest, HitTestCircle, HitTestSwept):
23 * - Broadphase: Fast rejection using the container's world-space AABB.
24 * - Coordinate Space Inversion: Query points, circles, and swept capsules are mapped
25 * into object-local space via the inverted affine matrix M^-1.
26 * - Narrowphase:
27 * a. BLPath::hit_test with BL_FILL_RULE_NON_ZERO for instant polygonal containment.
28 * b. StrokeCollisionEngine for segment-to-point and swept capsule continuous collision.
29 *
30 * 3. Affine Transform Baking (BakeTransform):
31 * - Geometric scale factor is computed via the determinant:
32 * scaleFactor = sqrt(|m00*m11 - m01*m10|)
33 * - All segment endpoints and centerline vertices are mapped through M, and thickness is
34 * scaled by scaleFactor.
35 * - StrokeOutlineBuilder rebuilds fresh, artifact-free 2D closed polygon contours (BLPath),
36 * and the transform matrix is reset to identity M = I.
37 *
38 * 4. Eraser Slicing (SliceStrokeAt):
39 * - Maps the circular eraser kernel into local space, scaling radius by 1 / hypot(m00, m01).
40 * - StrokeSlicer splits segment chains into surviving continuous sub-chains.
41 * - Surviving sub-chains are re-tessellated into smooth 2D closed polygon contours, with
42 * newly spawned fragments added to outNewFragments.
43 *
44 * 5. High-Fidelity Rendering (Render):
45 * - Viewport frustum culling discards off-screen containers before context dispatch.
46 * - Context transform is isolated via save() / restore().
47 * - Uses BL_FILL_RULE_NON_ZERO to ensure overlapping cursive loops fuse cleanly without hollows.
48 * - Renders with per-stroke blend modes (SrcOver, Multiply for Highlighters, Plus for Glow).
49 */

96void InkContainer::UpdateBounds() {
97 if (strokes.empty()) {
98 bounds = AABB{};
99 worldX = worldY = worldWidth = worldHeight = 0.0;
100 return;
101 }
102
103 double minX = 1e20, minY = 1e20, maxX = -1e20, maxY = -1e20;
104
105 for (auto& stroke : strokes) {
106 if (!stroke.outlinePath.is_empty()) {
107 BLBox box;
108 if (stroke.outlinePath.get_bounding_box(&box) == BL_SUCCESS) {
109 stroke.bounds = AABB{box.x0, box.y0, box.x1, box.y1};
110 minX = (std::min)(minX, box.x0);
111 minY = (std::min)(minY, box.y0);
112 maxX = (std::max)(maxX, box.x1);
113 maxY = (std::max)(maxY, box.y1);
114 }
115 } else if (!stroke.points.empty()) {
116 double sMinX = 1e20, sMinY = 1e20, sMaxX = -1e20, sMaxY = -1e20;
117 for (const auto& pt : stroke.points) {
118 sMinX = (std::min)(sMinX, pt.x - pt.r);
119 sMinY = (std::min)(sMinY, pt.y - pt.r);
120 sMaxX = (std::max)(sMaxX, pt.x + pt.r);
121 sMaxY = (std::max)(sMaxY, pt.y + pt.r);
122 }
123 stroke.bounds = AABB{sMinX, sMinY, sMaxX, sMaxY};
124 minX = (std::min)(minX, sMinX);
125 minY = (std::min)(minY, sMinY);
126 maxX = (std::max)(maxX, sMaxX);
127 maxY = (std::max)(maxY, sMaxY);
128 } else {
129 double sMinX = 1e20, sMinY = 1e20, sMaxX = -1e20, sMaxY = -1e20;
130 for (const auto& seg : stroke.segments) {
131 double hw = seg.width * 0.5;
132 sMinX = (std::min)({sMinX, seg.p0.x - hw, seg.p1.x - hw});
133 sMinY = (std::min)({sMinY, seg.p0.y - hw, seg.p1.y - hw});
134 sMaxX = (std::max)({sMaxX, seg.p0.x + hw, seg.p1.x + hw});
135 sMaxY = (std::max)({sMaxY, seg.p0.y + hw, seg.p1.y + hw});
136 }
137 stroke.bounds = AABB{sMinX, sMinY, sMaxX, sMaxY};
138 minX = (std::min)(minX, sMinX);
139 minY = (std::min)(minY, sMinY);
140 maxX = (std::max)(maxX, sMaxX);
141 maxY = (std::max)(maxY, sMaxY);
142 }
143 }
144
145 if (minX > maxX || minY > maxY) {
146 bounds = AABB{};
147 worldX = worldY = worldWidth = worldHeight = 0.0;
148 return;
149 }
150
151 constexpr double pad = 1.0;
152 BLPoint corners[4] = {
153 transform.map_point(minX - pad, minY - pad),
154 transform.map_point(maxX + pad, minY - pad),
155 transform.map_point(minX - pad, maxY + pad),
156 transform.map_point(maxX + pad, maxY + pad)
157 };
158
159 bounds = AABB{
160 (std::min)({corners[0].x, corners[1].x, corners[2].x, corners[3].x}),
161 (std::min)({corners[0].y, corners[1].y, corners[2].y, corners[3].y}),
162 (std::max)({corners[0].x, corners[1].x, corners[2].x, corners[3].x}),
163 (std::max)({corners[0].y, corners[1].y, corners[2].y, corners[3].y})
164 };
165
166 worldX = bounds.minX;
167 worldY = bounds.minY;
168 worldWidth = bounds.Width();
169 worldHeight = bounds.Height();
170}

172bool InkContainer::HitTest(double x, double y) const noexcept {
173 if (!isVisible || !isSelectable) return false;
174 if (!bounds.Contains(x, y)) return false;
175 if (isSelected) return true;
176
177 BLMatrix2D invTransform;
178 if (BLMatrix2D::invert(invTransform, transform) != BL_SUCCESS) return false;
179 BLPoint localPt = invTransform.map_point(x, y);
180
181 for (const auto& stroke : strokes) {
182 if (!stroke.bounds.Contains(localPt.x, localPt.y)) continue;
183
184 if (!stroke.outlinePath.is_empty()) {
185 BLHitTest hit = stroke.outlinePath.hit_test(localPt, BL_FILL_RULE_NON_ZERO);
186 if (hit == BL_HIT_TEST_IN) return true;
187 }
188
189 auto hit = StrokeCollisionEngine::HitTestStroke(stroke.segments, localPt.x, localPt.y, 1.5);
190 if (hit.hit) return true;
191 }
192 return false;
193}

195bool InkContainer::HitTestCircle(double cx, double cy, double radiusMm) const noexcept {
196 if (!isVisible) return false;
197
198 AABB queryBox(cx - radiusMm, cy - radiusMm, cx + radiusMm, cy + radiusMm);
199 if (!bounds.Intersects(queryBox)) return false;
200
201 BLMatrix2D invTransform;
202 if (BLMatrix2D::invert(invTransform, transform) != BL_SUCCESS) return false;
203 BLPoint localPt = invTransform.map_point(cx, cy);
204
205 double scale = std::hypot(transform.m00, transform.m01);
206 double localRadius = (scale > 1e-6) ? (radiusMm / scale) : radiusMm;
207
208 for (const auto& stroke : strokes) {
209 auto hit = StrokeCollisionEngine::HitTestStroke(stroke.segments, localPt.x, localPt.y, localRadius);
210 if (hit.hit) return true;
211
212 if (!stroke.outlinePath.is_empty()) {
213 BLHitTest blHit = stroke.outlinePath.hit_test(localPt, BL_FILL_RULE_NON_ZERO);
214 if (blHit == BL_HIT_TEST_IN) return true;
215 }
216 }
217 return false;
218}

220bool InkContainer::HitTestSwept(const Point2D& w0, const Point2D& w1, double radiusMm) const {
221 if (!isVisible) return false;
222
223 AABB sweptBox(
224 (std::min)(w0.x, w1.x) - radiusMm,
225 (std::min)(w0.y, w1.y) - radiusMm,
226 (std::max)(w0.x, w1.x) + radiusMm,
227 (std::max)(w0.y, w1.y) + radiusMm
228 );
229 if (!bounds.Intersects(sweptBox)) return false;
230
231 BLMatrix2D invTransform;
232 if (BLMatrix2D::invert(invTransform, transform) != BL_SUCCESS) return false;
233
234 BLPoint local0 = invTransform.map_point(w0.x, w0.y);
235 BLPoint local1 = invTransform.map_point(w1.x, w1.y);
236 Point2D lw0{ local0.x, local0.y };
237 Point2D lw1{ local1.x, local1.y };
238
239 double scale = std::hypot(transform.m00, transform.m01);
240 double localRadius = (scale > 1e-6) ? (radiusMm / scale) : radiusMm;
241
242 for (const auto& stroke : strokes) {
243 if (StrokeCollisionEngine::HitTestStrokeSwept(stroke.segments, lw0, lw1, localRadius)) {
244 return true;
245 }
246 }
247 return false;
248}

260void InkContainer::BakeTransform() {
261 if (transform.m00 == 1.0 && transform.m11 == 1.0 &&
262 transform.m01 == 0.0 && transform.m10 == 0.0 &&
263 transform.m20 == 0.0 && transform.m21 == 0.0) {
264 return;
265 }
266
267 double det = std::abs(transform.m00 * transform.m11 - transform.m01 * transform.m10);
268 double scaleFactor = (det > 1e-6) ? std::sqrt(det) : 1.0;
269
270 for (auto& stroke : strokes) {
271 for (auto& seg : stroke.segments) {
272 BLPoint p0 = transform.map_point(seg.p0.x, seg.p0.y);
273 BLPoint p1 = transform.map_point(seg.p1.x, seg.p1.y);
274 seg.p0 = Point2D(p0.x, p0.y);
275 seg.p1 = Point2D(p1.x, p1.y);
276 seg.width = static_cast<float>(seg.width * scaleFactor);
277 }
278
279 for (auto& pt : stroke.points) {
280 BLPoint p = transform.map_point(pt.x, pt.y);
281 pt.x = p.x;
282 pt.y = p.y;
283 pt.r = static_cast<float>(pt.r * scaleFactor);
284 }
285
286 stroke.baseWidthMm *= static_cast<float>(scaleFactor);
287
288 if (!stroke.points.empty()) {
289 std::vector<StrokeOutlineBuilder::InputPoint> pts;
290 pts.reserve(stroke.points.size());
291 for (const auto& pt : stroke.points) {
292 pts.push_back({ pt.x, pt.y, pt.r * 2.0f });
293 }
294 stroke.outlinePath = StrokeOutlineBuilder::BuildOutline(pts, stroke.capType, stroke.strokePattern);
295 }
296 }
297
298 transform = BLMatrix2D::make_identity();
299 renderDirty = true;
300 UpdateBounds();
301}

307void InkContainer::Render(BLContext& ctx, const Viewport& viewport) const {
308 if (!isVisible || opacity <= 0.0f) return;
309 if (!bounds.Intersects(viewport.bounds)) return;
310
311 renderDirty = false;
312
313 ctx.save();
314 ctx.apply_transform(transform);
315 ctx.set_fill_rule(BL_FILL_RULE_NON_ZERO);
316
317 for (const auto& stroke : strokes) {
318 if (stroke.outlinePath.is_empty() && stroke.segments.empty()) continue;
319
320 BLCompOp compOp = (isHighlighter || stroke.blendMode == BlendMode::Multiply)
321 ? BL_COMP_OP_MULTIPLY
322 : stroke.GetBlend2DCompOp();
323 ctx.set_comp_op(compOp);
324 ctx.set_fill_style(stroke.color);
325
326 if (!stroke.outlinePath.is_empty()) {
327 ctx.fill_path(stroke.outlinePath);
328 } else if (!stroke.points.empty()) {
329 std::vector<StrokeOutlineBuilder::InputPoint> pts;
330 pts.reserve(stroke.points.size());
331 for (const auto& pt : stroke.points) {
332 pts.push_back({ pt.x, pt.y, pt.r * 2.0f });
333 }
334 BLPath fallbackOutline = StrokeOutlineBuilder::BuildOutline(pts, stroke.capType, stroke.strokePattern);
335 ctx.fill_path(fallbackOutline);
336 } else if (stroke.segments.size() == 1) {
337 ctx.set_stroke_caps(BL_STROKE_CAP_ROUND);
338 ctx.set_stroke_width(stroke.segments[0].width);
339 ctx.stroke_line(stroke.segments[0].p0.x, stroke.segments[0].p0.y,
340 stroke.segments[0].p1.x, stroke.segments[0].p1.y);
341 }
342 }
343
344 ctx.restore();
345}

375bool InkContainer::SliceStrokeAt(double worldX, double worldY, double radius,
376 std::vector<std::shared_ptr<InkContainer>>& outNewFragments) {
377 if (strokes.empty()) return false;
378
379 AABB eraserAABB(worldX - radius, worldY - radius, worldX + radius, worldY + radius);
380 if (!bounds.Intersects(eraserAABB)) return false;
381
382 BLMatrix2D invTransform;
383 if (BLMatrix2D::invert(invTransform, transform) != BL_SUCCESS) return false;
384 BLPoint localCenter = invTransform.map_point(worldX, worldY);
385
386 double scale = std::hypot(transform.m00, transform.m01);
387 double localRadius = (scale > 1e-6) ? (radius / scale) : radius;
388
389 std::vector<Stroke> newStrokes;
390 bool anyModified = false;
391
392 for (const auto& stroke : strokes) {
393 std::vector<std::vector<Segment1D>> subChains;
394 bool strokeModified = StrokeSlicer::SliceSegments(
395 stroke.segments, localCenter.x, localCenter.y, localRadius, subChains);
396
397 if (strokeModified) {
398 anyModified = true;
399 for (size_t subIdx = 0; subIdx < subChains.size(); ++subIdx) {
400 auto& subChain = subChains[subIdx];
401 if (subChain.empty()) continue;
402
403 Stroke newStroke;
404 newStroke.color = stroke.color;
405 newStroke.baseWidthMm = stroke.baseWidthMm;
406 newStroke.strokePattern = stroke.strokePattern;
407 newStroke.capType = stroke.capType;
408 newStroke.blendMode = stroke.blendMode;
409 newStroke.penType = stroke.penType;
410 newStroke.opacity = stroke.opacity;
411 newStroke.segments = std::move(subChain);
412
413 // Reconstruct points and polygon outline
414 std::vector<StrokeOutlineBuilder::InputPoint> pts;
415 pts.reserve(newStroke.segments.size() + 1);
416 pts.push_back({ newStroke.segments[0].p0.x, newStroke.segments[0].p0.y, newStroke.segments[0].width });
417 newStroke.points.push_back({ newStroke.segments[0].p0.x, newStroke.segments[0].p0.y, newStroke.segments[0].width * 0.5f });
418
419 for (const auto& s : newStroke.segments) {
420 pts.push_back({ s.p1.x, s.p1.y, s.width });
421 newStroke.points.push_back({ s.p1.x, s.p1.y, s.width * 0.5f });
422 }
423
424 newStroke.outlinePath = StrokeOutlineBuilder::BuildOutline(pts, newStroke.capType, newStroke.strokePattern);
425 newStrokes.push_back(std::move(newStroke));
426 }
427 } else {
428 newStrokes.push_back(stroke);
429 }
430 }
431
432 if (anyModified) {
433 strokes = std::move(newStrokes);
434 renderDirty = true;
435 UpdateBounds();
436 return true;
437 }
438 return false;
439}
