24 if (!engine) return;
25 if (clearSelection) {

167BatchEraseCommand::BatchEraseCommand(
168 std::vector<std::shared_ptr<CanvasObject>> deleted,

170) : deletedObjects(std::move(deleted)), slicedStrokes(std::move(sliced)) {}

172bool BatchEraseCommand::IsEmpty() const {
173 return deletedObjects.empty() && slicedStrokes.empty();
174}

176void BatchEraseCommand::Execute(CanvasPage& page, CanvasEngine* engine) {
177 for (const auto& del : deletedObjects) {
178 if (del) {
179 del->isVisible = 0;
180 del->isSelectable = 0;
181 del->isSelected = 0;
182 page.RemoveObject(del);
183 }
184 }
185 for (const auto& s : slicedStrokes) {
186 if (s.originalStroke) {
187 s.originalStroke->isVisible = 0;
188 s.originalStroke->isSelectable = 0;
189 s.originalStroke->isSelected = 0;
190 page.RemoveObject(s.originalStroke);
191 }
192 for (const auto& frag : s.generatedFragments) {
193 if (frag) {
194 frag->isVisible = 1;
195 frag->isSelectable = 1;
196 frag->isSelected = 0;
197 page.AddObject(frag);
198 }
199 }
200 }
201 InvalidateEngine(engine, /*clearSelection=*/true);
202}
203
204void BatchEraseCommand::Undo(CanvasPage& page, CanvasEngine* engine) {
205 // Reverse slices: Remove generated fragments, restore original stroke
206 for (const auto& s : slicedStrokes) {
207 for (const auto& frag : s.generatedFragments) {
208 if (frag) {
209 frag->isVisible = 0;
210 frag->isSelectable = 0;
211 frag->isSelected = 0;
212 page.RemoveObject(frag);
213 }
214 }
215 if (s.originalStroke) {
216 s.originalStroke->isVisible = 1;
217 s.originalStroke->isSelectable = 1;
218 s.originalStroke->isSelected = 0;
219 page.AddObject(s.originalStroke);
220 }
221 }
222 // Restore deleted objects
223 for (const auto& del : deletedObjects) {
224 if (del) {
225 del->isVisible = 1;
226 del->isSelectable = 1;
227 del->isSelected = 0;
228 page.AddObject(del);
229 }
230 }
231 InvalidateEngine(engine, /*clearSelection=*/true);
232}

238 }
239 for (const auto& s : slicedStrokes) {
240 if (s.originalStroke) box.Merge(s.originalStroke->bounds);
241 }
242 return box;
243}

485ModifyTextCommand::ModifyTextCommand(uint32_t uid, std::string prevText, std::string nextText,
486 double prevW, double nextW, double prevH, double nextH)
487 : textBoxUid(uid),
488 previousText(std::move(prevText)),
489 newText(std::move(nextText)),
490 previousWidth(prevW),
491 newWidth(nextW),

494
495void ModifyTextCommand::Execute(CanvasPage& page, CanvasEngine* engine) {
496 auto obj = page.FindObjectByUid(textBoxUid);
497 if (obj && obj->type == ObjectType::Text) {
498 auto tb = std::static_pointer_cast<TextBoxObject>(obj);
499 tb->text = newText;
500 tb->worldWidth = newWidth;
501 tb->worldHeight = newHeight;
502 tb->SyncTextToRuns();
503 tb->UpdateBounds();
504 page.spatialIndex.Remove(tb->uid);
505 page.spatialIndex.Insert(tb->uid, tb->bounds);
506 page.isModified = true;
507 if (engine && engine->textEditor.GetTarget() == tb.get()) {
508 engine->textEditor.ReflowLayout();
509 }
510 }
511 InvalidateEngine(engine, /*clearSelection=*/false);
512}

558void RelinkAttachmentCommand::Execute(CanvasPage& page, CanvasEngine* engine) {
559 auto obj = page.FindObjectByUid(attachmentUid);
560 if (obj && obj->type == ObjectType::AttachmentFile) {
561 auto attach = std::static_pointer_cast<AttachmentObject>(obj);
562 attach->filePath = newPath;
563 if (!newDisplayName.empty()) {
564 attach->displayName = newDisplayName;
565 }
566 attach->isEmbedded = newEmbedded;
567 std::string resolved = attach->isEmbedded ? FileManager::ResolveAssetPath(attach->filePath) : attach->filePath;
568 attach->SetFileValid(!resolved.empty() && FileManager::Exists(resolved));
569 page.isModified = true;
570 }
571 InvalidateEngine(engine, /*clearSelection=*/false);
572}
