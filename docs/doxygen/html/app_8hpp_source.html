118 /**
119 * @brief Pre-indexes and ensures existence of the dedicated user Imports directory.

146 /**
147 * @brief Stages an external document into the Imports folder and presents the import placement dialog.
148 *
149 * GENERAL WORKING PROCESS:
150 * 1. Inspects the source file to verify existence and `.pdf` format.
151 * 2. If the file is not already inside `importsDirectory`, copies it into `importsDirectory`.
152 * If a file with the same filename already exists, appends an incremental numeric suffix
153 * (e.g., `document_1.pdf`) to prevent accidental overwrites of existing imports.
154 * 3. Marks the destination canonical path in `knownImportedPdfs` so background polling ignores it.
155 * 4. Calls `pdfImportModal.Open(dest, &session)` which launches the interactive UI modal
156 * asking the user where to place the document (which notebook, section, page, or standalone viewer).
157 *
158 * @param sourcePath Canonical or relative filesystem path to the external PDF document.
159 * @return true if the document was successfully staged and modal was opened; false otherwise.
160 */
161 bool ImportExternalPdf(const std::string& sourcePath) {
162 if (sourcePath.empty()) return false;
163 std::error_code ec;
164 std::filesystem::path src(sourcePath);
165 if (!std::filesystem::exists(src, ec)) return false;
166
167 std::filesystem::path dest = src;
168 if (!importsDirectory.empty()) {
169 std::filesystem::path target = importsDirectory / src.filename();
170 // Avoid copying onto itself if the file was already placed directly into Imports
171 if (std::filesystem::canonical(src, ec) != std::filesystem::canonical(target, ec)) {
172 int counter = 1;
173 while (std::filesystem::exists(target, ec)) {
174 std::string stem = src.stem().string() + "_" + std::to_string(counter++);
175 target = importsDirectory / (stem + src.extension().string());
176 }
177 std::filesystem::copy_file(src, target, std::filesystem::copy_options::overwrite_existing, ec);
178 if (!ec) dest = target;
179 }
180 }
181
182 std::string canonicalDest = std::filesystem::canonical(dest, ec).string();
183 if (canonicalDest.empty()) canonicalDest = dest.string();
184 knownImportedPdfs.insert(canonicalDest);
185
186 pdfImportModal.Open(dest.string(), &session);
187 return true;
188 }

190 /**
191 * @brief Scans the Imports directory for incoming documents (e.g. from 'Print to FolioNote').
192 *
193 * GENERAL WORKING PROCESS:
194 * 1. Guard check: If `pdfImportModal` is already open, avoids popping an overlapping modal.
195 * 2. Iterates over regular `.pdf` files located in `importsDirectory`.
196 * 3. For any file not present in `knownImportedPdfs`:
197 * - Checks that the file size is greater than 0 bytes.
198 * - Verifies file is not locked: Attempts an exclusive binary stream open. If locked by the
199 * Windows Print Spooler or browser during an active print write, defers until next scan cycle.
200 * - Adds path to `knownImportedPdfs` and triggers `pdfImportModal.Open(...)`.
201 * - Returns true immediately to process one document at a time.
202 *
203 * @return true if a new incoming PDF document was detected and opened; false if none.
204 */
205 bool CheckImportFolder() {
206 if (pdfImportModal.isOpen) return false;
207 if (importsDirectory.empty()) return false;
208
209 std::error_code ec;
210 if (!std::filesystem::exists(importsDirectory, ec)) return false;
211
212 for (const auto& entry : std::filesystem::directory_iterator(importsDirectory, ec)) {
213 if (!entry.is_regular_file()) continue;
214
215 std::string ext = entry.path().extension().string();
216 for (auto& c : ext) c = static_cast<char>(::tolower(static_cast<unsigned char>(c)));
217 if (ext != ".pdf") continue;
218
219 std::string canonicalPath = std::filesystem::canonical(entry.path(), ec).string();
220 if (canonicalPath.empty()) canonicalPath = entry.path().string();
221
222 if (knownImportedPdfs.find(canonicalPath) != knownImportedPdfs.end()) {
223 continue;
224 }
225
226 // Ensure file is finished writing (not locked by print spooler / browser download)
227 auto fileSize = std::filesystem::file_size(entry.path(), ec);
228 if (ec || fileSize == 0) {
229 continue; // Still spooling or 0 bytes
230 }
231
232 // Test if file can be opened for reading
233 std::ifstream testStream(entry.path(), std::ios::binary | std::ios::in);
234 if (!testStream.is_open()) {
235 continue; // File locked by another process
236 }
237 testStream.close();
238
239 // Register and launch import modal
240 knownImportedPdfs.insert(canonicalPath);
241 pdfImportModal.Open(entry.path().string(), &session);
242 devTelemetry.LogEvent("Discovered incoming virtual print document: " + entry.path().filename().string(), LogCategory::System);
243 return true;
244 }
245 return false;
246 }

735 /**
736 * @brief Dispatches global keyboard shortcuts cleanly through UIActionRegistry.
737 *
738 * Evaluates keydown combinations when ImGui keyboard capture is not asserted.
739 * Prevents duplicate shortcut handling logic across multiple frame event polling loops.
740 *
741 * @param event The active SDL event of type SDL_EVENT_KEY_DOWN.
742 * @return true if an action or shortcut was recognized and handled; false otherwise.
743 */
744 bool HandleGlobalShortcut(const SDL_Event& event) {
745 if (event.type != SDL_EVENT_KEY_DOWN) return false;
746
747 const SDL_Keymod mod = SDL_GetModState();
748 const bool isCtrl = (mod & SDL_KMOD_CTRL) != 0;
749 const bool isShift = (mod & SDL_KMOD_SHIFT) != 0;
750 const auto key = event.key.key;
751 auto& reg = Folio::UIActionRegistry::Instance();
752
753 // Spotlight Command Palette (Ctrl+K or Ctrl+P) - allowed even when ImGui has focus
754 if (isCtrl && (key == SDLK_K || key == SDLK_P)) {
755 reg.Execute("tools.command_palette");
756 return true;
757 }
758
759 // Developer overlays and diagnostic toggles
760 if (key == SDLK_F3) { reg.Execute("tools.telemetry"); return true; }
761 if (key == SDLK_F4) { canvas.devMode = !canvas.devMode; canvas.isDirty = true; return true; }
762 if (key == SDLK_F5) { reg.Execute("tools.tuning"); return true; }
763 if (key == SDLK_F6) { reg.Execute("tools.toolbar_demo"); return true; }
764 if (key == SDLK_F11) { reg.Execute("view.toggle_fullscreen"); return true; }
765 if (isCtrl && key == SDLK_F1) { reg.Execute("view.cycle_ribbon_mode"); return true; }
766
767 // If Dear ImGui active widgets (like text boxes) want keyboard focus, bypass general editor hotkeys
768 if (ImGui::GetIO().WantCaptureKeyboard) {
769 return false;
770 }
771
772 // Clipboard & Selection hotkeys
773 if (isCtrl && key == SDLK_C) { return reg.Execute("edit.copy"); }
774 if (isCtrl && key == SDLK_X) { return reg.Execute("edit.cut"); }
775 if (isCtrl && key == SDLK_V) { return reg.Execute("edit.paste"); }
776 if (isCtrl && key == SDLK_D) { return reg.Execute("edit.duplicate"); }
777 if (isCtrl && key == SDLK_A) { return reg.Execute("edit.select_all"); }
778 if (key == SDLK_ESCAPE) { return reg.Execute("edit.clear_selection"); }
779 if (key == SDLK_DELETE) { return reg.Execute("edit.delete"); }
780
781 // Undo / Redo
782 if (isCtrl && key == SDLK_Z) {
783 return isShift ? reg.Execute("edit.redo") : reg.Execute("edit.undo");
784 }
785 if (isCtrl && key == SDLK_Y) { return reg.Execute("edit.redo"); }
786
787 // Document & Navigation hotkeys
788 if (isCtrl && key == SDLK_N) { return reg.Execute("page.new"); }
789 if (isCtrl && key == SDLK_S) { return reg.Execute("doc.save"); }
790 if (isCtrl && key == SDLK_H) { return reg.Execute("view.notebook_hub"); }
791 if (isCtrl && key == SDLK_I) { return reg.Execute("canvas.invert"); }
792 if (key == SDLK_PAGEDOWN) { return reg.Execute("page.next"); }
793 if (key == SDLK_PAGEUP) { return reg.Execute("page.prev"); }
794 if (key == SDLK_HOME) { return reg.Execute("canvas.reset_view"); }
795
796 // Canvas Viewport hotkeys
797 if (isCtrl && (key == SDLK_PLUS || key == SDLK_EQUALS)) { return reg.Execute("canvas.zoom_in"); }
798 if (isCtrl && key == SDLK_MINUS) { return reg.Execute("canvas.zoom_out"); }
799 if (isCtrl && key == SDLK_0) { return reg.Execute("canvas.zoom_reset"); }
800
801 return false;
802 }
