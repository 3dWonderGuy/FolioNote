14bool NotebookSearchIndex::InitSchema(sqlite3* db) {
15 if (!db) return false;
16
17 // 1. Relational content index table (stores spatial coordinates, timestamps, and foreign keys)
18 const char* schema = R"(
19 CREATE TABLE IF NOT EXISTS search_catalog (
20 id INTEGER PRIMARY KEY AUTOINCREMENT,
21 notebook_guid TEXT NOT NULL,
22 section_guid TEXT NOT NULL,
23 page_guid TEXT NOT NULL,
24 object_guid TEXT NOT NULL,
25 content_type INTEGER NOT NULL,
26 raw_content TEXT NOT NULL,
27 tags TEXT NOT NULL,
28 min_x REAL NOT NULL DEFAULT 0.0,
29 min_y REAL NOT NULL DEFAULT 0.0,
30 max_x REAL NOT NULL DEFAULT 0.0,
31 max_y REAL NOT NULL DEFAULT 0.0,
32 created_at INTEGER NOT NULL
33 );
34
35 CREATE INDEX IF NOT EXISTS idx_search_page ON search_catalog(page_guid);
36 CREATE INDEX IF NOT EXISTS idx_search_section ON search_catalog(section_guid);
37 CREATE INDEX IF NOT EXISTS idx_search_notebook ON search_catalog(notebook_guid);
38 CREATE INDEX IF NOT EXISTS idx_search_tags ON search_catalog(tags);
39
40 -- FTS5 Full-Text Search Virtual Table
41 CREATE VIRTUAL TABLE IF NOT EXISTS search_fts USING fts5(
42 raw_content,
43 tags,
44 content='search_catalog',
45 content_rowid='id',
46 tokenize='porter unicode61'
47 );
48
49 -- Sync Triggers to keep FTS5 synchronized with search_catalog automatically
50 CREATE TRIGGER IF NOT EXISTS trg_search_catalog_ai AFTER INSERT ON search_catalog BEGIN
51 INSERT INTO search_fts(rowid, raw_content, tags) VALUES (new.id, new.raw_content, new.tags);
52 END;
53
54 CREATE TRIGGER IF NOT EXISTS trg_search_catalog_ad AFTER DELETE ON search_catalog BEGIN
55 INSERT INTO search_fts(search_fts, rowid, raw_content, tags) VALUES('delete', old.id, old.raw_content, old.tags);
56 END;
57
58 CREATE TRIGGER IF NOT EXISTS trg_search_catalog_au AFTER UPDATE ON search_catalog BEGIN
59 INSERT INTO search_fts(search_fts, rowid, raw_content, tags) VALUES('delete', old.id, old.raw_content, old.tags);
60 INSERT INTO search_fts(rowid, raw_content, tags) VALUES (new.id, new.raw_content, new.tags);
61 END;
62 )";
63
64 char* err = nullptr;
65 int rc = sqlite3_exec(db, schema, nullptr, nullptr, &err);
66 if (rc != SQLITE_OK) {
67 LOG_ERROR(SearchIndex, "Failed to initialize search index schema: " + std::string(err ? err : "Unknown"));
68 if (err) sqlite3_free(err);
69 return false;
70 }
71
72 LOG_INFO(SearchIndex, "Search FTS5 and catalog schema initialized successfully.");
73 return true;
74}

79std::vector<std::string> NotebookSearchIndex::ExtractHashtags(const std::string& text) {
80 std::vector<std::string> tags;
81 static const std::regex tagRegex(R"(#([a-zA-Z0-9_-]+))");
82 auto words_begin = std::sregex_iterator(text.begin(), text.end(), tagRegex);
83 auto words_end = std::sregex_iterator();
84
85 for (std::sregex_iterator i = words_begin; i != words_end; ++i) {
86 std::smatch match = *i;
87 tags.push_back(match.str());
88 }
89 return tags;

136bool NotebookSearchIndex::IndexPage(sqlite3* db, const std::string& notebookGuid,
137 const std::string& sectionGuid, const CanvasPage& page) {
138 if (!db) return false;
139
140 // 1. Remove prior entries for this page to prevent duplicate search results
141 RemovePage(db, page.guid);
142
143 // 2. Index Page Title
144 if (!page.title.empty()) {
145 SearchIndexEntry titleEntry;
146 titleEntry.notebookGuid = notebookGuid;
147 titleEntry.sectionGuid = sectionGuid;
148 titleEntry.pageGuid = page.guid;
149 titleEntry.objectGuid = "";
150 titleEntry.contentType = ContentType::PageTitle;
151 titleEntry.rawContent = page.title;
152
153 auto titleTags = ExtractHashtags(page.title);
154 std::ostringstream tagStream;
155 for (const auto& t : titleTags) tagStream << t << " ";
156 titleEntry.tags = tagStream.str();
157 titleEntry.spatialBounds = AABB(0, 0, 1000, 100);
158
159 IndexEntry(db, titleEntry);
160 }
161
162 // 3. Scan Canvas Objects (Text Boxes, Tags, future OCR)
163 for (const auto& obj : page.objects) {
164 if (!obj) continue;
165
166 if (obj->type == ObjectType::Text) {
167 auto txt = std::static_pointer_cast<TextBoxObject>(obj);
168 if (!txt->text.empty()) {
169 SearchIndexEntry textEntry;
170 textEntry.notebookGuid = notebookGuid;
171 textEntry.sectionGuid = sectionGuid;
172 textEntry.pageGuid = page.guid;
173 textEntry.objectGuid = txt->guuid;
174 textEntry.contentType = ContentType::TextBox;
175 textEntry.rawContent = txt->text;
176
177 auto extracted = ExtractHashtags(txt->text);
178 std::ostringstream tagStream;
179 for (const auto& t : extracted) tagStream << t << " ";
180 textEntry.tags = tagStream.str();
181 textEntry.spatialBounds = txt->bounds;
182
183 IndexEntry(db, textEntry);
184 }
185 }
186 }
187
188 LOG_INFO(SearchIndex, "Indexed page content for: '" + page.title + "' (" + page.guid + ")");
189 return true;

225 * @brief Performs ranked full-text search with spatial coordinates and snippet extraction.
226 */
227std::vector<SearchResult> NotebookSearchIndex::Search(sqlite3* db,
228 const std::string& notebookGuid,
229 const std::string& queryText,
230 const std::string& tagFilter,
231 int limit) {
232 std::vector<SearchResult> results;
233 if (!db || (queryText.empty() && tagFilter.empty())) return results;
234
235 std::string sql;
236 sqlite3_stmt* stmt = nullptr;
237
238 if (!queryText.empty()) {
239 // FTS5 MATCH query with BM25 ranking and snippet generation
240 sql = R"(
241 SELECT
242 c.notebook_guid, c.section_guid, s.name as section_name,
243 c.page_guid, p.title as page_title, c.object_guid,
244 c.content_type, snippet(search_fts, 0, '<b>', '</b>', '...', 12) as snippet_text,
245 c.raw_content, c.tags,
246 c.min_x, c.min_y, c.max_x, c.max_y,
247 bm25(search_fts) as rank
248 FROM search_fts
249 JOIN search_catalog c ON c.id = search_fts.rowid
250 LEFT JOIN sections s ON s.guid = c.section_guid
251 LEFT JOIN pages p ON p.guid = c.page_guid
252 WHERE search_fts MATCH ?
253 AND c.notebook_guid = ?
254 ORDER BY rank
255 LIMIT ?;
256 )";
257
258 if (sqlite3_prepare_v2(db, sql.c_str(), -1, &stmt, nullptr) != SQLITE_OK) {
259 LOG_ERROR(SearchIndex, "Failed to prepare FTS search query: " + std::string(sqlite3_errmsg(db)));
260 return results;
261 }
262
263 // Format query for prefix search (e.g. "calc" -> "calc*")
264 std::string formattedQuery = queryText;
265 if (!formattedQuery.empty() && formattedQuery.back() != '*' && formattedQuery.find(' ') == std::string::npos) {
266 formattedQuery += "*";
267 }
268
269 sqlite3_bind_text(stmt, 1, formattedQuery.c_str(), -1, SQLITE_TRANSIENT);
270 sqlite3_bind_text(stmt, 2, notebookGuid.c_str(), -1, SQLITE_TRANSIENT);
271 sqlite3_bind_int(stmt, 3, limit);
272
273 } else {
274 // Tag-only lookup
275 sql = R"(
276 SELECT
277 c.notebook_guid, c.section_guid, s.name as section_name,
278 c.page_guid, p.title as page_title, c.object_guid,
279 c.content_type, c.raw_content as snippet_text,
280 c.raw_content, c.tags,
281 c.min_x, c.min_y, c.max_x, c.max_y,
282 0.0 as rank
283 FROM search_catalog c
284 LEFT JOIN sections s ON s.guid = c.section_guid
285 LEFT JOIN pages p ON p.guid = c.page_guid
286 WHERE c.tags LIKE ?
287 AND c.notebook_guid = ?
288 LIMIT ?;
289 )";
290
291 if (sqlite3_prepare_v2(db, sql.c_str(), -1, &stmt, nullptr) != SQLITE_OK) {
292 LOG_ERROR(SearchIndex, "Failed to prepare tag search query: " + std::string(sqlite3_errmsg(db)));
293 return results;
294 }
295
296 std::string likeTag = "%" + tagFilter + "%";
297 sqlite3_bind_text(stmt, 1, likeTag.c_str(), -1, SQLITE_TRANSIENT);
298 sqlite3_bind_text(stmt, 2, notebookGuid.c_str(), -1, SQLITE_TRANSIENT);
299 sqlite3_bind_int(stmt, 3, limit);
300 }
301
302 while (sqlite3_step(stmt) == SQLITE_ROW) {
303 SearchResult res;
304 res.notebookGuid = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 0));
305 res.sectionGuid = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 1));
306 const char* secName = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 2));
307 res.sectionName = secName ? secName : "Untitled Section";
308 res.pageGuid = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 3));
309 const char* pTitle = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 4));
310 res.pageTitle = pTitle ? pTitle : "Untitled Page";
311 res.objectGuid = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 5));
312 res.contentType = static_cast<ContentType>(sqlite3_column_int(stmt, 6));
313
314 const char* snip = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 7));
315 res.snippet = snip ? snip : "";
316 const char* raw = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 8));
317 res.rawContent = raw ? raw : "";
318 const char* tg = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 9));
319 res.tags = tg ? tg : "";
320
321 res.spatialBounds.minX = sqlite3_column_double(stmt, 10);
322 res.spatialBounds.minY = sqlite3_column_double(stmt, 11);
323 res.spatialBounds.maxX = sqlite3_column_double(stmt, 12);
324 res.spatialBounds.maxY = sqlite3_column_double(stmt, 13);
325 res.rankScore = sqlite3_column_double(stmt, 14);
326
327 results.push_back(std::move(res));
328 }
329
330 sqlite3_finalize(stmt);
331 LOG_INFO(SearchIndex, "Search query '" + queryText + "' returned " + std::to_string(results.size()) + " matches.");
332 return results;
333}

335/**
336 * @brief Returns tag summary aggregation for tag clouds in the sidebar.
337 */
338std::vector<TagSummary> NotebookSearchIndex::GetNotebookTags(sqlite3* db, const std::string& notebookGuid) {
339 std::vector<TagSummary> summaries;
340 if (!db) return summaries;
341
342 const char* sql = R"(
343 SELECT tags, page_guid FROM search_catalog
344 WHERE notebook_guid = ? AND tags != '';
345 )";
346
347 sqlite3_stmt* stmt = nullptr;
348 if (sqlite3_prepare_v2(db, sql, -1, &stmt, nullptr) != SQLITE_OK) return summaries;
349
350 sqlite3_bind_text(stmt, 1, notebookGuid.c_str(), -1, SQLITE_TRANSIENT);
351
352 std::unordered_map<std::string, TagSummary> tagMap;
353
354 while (sqlite3_step(stmt) == SQLITE_ROW) {
355 std::string tagsStr = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 0));
356 std::string pageGuid = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 1));
357
358 std::istringstream iss(tagsStr);
359 std::string tag;
360 while (iss >> tag) {
361 if (tag.empty()) continue;
362 auto& summary = tagMap[tag];
363 summary.tag = tag;
364 summary.count++;
365 if (std::find(summary.pageGuids.begin(), summary.pageGuids.end(), pageGuid) == summary.pageGuids.end()) {
366 summary.pageGuids.push_back(pageGuid);
367 }
368 }
369 }
370
371 sqlite3_finalize(stmt);
372
373 summaries.reserve(tagMap.size());
374 for (auto& pair : tagMap) {
375 summaries.push_back(std::move(pair.second));
376 }
377 return summaries;
378}
