23/**
24 * @brief Termination style drawn at each endpoint of a SmartArrowObject.
25 *
26 * Drawing math (all computed relative to line direction angle θ):
27 * cosA = cos(θ), sinA = sin(θ), perpX = -sinA, perpY = cosA
28 *
29 * Triangle: filled solid triangle — tip at endpoint, base at (tip - size * dir)
30 * Stealth: winged arrowhead with notched indentation at 75% shaft
31 * Open: two-stroke open chevron, no fill (wireframe V)
32 * Circle: filled circle at endpoint tip, radius = size * 0.4
33 */
34enum class ArrowHeadType : uint8_t {

36 Triangle = 1, ///< Classic solid filled equilateral arrowhead

38 Open = 3, ///< Two-stroke open chevron (wireframe, no fill)
39 Circle = 4, ///< Circular terminal dot
40};
