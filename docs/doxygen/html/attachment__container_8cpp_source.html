91/**
92 * @brief Applies an affine transformation matrix to the attachment chip.
93 *
94 * MATHEMATICAL PROCESS & INVARIANTS:
95 * -----------------------------------
96 * Attachment chips are fixed-dimension UI cards (GizmoStyle::MoveOnly). They must
97 * never scale, stretch, or rotate with the canvas camera or group transformations:
98 * 1. Project the current top-left anchor point (worldX, worldY) through the incoming
99 * 2x3 affine matrix `M`:
100 * newX = (worldX * M.m00) + (worldY * M.m10) + M.m20
101 * newY = (worldX * M.m01) + (worldY * M.m11) + M.m21
102 * 2. Calculate purely translational displacement deltas:
103 * dx = newX - worldX
104 * dy = newY - worldY
105 * 3. Construct a translation-only affine matrix `T = BLMatrix2D::make_translation(dx, dy)`.
106 * This completely discards scale, shear, and rotation components while keeping the
107 * chip position pinned accurately in world space.
108 * 4. Post-multiplies `T` into the local `transform` and recomputes axis-aligned bounding box.
109 *
110 * @param[in] matrix Incoming 2D transformation matrix from canvas or gizmo manipulation.
111 */
112void AttachmentObject::ApplyTransform(const BLMatrix2D &matrix) {
113 const double newX = (worldX * matrix.m00) + (worldY * matrix.m10) + matrix.m20;
114 const double newY = (worldX * matrix.m01) + (worldY * matrix.m11) + matrix.m21;
115
116 const double dx = newX - worldX;
117 const double dy = newY - worldY;
118
119 BLMatrix2D translationOnly = BLMatrix2D::make_translation(dx, dy);
120 transform.post_transform(translationOnly);
121 UpdateBounds();
122}

124/**
125 * @brief Commits accumulated translation deltas into world coordinates and resets transform to identity.
126 *
127 * MATHEMATICAL PROCESS:
128 * Extracts translation offsets (m20, m21) from the affine transform matrix:
129 * worldX += transform.m20;
130 * worldY += transform.m21;
131 * Resets `transform` to identity to prevent unbounded matrix accumulation.
132 */
133void AttachmentObject::BakeTransform() {
134 worldX += transform.m20;
135 worldY += transform.m21;
136 transform = BLMatrix2D::make_identity();
137 UpdateBounds();
138}

140/**
141 * @brief Renders the interactive attachment file card into the Blend2D context.
142 *
143 * GEOMETRIC LAYOUT & MATHEMATICAL COMPOSITION (Physical Canvas Millimeters):
144 * --------------------------------------------------------------------------
145 * The attachment card is rendered in world millimeter space:
146 * - Origin: (x, y) = (worldX, worldY)
147 * - Dimensions: w x h (typically 50.0mm x 18.0mm from ObjectConfig)
148 * - Corner Radius: r = cfg.cardCornerRadiusMm (2.0mm)
149 * - Type Band: Left strip of width bw = cfg.cardTypeBandWidthMm (8.0mm)
150 *
151 * VISUAL LAYERING (Back-to-Front Painter's Algorithm):
152 * 1. Base Card Surface: Rounded rectangle filled with theme-aware card color (dark/light mode).
153 * 2. Type Band Strip: Left-aligned colored pill based on file extension (PDF=Red, Office=Blue/Green, etc.)
154 * clipped to the card's rounded boundary.
155 * 3. Type Extension Badge: Bold uppercase text centered in the type band.
156 * 4. Filename Label: Scissor-clipped label preventing text bleed outside the right margin.
157 * 5. State Indicator Pill: Bottom-right status badge:
158 * - Embedded ('E' Teal): Safe copy stored inside the notebook sidecar bundle.
159 * - Link ('L' Amber): External filesystem path that may break if moved.
160 * - Broken ('!' Fluent Red): File was not found at specified path.
161 *
162 * @param[in,out] ctx Target Blend2D drawing context.
163 * @param[in] viewport Active camera viewport for zoom and scale references.
164 */
165void AttachmentObject::Render(BLContext &ctx, const Viewport & /*viewport*/) const {
166 if (!isVisible) return;
167
168 const auto& cfg = GetConfig();
169
170 ctx.save();
171 ctx.apply_transform(transform);
172
173 const double x = worldX;
174 const double y = worldY;
175 const double w = (worldWidth > 0.0) ? worldWidth : GetDefaultChipWidth();
176 const double h = (worldHeight > 0.0) ? worldHeight : GetDefaultChipHeight();
177 const double r = cfg.cardCornerRadiusMm;
178 const double bw = cfg.cardTypeBandWidthMm;
179
180 // -------------------------------------------------------------------------
181 // 1. BASE CARD SURFACE
182 // -------------------------------------------------------------------------
183 ctx.set_fill_style(cfg.GetCardBackgroundColor(opacity));
184 ctx.fill_round_rect(BLRoundRect(x, y, w, h, r, r));
185
186 // -------------------------------------------------------------------------
187 // 2. FILE TYPE COLOR BAND (Left Edge)
188 // -------------------------------------------------------------------------
189 BLRgba32 bandCol = GetTypeColor();
190 ctx.save();
191 ctx.clip_to_rect(BLRect(x, y, bw, h));
192 ctx.set_fill_style(bandCol);
193 ctx.fill_round_rect(BLRoundRect(x, y, w, h, r, r));
194 ctx.restore();
195
196 // -------------------------------------------------------------------------
197 // 3. CARD BORDER & STATUS OUTLINE
198 // -------------------------------------------------------------------------
199 bool isBroken = !IsFileValid();
200 ctx.set_stroke_style(cfg.GetCardBorderColor(isBroken, opacity));
201 ctx.set_stroke_width(isBroken ? 0.6 : 0.4);
202 ctx.stroke_round_rect(BLRoundRect(x, y, w, h, r, r));
203
204 // -------------------------------------------------------------------------
205 // 4. TYPE BADGE TEXT (Centered in left band)
206 // -------------------------------------------------------------------------
207 std::string ext = "";
208 auto dot = displayName.rfind('.');
209 if (dot != std::string::npos && dot + 1 < displayName.size()) {
210 ext = displayName.substr(dot + 1);
211 for (auto &c : ext) {
212 c = static_cast<char>(std::toupper(static_cast<unsigned char>(c)));
213 }
214 if (ext.size() > 4) ext = ext.substr(0, 4);
215 } else {
216 ext = "FILE";
217 }
218
219 BLFont badgeFont = FontManager::Instance().GetFont("Segoe UI", cfg.badgeFontSizePt, true);
220 ctx.fill_utf8_text(BLPoint(x + 1.2, y + h * 0.5 + 1.1), badgeFont,
221 ext.data(), ext.size(), BLRgba32(0xFF, 0xFF, 0xFF, 0xFF));
222
223 // -------------------------------------------------------------------------
224 // 5. FILENAME DISPLAY LABEL (Scissor-Clipped)
225 // -------------------------------------------------------------------------
226 BLFont labelFont = FontManager::Instance().GetFont("Segoe UI", cfg.labelFontSizePt, false);
227 BLRgba32 labelCol = cfg.GetCardTextColor(isBroken, opacity);
228
229 ctx.save();
230 ctx.clip_to_rect(BLRect(x + bw + 2.0, y + 1.0, w - bw - 4.0, h - 2.0));
231 ctx.fill_utf8_text(BLPoint(x + bw + 2.5, y + h * 0.5 + 1.2), labelFont,
232 displayName.data(), displayName.size(), labelCol);
233 ctx.restore();
234
235 // -------------------------------------------------------------------------
236 // 6. EMBEDDED / LINK / BROKEN BADGE (Bottom-Right Corner)
237 // -------------------------------------------------------------------------
238 {
239 const double badgeR = 2.8; // Badge circle radius (mm)
240 const double badgeCX = x + w - badgeR - 1.2; // Center X
241 const double badgeCY = y + h - badgeR - 1.0; // Center Y
242
243 if (isBroken) {
244 // Filled warning red circle with centered white '!'
245 ctx.set_fill_style(BLRgba32(0xE8, 0x11, 0x23, 235));
246 ctx.fill_circle(BLCircle(badgeCX, badgeCY, badgeR));
247 BLFont warnFont = FontManager::Instance().GetFont("Segoe UI", 2.6f, true);
248 ctx.fill_utf8_text(BLPoint(badgeCX - 0.7, badgeCY + 0.9), warnFont, "!",
249 1, BLRgba32(0xFF, 0xFF, 0xFF, 255));
250 } else if (isEmbedded) {
251 // Filled teal circle with white 'E' (Embedded)
252 ctx.set_fill_style(BLRgba32(0x00, 0xB3, 0x9A, 210));
253 ctx.fill_circle(BLCircle(badgeCX, badgeCY, badgeR));
254 BLFont tinyFont = FontManager::Instance().GetFont("Segoe UI", 2.6f, true);
255 ctx.fill_utf8_text(BLPoint(badgeCX - 1.5, badgeCY + 1.0), tinyFont, "E",
256 1, BLRgba32(0xFF, 0xFF, 0xFF, 230));
257 } else {
258 // Hollow warning-amber circle with amber 'L' (Link)
259 ctx.set_stroke_style(BLRgba32(0xE8, 0xB3, 0x00, 190));
260 ctx.set_stroke_width(0.5);
261 ctx.stroke_circle(BLCircle(badgeCX, badgeCY, badgeR));
262 BLFont tinyFont = FontManager::Instance().GetFont("Segoe UI", 2.6f, true);
263 ctx.fill_utf8_text(BLPoint(badgeCX - 1.3, badgeCY + 1.0), tinyFont, "L",
264 1, BLRgba32(0xE8, 0xB3, 0x00, 200));
265 }
266 }
267
268 ctx.restore();
269}

307 relinkAct.iconKey = "warning";
308 relinkAct.order = 10;
309 relinkAct.isEnabled = true;

311 actions.push_back(std::move(relinkAct));

313
314 Folio::ContextMenuItem copyPathAct;
315 copyPathAct.label = "Copy File Path";
316 copyPathAct.icon = "📋";
317 copyPathAct.iconKey = "copy";
318 copyPathAct.order = 30;
319 copyPathAct.onTrigger = [this]() {
320 SDL_SetClipboardText(filePath.c_str());
321 };
322 actions.push_back(std::move(copyPathAct));
323
324 Folio::ContextMenuItem explorerAct;
325 explorerAct.label = "Show in File Explorer";
326 explorerAct.icon = "📁";
327 explorerAct.iconKey = "folder";
328 explorerAct.order = 40;
329 explorerAct.isEnabled = isFileValid;
330 explorerAct.onTrigger = [this]() {
331 std::string parentDir = FileManager::GetParentPath(filePath);
332 if (!parentDir.empty() && FileManager::Exists(parentDir)) {
333 FileManager::OpenWithDefaultApp(parentDir);
334 }
335 };
336 actions.push_back(std::move(explorerAct));
337}

339/**
340 * @brief Resolves color-coding for the file extension badge based on MIME and file extension.
341 *
342 * Color Mapping Scheme:
343 * - PDF: Fluent Red (0xE81123)
344 * - Spreadsheets (XLSX, XLS, CSV): Excel Green (0x107C41)
345 * - Documents (DOCX, DOC): Word Blue (0x0078D4)
346 * - Presentations (PPTX, PPT): PowerPoint Orange (0xD83B01)
347 * - Archives (ZIP, RAR, 7Z): Amber Yellow (0xF0CC00)
348 * - Media (PNG, JPG, SVG): Teal (0x008272)
349 * - Audio (MP3, WAV, FLAC): Purple (0x5C2D91)
350 * - Video (MP4, MKV, MOV): Indigo (0x0066CC)
351 * - Plaintext (TXT, MD): Charcoal Grey (0x606060)
352 * - Fallback: Neutral Slate (0x404452)
353 *
354 * @return BLRgba32 Accent color for the badge strip.
355 */
356BLRgba32 AttachmentObject::GetTypeColor() const {
357 auto ext = std::string{};
358 auto dot = displayName.rfind('.');
359 if (dot != std::string::npos) {
360 ext = displayName.substr(dot + 1);
361 for (auto &c : ext) {
362 c = static_cast<char>(std::tolower(static_cast<unsigned char>(c)));
363 }
364 }
365
366 if (ext == "pdf") return BLRgba32(0xE8, 0x11, 0x23, 0xFF); // Red
367 if (ext == "xlsx" || ext == "xls" || ext == "csv") return BLRgba32(0x10, 0x7C, 0x41, 0xFF); // Green
368 if (ext == "docx" || ext == "doc") return BLRgba32(0x00, 0x78, 0xD4, 0xFF); // Blue
369 if (ext == "pptx" || ext == "ppt") return BLRgba32(0xD8, 0x3B, 0x01, 0xFF); // Orange
370 if (ext == "zip" || ext == "rar" || ext == "7z") return BLRgba32(0xF0, 0xCC, 0x00, 0xFF); // Yellow
371 if (ext == "png" || ext == "jpg" || ext == "jpeg" || ext == "gif" || ext == "svg") return BLRgba32(0x00, 0x82, 0x72, 0xFF); // Teal
372 if (ext == "mp3" || ext == "wav" || ext == "flac" || ext == "aac") return BLRgba32(0x5C, 0x2D, 0x91, 0xFF); // Purple
373 if (ext == "mp4" || ext == "mov" || ext == "avi" || ext == "mkv") return BLRgba32(0x00, 0x66, 0xCC, 0xFF); // Dark blue
374 if (ext == "txt" || ext == "md") return BLRgba32(0x60, 0x60, 0x60, 0xFF); // Grey
375
376 return BLRgba32(0x40, 0x44, 0x52, 0xFF); // Default dark grey
377}
