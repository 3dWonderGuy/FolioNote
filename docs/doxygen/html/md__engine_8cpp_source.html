374 break;
375 case SDLK_A:

378 case SDLK_Z:

383 }
384 break;

386 if (ctrl) {
387 editor.Redo();
388 isDirty = true;
389 }
390 break;
391 case SDLK_B:

393 editor.ToggleBold();
394 isDirty = true;
