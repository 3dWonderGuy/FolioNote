34/**
35 * @brief Deduplicates and persists an image into the active notebook's package directory.
36 *
37 * Mathematical Basis:
38 * Uses 64-bit FNV-1a hashing (Offset Basis: 14695981039346656037ULL, Prime: 1099511628211ULL)
39 * to compute a content hash over byte streams for fast O(1) content deduplication.
40 *
41 * Inputs:
42 * - srcPath: Optional path or content:// URI to the source image file on disk.
43 * - data: Optional pointer to raw in-memory image bytes.
44 * - size: Byte size of in-memory image buffer.
45 * - session: Pointer to active DocumentSession for notebook directory resolution.
46 * - ext: File extension (e.g., ".png", ".jpg").
47 *
48 * Output:
49 * - Relative package path (e.g., "imports/images/img_<hash>.png") or empty string on failure.
50 */
51std::string CanvasEngine::DeduplicateAndSaveImage(const std::string& srcPath, const void* data, size_t size, DocumentSession* session, const std::string& ext) {
52 if (!session) return "";
53 auto activeNb = session->workspace.GetActiveNotebook();
54 if (!activeNb || activeNb->filePath.empty()) return "";
55
56 std::filesystem::path pkgPath(activeNb->filePath);
57 std::filesystem::path imgDir = pkgPath / "imports" / "images";
58 Folio::FileManager::CreateDirectories(imgDir.string());
59
60 uint64_t hash = 14695981039346656037ULL;
61 if (data && size > 0) {
62 const uint8_t* p = static_cast<const uint8_t*>(data);
63 for (size_t i = 0; i < size; ++i) {
64 hash ^= p[i];
65 hash *= 1099511628211ULL;
66 }
67 } else if (!srcPath.empty() && Folio::FileManager::Exists(srcPath)) {
68 std::vector<uint8_t> buf;
69 if (Folio::FileManager::ReadBinary(srcPath, buf) && !buf.empty()) {
70 for (size_t i = 0; i < buf.size(); ++i) {
71 hash ^= buf[i];
72 hash *= 1099511628211ULL;
73 }
74 } else {
75 return "";
76 }
77 } else {
78 return "";
79 }
80
81 char hashStr[32];
82 std::snprintf(hashStr, sizeof(hashStr), "%016llx", static_cast<unsigned long long>(hash));
83 std::string filename = std::string("img_") + hashStr + ext;
84 std::filesystem::path destFile = imgDir / filename;
85
86 if (!Folio::FileManager::Exists(destFile.string())) {
87 if (data && size > 0) {
88 std::ofstream out(destFile, std::ios::binary | std::ios::trunc);
89 if (out.is_open()) {
90 out.write(static_cast<const char*>(data), static_cast<std::streamsize>(size));
91 }
92 } else if (!srcPath.empty() && Folio::FileManager::Exists(srcPath)) {
93 Folio::FileManager::CopySingleFile(srcPath, destFile.string(), true);
94 }
95 }
96
97 return (std::filesystem::path("imports") / "images" / filename).string();
98}

100/**
101 * @brief Asynchronous callback invoked by SDL when an image file is picked by the user.
102 *
103 * Working Process:
104 * 1. Reads binary payload using Folio::FileManager::ReadBinary (supporting Android SAF content:// URIs).
105 * 2. On failure, triggers an error toast notification without silent dropping.
106 * 3. Detects/sniffs magic byte headers if the file extension is absent or generic.
107 * 4. Deduplicates and caches image in notebook package.
108 * 5. Decodes image in background; if decoding fails, raises error toast.
109 * 6. Dispatches UI/session mutation to main thread via EnqueueMainThreadTask to prevent race conditions.
110 */
111void SDLCALL CanvasEngine::OnImageFileSelected(void* userdata, const char* const* filelist, int) {
112 auto* ctx = static_cast<ImageFileDialogContext*>(userdata);
113 if (!ctx) return;
114
115 if (filelist && filelist[0] && filelist[0][0] != '\0') {
116 std::string selectedPath = filelist[0];
117 std::vector<uint8_t> rawBytes;
118 bool readOk = Folio::FileManager::ReadBinary(selectedPath, rawBytes);
119 if (!readOk || rawBytes.empty()) {
120 LOG_ERROR(CanvasEngine, "Failed to read image bytes from path: " + selectedPath);
121 if (ctx->canvas) {
122 ctx->canvas->EnqueueMainThreadTask([selectedPath]() {
123 Folio::UI::UIOverlayHost::Instance().ShowErrorToast(
124 "Image Import Failed", "Unable to read image file from storage: " + selectedPath);
125 });
126 }
127 delete ctx;
128 return;
129 }
130
131 std::string ext = Folio::FileManager::GetExtension(selectedPath);
132 // Sniff image magic bytes if extension is absent (standard for Android SAF content:// URIs)
133 if (ext.empty() || ext == "bin") {
134 if (rawBytes.size() >= 8 && rawBytes[0] == 0x89 && rawBytes[1] == 'P' && rawBytes[2] == 'N' && rawBytes[3] == 'G') {
135 ext = ".png";
136 } else if (rawBytes.size() >= 3 && rawBytes[0] == 0xFF && rawBytes[1] == 0xD8 && rawBytes[2] == 0xFF) {
137 ext = ".jpg";
138 } else if (rawBytes.size() >= 12 && rawBytes[0] == 'R' && rawBytes[1] == 'I' && rawBytes[2] == 'F' && rawBytes[3] == 'F' &&
139 rawBytes[8] == 'W' && rawBytes[9] == 'E' && rawBytes[10] == 'B' && rawBytes[11] == 'P') {
140 ext = ".webp";
141 } else if (rawBytes.size() >= 6 && rawBytes[0] == 'G' && rawBytes[1] == 'I' && rawBytes[2] == 'F') {
142 ext = ".gif";
143 } else if (rawBytes.size() >= 2 && rawBytes[0] == 'B' && rawBytes[1] == 'M') {
144 ext = ".bmp";
145 } else {
146 ext = ".png";
147 }
148 }
149 if (!ext.empty() && ext[0] != '.') ext = "." + ext;
150
151 std::string storedPath = ctx->canvas->DeduplicateAndSaveImage(selectedPath, rawBytes.data(), rawBytes.size(), ctx->session, ext);
152
153 const double screenDpi = ctx->canvas->transform.pixelsPerMm * 25.4;
154 auto img = std::make_shared<Folio::ImageObject>(rawBytes.data(), rawBytes.size(), storedPath.empty() ? selectedPath : storedPath, screenDpi);
155 if (!img->isLoaded) {
156 LOG_ERROR(CanvasEngine, "Failed to decode image from path: " + selectedPath);
157 if (ctx->canvas) {
158 ctx->canvas->EnqueueMainThreadTask([selectedPath]() {
159 Folio::UI::UIOverlayHost::Instance().ShowErrorToast(
160 "Image Decode Failed", "Unsupported or corrupted image format: " + selectedPath);
161 });
162 }
163 delete ctx;
164 return;
165 }
166
167 CanvasEngine* canvas = ctx->canvas;
168 DocumentSession* session = ctx->session;
169 Point2D insertPos = ctx->insertPosWorld;
170
171 // Dispatch canvas insertion to the main UI thread
172 canvas->EnqueueMainThreadTask([canvas, session, img, insertPos, selectedPath, storedPath]() {
173 if (!session) return;
174 img->worldX = insertPos.x - img->worldWidth * 0.5;
175 img->worldY = insertPos.y - img->worldHeight * 0.5;
176 img->UpdateBounds();
177
178 session->AddImage(img);
179 canvas->InvalidateLayer();
180 LOG_INFO(CanvasEngine, "Imported image from '" + selectedPath + "' -> '" + (storedPath.empty() ? selectedPath : storedPath) + "' (" + Folio::ImageFormatToString(img->imageFormat).data() + ")");
181 });
182 }
183
184 delete ctx;
185}

307/**
308 * @brief Commits a selected attachment file, either copying into the package attachments sidecar
309 * or retaining it as an external filesystem/URL link.
310 *
311 * Inputs:
312 * - embed: If true, copies the file into <notebook>/attachments/ and records relative path.
313 * - session: Pointer to active DocumentSession.
314 */
315void CanvasEngine::CommitAttachment(bool embed, DocumentSession* session) {
316 if (m_pendingAttachPath.empty() || !session) {
317 m_attachModalOpen = false;
318 return;
319 }
320
321 std::string finalPath = m_pendingAttachPath;
322 bool embeddedOk = false;
323
324 if (embed) {
325 auto activeNb = session->GetActiveNotebook();
326 std::string notebookDir = (activeNb) ? activeNb->filePath : "";
327 if (!notebookDir.empty()) {
328 std::string attachDir = Folio::FileManager::JoinPath(notebookDir, "attachments");
329 if (Folio::FileManager::CreateDirectories(attachDir)) {
330 std::string safeFilename = GUIDGenerator::GenerateV4().substr(0, 8)
331 + "_" + Folio::FileManager::SanitizeFileName(m_pendingAttachName);
332 std::string destPath = Folio::FileManager::JoinPath(attachDir, safeFilename);
333
334 if (Folio::FileManager::CopySingleFile(m_pendingAttachPath, destPath, true)) {
335 finalPath = Folio::FileManager::JoinPath("attachments", safeFilename);
336 embeddedOk = true;
337 LOG_INFO(CanvasEngine, "Embedded attachment: copied '" + m_pendingAttachName +
338 "' to sidecar as '" + safeFilename + "'");
339 } else {
340 LOG_ERROR_CODE(CanvasEngine, Folio::FolioErrorCode::SysFileWriteFailed,
341 "Failed to copy attachment to sidecar: " + m_pendingAttachPath +
342 " — falling back to link mode.");
343 Folio::UI::UIOverlayHost::Instance().ShowErrorToast(
344 "Attachment Copy Failed", "System copy failed for attachment file. Falling back to link mode.");
345 finalPath = m_pendingAttachPath;
346 embeddedOk = false;
347 }
348 } else {
349 LOG_ERROR_CODE(CanvasEngine, Folio::FolioErrorCode::SysDirectoryCreateFailed,
350 "Failed to create attachments sidecar directory — falling back to link mode.");
351 Folio::UI::UIOverlayHost::Instance().ShowErrorToast(
352 "Folder Creation Failed", "Could not create attachments folder in notebook package.");
353 }
354 } else {
355 LOG_WARN(CanvasEngine,
356 "Cannot embed attachment: notebook has no directory yet. Using link mode.");
357 }
358 }
359
360 Point2D centerWorld = transform.ScreenToWorld(
361 static_cast<float>(viewportW) * 0.5f,
362 static_cast<float>(viewportH) * 0.5f);
363
364 auto attachObj = std::make_shared<Folio::AttachmentObject>(
365 finalPath, m_pendingAttachName, "", (embed && embeddedOk));
366 attachObj->isFileValid = Folio::FileManager::Exists(finalPath);
367 attachObj->worldX = centerWorld.x - Folio::AttachmentObject::chipW * 0.5;
368 attachObj->worldY = centerWorld.y - Folio::AttachmentObject::chipH * 0.5;
369 attachObj->guuid = GUIDGenerator::GenerateV4();
370 attachObj->uid = UIDGenerator::Next();
371 attachObj->UpdateBounds();
372
373 auto activePage = session->GetActivePage();
374 if (activePage) {
375 activePage->AddObject(attachObj, true);
376 session->RecordHistoryCommand(
377 activePage,
378 std::make_unique<Folio::AddObjectCommand>(attachObj));
379 activePage->isModified = true;
380 session->NotifyPageModified(activePage);
381 }
382
383 InvalidateLayer();
384
385 LOG_INFO(CanvasEngine, "Committed attachment '" + m_pendingAttachName +
386 "' mode=" + std::string(embed && embeddedOk ? "embedded" : "link") +
387 " path='" + finalPath + "'");
388
389 m_pendingAttachPath.clear();
390 m_pendingAttachName.clear();
391 m_pendingAttachSession = nullptr;
392 m_attachModalOpen = false;
393}

500bool CanvasEngine::InsertVideoFromFile(const std::string& filePath, DocumentSession* session) {
501 if (!session) return false;
502 auto activePage = session->GetActivePage();
503 if (!activePage) return false;
504
505 std::filesystem::path fspath(filePath);
506 if (!std::filesystem::exists(fspath)) {
507 LOG_ERROR(CanvasEngine, "InsertVideoFromFile: File not found: " + filePath);
508 return false;
509 }
510
511 std::string ext = fspath.extension().string();
512 for (auto& c : ext) c = static_cast<char>(::tolower(static_cast<unsigned char>(c)));
513
514 static const std::initializer_list<const char*> kSupportedVideoExts = {
515 ".mp4", ".mkv", ".webm", ".mov", ".avi", ".ts", ".m2ts",
516 ".mpg", ".mpeg", ".wmv", ".flv", ".3gp", ".m4v", ".ogv"
517 };
518 bool supported = false;
519 for (const char* e : kSupportedVideoExts) {
520 if (ext == e) { supported = true; break; }
521 }
522 if (!supported) {
523 LOG_WARN(CanvasEngine, "InsertVideoFromFile: Unsupported video format: " + ext);
524 return false;
525 }
526
527 auto& cfg = Folio::ObjectConfig::Get();
528 std::error_code ec;
529 auto fileBytes = std::filesystem::file_size(fspath, ec);
530 if (!ec && fileBytes > cfg.maxVideoFileSizeBytes) {
531 LOG_WARN(CanvasEngine, "InsertVideoFromFile: File too large (" +
532 std::to_string(fileBytes / (1024*1024)) + " MB)");
533 return false;
534 }
535
536 std::string importedPath = filePath;
537 try {
538 auto activeNb = session->workspace.GetActiveNotebook();
539 if (activeNb && !activeNb->filePath.empty()) {
540 std::filesystem::path pkgPath(activeNb->filePath);
541 std::filesystem::path importDir = pkgPath / "imports" / "videos";
542 std::filesystem::create_directories(importDir, ec);
543 if (!ec) {
544 std::filesystem::path destPath = importDir / fspath.filename();
545 if (!std::filesystem::exists(destPath)) {
546 std::filesystem::copy_file(fspath, destPath,
547 std::filesystem::copy_options::overwrite_existing, ec);
548 }
549 if (!ec) {
550 importedPath = destPath.string();
551 }
552 }
553 }
554 } catch (...) {
555 importedPath = filePath;
556 LOG_WARN(CanvasEngine, "InsertVideoFromFile: Could not copy to sidecar; streaming from original path");
557 }
558
559 double w = cfg.defaultVideoWidthMm;
560 double h = cfg.defaultVideoHeightMm;
561
562 auto vid = std::make_shared<Folio::VideoObject>(importedPath, fspath.filename().string(), w, h);
563
564 Point2D centerWorld = transform.ScreenToWorld(
565 static_cast<float>(viewportW) * 0.5f,
566 static_cast<float>(viewportH) * 0.5f
567 );
568 vid->worldX = centerWorld.x - w * 0.5;
569 vid->worldY = centerWorld.y - h * 0.5;
570 vid->UpdateBounds();
571
572 vid->Play([this]() {
573 InvalidateLayer();
574 });
575 vid->Pause();
576
577 session->AddVideo(vid);
578 InvalidateLayer();
579

667bool CanvasEngine::InsertAudioFromFile(const std::string& filePath, DocumentSession* session) {
668 if (!session) return false;
669 auto activePage = session->GetActivePage();
670 if (!activePage) return false;
671
672 std::filesystem::path fspath(filePath);
673 if (!std::filesystem::exists(fspath)) {
674 LOG_ERROR(CanvasEngine, "InsertAudioFromFile: File not found: " + filePath);
675 return false;
676 }
677
678 std::string ext = fspath.extension().string();
679 for (auto& c : ext) c = static_cast<char>(::tolower(static_cast<unsigned char>(c)));
680
681 static const std::initializer_list<const char*> kSupportedAudioExts = {
682 ".mp3", ".wav", ".m4a", ".flac", ".ogg", ".aac", ".opus", ".wma", ".aiff"
683 };
684 bool supported = false;
685 for (const char* e : kSupportedAudioExts) {
686 if (ext == e) { supported = true; break; }
687 }
688 if (!supported) {
689 LOG_WARN(CanvasEngine, "InsertAudioFromFile: Unsupported audio format: " + ext);
690 return false;
691 }
692
693 std::string importedPath = filePath;
694 try {
695 auto activeNb = session->workspace.GetActiveNotebook();
696 if (activeNb && !activeNb->filePath.empty()) {
697 std::error_code ec;
698 std::filesystem::path pkgPath(activeNb->filePath);
699 std::filesystem::path importDir = pkgPath / "imports" / "audio";
700 std::filesystem::create_directories(importDir, ec);
701 if (!ec) {
702 std::filesystem::path destPath = importDir / fspath.filename();
703 if (!std::filesystem::exists(destPath)) {
704 std::filesystem::copy_file(fspath, destPath,
705 std::filesystem::copy_options::overwrite_existing, ec);
706 }
707 if (!ec) {
708 importedPath = destPath.string();
709 }
710 }
711 }
712 } catch (...) {
713 importedPath = filePath;
714 }
715
716 auto audioObj = std::make_shared<Folio::AudioObject>(importedPath, fspath.filename().string());
717
718 Point2D centerWorld = transform.ScreenToWorld(
719 static_cast<float>(viewportW) * 0.5f,
720 static_cast<float>(viewportH) * 0.5f
721 );
722 audioObj->worldX = centerWorld.x - Folio::AudioObject::chipW * 0.5;
723 audioObj->worldY = centerWorld.y - Folio::AudioObject::chipH * 0.5;
724 audioObj->UpdateBounds();
725
726 session->AddObject(audioObj);
727 InvalidateLayer();
728
729 LOG_INFO(CanvasEngine, "Inserted AudioObject: " + importedPath);
730 return true;
731}
