122 /**
123 * @brief Parses page range strings (e.g., "1, 3-5, 20" or "3-25") into 0-indexed page indices.
124 */
125 static std::vector<int> ParsePageRange(const std::string& input, int totalPages) {
126 std::vector<int> pages;
127 if (totalPages <= 0) return pages;
128
129 std::stringstream ss(input);
130 std::string token;
131
132 while (std::getline(ss, token, ',')) {
133 // Trim whitespace
134 size_t start = token.find_first_not_of(" \t\r\n");
135 size_t end = token.find_last_not_of(" \t\r\n");
136 if (start == std::string::npos) continue;
137 std::string part = token.substr(start, end - start + 1);
138
139 size_t dash = part.find('-');
140 if (dash != std::string::npos) {
141 std::string sStart = part.substr(0, dash);
142 std::string sEnd = part.substr(dash + 1);
143 try {
144 int p1 = std::clamp(std::stoi(sStart), 1, totalPages);
145 int p2 = std::clamp(std::stoi(sEnd), 1, totalPages);
146 if (p1 > p2) std::swap(p1, p2);
147 for (int p = p1; p <= p2; ++p) {
148 pages.push_back(p - 1);
149 }
150 } catch (...) {}
151 } else {
152 try {
153 int p = std::clamp(std::stoi(part), 1, totalPages);
154 pages.push_back(p - 1);
155 } catch (...) {}
156 }
157 }
158
159 // Sort and remove duplicates
160 std::sort(pages.begin(), pages.end());
161 pages.erase(std::unique(pages.begin(), pages.end()), pages.end());
162 return pages;
163 }

513 void CommitImport(DocumentSession& session, CanvasEngine& canvas, const std::vector<int>& pagesToImport) {
514 if (pagesToImport.empty()) return;
515
516 PdfDocumentInfo docInfo;
517 if (!PdfStorage::IngestPdf(currentDoc.originalPath, &session, importMode, docInfo)) {
518 LOG_ERROR(PdfStorage, "CommitImport failed: ingestion failed for " + currentDoc.originalPath);
519 Folio::UI::UIOverlayHost::Instance().ShowErrorToast(
520 "PDF Ingest Failed", "Could not copy PDF file into notebook package: " + currentDoc.originalPath);
521 return;
522 }
523
524 // Find target notebook
525 std::shared_ptr<Notebook> targetNb = nullptr;
526 for (const auto& nb : session.workspace.notebooks) {
527 if (nb && nb->guid == targetNotebookGuid) {
528 targetNb = nb;
529 break;
530 }
531 }
532 if (!targetNb) targetNb = session.workspace.GetActiveNotebook();
533 if (!targetNb) return;
534
535 // Set active notebook in workspace
536 for (size_t i = 0; i < session.workspace.notebooks.size(); ++i) {
537 if (session.workspace.notebooks[i] == targetNb) {
538 session.workspace.activeNotebookIndex = i;
539 break;
540 }
541 }
542
543 auto targetSec = targetNb->FindSectionByGuid(targetSectionGuid);
544 if (!targetSec) targetSec = targetNb->GetActiveSection();
545 if (!targetSec) return;
546 targetNb->SetActiveSection(targetSec);
547
548 std::string baseDocTitle = docInfo.originalFileName;
549 if (baseDocTitle.size() > 4 && baseDocTitle.substr(baseDocTitle.size() - 4) == ".pdf") {
550 baseDocTitle = baseDocTitle.substr(0, baseDocTitle.size() - 4);
551 }
552
553 // =========================================================================
554 // CASE A: INSERT INTO EXISTING PAGE BELOW LAST KNOWN STROKE / CONTENT
555 // =========================================================================
556 if (isTargetingExistingPage) {
557 std::shared_ptr<CanvasPage> targetPg = nullptr;
558 size_t pgIdx = 0;
559 for (size_t i = 0; i < targetSec->pages.size(); ++i) {
560 if (targetSec->pages[i] && targetSec->pages[i]->guid == targetPageGuid) {
561 targetPg = targetSec->pages[i];
562 pgIdx = i;
563 break;
564 }
565 }
566 if (!targetPg) targetPg = targetSec->GetActivePage();
567
568 if (targetPg) {
569 targetSec->activePageIndex = pgIdx;
570
571 // Detect bottom of existing content on this canvas
572 double startY = DEFAULT_MARGIN_TOP_MM;
573 bool hasContent = false;
574 double maxBottom = 0.0;
575
576 for (const auto& obj : targetPg->objects) {
577 if (!obj || !obj->isVisible) continue;
578 double b = obj->bounds.maxY;
579 if (!hasContent || b > maxBottom) {
580 maxBottom = b;
581 }
582 hasContent = true;
583 }
584
585 if (hasContent) {
586 startY = maxBottom + DEFAULT_GAP_Y_MM;
587 }
588
589 double curY = startY;
590 for (size_t i = 0; i < pagesToImport.size(); ++i) {
591 int pIdx = pagesToImport[i];
592 auto pdfObj = std::make_shared<PdfContainer>(
593 docInfo.packagePath, docInfo.originalFileName, pIdx, docInfo.pageCount,
594 DEFAULT_MARGIN_LEFT_MM, curY, 210.0, 297.0, setAsBackground
595 );
596 pdfObj->resolvedDiskPath = docInfo.diskPath;
597 pdfObj->isExternalLink = docInfo.isExternal;
598 pdfObj->guuid = GUIDGenerator::GenerateV4();
599 pdfObj->uid = UIDGenerator::Next();
600 pdfObj->EnsurePageLoaded();
601 pdfObj->UpdateBounds();
602
603 targetPg->AddObject(pdfObj);
604 curY += (pdfObj->worldHeight + DEFAULT_GAP_Y_MM);
605 }
606
607 LOG_INFO(PdfStorage, "Inserted " + std::to_string(pagesToImport.size()) +
608 " PDF page(s) into existing page '" + targetPg->title + "' starting at Y=" +
609 std::to_string(static_cast<int>(startY)) + " mm");
610 }
611 }
612 // =========================================================================
613 // CASE B: ONE CANVAS PAGE PER PDF PAGE
614 // =========================================================================
615 else if (placementMode == PdfPlacementMode::OnePagePerCanvas) {
616 size_t firstCreatedIdx = targetSec->pages.size();
617 for (size_t i = 0; i < pagesToImport.size(); ++i) {
618 int pIdx = pagesToImport[i];
619 std::string pageTitle = baseDocTitle + " (Page " + std::to_string(pIdx + 1) + ")";
620
621 auto newPage = std::make_shared<CanvasPage>(pageTitle);
622 newPage->guid = GUIDGenerator::GenerateV4();
623
624 auto pdfObj = std::make_shared<PdfContainer>(
625 docInfo.packagePath, docInfo.originalFileName, pIdx, docInfo.pageCount,
626 DEFAULT_MARGIN_LEFT_MM, DEFAULT_MARGIN_TOP_MM, 210.0, 297.0, setAsBackground
627 );
628 pdfObj->resolvedDiskPath = docInfo.diskPath;
629 pdfObj->isExternalLink = docInfo.isExternal;
630 pdfObj->guuid = GUIDGenerator::GenerateV4();
631 pdfObj->uid = UIDGenerator::Next();
632 pdfObj->EnsurePageLoaded();
633 pdfObj->UpdateBounds();
634
635 newPage->AddObject(pdfObj);
636 targetSec->AddPage(newPage);
637 }
638 targetSec->activePageIndex = firstCreatedIdx;
639 }
640 // =========================================================================
641 // CASE C: ALL PAGES STACKED ON SINGLE CANVAS
642 // =========================================================================
643 else if (placementMode == PdfPlacementMode::AllPagesStacked) {
644 std::string pageTitle = baseDocTitle + " (" + std::to_string(pagesToImport.size()) + " Pages)";
645 auto newPage = std::make_shared<CanvasPage>(pageTitle);
646 newPage->guid = GUIDGenerator::GenerateV4();
647
648 double curY = DEFAULT_MARGIN_TOP_MM;
649 for (size_t i = 0; i < pagesToImport.size(); ++i) {
650 int pIdx = pagesToImport[i];
651 auto pdfObj = std::make_shared<PdfContainer>(
652 docInfo.packagePath, docInfo.originalFileName, pIdx, docInfo.pageCount,
653 DEFAULT_MARGIN_LEFT_MM, curY, 210.0, 297.0, setAsBackground
654 );
655 pdfObj->resolvedDiskPath = docInfo.diskPath;
656 pdfObj->isExternalLink = docInfo.isExternal;
657 pdfObj->guuid = GUIDGenerator::GenerateV4();
658 pdfObj->uid = UIDGenerator::Next();
659 pdfObj->EnsurePageLoaded();
660 pdfObj->UpdateBounds();
661
662 newPage->AddObject(pdfObj);
663 curY += (pdfObj->worldHeight + DEFAULT_GAP_Y_MM);
664 }
665
666 targetSec->AddPage(newPage);
667 targetSec->activePageIndex = targetSec->pages.size() - 1;
668 }
669 // =========================================================================
670 // CASE D: STANDALONE PDF VIEWER PAGE
671 // =========================================================================
672 else if (placementMode == PdfPlacementMode::StandaloneViewer) {
673 std::string pageTitle = "[PDF] " + baseDocTitle;
674 auto newPage = std::make_shared<CanvasPage>(pageTitle);
675 newPage->guid = GUIDGenerator::GenerateV4();
676 newPage->isDedicatedPdf = true;
677 // Store portable package-relative path for local copies, or absolute path for external links
678 newPage->dedicatedPdfPath = docInfo.isExternal ? docInfo.diskPath : docInfo.packagePath;
679 newPage->isModified = true;
680
681 auto pdfObj = std::make_shared<PdfContainer>(
682 docInfo.packagePath, docInfo.originalFileName, 0, docInfo.pageCount,
683 DEFAULT_MARGIN_LEFT_MM, DEFAULT_MARGIN_TOP_MM, 210.0, 297.0, true // locked background
684 );
685 pdfObj->resolvedDiskPath = docInfo.diskPath;
686 pdfObj->isExternalLink = docInfo.isExternal;
687 pdfObj->guuid = GUIDGenerator::GenerateV4();
688 pdfObj->uid = UIDGenerator::Next();
689 pdfObj->EnsurePageLoaded();
690 pdfObj->UpdateBounds();
691
692 newPage->AddObject(pdfObj);
693 targetSec->AddPage(newPage);
694 targetSec->activePageIndex = targetSec->pages.size() - 1;
695 }
696
697 session.workspace.FlushActiveNotebookAsync();
698 canvas.needsFullRebake = true;
699 canvas.isDirty = true;
700 LOG_INFO(PdfStorage, "Successfully imported PDF '" + docInfo.originalFileName +
701 "' with " + std::to_string(pagesToImport.size()) + " pages into section '" + targetSec->name + "'");
702 }
703};
