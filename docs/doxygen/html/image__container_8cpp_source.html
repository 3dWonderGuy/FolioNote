65bool ImageObject::LoadFromFile(const std::string& filePath, double dpi) {
66 loadFailed = false;
67
68 if (filePath.empty()) {
69 isLoaded = false;
70 loadFailed = true;
71 LOG_WARN(CanvasObject, "ImageObject::LoadFromFile rejected: File path is empty.");
72 return false;
73 }
74
75 imagePath = filePath;
76
77 // Resolve relative companion package path or absolute filesystem disk path
78 const std::string resolvedPath = FileManager::ResolveAssetPath(filePath);
79 if (!FileManager::Exists(resolvedPath)) {
80 isLoaded = false;
81 loadFailed = true;
82 LOG_ERROR(CanvasObject, "ImageObject::LoadFromFile: File does not exist: " + resolvedPath);
83 return false;
84 }
85
86 std::vector<uint8_t> fileBuffer;
87 if (!FileManager::ReadBinary(resolvedPath, fileBuffer) || fileBuffer.empty()) {
88 isLoaded = false;
89 loadFailed = true;
90 LOG_ERROR(CanvasObject, "ImageObject::LoadFromFile: Failed to read binary bytes from: " + resolvedPath);
91 return false;
92 }
93
94 return LoadFromMemory(fileBuffer.data(), fileBuffer.size(), filePath, dpi);
95}

97bool ImageObject::LoadFromMemory(const uint8_t* data, size_t size, std::string_view filenameHint, double dpi) {
98 loadFailed = false;
99
100 if (!data || size == 0) {
101 isLoaded = false;
102 loadFailed = true;
103 LOG_WARN(CanvasObject, "ImageObject::LoadFromMemory rejected: Buffer pointer is null or size is 0 bytes.");
104 return false;
105 }
106
107 if (size > ObjectConfig::Get().maxImageFileSizeBytes) {
108 isLoaded = false;
109 loadFailed = true;
110 LOG_WARN(CanvasObject, "ImageObject::LoadFromMemory rejected: Buffer size (" +
111 std::to_string(size) + " bytes) exceeds maximum configured limit (" +
112 std::to_string(ObjectConfig::Get().maxImageFileSizeBytes) + " bytes).");
113 return false;
114 }
115
116 if (!filenameHint.empty()) {
117 ImageFormat detected = ImageFormatFromExtension(filenameHint);
118 if (detected != ImageFormat::Unknown) {
119 imageFormat = detected;
120 }
121
122 // Only adopt filenameHint as persistent imagePath if it represents an actual filename or path,
123 // rather than just a bare extension hint (e.g. ".png" or "webp").
124 const bool isBareExtension = (filenameHint.front() == '.' && filenameHint.find_first_of("/\\") == std::string_view::npos) ||
125 (filenameHint.find_first_of("./\\") == std::string_view::npos);
126 if (!isBareExtension) {
127 imagePath = std::string(filenameHint);
128 }
129 }
130
131 // Retain binary buffer for document persistence/serialization
132 embeddedData.assign(data, data + size);
133
134 DecodedImageResult result = ImageDecoder::DecodeFromMemory(data, size, filenameHint);
135 if (!result.success || (result.image.is_empty() && result.frames.empty())) {
136 isLoaded = false;
137 loadFailed = true;
138 const std::string hintStr = filenameHint.empty() ? "<in-memory payload>" : std::string(filenameHint);
139 const std::string reason = result.errorMessage.empty() ? "Unsupported format or corrupted payload" : result.errorMessage;
140 LOG_ERROR(CanvasObject, "ImageObject::LoadFromMemory failed to decode " + hintStr + " (" + std::to_string(size) + " bytes): " + reason);
141 return false;
142 }
143
144 if (result.isAnimated && result.frames.size() > 1) {
145 isAnimated = true;
146 isAnimationPaused = false;
147 frames = std::move(result.frames);
148 cachedBlImage = frames[0].image;
149 currentFrameIndex = 0;
150 lastFrameTickMs = 0;
151 } else {
152 isAnimated = false;
153 isAnimationPaused = false;
154 frames.clear();
155 cachedBlImage = std::move(result.image);
156 }
157
158 naturalWidth = result.naturalWidth;
159 naturalHeight = result.naturalHeight;
160 if (result.format != ImageFormat::Unknown) {
161 imageFormat = result.format;
162 }
163 isLoaded = true;
164
165 CalculateDimensionsFromDpi(dpi);
166 UpdateBounds();
167 return true;
168}

188bool ImageObject::EnsureLoaded() {
189 if (isLoaded && !cachedBlImage.is_empty()) {
190 return true;
191 }
192
193 // Suppress repeated reload thrashing and 120 FPS console log spamming if previous load failed
194 if (loadFailed) {
195 return false;
196 }
197
198 // Preserve custom bounds if already deserialized or set
199 const double savedW = worldWidth;
200 const double savedH = worldHeight;
201 bool ok = false;
202
203 if (!embeddedData.empty()) {
204 ok = LoadFromMemory(embeddedData.data(), embeddedData.size(), imagePath);
205 } else if (!imagePath.empty()) {
206 // Fallback: reload from package companion asset or physical filesystem path
207 ok = LoadFromFile(imagePath);
208 } else {
209 LOG_WARN(CanvasObject, "ImageObject::EnsureLoaded: No embedded buffer or valid image path available to restore image surface.");
210 }
211
212 if (ok && savedW > 0.0 && savedH > 0.0) {
213 worldWidth = savedW;
214 worldHeight = savedH;
215 UpdateBounds();
216 } else if (!ok) {
217 loadFailed = true;
218 LOG_ERROR(CanvasObject, "ImageObject::EnsureLoaded: Failed to restore surface for '" + imagePath + "'. Displaying placeholder fallback.");
219 }
220
221 return ok;
222}

228void ImageObject::CalculateDimensionsFromDpi(double dpi) {
229 // Robust dimension resolution: if naturalWidth/naturalHeight are 0, recover from frames or cached surface
230 if ((naturalWidth == 0 || naturalHeight == 0) && !frames.empty() && !frames[0].image.is_empty()) {
231 naturalWidth = static_cast<uint32_t>(frames[0].image.width());
232 naturalHeight = static_cast<uint32_t>(frames[0].image.height());
233 }
234 if ((naturalWidth == 0 || naturalHeight == 0) && !cachedBlImage.is_empty()) {
235 naturalWidth = static_cast<uint32_t>(cachedBlImage.width());
236 naturalHeight = static_cast<uint32_t>(cachedBlImage.height());
237 }
238
239 if (naturalWidth == 0 || naturalHeight == 0) {
240 return;
241 }
242
243 // Default to configured standard desktop display density fallback
244 const double fallbackDpi = ObjectConfig::Get().defaultImageDpi;
245 const double effectiveDpi = (dpi >= 10.0) ? dpi : (fallbackDpi >= 10.0 ? fallbackDpi : 96.0);
246
247 // Convert pixel dimensions to physical millimeters: 1 inch = 25.4 mm
248 const double mmPerPixel = 25.4 / effectiveDpi;
249 double rawW = static_cast<double>(naturalWidth) * mmPerPixel;
250 double rawH = static_cast<double>(naturalHeight) * mmPerPixel;
251
252 // 1. Hard maximum limit clamping (preserves exact source aspect ratio)
253 const double maxLimitMm = ObjectConfig::Get().maxImageCanvasDimensionMm;
254 const double maxDimension = (std::max)(rawW, rawH);
255 if (maxLimitMm > 0.0 && maxDimension > maxLimitMm) {
256 const double downscale = maxLimitMm / maxDimension;
257 rawW *= downscale;
258 rawH *= downscale;
259 }
260
261 // 2. Hard minimum limit clamping (prevents sub-millimeter disappearance)
262 const double minLimitMm = ObjectConfig::Get().minImageCanvasDimensionMm;
263 const double minDimension = (std::min)(rawW, rawH);
264 if (minLimitMm > 0.0 && minDimension < minLimitMm) {
265 const double upscale = minLimitMm / minDimension;
266 rawW *= upscale;
267 rawH *= upscale;
268 }
269
270 worldWidth = rawW;
271 worldHeight = rawH;
272}

278bool ImageObject::UpdateAnimation(uint64_t nowMs) {
279 if (isAnimationPaused || !isAnimated || frames.size() <= 1) {
280 return false;
281 }
282
283 if (lastFrameTickMs == 0) {
284 lastFrameTickMs = nowMs;
285 return false;
286 }
287
288 uint32_t delay = frames[currentFrameIndex].delayMs;
289 // Standard web browser clamp: delays <= 10ms are clamped to 100ms (10 FPS)
290 // to protect system scheduler from ancient broken GIF encoders.
291 if (delay <= 10) {
292 delay = 100;
293 }
294
295 if (nowMs - lastFrameTickMs >= delay) {
296 currentFrameIndex = (currentFrameIndex + 1) % frames.size();
297 cachedBlImage = frames[currentFrameIndex].image;
298 lastFrameTickMs = nowMs;
299 return true;
300 }
301
302 return false;
303}

331void ImageObject::Render(BLContext& ctx, const Viewport& viewport) const {
332 if (!isVisible) return;
333
334 if (!isLoaded || cachedBlImage.is_empty()) {
335 const_cast<ImageObject*>(this)->EnsureLoaded();
336 }
337
338 ctx.save();
339 ctx.apply_transform(transform);
340
341 if (opacity < 0.999f) {
342 ctx.set_global_alpha(static_cast<double>(opacity));
343 }
344
345 if (isLoaded && !cachedBlImage.is_empty()) {
346 // High-performance hardware-accelerated raster blit
347 ctx.blit_image(BLRect(worldX, worldY, worldWidth, worldHeight), cachedBlImage);
348 } else {
349 // Sleek placeholder frame when the image file is unloaded or missing
350 ctx.set_fill_style(BLRgba32(0x18, 0x1C, 0x24, static_cast<uint8_t>(opacity * 200)));
351 ctx.fill_round_rect(BLRoundRect(worldX, worldY, worldWidth, worldHeight, 4.0, 4.0));
352
353 ctx.set_stroke_style(BLRgba32(0x3B, 0x44, 0x54, static_cast<uint8_t>(opacity * 255)));
354 const double strokeScale = (viewport.zoom > 0.001) ? (1.0 / viewport.zoom) : 1.0;
355 ctx.set_stroke_width(strokeScale);
356 ctx.stroke_round_rect(BLRoundRect(worldX, worldY, worldWidth, worldHeight, 4.0, 4.0));
357 }
358
359 ctx.restore();
360}

366double ImageObject::GetAspectRatio() const noexcept {
367 if (naturalWidth > 0 && naturalHeight > 0) {
368 return static_cast<double>(naturalWidth) / static_cast<double>(naturalHeight);
369 }
370 return (worldHeight > 0.001) ? (worldWidth / worldHeight) : 1.0;
371}

373void ImageObject::SetWidthPreservingAspect(double newWidth) {
374 if (newWidth <= 0.0) return;
375 const double aspect = GetAspectRatio();
376 worldWidth = newWidth;
377 worldHeight = (aspect > 0.001) ? (newWidth / aspect) : worldHeight;
378 UpdateBounds();
379}
380
381void ImageObject::SetHeightPreservingAspect(double newHeight) {
382 if (newHeight <= 0.0) return;
383 const double aspect = GetAspectRatio();
384 worldHeight = newHeight;
385 worldWidth = (aspect > 0.001) ? (newHeight * aspect) : worldWidth;
386 UpdateBounds();
387}
