1/**
2 * @file text_editor_state.cpp
3 * @brief Implementation of the headless, Slint-proof TextEditorState controller.
4 *
5 * GENERAL WORKING PROCESS:
6 * ------------------------
7 * 1. Attaches to a TextBoxObject when the user clicks to type or edits an existing note.
8 * 2. On text input / key presses, updates the text buffer, handles UTF-8 byte sequences,
9 * and manages the cursor and selection range.
10 * 3. Calculates typographical text layout using FontManager metrics (ascent, descent,
11 * advance width) and computes word wrapping.
12 * 4. Yields world-space cursor positions and selection highlight bounding boxes for
13 * Blend2D to render without any UI-toolkit dependency.
14 *
15 * MATHEMATICAL FOUNDATIONS:
16 * -------------------------
17 * - Line Layout:
18 * Local Line Top: y_{line} = \sum_{k=0}^{i-1} height_k
19 * Baseline: baselineY = y_{line} + ascent
20 * - Word Wrapping:
21 * Greedy wrapping: advance = MeasureTextWidth(font, currentWord)
22 * If (lineWidth + advance > maxWidth) -> wrap to next line.
23 * - Point-to-Index Projection:
24 * Line is selected by: line.localY <= (queryY - worldY) < line.localY + line.height.
25 * Char offset is found by minimal horizontal distance: |x_{char} - queryX|.
26 */

107/**
108 * @brief Recomputes lines, word wrap, and total container dimensions.
109 *
110 * Mathematical & Typographical Flow:
111 * - Line metrics: height and ascent are computed via m_target->GetLineMetricsForRange(start, end).
112 * - Span advances: computed via m_target->MeasureRange(start, end).
113 * - Word boundaries: detected via ASCII space or paragraph breaks ('\n').
114 * - Dynamic expansion: expands m_target->worldHeight and worldWidth to enclose all content.
115 */
116void TextEditorState::ReflowLayout() {
117 m_lines.clear();
118 if (!m_target) return;
119
120 const std::string& fullText = m_target->text;
121 const double maxWrapWidth = (std::max)(10.0, m_target->worldWidth - 4.0); // 2mm internal margin on each side
122
123 double curY = 2.0; // 2mm top margin
124 double maxLineWidth = 0.0;
125
126 // Helper lambda to commit a line slice
127 auto CommitLine = [&](size_t start, size_t count, const std::string& lineStr, double advanceW) {
128 double lineAscent = 0.0;
129 double lineHeight = 0.0;
130 m_target->GetLineMetricsForRange(start, start + count, lineAscent, lineHeight);
131
132 TextLineLayout l;
133 l.startCharIndex = start;
134 l.charCount = count;
135 l.localY = curY;
136 l.height = lineHeight;
137 l.baselineY = curY + lineAscent;
138 l.width = advanceW;
139 l.text = lineStr;
140 m_lines.push_back(l);
141
142 curY += lineHeight;
143 if (advanceW > maxLineWidth) {
144 maxLineWidth = advanceW;
145 }
146 };
147
148 if (fullText.empty()) {
149 CommitLine(0, 0, "", 0.0);
150 } else {
151 size_t i = 0;
152 while (i < fullText.size()) {
153 size_t newlinePos = fullText.find('\n', i);
154 size_t paraEnd = (newlinePos != std::string::npos) ? newlinePos : fullText.size();
155 size_t paraLen = paraEnd - i;
156 size_t paraStart = i;
157
158 if (!m_target->isWrap || maxWrapWidth <= 10.0) {
159 // No word wrapping — measure entire paragraph range
160 double w = m_target->MeasureRange(paraStart, paraEnd);
161 CommitLine(paraStart, paraLen, fullText.substr(paraStart, paraLen), w);
162 } else {
163 // Word wrapping using per-span width measurements
164 size_t lineStartIdx = paraStart;
165 size_t scanIdx = paraStart;
166
167 while (scanIdx < paraEnd) {
168 size_t nextSpace = fullText.find(' ', scanIdx);
169 if (nextSpace > paraEnd) nextSpace = std::string::npos;
170 size_t wordEnd = (nextSpace != std::string::npos) ? nextSpace : paraEnd;
171 size_t candidateEnd = (nextSpace != std::string::npos) ? nextSpace + 1 : paraEnd;
172
173 double testW = m_target->MeasureRange(lineStartIdx, wordEnd);
174 if (testW <= maxWrapWidth || scanIdx == lineStartIdx) {
175 scanIdx = candidateEnd;
176 } else {
177 size_t actualLineEnd = scanIdx;
178 if (actualLineEnd > lineStartIdx && fullText[actualLineEnd - 1] == ' ') {
179 actualLineEnd--;
180 }
181 double curW = m_target->MeasureRange(lineStartIdx, actualLineEnd);
182 CommitLine(lineStartIdx, actualLineEnd - lineStartIdx, fullText.substr(lineStartIdx, actualLineEnd - lineStartIdx), curW);
183 lineStartIdx = scanIdx;
184 scanIdx = candidateEnd;
185 }
186 }
187
188 if (lineStartIdx < paraEnd) {
189 double curW = m_target->MeasureRange(lineStartIdx, paraEnd);
190 CommitLine(lineStartIdx, paraEnd - lineStartIdx, fullText.substr(lineStartIdx, paraEnd - lineStartIdx), curW);
191 }
192 }
193
194 if (newlinePos != std::string::npos) {
195 i = newlinePos + 1;
196 } else {
197 break;
198 }
199 }
200 }
201
202 // Dynamic vertical expansion: ensure text box contains all lines
203 double neededHeight = curY + 4.0; // 4mm bottom margin
204 if (neededHeight > m_target->worldHeight) {
205 m_target->worldHeight = neededHeight;
206 m_target->UpdateBounds();
207 }
208
209 // Dynamic horizontal expansion if not wrapping
210 if (!m_target->isWrap) {
211 double neededWidth = maxLineWidth + 8.0;
212 if (neededWidth > m_target->worldWidth) {
213 m_target->worldWidth = neededWidth;
214 m_target->UpdateBounds();
215 }
216 }
217}

257void TextEditorState::OnKeyDown(int32_t keycode, uint16_t keymod) {
258 if (!m_target) return;
259
260 bool shiftHeld = (keymod & SDL_KMOD_SHIFT) != 0;
261
262 switch (keycode) {
263 case SDLK_BACKSPACE: {
264 if (HasSelection()) {
265 DeleteSelection();
266 } else if (m_cursorIndex > 0) {
267 // Remove previous UTF-8 character
268 size_t prevIndex = m_cursorIndex - 1;
269 while (prevIndex > 0 && (m_target->text[prevIndex] & 0xC0) == 0x80) {
270 --prevIndex;
271 }
272 m_target->text.erase(prevIndex, m_cursorIndex - prevIndex);
273 m_cursorIndex = prevIndex;
274 m_selectionAnchor = m_cursorIndex;
275 m_target->SyncTextToRuns();
276 m_target->isDirty = true;
277 ReflowLayout();
278 }
279 break;
280 }
281
282 case SDLK_DELETE: {
283 if (HasSelection()) {
284 DeleteSelection();
285 } else if (m_cursorIndex < m_target->text.size()) {
286 size_t nextIndex = m_cursorIndex + 1;
287 while (nextIndex < m_target->text.size() && (m_target->text[nextIndex] & 0xC0) == 0x80) {
288 ++nextIndex;
289 }
290 m_target->text.erase(m_cursorIndex, nextIndex - m_cursorIndex);
291 m_target->SyncTextToRuns();
292 m_target->isDirty = true;
293 ReflowLayout();
294 }
295 break;
296 }
297
298 case SDLK_RETURN:
299 case SDLK_KP_ENTER: {
300 OnTextInput("\n");
301 break;
302 }
303
304 case SDLK_LEFT: {
305 if (!shiftHeld && HasSelection()) {
306 m_cursorIndex = SelectionStart();
307 ClearSelection();
308 } else if (m_cursorIndex > 0) {
309 size_t prevIndex = m_cursorIndex - 1;
310 while (prevIndex > 0 && (m_target->text[prevIndex] & 0xC0) == 0x80) {
311 --prevIndex;
312 }
313 MoveCursorTo(prevIndex, shiftHeld);
314 }
315 break;
316 }
317
318 case SDLK_RIGHT: {
319 if (!shiftHeld && HasSelection()) {
320 m_cursorIndex = SelectionEnd();
321 ClearSelection();
322 } else if (m_cursorIndex < m_target->text.size()) {
323 size_t nextIndex = m_cursorIndex + 1;
324 while (nextIndex < m_target->text.size() && (m_target->text[nextIndex] & 0xC0) == 0x80) {
325 ++nextIndex;
326 }
327 MoveCursorTo(nextIndex, shiftHeld);
328 }
329 break;
330 }
331
332 case SDLK_HOME: {
333 // Find start of current line
334 for (const auto& l : m_lines) {
335 if (m_cursorIndex >= l.startCharIndex && m_cursorIndex <= l.startCharIndex + l.charCount) {
336 MoveCursorTo(l.startCharIndex, shiftHeld);
337 break;
338 }
339 }
340 break;
341 }
342
343 case SDLK_END: {
344 // Find end of current line
345 for (const auto& l : m_lines) {
346 if (m_cursorIndex >= l.startCharIndex && m_cursorIndex <= l.startCharIndex + l.charCount) {
347 MoveCursorTo(l.startCharIndex + l.charCount, shiftHeld);
348 break;
349 }
350 }
351 break;
352 }
353
354 case SDLK_UP: {
355 Point2D pt = GetCursorWorldPos();
356 double lineHeight = GetCaretHeight();
357 OnMouseDown(pt.x, pt.y - lineHeight * 0.5, shiftHeld);
358 break;
359 }
360
361 case SDLK_DOWN: {
362 Point2D pt = GetCursorWorldPos();
363 double lineHeight = GetCaretHeight();
364 OnMouseDown(pt.x, pt.y + lineHeight * 1.5, shiftHeld);
365 break;
366 }
367
368 default:
369 break;
370 }
371
372 m_caretVisible = true;
373 m_lastBlinkTimeSec = 0.0;
374}

383size_t TextEditorState::FindClosestCharIndex(double worldX, double worldY) const {
384 if (!m_target || m_lines.empty()) return 0;
385
386 double localX = worldX - m_target->worldX - 2.0; // subtract left margin
387 double localY = worldY - m_target->worldY;
388
389 // 1. Locate line by localY
390 const TextLineLayout* bestLine = &m_lines[0];
391 for (const auto& l : m_lines) {
392 if (localY >= l.localY && localY < l.localY + l.height) {
393 bestLine = &l;
394 break;
395 }
396 if (localY >= l.localY + l.height) {
397 bestLine = &l; // furthest down so far
398 }
399 }
400
401 // 2. Locate character within bestLine using per-span font measurements
402 size_t closestOffset = 0;
403 double minDiff = 1e9;
404
405 for (size_t c = 0; c <= bestLine->charCount; ++c) {
406 double advance = m_target->MeasureRange(bestLine->startCharIndex, bestLine->startCharIndex + c);
407 double diff = std::abs(advance - localX);
408 if (diff < minDiff) {
409 minDiff = diff;
410 closestOffset = c;
411 }
412 }
413
414 return bestLine->startCharIndex + closestOffset;
415}

435Point2D TextEditorState::GetCursorWorldPos() const {
436 if (!m_target || m_lines.empty()) {
437 return Point2D{m_target ? m_target->worldX + 2.0 : 0.0, m_target ? m_target->worldY + 2.0 : 0.0};
438 }
439
440 for (const auto& l : m_lines) {
441 if (m_cursorIndex >= l.startCharIndex && m_cursorIndex <= l.startCharIndex + l.charCount) {
442 double adv = m_target->MeasureRange(l.startCharIndex, m_cursorIndex);
443 return Point2D{m_target->worldX + 2.0 + adv, m_target->worldY + l.localY};
444 }
445 }
446
447 // Fallback: end of last line
448 const auto& lastLine = m_lines.back();
449 double adv = m_target->MeasureRange(lastLine.startCharIndex, lastLine.startCharIndex + lastLine.charCount);
450 return Point2D{m_target->worldX + 2.0 + adv, m_target->worldY + lastLine.localY};
451}

460std::vector<AABB> TextEditorState::GetSelectionBoxes() const {
461 std::vector<AABB> boxes;
462 if (!HasSelection() || !m_target || m_lines.empty()) return boxes;
463
464 size_t selStart = SelectionStart();
465 size_t selEnd = SelectionEnd();
466
467 for (const auto& l : m_lines) {
468 size_t lineEnd = l.startCharIndex + l.charCount;
469 if (selEnd <= l.startCharIndex || selStart >= lineEnd) {
470 continue; // line not in selection
471 }
472
473 size_t boxStartIdx = (std::max)(selStart, l.startCharIndex);
474 size_t boxEndIdx = (std::min)(selEnd, lineEnd);
475
476 double x0 = m_target->MeasureRange(l.startCharIndex, boxStartIdx);
477 double x1 = m_target->MeasureRange(l.startCharIndex, boxEndIdx);
478
479 double wx0 = m_target->worldX + 2.0 + x0;
480 double wx1 = m_target->worldX + 2.0 + x1;
481 double wy0 = m_target->worldY + l.localY;
482 double wy1 = wy0 + l.height;
483
484 boxes.push_back(AABB(wx0, wy0, wx1, wy1));
485 }
486
487 return boxes;
488}
