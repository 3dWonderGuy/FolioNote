2/**
3 * @file link_object.hpp
4 * @brief Standalone canvas hyperlink — external URL or cross-note anchor chip.
5 *
6 * LinkObject is a non-resizable pill badge on the canvas that represents a
7 * hyperlink. It can point to:
8 *
9 * 1. External URL: "https://..." — opens in system default browser
10 * 2. Cross-note anchor: "folio://notebook-uuid/page-id#selection-id"
11 * — navigates within FolioNote to another page or selection
12 *
13 * This object is distinct from INLINE links inside a TextBoxObject:
14 * - TextRun.linkRef references a LinkObject by UUID for inline hyperlinks
15 * (the link is anchored to a word/phrase inside a text box)
16 * - LinkObject as a standalone canvas entity is a free-floating chip that
17 * you can drop anywhere on the canvas, independent of text content
18 *
19 * Interaction:
20 * - Non-resizable: fixed-height pill, width expands with displayText.
21 * - Moveable: body-drag via standard gizmo.
22 * - Double-click (Phase 2): triggers Navigate().
23 *
24 * Scalability:
25 * Add new URL scheme handlers in Navigate(). No other files change.
26 * When cross-note navigation is implemented, the "folio://" handler
27 * will call into the document session router.
28 */

144 // TRANSFORM — Translation Only (Locked Scale & Rotation)

147 /**
148 * @brief Applies a 2D affine transformation while locking card dimensions and rotation.
149 *
150 * Mathematical Process:
151 * Maps anchor (worldX, worldY) through matrix M:
152 * x_new = (worldX * m00) + (worldY * m10) + m20
153 * y_new = (worldX * m01) + (worldY * m11) + m21
154 * Extracts delta translation dx = x_new - worldX, dy = y_new - worldY and applies
155 * only pure translation T(dx, dy) to maintain fixed bookmark card geometry.
156 *
157 * @param matrix 2D affine transformation matrix.
158 */
159 void ApplyTransform(const BLMatrix2D& matrix) override {
160 const double newX = (worldX * matrix.m00) + (worldY * matrix.m10) + matrix.m20;
161 const double newY = (worldX * matrix.m01) + (worldY * matrix.m11) + matrix.m21;
162
163 const double dx = newX - worldX;
164 const double dy = newY - worldY;
165
166 BLMatrix2D translationOnly = BLMatrix2D::make_translation(dx, dy);
167 transform.post_transform(translationOnly);
168 UpdateBounds();
169 }
170
171 /**
172 * @brief Bakes accumulated translation into worldX/Y and resets transform to identity.
173 */
174 void BakeTransform() override {
175 worldX += transform.m20;
176 worldY += transform.m21;
177 transform = BLMatrix2D::make_identity();
178 UpdateBounds();
179 }
180

185 /**
186 * @brief Renders a pill-shaped hyperlink badge.
187 *
188 * External links: blue pill with chain-link icon placeholder
189 * Cross-note links: teal/green pill with internal-arrow icon placeholder
190 *
191 * Phase 2: replace placeholder icon with actual vector icon glyphs and
192 * BLFont displayText rendering.
193 */
194 void Render(BLContext& ctx, const Viewport& /*viewport*/) const override {
195 if (!isVisible) return;
196
197 ctx.save();
198 ctx.apply_transform(transform);
199
200 const double x = worldX;
201 const double y = worldY;
202 const double cw = GetChipWidth();
203 const double ch = chipH;
204 const double r = ch * 0.5; // Fully rounded pill
205
206 // Pill color: blue = external, teal = cross-note
207 BLRgba32 pillCol = IsCrossNote()
208 ? BLRgba32(0x00, 0x82, 0x72, static_cast<uint8_t>(opacity * 200))
209 : BLRgba32(0x00, 0x78, 0xD4, static_cast<uint8_t>(opacity * 200));
210
211 ctx.set_fill_style(pillCol);
212 ctx.fill_round_rect(BLRoundRect(x, y, cw, ch, r, r));
213
214 // Link icon placeholder (small circle on the left side of pill)
215 ctx.set_fill_style(BLRgba32(0xFF, 0xFF, 0xFF, 180));
216 ctx.fill_circle(x + r, y + ch * 0.5, 1.2);
217
218 // Subtle border
219 ctx.set_stroke_style(BLRgba32(0xFF, 0xFF, 0xFF, 40));
220 ctx.set_stroke_width(0.3);
221 ctx.stroke_round_rect(BLRoundRect(x, y, cw, ch, r, r));
222
223 // TODO Phase 2: render displayText using BLFont
224
225 ctx.restore();
226 }
