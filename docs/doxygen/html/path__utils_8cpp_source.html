19std::filesystem::path PathUtils::Utf8ToNativePath(const std::string& utf8Str) {
20 if (utf8Str.empty()) {
21 return std::filesystem::path();
22 }
23#if defined(_WIN32)
24 int sizeNeeded = MultiByteToWideChar(CP_UTF8, 0, utf8Str.data(), static_cast<int>(utf8Str.size()), nullptr, 0);
25 if (sizeNeeded <= 0) {
26 return std::filesystem::path(utf8Str);
27 }
28 std::wstring wstr(static_cast<size_t>(sizeNeeded), 0);
29 MultiByteToWideChar(CP_UTF8, 0, utf8Str.data(), static_cast<int>(utf8Str.size()), &wstr[0], sizeNeeded);
30
31 // If path is long (>= 240 chars) and is an absolute drive path ("C:\..."), normalize to backslashes and prepend "\\?\"
32 // to bypass legacy Win32 MAX_PATH (260 character) limits for deeply nested notebooks and pages.
33 if (wstr.size() >= 240 && wstr.rfind(L"\\\\?\\", 0) == std::wstring::npos) {
34 if (wstr.size() >= 3 && std::iswalpha(wstr[0]) && wstr[1] == L':' && (wstr[2] == L'\\' || wstr[2] == L'/')) {
35 std::replace(wstr.begin(), wstr.end(), L'/', L'\\');
36 wstr.insert(0, L"\\\\?\\");
37 }
38 }
39 return std::filesystem::path(std::move(wstr));
40#else
41 return std::filesystem::path(utf8Str);
42#endif
43}

45std::string PathUtils::NativePathToUtf8(const std::filesystem::path& p) {
46#if defined(_WIN32)
47 std::wstring wstr = p.native();
48 if (wstr.empty()) {
49 return std::string();
50 }
51 // Strip Win32 extended-length prefix if present so user-facing strings remain canonical
52 if (wstr.rfind(L"\\\\?\\", 0) == 0) {
53 wstr.erase(0, 4);
54 }
55 int sizeNeeded = WideCharToMultiByte(CP_UTF8, 0, wstr.data(), static_cast<int>(wstr.size()), nullptr, 0, nullptr, nullptr);
56 if (sizeNeeded <= 0) {
57 return p.string();
58 }
59 std::string str(static_cast<size_t>(sizeNeeded), 0);
60 WideCharToMultiByte(CP_UTF8, 0, wstr.data(), static_cast<int>(wstr.size()), &str[0], sizeNeeded, nullptr, nullptr);
61 return str;
62#else
63 return p.string();
64#endif
65}

67std::string PathUtils::NormalizeSeparators(const std::string& path) {
68 if (path.empty()) return "";
69 std::string normalized = path;
70 for (char& c : normalized) {
71 if (c == '\\') c = '/';
72 }
73 while (normalized.size() > 1 && normalized.back() == '/') {
74 if (normalized.size() == 3 && normalized[1] == ':') break;
75 normalized.pop_back();
76 }
77 return normalized;
78}

80std::string PathUtils::JoinPath(const std::string& base, const std::string& child) {
81 if (base.empty()) return NormalizeSeparators(child);
82 if (child.empty()) return NormalizeSeparators(base);
83
84 std::string b = NormalizeSeparators(base);
85 std::string c = NormalizeSeparators(child);
86
87 if (b.back() == '/') {
88 if (c.front() == '/') {
89 return b + c.substr(1);
90 }
91 return b + c;
92 }
93 if (c.front() == '/') {
94 return b + c;
95 }
96 return b + "/" + c;
97}

149std::string PathUtils::SanitizeFileName(const std::string& name, char replacement) {
150 if (name.empty()) return "Untitled";
151
152 std::string safe = name;
153 for (char& c : safe) {
154 if (c == '/' || c == '\\' || c == ':' || c == '*' || c == '?' ||
155 c == '"' || c == '<' || c == '>' || c == '|' ||
156 static_cast<unsigned char>(c) < 32) {
157 c = replacement;
158 }
159 }
160
161 while (!safe.empty() && (safe.back() == ' ' || safe.back() == '.')) {
162 safe.pop_back();
163 }
164 return safe.empty() ? "Untitled" : safe;
165}

189std::string PathUtils::DisambiguatePath(const std::string& parentDir, const std::string& baseStem, const std::string& extension) {
190 std::string cleanStem = SanitizeFileName(baseStem.empty() ? "Untitled" : baseStem);
191 std::string ext = extension;

193 ext = "." + ext;
