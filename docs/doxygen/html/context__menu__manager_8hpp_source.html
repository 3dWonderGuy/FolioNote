2/**
3 * =========================================================================================
4 * @file app/context_menu_manager.hpp
5 * @brief Centralized, Decoupled Context Menu Management & Presentation Service for FolioNote
6 * =========================================================================================
7 *
8 * ARCHITECTURAL DESIGN & PURPOSE:
9 * Context menus in FolioNote are an application-level UI service, not an object-level concern.
10 * A single unified system coordinates context interactions across disparate domains:
11 *
12 * 1. Canvas Background:
13 * Paste, Change Paper Style (Submenu), Toggle Grid, Reset Zoom, Reset View, Create Link to Page.
14 *
15 * 2. Canvas Objects:
16 * Universal transformations (Cut, Copy, Duplicate, Stacking/Layering Submenu, Delete)
17 * plus domain-specific actions (e.g., AttachmentObject: Open, Re-link, Copy Path).
18 *
19 * 3. Navigation Elements:
20 * Section tabs (Rename, Color tag, Password protection, Delete), Page entries (Duplicate, Move).
21 *
22 * 4. Toolbar & Ribbon Presets:
23 * Pen style customization, Favorite slot management.
24 *
25 * KEY PRINCIPLES:
26 * - Decoupled from CanvasObject: CanvasObject remains pure vector geometry & rendering without
27 * inheriting or depending on ImGui UI presentation code.
28 * - Single Styling Authority: All menus render through ContextMenuThemeScope, ensuring pixel-perfect
29 * theming, rounded corners, icons, and keyboard shortcuts across light and dark modes.
30 * - Dynamic Action Dispatch: Actions are encapsulated in std::function<void()> callbacks,
31 * supporting undo/redo history recording and seamless async modal triggering.
32 * - Hierarchical Submenus: Automatically detects when an action contains `subActions` and
33 * recursively renders ImGui cascading submenus (`ImGui::BeginMenu` / `ImGui::EndMenu`).
34 */

171 /**
172 * @brief Recursively renders a hierarchy of action items into ImGui popups and submenus.
173 *
174 * GENERAL WORKING PROCESS:
175 * 1. Evaluates per-item dynamic visibility: skips items where `item.IsVisible() == false`.
176 * 2. Renders horizontal separators if `item.isSeparatorBefore` is true.
177 * 3. Checks if `item.HasSubActions()`:
178 * - If child actions exist, invokes `ImGui::BeginMenu(label, isEnabled)`.
179 * - Recursively calls `RenderActionList(item.subActions)`.
180 * - Concludes with `ImGui::EndMenu()`.
181 * 4. If leaf action, renders `ImGui::MenuItem(label, shortcut, isChecked, isEnabled)`.
182 * - On activation, executes `item.Execute()`.
183 *
184 * @param items List of actions or sub-actions to render.
185 */
186 template <typename ActionContainer>
187 void RenderActionList(const ActionContainer& items) {
188 for (const auto& item : items) {
189 if (!item.IsVisible()) {
190 continue;
191 }
192
193 if (item.isSeparatorBefore) {
194 ImGui::Separator();
195 }
196
197 // Push destructive red styling if marked
198 if (item.isDestructive) {
199 ImGui::PushStyleColor(ImGuiCol_Text, ImVec4(0.95f, 0.28f, 0.28f, 1.0f));
200 }
201
202 std::string displayLabel = (item.icon.empty() ? "" : item.icon + " ") + item.label;
203 const char* shortcutCStr = item.shortcut.empty() ? nullptr : item.shortcut.c_str();
204
205 if (item.HasSubActions()) {
206 // Cascading Submenu
207 if (ImGui::BeginMenu(displayLabel.c_str(), item.IsEnabled())) {
208 RenderActionList(item.subActions);
209 ImGui::EndMenu();
210 }
211 } else {
212 // Leaf Command Item
213 if (ImGui::MenuItem(displayLabel.c_str(), shortcutCStr, item.IsChecked(), item.IsEnabled())) {
214 item.Execute();
215 }
216 }
217
218 if (item.isDestructive) {
219 ImGui::PopStyleColor();
220 }
221 }
222 }

224 /**
225 * @brief Renders the active context menu using unified ImGui theming.
226 *
227 * Working Process:
228 * 1. If `m_requestOpen` is flagged, triggers `ImGui::OpenPopup(m_popupId)`.
229 * 2. Applies `ContextMenuThemeScope` for smooth padding, rounded corners, and consistent dark/light styling.
230 * 3. If open, renders header badge (if configured).
231 * 4. Delegates to `RenderActionList` to render all items and submenus hierarchically.
232 * 5. Resets state upon popup closure.
233 *
234 * @param themeManager Active application ThemeManager reference.
235 */
236 void Render(const ThemeManager& themeManager) {
237 if (m_requestOpen) {
238 ImGui::OpenPopup(m_popupId.c_str());
239 m_requestOpen = false;
240 }
241
242 ContextMenuThemeScope ctxScope(themeManager);
243 if (ImGui::BeginPopup(m_popupId.c_str())) {
244 // Optional Header Title
245 if (!m_headerTitle.empty()) {
246 ImGui::PushFont(FolioTheme::FontNavBoldLarge ? FolioTheme::FontNavBoldLarge : FolioTheme::FontBold);
247 std::string headerText = (m_headerIcon.empty() ? "" : m_headerIcon + " ") + m_headerTitle;
248 ImGui::TextColored(themeManager.colorPrimary, "%s", headerText.c_str());
249 ImGui::PopFont();
250 ImGui::Separator();
251 }
252
253 // Render all items hierarchically
254 RenderActionList(m_items);
255
256 ImGui::EndPopup();
257 } else {
258 // Popup closed by clicking outside
259 if (m_target != ContextMenuTarget::None && !m_requestOpen) {
260 m_target = ContextMenuTarget::None;
261 m_targetObject = nullptr;
262 m_items.clear();
263 }
264 }
265 }

275 /**
276 * @brief Populates menu actions for empty canvas background clicks.
277 *
278 * WORKING PROCESS & UX FLOW:
279 * 1. Detects clipboard availability across both session clipboard (CanvasObjects) and OS text clipboard.
280 * 2. Paste action intelligently prioritizes session geometric objects; falls back to creating a TextBox.
281 * 3. Provides global selection actions (Select All, Clear Selection).
282 * 4. Provides hierarchical submenus for:
283 * - Paper Style (Blank, Ruled, Grid, Dots)
284 * - View Controls (Reset Zoom, 100%, Home Origin)
285 *
286 * @param session Active document session providing selection, clipboard, and page models.
287 * @param canvas Canvas engine providing viewport transformations and rendering state.
288 */
289 void BuildBackgroundMenu(DocumentSession& session, CanvasEngine& canvas) {
290 auto activePg = session.GetActivePage();
291
292 // 1. Paste (handles either copied CanvasObjects or OS clipboard plain text)
293 bool canPaste = session.HasClipboardContent() || SDL_HasClipboardText();
294 m_items.push_back(ContextMenuItem{
295 "Paste", "Ctrl+V", "📋", false, canPaste, false,
296 [&session, &canvas, clickPos = m_clickWorldPos]() {
297 auto pg = session.GetActivePage();
298 if (!pg) return;
299 if (session.HasClipboardContent()) {
300 session.PasteObjects(clickPos.x, clickPos.y);
301 canvas.isDirty = true;
302 canvas.needsFullRebake = true;
303 } else if (SDL_HasClipboardText()) {
304 char* clipText = SDL_GetClipboardText();
305 if (clipText) {
306 if (clipText[0] != '\0') {
307 auto tb = std::make_shared<Folio::TextBoxObject>();
308 tb->worldX = clickPos.x;
309 tb->worldY = clickPos.y;
310 tb->text = clipText;
311 tb->UpdateBounds();
312 session.AddTextBox(tb);
313 canvas.isDirty = true;
314 canvas.needsFullRebake = true;
315 }
316 SDL_free(clipText);
317 }
318 }
319 }
320 });
321
322 // 2. Select All
323 m_items.push_back(ContextMenuItem{
324 "Select All", "Ctrl+A", "⬚", false, (activePg && !activePg->objects.empty()), false,
325 [&session, &canvas]() {
326 auto allObjs = session.QueryVisible(canvas.GetViewport());
327 for (auto& obj : allObjs) {
328 if (obj) obj->isSelected = 1;
329 }
330 canvas.selectionGizmo.SetSelectedObjects(allObjs);
331 canvas.isDirty = true;
332 }
333 });
334
335 // 3. Clear Selection (if selection active)
336 if (canvas.selectionGizmo.HasSelection()) {
337 m_items.push_back(ContextMenuItem{
338 "Clear Selection", "Esc", "✕", false, true, false,
339 [&canvas]() {
340 canvas.selectionGizmo.ClearSelection();
341 canvas.isDirty = true;
342 }
343 });
344 }
345
346 // 4. Submenu: Paper Style
347 if (activePg) {
348 ContextMenuItem paperSubmenu;
349 paperSubmenu.label = "Paper Style";
350 paperSubmenu.icon = "📝";
351 paperSubmenu.isSeparatorBefore = true;
352
353 const struct { const char* label; PaperStyle style; } styles[] = {
354 { "Blank / Plain", PaperStyle::Blank },
355 { "Ruled / Lined", PaperStyle::Lined },
356 { "Grid / Graph", PaperStyle::Grid },
357 { "Dotted", PaperStyle::Dotted },
358 { "Cornell Notes", PaperStyle::Cornell }
359 };
360
361 for (const auto& s : styles) {
362 ContextMenuItem styleItem;
363 styleItem.label = s.label;
364 styleItem.isSticky = (activePg->paperStyle == s.style);
365 styleItem.onTrigger = [activePg, &canvas, st = s.style]() {
366 activePg->paperStyle = st;
367 canvas.currentPaperStyle = st;
368 canvas.isDirty = true;
369 canvas.needsFullRebake = true;
370 };
371 paperSubmenu.AddSubAction(std::move(styleItem));
372 }
373 m_items.push_back(std::move(paperSubmenu));
374 }
375
376 // 5. Submenu: View Controls
377 {
378 ContextMenuItem viewSubmenu;
379 viewSubmenu.label = "View";
380 viewSubmenu.icon = "👁";
381 viewSubmenu.isSeparatorBefore = false;
382
383 ContextMenuItem resetZoom;
384 resetZoom.label = "Reset Zoom (100%)";
385 resetZoom.shortcut = "Ctrl+0";
386 resetZoom.icon = "🔍";
387 resetZoom.onTrigger = [&canvas]() {
388 canvas.transform.zoom = 1.0;
389 canvas.isDirty = true;
390 canvas.needsFullRebake = true;
391 };
392 viewSubmenu.AddSubAction(std::move(resetZoom));
393
394 ContextMenuItem resetOrigin;
395 resetOrigin.label = "Reset View to Origin";
396 resetOrigin.shortcut = "Home";
397 resetOrigin.icon = "⌂";
398 resetOrigin.onTrigger = [&canvas]() {
399 canvas.transform.panXMm = 0.0;
400 canvas.transform.panYMm = 0.0;
401 canvas.transform.zoom = 1.0;
402 canvas.isDirty = true;
403 canvas.needsFullRebake = true;
404 };
405 viewSubmenu.AddSubAction(std::move(resetOrigin));
406
407 m_items.push_back(std::move(viewSubmenu));
408 }
409 }

411 /**
412 * @brief Populates menu actions for a clicked or selected canvas object.
413 *
414 * WORKING PROCESS & ARCHITECTURAL DECOUPLING:
415 * 1. Extracts contextual metadata (display name, type glyph) for header display.
416 * 2. Configures a pure CanvasActionContext providing UI-agnostic functors for:
417 * - Bring to Front / Send to Back (via CanvasPage stacking manipulation or Order submenu)
418 * - Object deletion with undo history recording
419 * - Object duplication
420 * - Attachment file replacement dialogs
421 * 3. Dispatches BuildActionsForObject() to ObjectActionRegistry.
422 *
423 * @param obj Target CanvasObject clicked or selected.
424 * @param session Active document session providing document operations and command history.
425 * @param canvas Canvas engine providing spatial coordinates and rendering state.
426 */
427 void BuildObjectMenu(std::shared_ptr<CanvasObject> obj, DocumentSession& session, CanvasEngine& canvas) {
428 if (!obj) return;
429 auto activePg = session.GetActivePage();
430
431 if (obj->type == ObjectType::AttachmentFile) {
432 auto attach = std::dynamic_pointer_cast<AttachmentObject>(obj);
433 if (attach) {
434 m_headerTitle = attach->displayName;
435 m_headerIcon = attach->isEmbedded ? "📦" : "📎";
436 }
437 } else {
438 m_headerTitle = "Selection";
439 m_headerIcon = "⬚";
440 }
441
442 // Delegate to ObjectActionRegistry in core/objects:
443 // Automatically provides default actions (Delete, Order Submenu, Duplicate)
444 // and queries obj->CustomizeActions(actions) for domain-specific commands.
445 std::vector<ContextMenuItem> actions = ObjectActionRegistry::BuildActionsForObject(
446 obj, activePg, canvas, session
447 );
448
449 for (auto& act : actions) {
450 m_items.push_back(std::move(act));
451 }
452 }
453
454 // =========================================================================
455 // STATE FIELDS
456 // =========================================================================
457 ContextMenuTarget m_target = ContextMenuTarget::None;
458 Point2D m_clickWorldPos{0.0, 0.0};
459 std::shared_ptr<CanvasObject> m_targetObject = nullptr;
460 std::string m_headerTitle = "";
461 std::string m_headerIcon = "";
462 std::vector<ContextMenuItem> m_items;
463 bool m_requestOpen = false;
464 const std::string m_popupId = "##GlobalFolioContextMenu";
465};
