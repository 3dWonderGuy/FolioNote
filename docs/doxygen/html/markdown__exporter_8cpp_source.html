33std::string MarkdownExporter::FormatMarkdownText(const std::string& rawText, const ExportOptions& options) {
34 if (!options.enableLinks) {
35 return rawText;
36 }
37
38 // Convert URLs into Markdown links: [url](url)
39 static const std::regex urlRegex(R"((https?://[^\s<>"]+|folionote://[^\s<>"]+))");
40 std::string result;
41 auto wordsBegin = std::sregex_iterator(rawText.begin(), rawText.end(), urlRegex);
42 auto wordsEnd = std::sregex_iterator();
43
44 size_t lastPos = 0;
45 for (auto it = wordsBegin; it != wordsEnd; ++it) {
46 std::smatch match = *it;
47 result += rawText.substr(lastPos, match.position() - lastPos);
48
49 std::string url = match.str();
50 if (url.rfind("folionote://", 0) == 0) {
51 if (options.enableAppDeepLinks) {
52 result += "[Open in FolioNote](" + url + ")";
53 } else {
54 result += url;
55 }
56 } else if (options.enableWebLinks) {
57 result += "[" + url + "](" + url + ")";
58 } else {
59 result += url;
60 }
61
62 lastPos = match.position() + match.length();
63 }
64 result += rawText.substr(lastPos);
65 return result;
66}

68std::string MarkdownExporter::ExportInkToSvg(const CanvasPage& page, const std::string& assetsDir) {
69 if (page.objects.empty()) return "";
70
71 std::ostringstream svgStrokes;
72 int strokeCount = 0;
73 double minX = 1e9, minY = 1e9, maxX = -1e9, maxY = -1e9;
74
75 for (const auto& obj : page.objects) {
76 if (auto ink = std::dynamic_pointer_cast<InkContainer>(obj)) {
77 for (const auto& stroke : ink->strokes) {
78 if (stroke.segments.empty()) continue;
79 strokeCount++;
80
81 uint32_t val = stroke.color.value;
82 uint8_t a = (val >> 24) & 0xFF;
83 uint8_t r = (val >> 16) & 0xFF;
84 uint8_t g = (val >> 8) & 0xFF;
85 uint8_t b = val & 0xFF;
86 float alpha = a / 255.0f;
87 float w = std::max(0.5f, static_cast<float>(stroke.baseWidth));
88
89 svgStrokes << " <polyline fill=\"none\" stroke=\"rgba(" << (int)r << "," << (int)g << "," << (int)b << "," << alpha << ")\" "
90 << "stroke-width=\"" << w << "\" stroke-linecap=\"round\" stroke-linejoin=\"round\" points=\"";
91 svgStrokes << stroke.segments[0].p0.x << "," << stroke.segments[0].p0.y << " ";
92
93 minX = std::min(minX, stroke.segments[0].p0.x);
94 minY = std::min(minY, stroke.segments[0].p0.y);
95 maxX = std::max(maxX, stroke.segments[0].p0.x);
96 maxY = std::max(maxY, stroke.segments[0].p0.y);
97
98 for (const auto& seg : stroke.segments) {
99 svgStrokes << seg.p1.x << "," << seg.p1.y << " ";
100 minX = std::min(minX, seg.p1.x);
101 minY = std::min(minY, seg.p1.y);
102 maxX = std::max(maxX, seg.p1.x);
103 maxY = std::max(maxY, seg.p1.y);
104 }
105 svgStrokes << "\" />\n";
106 }
107 }
108 }
109
110 if (strokeCount == 0) return "";
111
112 double padding = 20.0;
113 minX -= padding;
114 minY -= padding;
115 double width = std::max(100.0, (maxX - minX) + padding * 2.0);
116 double height = std::max(100.0, (maxY - minY) + padding * 2.0);
117
118 std::ostringstream svg;
119 svg << "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n";
120 svg << "<svg xmlns=\"http://www.w3.org/2000/svg\" viewBox=\""
121 << minX << " " << minY << " " << width << " " << height
122 << "\" width=\"100%\" height=\"auto\">\n";
123 svg << svgStrokes.str();
124 svg << "</svg>\n";
125
126 FileManager::CreateDirectories(assetsDir);
127 std::string svgFilename = page.guid + "_drawing.svg";
128 std::string svgPath = FileManager::JoinPath(assetsDir, svgFilename);
129
130 FileManager::WriteTextAtomic(svgPath, svg.str());
131 return "assets/" + svgFilename;
132}

134std::string MarkdownExporter::ExportPage(
135 const std::shared_ptr<CanvasPage>& page,
136 const std::string& sectionName,
137 const std::string& notebookName,
138 const std::string& outputDirectory,
139 const ExportOptions& options
140) {
141 if (!page) return "";
142 std::ostringstream ss;
143
144 ss << "# " << page->title << "\n\n";
145 ss << "> **Location:** " << notebookName << " &rsaquo; " << sectionName << " \n";
146 if (!page->createdDateStr.empty()) {
147 ss << "> **Created:** " << page->createdDateStr << " " << page->createdTimeStr << "\n\n";
148 }
149
150 // Export hand-drawn vector strokes as SVG asset
151 std::string assetsDir = FileManager::JoinPath(outputDirectory, "assets");
152 std::string svgRelPath = ExportInkToSvg(*page, assetsDir);
153 if (!svgRelPath.empty()) {
154 ss << "![Handwritten Ink Notes](" << svgRelPath << ")\n\n";
155 }
156
157 // Text box contents
158 for (const auto& obj : page->objects) {
159 if (auto tb = std::dynamic_pointer_cast<Folio::TextBoxObject>(obj)) {
160 ss << FormatMarkdownText(tb->text, options) << "\n\n";
161 }
162 }
163
164 return ss.str();
165}

178 ss << "---\n\n";
179
180 for (const auto& pg : section->pages) {
181 if (pg) {
182 ss << ExportPage(pg, section->name, notebookName, outputDirectory, options);
183 ss << "\n---\n\n";
