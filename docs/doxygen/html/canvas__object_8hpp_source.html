113 /**
114 * @brief Recalculates the world-space AABB incorporating the active affine transform.
115 *
116 * MATHEMATICAL PROCESS:
117 * Transforms the four untransformed rectangle corners [ (worldX, worldY),
118 * (worldX + worldWidth, worldY), (worldX + worldWidth, worldY + worldHeight),
119 * (worldX, worldY + worldHeight) ] by `transform` (2x3 affine matrix), then computes
120 * the axis-aligned envelope [minX, minY, maxX, maxY].
121 */
122 virtual void UpdateBounds() {
123 bounds = Folio::AABBUtils::ComputeTransformedBounds(worldX, worldY, worldWidth, worldHeight, transform);
124 }

126 /**
127 * @brief Tests if a world-space point (x, y) intersects the object's boundary.
128 *
129 * Default implementation performs an AABB bounding check. Derived classes with
130 * non-rectangular geometry (e.g. SmartArrow, Shape, Ink) override this for tight geometry testing.
131 *
132 * @param x World X coordinate in millimeters.
133 * @param y World Y coordinate in millimeters.
134 * @return true if point is inside object boundary; false otherwise.
135 */
136 [[nodiscard]] virtual bool HitTest(double x, double y) const {
137 return bounds.Contains(x, y);
138 }
139
140 /**
141 * @brief Tests if a world-space circle (stylus tip, touch contact, or eraser reticle) intersects this object.
142 *
143 * MATHEMATICAL PROCESS:
144 * Clamps circle center (cx, cy) to closest point on bounds [minX..maxX, minY..maxY],
145 * then verifies if squared Euclidean distance <= radiusMm * radiusMm.
146 *
147 * @param cx Circle center X in millimeters.
148 * @param cy Circle center Y in millimeters.
149 * @param radiusMm Radius of intersection circle in millimeters.
150 * @return true if circle overlaps or encloses any part of this object.
151 */
152 [[nodiscard]] virtual bool HitTestCircle(double cx, double cy, double radiusMm) const {
153 return Folio::AABBUtils::IntersectsCircle(bounds, cx, cy, radiusMm);
154 }

184 */
185 virtual void BakeTransform() {

188 }
