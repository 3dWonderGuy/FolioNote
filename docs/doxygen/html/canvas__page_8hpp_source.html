19/**
20 * @brief Formats the current system clock time into human-readable date and time strings.
21 *
22 * MATHEMATICAL WORKING PROCESS & TIME RESOLUTION:
23 * - Samples system wall-clock time via `std::chrono::system_clock::now()` (sub-microsecond resolution).
24 * - Converts to calendar breakdown `std::tm` using thread-safe platform primitives (`localtime_s` on Windows,
25 * `localtime_r` on POSIX).
26 * - Generates formatted strings:
27 * - Date: "%B %d, %Y" (e.g. "September 19, 2026")
28 * - Time: "%I:%M %p" (e.g. "01:05 AM")
29 *
30 * @param[out] outDate Formatted calendar date string.
31 * @param[out] outTime Formatted 12-hour clock time string with AM/PM indicator.
32 */
33inline void PopulateCurrentDateTime(std::string& outDate, std::string& outTime) {
34 auto now = std::chrono::system_clock::now();
35 std::time_t tt = std::chrono::system_clock::to_time_t(now);
36 std::tm tm{};
37#if defined(_WIN32)
38 localtime_s(&tm, &tt);
39#else
40 localtime_r(&tt, &tm);
41#endif
42 char dateBuf[64];
43 char timeBuf[32];
44 std::strftime(dateBuf, sizeof(dateBuf), "%B %d, %Y", &tm);
45 std::strftime(timeBuf, sizeof(timeBuf), "%I:%M %p", &tm);
46 outDate = dateBuf;
47 outTime = timeBuf;
48}

50/**
51 * =========================================================================================
52 * @file canvas_page.hpp
53 * @brief Represents an infinite or bounded 2D canvas drawing surface within a notebook section.
54 * =========================================================================================
55 *
56 * ARCHITECTURAL ROLE:
57 * - The CanvasPage is the core content container of the application.
58 * - Each Section contains an ordered list of CanvasPages.
59 * - An infinite 2D coordinate system is used: objects (ink strokes, images, text boxes, etc)
60 * can be placed anywhere in world-space coordinates (double precision millimeters).
61 *
62 * KEY SUBSYSTEMS INTEGRATED:
63 * 1. Polymorphic Object Storage & Fast Lookups:
64 * Maintains `std::vector<std::shared_ptr<CanvasObject>> objects` for strict z-order rendering,
65 * alongside `std::unordered_map<uint32_t, std::shared_ptr<CanvasObject>> objectMap` providing
66 * instant O(1) UID lookups and eliminating O(K * N) bottlenecks during viewport culling.
67 * 2. Spatial Indexing (R-Tree):
68 * Uses a dynamic R-Tree (`RTree spatialIndex`) storing AABBs (Axis-Aligned Bounding Boxes).
69 * During rendering, `QueryVisible(viewport)` performs an O(log N + K) frustum query so that only
70 * objects visible on the user's screen are processed by Blend2D, guaranteeing 120+ FPS even
71 * with 50,000+ strokes on a single page.
72 * 3. Undo / Redo History:
73 * Owns an isolated `CommandHistory history` instance. Each page tracks its own undo/redo
74 * action stack, preventing command pollution across pages.
75 * 4. Hierarchical Sub-Page Tree (OneNote Style):
76 * Supports tree nesting via `parentPageGuid` and `nestingLevel`:
77 * - Level 0: Main page
78 * - Level 1: Sub-page
79 * - Level 2: Sub-sub-page
80 * Also tracks UI collapse state (`isCollapsed`) for folding child pages in the navigation sidebar.
81 * 5. Per-Page Layout & Paper Templates:
82 * Tracks page-level infinity mode (SemiInfinity, FullInfinity, VerticalScroll, HorizontalScroll),
83 * paper styling (Grid, Lined, Blank, Dotted, Cornell), dimensions (Letter, A4, Custom),
84 * border styles, and DPI calibration factor.
85 * 6. Memory Management & LRU Eviction:
86 * Pages track `lastAccessTimeMs`, `isLoaded`, and `isModified`. When inactive, the storage
87 * repository evicts stroke data from RAM to conserve working set memory while preserving
88 * the lightweight metadata stub.
89 */

318 /**
319 * @brief Finds the highest zOrder of any object spatially intersecting the given query bounds.
320 *
321 * MATHEMATICAL & SPATIAL PROCESS:
322 * -------------------------------
323 * 1. Performs an O(log N) R-Tree bounding box query `spatialIndex.Query(queryBounds)`
324 * to rapidly retrieve candidate UIDs overlapping the new object's footprint.
325 * 2. Resolves each UID via O(1) hash table lookup in `objectMap`.
326 * 3. Calculates max(zOrder) across all overlapping visible objects.
327 *
328 * @param queryBounds The bounding box of the query area / new object.
329 * @return int32_t Highest zOrder found among overlapping objects, or 0 if empty canvas.
330 */
331 [[nodiscard]] int32_t GetHighestZUnder(const AABB& queryBounds) const {
332 if (queryBounds.IsEmpty()) return 0;
333
334 std::vector<uint32_t> candidateUids = spatialIndex.Query(queryBounds);
335 int32_t maxZ = 0;
336
337 for (uint32_t uid : candidateUids) {
338 auto it = objectMap.find(uid);
339 if (it != objectMap.end() && it->second && it->second->isVisible) {
340 if (it->second->bounds.Intersects(queryBounds)) {
341 maxZ = (std::max)(maxZ, it->second->zOrder);
342 }
343 }
344 }
345 return maxZ;
346 }

362 /**
363 * @brief Adds a canvas object to the page, registers it in the R-Tree, and inserts it into the fast UID map.
364 *
365 * MATHEMATICAL & TIME COMPLEXITY PROCESS:
366 * - Auto-Stacking: If `autoStackZ` is true and `obj->zOrder <= 1`, queries `GetHighestZUnder(obj->bounds)`
367 * to place this object directly on top of whatever lies physically underneath it (z = highestUnder + 1).
368 * - Vector Append: O(1) amortized insertion into `objects` to maintain rendering z-order.
369 * - Fast UID Map: O(1) hash insertion into `objectMap[obj->uid] = obj`.
370 * - Spatial Index: O(log N) R-Tree insertion with Axis-Aligned Bounding Box (AABB) expansion.
371 * - Marks page dirty (`isModified = true`) and updates LRU timestamp.
372 *
373 * @param obj Shared pointer to any derived CanvasObject (Ink, Image, TextBox, PDF).
374 * @param autoStackZ If true and obj->zOrder <= 1, auto-increments zOrder above overlapping objects beneath it.
375 */
376 void AddObject(const std::shared_ptr<CanvasObject>& obj, bool autoStackZ = false) {
377 if (!obj) return;
378 if (obj->uid == 0) {
379 obj->uid = UIDGenerator::Next();
380 }
381 auto existingIt = objectMap.find(obj->uid);
382 if (existingIt != objectMap.end()) {
383 if (existingIt->second == obj) {
384 // Object is already registered on this page; avoid duplicate entry
385 return;
386 }
387 // Collision with another object: assign fresh unique runtime UID
388 obj->uid = UIDGenerator::Next();
389 }
390 if (autoStackZ && obj->zOrder <= 1) {
391 int32_t highest = GetHighestZUnder(obj->bounds);
392 if (highest >= 1) {
393 obj->zOrder = highest + 1;
394 }
395 }
396 obj->pageIndex = static_cast<uint32_t>(objects.size());
397 objects.push_back(obj);
398 objectMap[obj->uid] = obj;
399 spatialIndex.Insert(obj->uid, obj->bounds);
400 isModified = true;
401 Touch();
402 }

409 * @param newObj New canvas object to insert into the document.

481 /**
482 * @brief Moves an object to the absolute front of the visual stacking order (highest zOrder).
483 *
484 * MATHEMATICAL & STACKING PROCESS:
485 * -------------------------------
486 * 1. Traverses existing page objects to evaluate the current supremum:
487 * z_max = max_{o in Objects}(o.zOrder)
488 * 2. Sets target.zOrder = z_max + 1 to guarantee precedence during stable zOrder sort.
489 * 3. Relocates target pointer to the end of the `objects` vector, ensuring O(N) cache-friendly
490 * in-order iteration naturally renders this object last (on top).
491 * 4. Flags page as modified and refreshes LRU timestamp.
492 *
493 * @param uid The unique 32-bit runtime ID of the target object.
494 */
495 void BringToFront(uint32_t uid) {
496 auto it = objectMap.find(uid);
497 if (it == objectMap.end() || !it->second) return;
498 auto obj = it->second;
499
500 int32_t maxZ = 1;
501 for (const auto& o : objects) {
502 if (o) maxZ = (std::max)(maxZ, o->zOrder);
503 }
504 obj->zOrder = maxZ + 1;
505
506 auto vit = std::find(objects.begin(), objects.end(), obj);
507 if (vit != objects.end()) {
508 objects.erase(vit);
509 objects.push_back(obj);
510 ReindexObjects();
511 }
512 isModified = true;
513 Touch();

518 *
519 * MATHEMATICAL & STACKING PROCESS:
520 * -------------------------------
521 * 1. Traverses existing page objects (excluding target) to evaluate the current infimum:
522 * z_min = min_{o in Objects \ {target}}(o.zOrder)
523 * 2. Sets target.zOrder = max(1, z_min - 1) ensuring non-negative layer order above canvas background.
524 * 3. Relocates target pointer to the beginning of the `objects` vector, ensuring O(N) cache-friendly
525 * in-order iteration naturally renders this object first (underneath all others).

551 /**
552 * @brief Steps an object forward by one layer in the visual stacking sequence.
553 *
554 * MATHEMATICAL & ORDERING PROCESS:
555 * -------------------------------
556 * 1. Locates the iterator index `vit` of the target object in the `objects` array.
557 * 2. If `vit + 1 != objects.end()`, swaps `*vit` with `*(vit + 1)`.
558 * 3. Reconciles layer integer: ensures `target.zOrder >= neighbor.zOrder` to guarantee
559 * stable sorting precedence while preserving relative layer distances.
560 * 4. Flags page as modified and touches LRU access timestamp.
561 *
562 * @param uid Unique 32-bit runtime ID of target object.
563 */
564 void ShiftFront(uint32_t uid) {
565 auto it = objectMap.find(uid);
566 if (it == objectMap.end() || !it->second) return;
567 auto obj = it->second;
568
569 auto vit = std::find(objects.begin(), objects.end(), obj);
570 if (vit != objects.end() && (vit + 1) != objects.end()) {
571 auto nextIt = vit + 1;
572 std::iter_swap(vit, nextIt);
573 if ((*nextIt)->zOrder < (*vit)->zOrder) {
574 (*nextIt)->zOrder = (*vit)->zOrder;
575 }
576 ReindexObjects();
577 isModified = true;
578 Touch();
579 }

599

601 if (vit != objects.end() && vit != objects.begin()) {
602 auto prevIt = vit - 1;
603 std::iter_swap(vit, prevIt);
604 if ((*prevIt)->zOrder > (*vit)->zOrder) {
605 (*prevIt)->zOrder = (*vit)->zOrder;
606 }
607 ReindexObjects();
608 isModified = true;
609 Touch();
610 }
611 }

613 /**
614 * @brief Replaces an existing object in-place, updating the fast UID map and R-Tree spatial index.
615 * Maintains the exact rendering z-order position in the objects array.
616 *
617 * @param uid Runtime UID of the object to replace.
618 * @param newObj Shared pointer to the replacement CanvasObject.

623 if (existing) {
624 newObj->uid = uid;
625 newObj->zOrder = existing->zOrder;

630 if (it != objects.end()) {
631 newObj->pageIndex = static_cast<uint32_t>(std::distance(objects.begin(), it));
632 *it = newObj;
633 }
634 isModified = true;
635 Touch();
636 } else {
637 AddObject(newObj);
638 }

672 /**
673 * @brief Resolves single-click selection at world coordinates (clickX, clickY) using smallest-area priority.
674 *
675 * MATHEMATICAL HEURISTIC & PRIORITY ARBITRATION:
676 * ----------------------------------------------
677 * Problem: When small notes or ink strokes sit inside a giant container (like a ShapeObject rectangle),
678 * a naive z-order or first-hit query can "trap" the smaller items behind the container's broad bounds.
679 *
680 * Mathematical Model:
681 * - Candidate set C = { obj \in objects | obj->isVisible \land obj->isSelectable \land obj->HitTest(clickX, clickY) }
682 * - For each candidate obj \in C:
683 * effectiveArea = obj->bounds.Area()
684 * if (obj->type == ObjectType::InkContainer || obj->type == ObjectType::Text) {
685 * effectiveArea *= 0.5; // 0.5x bias factor giving high priority to notes and strokes
686 * }
687 * - Result = \argmin_{obj \in C} (effectiveArea)
688 *
689 * @param clickX World X coordinate in millimeters.
690 * @param clickY World Y coordinate in millimeters.
691 * @param circleRadiusMm Proximity radius for fine strokes (e.g. config.objectHitTestRadiusMm).
692 * @return std::shared_ptr<CanvasObject> Selected object with minimal effective area, or nullptr.
693 */
694 [[nodiscard]] std::shared_ptr<CanvasObject> HitTestSingleClick(double clickX, double clickY, double circleRadiusMm = 0.0) const {
695 std::vector<std::shared_ptr<CanvasObject>> candidates;
696
697 for (auto it = objects.rbegin(); it != objects.rend(); ++it) {
698 auto& obj = *it;
699 if (obj && obj->isVisible && obj->isSelectable) {
700 bool hit = obj->HitTest(clickX, clickY);
701 if (!hit && circleRadiusMm > 0.0) {
702 hit = obj->HitTestCircle(clickX, clickY, circleRadiusMm);
703 }
704 if (hit) {
705 candidates.push_back(obj);
706 }
707 }
708 }
709
710 if (candidates.empty()) return nullptr;
711 if (candidates.size() == 1) return candidates[0];
712
713 // Select candidate with smallest effective area
714 std::shared_ptr<CanvasObject> bestObj = nullptr;
715 double bestEffectiveArea = 1e18;
716
717 for (const auto& obj : candidates) {
718 double area = obj->bounds.Area();
719 if (area <= 0.0) area = 0.01;
720
721 // Apply 0.5x bias factor to InkContainer and Text
722 if (obj->type == ObjectType::InkContainer || obj->type == ObjectType::Text) {
723 area *= 0.5;
724 }
725
726 if (area < bestEffectiveArea) {
727 bestEffectiveArea = area;
728 bestObj = obj;
729 }
730 }
731
732 return bestObj;
733 }

735 /**
736 * @brief Specialized hit-test for context menus that finds selectable objects first,
737 * then falls back to hit-testing pinned background objects.
738 *
739 * Working Process:
740 * 1. Queries standard selectable objects via HitTestSingleClick.
741 * 2. If a foreground object (stroke, shape, text, chip) is hit, it is prioritized.
742 * 3. If no foreground object was hit, scans in reverse Z-order for pinned background
743 * objects (isBackground == 1) so users can right-click them to restore/unlock.
744 *
745 * @param clickX World X coordinate in millimeters.
746 * @param clickY World Y coordinate in millimeters.
747 * @param circleRadiusMm Proximity radius in millimeters.
748 * @return Hit CanvasObject (foreground or background), or nullptr.
749 */
750 [[nodiscard]] std::shared_ptr<CanvasObject> HitTestForContextMenu(double clickX, double clickY, double circleRadiusMm = 0.0) const {
751 auto hit = HitTestSingleClick(clickX, clickY, circleRadiusMm);
752 if (hit) return hit;
753
754 // 2. Second priority: background / non-selectable objects (e.g. pinned background images)
755 for (auto it = objects.rbegin(); it != objects.rend(); ++it) {
756 auto& obj = *it;
757 if (obj && obj->isVisible && !obj->isSelectable) {
758 bool isHit = obj->HitTest(clickX, clickY);
759 if (!isHit && circleRadiusMm > 0.0) {
760 isHit = obj->HitTestCircle(clickX, clickY, circleRadiusMm);
761 }
762 if (isHit) {
763 return obj;
764 }
765 }
766 }
767
768 return nullptr;
769 }

771 /**
772 * @brief Queries all objects intersecting the camera viewport frustum using spatial culling.
773 *
774 * MATHEMATICAL CULLING & RENDERING PIPELINE:
775 * - Viewport Frustum: AABB bounds computed from screen dimensions inverted through the camera matrix:
776 * minX = ScreenToWorld(0, 0).x, maxX = ScreenToWorld(W, H).x
777 * minY = ScreenToWorld(0, 0).y, maxY = ScreenToWorld(W, H).y
778 * - R-Tree Frustum Search: O(log N + K) hierarchy traversal where K is the number of visible candidates.
779 * - UID Map Resolution: Eliminates the prior O(K * N) linear vector scan by resolving each candidate UID
780 * in O(1) via `objectMap`. Total resolution time is O(K) instead of O(K * N).
781 *
782 * @param viewport Active rendering camera viewport with physical bounds.
783 * @return Vector of visible canvas objects ready for rasterization.
784 */
785 [[nodiscard]] std::vector<std::shared_ptr<CanvasObject>> QueryVisible(const Viewport& viewport) {
786 Touch();
787 std::vector<uint32_t> visibleUids = spatialIndex.Query(viewport.bounds);
788 std::vector<std::shared_ptr<CanvasObject>> visible;
789 visible.reserve(visibleUids.size());
790
791 for (uint32_t id : visibleUids) {
792 auto it = objectMap.find(id);
793 if (it != objectMap.end() && it->second && it->second->isVisible) {
794 visible.push_back(it->second);
795 }
796 }
797
798 // Sort strictly by ascending zOrder, tie-broken by document pageIndex for deterministic painter's algorithm rendering.
799 std::sort(visible.begin(), visible.end(), [](const auto& a, const auto& b) {
800 if (a->zOrder != b->zOrder) return a->zOrder < b->zOrder;
801 return a->pageIndex < b->pageIndex;
802 });
803
804 return visible;
805 }

807 /**
808 * @brief Permanently purges soft-deleted / invisible objects from RAM, UID maps, and spatial index.
809 *
810 * MATHEMATICAL & MEMORY RECLAMATION PROCESS:
811 * - Traverses the resident `objects` vector in O(N) time.
812 * - Identifies soft-deleted tombstones marked with `!obj->isVisible`.
813 * - De-indexes them from `spatialIndex` (O(log N) R-Tree leaf purge) and `objectMap` (O(1) hash erase).
814 * - Uses the erase-remove idiom to shift surviving valid objects in-place and shrink the vector.
815 * - Marks the page dirty (`isModified = true`) to reflect permanent truncation in persistent storage.
816 *
817 * @return size_t Total number of tombstone objects permanently purged from the page.
818 */
819 size_t PurgeInvisibleObjects() {
820 size_t purgedCount = 0;
821 auto it = objects.begin();
822 while (it != objects.end()) {
823 if (*it && !(*it)->isVisible) {
824 spatialIndex.Remove((*it)->uid);
825 objectMap.erase((*it)->uid);
826 it = objects.erase(it);
827 ++purgedCount;
828 } else {
829 ++it;
830 }
831 }
832 if (purgedCount > 0) {
833 ReindexObjects();
834 isModified = true;
835 Touch();
836 }
837 return purgedCount;
838 }

840 /**
841 * @brief Rebuilds the fast UID lookup map and R-Tree spatial index from scratch.
842 * Guarantees spatial and associative coherence if the objects array was modified directly.
843 */
844 void RebuildSpatialIndex() {
845 objectMap.clear();
846 std::vector<std::pair<uint32_t, AABB>> items;
847 items.reserve(objects.size());
848 for (const auto& obj : objects) {
849 if (obj) {
850 objectMap[obj->uid] = obj;
851 items.emplace_back(obj->uid, obj->bounds);
852 }
853 }
854 spatialIndex.Rebuild(items);
855 }
