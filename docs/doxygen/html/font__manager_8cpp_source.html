1/**
2 * @file font_manager.cpp
3 * @brief Implementation of the Blend2D FontManager for vector typography resolution and metrics.
4 *
5 * GENERAL WORKING PROCESS:
6 * ------------------------
7 * 1. Request font via GetFont(family, sizePt, bold, italic).
8 * 2. Check if the (family, sizePt, bold, italic) combination is already cached in m_fontCache.
9 * If found, return immediately without locking or parsing overhead.
10 * 3. If uncached, look up the file path on disk matching family + style:
11 * - Windows: C:/Windows/Fonts/ (segoeui.ttf, segoeuib.ttf, arial.ttf, etc.)
12 * - Linux: /usr/share/fonts/
13 * - Local fallback: assets/fonts/
14 * 4. Load the font face via BLFontFace::create_from_file. Cache the BLFontFace by path in m_faceCache
15 * so multiple font sizes reuse the same underlying face descriptor.
16 * 5. Instantiate BLFont::create_from_face(face, sizePt), cache, and return.
17 *
18 * MATHEMATICAL FOUNDATIONS:
19 * -------------------------
20 * - Typographical Point to Metric conversion:
21 * 1 pt = 1/72 inch.
22 * Blend2D accepts font size in user-space points / coordinates directly.
23 * - Advance Width:
24 * advanceX = sum_{i} glyph_advance_x(g_i)
25 * - Line Spacing:
26 * LineHeight = ascent + descent + lineGap
27 */

45 std::string FindFirstExisting(const std::vector<std::string>& candidates) {
46 std::error_code ec;
47 const char* home = std::getenv("HOME");
48 for (const auto& path : candidates) {
49 std::string resolved = path;
50 if (!resolved.empty() && resolved[0] == '~' && home) {
51 resolved = std::string(home) + resolved.substr(1);
52 }
53 if (fs::exists(resolved, ec) && !ec) {
54 return resolved;
55 }
56 }
57 return "";
58 }
59}

293BLFontFace FontManager::LoadFontFace(const std::string& filePath) {
294 auto it = m_faceCache.find(filePath);
295 if (it != m_faceCache.end()) {
296 return it->second;
297 }
298
299 BLFontFace face;
300 if (face.create_from_file(filePath.c_str()) == BL_SUCCESS) {
301 m_faceCache[filePath] = face;
302 return face;
303 }
304
305 return m_fallbackFace;
306}

308BLFont FontManager::GetFont(const std::string& family, float sizePt, bool bold, bool italic) {
309 std::lock_guard<std::mutex> lock(m_mutex);
310
311 FontKey key{family, sizePt, bold, italic};
312 auto it = m_fontCache.find(key);
313 if (it != m_fontCache.end()) {
314 return it->second;
315 }
316
317 std::string path = ResolveFontPath(family, bold, italic);
318 BLFontFace face;
319 if (!path.empty() && fs::exists(path)) {
320 face = LoadFontFace(path);
321 } else if (m_fallbackLoaded) {
322 face = m_fallbackFace;
323 }
324
325 BLFont font;
326 if (face.is_valid()) {
327 font.create_from_face(face, sizePt);
328 }
329
330 m_fontCache[key] = font;
331 return font;
332}

341double FontManager::MeasureTextWidth(const BLFont& font, const std::string& text) {
342 if (!font.is_valid() || text.empty()) {
343 return 0.0;
344 }
345
346 BLGlyphBuffer gb;
347 gb.set_utf8_text(text.data(), text.size());
348
349 BLTextMetrics tm;
350 if (font.get_text_metrics(gb, tm) == BL_SUCCESS) {
351 return tm.advance.x;
352 }
353
354 // Fallback heuristic: 0.55 * sizePt per character if shaping fails
355 const auto& fm = font.metrics();
356 return text.size() * (fm.ascent * 0.55);
357}
