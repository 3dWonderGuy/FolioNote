1/**
2 * =========================================================================================
3 * @file core/clipboard/clipboard_manager.cpp
4 * @brief Implementation of Multi-Format Clipboard Coordinator
5 * =========================================================================================
6 *
7 * ARCHITECTURAL DESIGN & INVERSION OF CONTROL:
8 * --------------------------------------------
9 * ClipboardManager isolates all platform clipboard communication away from CanvasEngine.
10 *
11 * It provides:
12 * 1. Multi-format packaging on Copy:
13 * - Deep clone of objects (preserves full parameters).
14 * - Vector SVG output (<polyline>, <rect>, <text> tags) formatted for Microsoft OneNote,
15 * Word, PowerPoint, and Illustrator.
16 * - High-resolution transparent PNG bitmap (rendered via Blend2D at 192 DPI) for
17 * universal paste into Discord, Slack, Paint, Teams, browser chat, etc.
18 * - Plaintext / URL extraction (video URLs, audio paths, textbox text) pushed to SDL/OS clipboard.
19 * 2. Smart Ingestion on Paste:
20 * - If internal package is available, pastes clones centered at target world coordinates.
21 * - Else if OS clipboard has an image (PNG/JPEG/BMP/WEBP), creates an ImageObject.
22 * - Else if OS clipboard has text:
23 * * Checks for YouTube URLs -> creates VideoObject.
24 * * Checks for audio/video file paths -> creates AudioObject or VideoObject.
25 * * Falls back to TextBoxObject with the plain text.
26 */

65bool ClipboardManager::CopyObjects(const std::vector<std::shared_ptr<CanvasObject>>& objects) {
66 if (objects.empty()) {
67 return false;
68 }
69
70 currentPackage.Clear();
71 currentPackage.clonedObjects.reserve(objects.size());
72
73 // 1. In-Memory Deep Clone & AABB Union Calculation
74 AABB totalBox;
75 bool hasBounds = false;
76
77 for (const auto& obj : objects) {
78 if (!obj) continue;
79 auto cloned = std::shared_ptr<CanvasObject>(obj->Clone().release());
80 if (cloned) {
81 currentPackage.clonedObjects.push_back(cloned);
82 if (!hasBounds) {
83 totalBox = obj->bounds;
84 hasBounds = true;
85 } else {
86 totalBox.Merge(obj->bounds);
87 }
88 }
89 }
90
91 if (currentPackage.clonedObjects.empty()) {
92 return false;
93 }
94
95 currentPackage.objectCount = static_cast<uint32_t>(currentPackage.clonedObjects.size());
96 currentPackage.totalBounds = totalBox;
97 currentPackage.originX = (totalBox.minX + totalBox.maxX) * 0.5;
98 currentPackage.originY = (totalBox.minY + totalBox.maxY) * 0.5;
99
100 // 2. Generate SVG Representation (OneNote / Office / Vector Apps)
101 currentPackage.svgXml = GenerateSvgXml(currentPackage.clonedObjects, totalBox);
102
103 // 3. Render High-Resolution Raster Image (PNG) with Blend2D (Discord / Paint / Slack)
104 // Optimization for single ImageObject: retain pristine source bytes without re-encoding
105 BLImage blSurface;
106 if (currentPackage.clonedObjects.size() == 1) {
107 if (auto imgObj = std::dynamic_pointer_cast<ImageObject>(currentPackage.clonedObjects[0])) {
108 if (!imgObj->embeddedData.empty() && imgObj->imageFormat == ImageFormat::PNG) {
109 currentPackage.pngBytes = imgObj->embeddedData;
110 } else if (!imgObj->cachedBlImage.is_empty()) {
111 BLArray<uint8_t> pngBuffer;
112 BLImageCodec codec;
113 codec.find_by_name("PNG");
114 if (codec.is_valid()) {
115 imgObj->cachedBlImage.write_to_data(pngBuffer, codec);
116 currentPackage.pngBytes.assign(pngBuffer.data(), pngBuffer.data() + pngBuffer.size());
117 }
118 }
119 blSurface = imgObj->cachedBlImage;
120 }
121 }
122
123 if (currentPackage.pngBytes.empty()) {
124 currentPackage.pngBytes = RenderToPng(currentPackage.clonedObjects, totalBox, 192.0, &blSurface);
125 }
126
127 // 4. Extract Plaintext / URLs
128 currentPackage.plainText = ExtractPlainText(currentPackage.clonedObjects);
129
130 // 5. Publish to Platform & OS Clipboard
131#if defined(_WIN32)
132 PlatformWin32Clipboard::PublishToWindowsClipboard(
133 currentPackage.clonedObjects,
134 currentPackage.svgXml,
135 currentPackage.pngBytes,
136 blSurface,
137 currentPackage.plainText,
138 lastClipboardSequenceNumber
139 );
140#else
141 if (!currentPackage.plainText.empty()) {
142 SDL_SetClipboardText(currentPackage.plainText.c_str());
143 }
144
145 // Publish SVG mime type to OS clipboard if supported
146 if (!currentPackage.svgXml.empty()) {
147 static const char* s_svgMimeTypes[] = { "image/svg+xml" };
148 SDL_SetClipboardData([](void* userdata, const char* mime_type, size_t* size) -> const void* {
149 auto* pkg = static_cast<ClipboardDataPackage*>(userdata);
150 if (std::string(mime_type) == "image/svg+xml") {
151 *size = pkg->svgXml.size();
152 return pkg->svgXml.data();
153 }
154 return nullptr;
155 }, nullptr, &currentPackage, s_svgMimeTypes, 1);
156 }
157#endif
158
159 LOG_INFO(General, "Copied " + std::to_string(currentPackage.objectCount) +
160 " objects (Bounds: " + std::to_string(totalBox.Width()) + "x" +
161 std::to_string(totalBox.Height()) + " mm). Generated SVG: " +
162 std::to_string(currentPackage.svgXml.size()) + " B, PNG: " +
163 std::to_string(currentPackage.pngBytes.size()) + " B.");
164
165 return true;
166}

168bool ClipboardManager::CutObjects(const std::vector<std::shared_ptr<CanvasObject>>& objects,
169 DocumentSession& session) {
170 if (!CopyObjects(objects)) {
171 return false;
172 }
173
174 auto activePage = session.GetActivePage();
175 if (!activePage) return false;
176
177 std::vector<std::shared_ptr<CanvasObject>> toRemove;
178 toRemove.reserve(objects.size());
179
180 for (const auto& obj : objects) {
181 if (obj) {
182 toRemove.push_back(obj);
183 }
184 }
185
186 if (!toRemove.empty()) {
187 session.RecordHistoryCommand(activePage, std::make_unique<Folio::RemoveObjectsCommand>(toRemove));
188 for (const auto& obj : toRemove) {
189 activePage->RemoveObject(obj);
190 }
191 activePage->isModified = true;
192 session.NotifyPageModified(activePage);
193 LOG_INFO(General, "Cut " + std::to_string(toRemove.size()) + " objects from active page.");
194 }
195
196 return true;
197}

203std::vector<std::shared_ptr<CanvasObject>> ClipboardManager::Paste(DocumentSession& session,
204 CanvasEngine& engine,
205 double targetWorldX,
206 double targetWorldY) {
207 auto activePage = session.GetActivePage();
208 if (!activePage) return {};
209
210 std::vector<std::shared_ptr<CanvasObject>> createdObjects;
211
212#if defined(_WIN32)
213 const DWORD currentSeq = ::GetClipboardSequenceNumber();
214 const bool externalClipboardChanged = (currentSeq != lastClipboardSequenceNumber);
215
216 // If external clipboard was modified since our last Copy, attempt to ingest external OS clipboard first!
217 if (externalClipboardChanged || currentPackage.clonedObjects.empty()) {
218 if (PlatformWin32Clipboard::IngestFromWindowsClipboard(session, engine, targetWorldX, targetWorldY, createdObjects)) {
219 // Update sequence number to prevent redundant re-ingestion
220 lastClipboardSequenceNumber = ::GetClipboardSequenceNumber();
221 return createdObjects;
222 }
223 }
224#endif
225
226 // --- Priority 1: Native In-Memory Cloned Objects ---
227 if (!currentPackage.clonedObjects.empty()) {
228 double dx = targetWorldX - currentPackage.originX;
229 double dy = targetWorldY - currentPackage.originY;
230 BLMatrix2D trans = BLMatrix2D::make_translation(dx, dy);
231
232 createdObjects.reserve(currentPackage.clonedObjects.size());
233 for (const auto& obj : currentPackage.clonedObjects) {
234 if (!obj) continue;
235 auto clone = std::shared_ptr<CanvasObject>(obj->Clone().release());
236 if (!clone) continue;
237
238 clone->uid = UIDGenerator::Next();
239 clone->guuid = GUIDGenerator::GenerateV4();
240 clone->ApplyTransform(trans);
241 clone->UpdateBounds();
242 clone->isSelected = 1;
243
244 activePage->AddObject(clone);
245 createdObjects.push_back(clone);
246 }
247
248 if (!createdObjects.empty()) {
249 session.RecordHistoryCommand(activePage, std::make_unique<Folio::AddObjectsCommand>(createdObjects));
250 activePage->isModified = true;
251 session.NotifyPageModified(activePage);
252 engine.selectionGizmo.SetSelectedObjects(createdObjects);
253 engine.isDirty = true;
254 engine.needsFullRebake = true;
255 LOG_INFO(General, "Pasted " + std::to_string(createdObjects.size()) +
256 " native objects at (" + std::to_string(targetWorldX) + ", " +
257 std::to_string(targetWorldY) + ") mm");
258 return createdObjects;
259 }
260 }
261
262 // --- Priority 2: OS Clipboard Image (PNG, JPEG, BMP, etc.) ---
263 const char* imageMimes[] = { "image/png", "image/jpeg", "image/bmp", "image/gif", "image/webp", "image/svg+xml" };
264 for (const char* mime : imageMimes) {
265 if (SDL_HasClipboardData(mime)) {
266 size_t dataSize = 0;
267 void* clipData = SDL_GetClipboardData(mime, &dataSize);
268 if (clipData && dataSize > 0) {
269 std::string relPath = CanvasEngine::DeduplicateAndSaveImage("", clipData, dataSize, &session, ".png");
270 const double screenDpi = engine.transform.pixelsPerMm * 25.4;
271 auto img = std::make_shared<Folio::ImageObject>(static_cast<const uint8_t*>(clipData), dataSize, relPath, screenDpi);
272
273 if (img->isLoaded) {
274 img->uid = UIDGenerator::Next();
275 img->guuid = GUIDGenerator::GenerateV4();
276 img->worldX = targetWorldX - img->worldWidth * 0.5;
277 img->worldY = targetWorldY - img->worldHeight * 0.5;
278 img->UpdateBounds();
279 img->isSelected = 1;
280
281 session.AddImage(img);
282 createdObjects.push_back(img);
283 engine.selectionGizmo.SetSelectedObjects({img});
284 engine.needsFullRebake = true;
285 engine.isDirty = true;
286 SDL_free(clipData);
287 LOG_INFO(General, "Pasted image from OS clipboard (" +
288 std::to_string(img->naturalWidth) + "x" +
289 std::to_string(img->naturalHeight) + " px)");
290 return createdObjects;
291 }
292 SDL_free(clipData);
293 }
294 }
295 }
296
297 // --- Priority 3: OS Clipboard Text / URL ---
298 if (SDL_HasClipboardText()) {
299 char* textPtr = SDL_GetClipboardText();
300 if (textPtr) {
301 std::string clipText(textPtr);
302 SDL_free(textPtr);
303
304 if (!clipText.empty()) {
305 // Check if text is a YouTube or Web URL
306 bool isHttp = (clipText.rfind("http://", 0) == 0 || clipText.rfind("https://", 0) == 0);
307 bool isYouTube = (clipText.find("youtube.com") != std::string::npos || clipText.find("youtu.be") != std::string::npos);
308
309 if (isYouTube) {
310 // Create a VideoObject for YouTube URLs
311 double cardW = 160.0;
312 double cardH = 95.0;
313 auto vid = std::make_shared<Folio::VideoObject>(
314 clipText, "YouTube Video", cardW, cardH
315 );
316 vid->uid = UIDGenerator::Next();
317 vid->guuid = GUIDGenerator::GenerateV4();
318 vid->worldX = targetWorldX - cardW * 0.5;
319 vid->worldY = targetWorldY - cardH * 0.5;
320 vid->UpdateBounds();
321 vid->isSelected = 1;
322
323 session.AddObject(vid);
324 createdObjects.push_back(vid);
325 engine.selectionGizmo.SetSelectedObjects({vid});
326 engine.needsFullRebake = true;
327 engine.isDirty = true;
328 LOG_INFO(General, "Pasted YouTube URL as VideoObject: " + clipText);
329 return createdObjects;
330 }
331
332 // Fallback: Create a TextBoxObject for plain text
333 auto tb = std::make_shared<Folio::TextBoxObject>();
334 tb->uid = UIDGenerator::Next();
335 tb->guuid = GUIDGenerator::GenerateV4();
336 tb->text = clipText;
337 tb->worldX = targetWorldX - 40.0;
338 tb->worldY = targetWorldY - 15.0;
339 tb->worldWidth = 80.0;
340 tb->worldHeight = 30.0;
341 tb->UpdateBounds();
342 tb->isSelected = 1;
343
344 session.AddTextBox(tb);
345 createdObjects.push_back(tb);
346 engine.selectionGizmo.SetSelectedObjects({tb});
347 engine.needsFullRebake = true;
348 engine.isDirty = true;
349 LOG_INFO(General, "Pasted text from clipboard into TextBoxObject (" + std::to_string(clipText.size()) + " chars)");
350 return createdObjects;
351 }
352 }
353 }
354
355 return createdObjects;
356}

358bool ClipboardManager::HasPasteableContent() const {
359#if defined(_WIN32)
360 const DWORD currentSeq = ::GetClipboardSequenceNumber();
361 if (currentSeq == lastClipboardSequenceNumber && !currentPackage.clonedObjects.empty()) {
362 return true;
363 }
364 if (PlatformWin32Clipboard::HasWindowsClipboardContent()) {
365 return true;
366 }
367#endif
368 if (!currentPackage.clonedObjects.empty()) return true;

370
371 const char* imageMimes[] = { "image/png", "image/jpeg", "image/bmp", "image/gif", "image/webp", "image/svg+xml" };
372 for (const char* mime : imageMimes) {
373 if (SDL_HasClipboardData(mime)) return true;
374 }
375
376 return false;

383std::string ClipboardManager::GenerateSvgXml(const std::vector<std::shared_ptr<CanvasObject>>& objects,
384 const AABB& bounds) {
385 double w = std::max(1.0, bounds.Width());
386 double h = std::max(1.0, bounds.Height());
387
388 std::ostringstream ss;
389 ss << "<?xml version=\"1.0\" encoding=\"UTF-8\" standalone=\"no\"?>\n";
390 ss << "<svg xmlns=\"http://www.w3.org/2000/svg\" version=\"1.1\" "
391 << "viewBox=\"" << bounds.minX << " " << bounds.minY << " " << w << " " << h << "\" "
392 << "width=\"" << w << "mm\" height=\"" << h << "mm\">\n";
393 ss << " <!-- Generated by FolioNote Multi-Format Clipboard -->\n";
394
395 for (const auto& obj : objects) {
396 if (!obj) continue;
397
398 // Ink strokes
399 if (auto ink = std::dynamic_pointer_cast<InkContainer>(obj)) {
400 for (const auto& stroke : ink->strokes) {
401 if (stroke.segments.empty()) continue;
402
403 uint32_t val = stroke.color.value;
404 uint8_t a = (val >> 24) & 0xFF;
405 uint8_t r = (val >> 16) & 0xFF;
406 uint8_t g = (val >> 8) & 0xFF;
407 uint8_t b = val & 0xFF;
408 float alpha = a / 255.0f;
409 float strokeW = std::max(0.2f, static_cast<float>(stroke.baseWidth));
410
411 ss << " <polyline fill=\"none\" stroke=\"rgba(" << (int)r << "," << (int)g << "," << (int)b << "," << alpha << ")\" "
412 << "stroke-width=\"" << strokeW << "\" stroke-linecap=\"round\" stroke-linejoin=\"round\" points=\"";
413 ss << stroke.segments[0].p0.x << "," << stroke.segments[0].p0.y << " ";
414 for (const auto& seg : stroke.segments) {
415 ss << seg.p1.x << "," << seg.p1.y << " ";
416 }
417 ss << "\" />\n";
418 }
419 }
420 // Text boxes
421 else if (auto tb = std::dynamic_pointer_cast<TextBoxObject>(obj)) {
422 ss << " <text x=\"" << tb->worldX << "\" y=\"" << (tb->worldY + tb->fontSize * 0.35)
423 << "\" font-family=\"" << tb->fontFamily << "\" font-size=\"" << tb->fontSize
424 << "pt\" fill=\"#1f2937\">" << tb->text << "</text>\n";
425 }
426 // Raster Images (embed base64 PNG for SVG consumers)
427 else if (auto img = std::dynamic_pointer_cast<ImageObject>(obj)) {
428 if (img->isLoaded && !img->cachedBlImage.is_empty()) {
429 BLArray<uint8_t> pngBuf;
430 BLImageCodec codec;
431 codec.find_by_name("PNG");
432 if (codec.is_valid() && img->cachedBlImage.write_to_data(pngBuf, codec) == BL_SUCCESS && !pngBuf.is_empty()) {
433 static const char* b64Chars = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
434 std::string b64;
435 const uint8_t* p = pngBuf.data();
436 const size_t len = pngBuf.size();
437 b64.reserve(((len + 2) / 3) * 4);
438 for (size_t i = 0; i < len; i += 3) {
439 const uint32_t b = (p[i] << 16) | ((i + 1 < len ? p[i + 1] : 0) << 8) | (i + 2 < len ? p[i + 2] : 0);
440 b64.push_back(b64Chars[(b >> 18) & 0x3F]);
441 b64.push_back(b64Chars[(b >> 12) & 0x3F]);
442 b64.push_back(i + 1 < len ? b64Chars[(b >> 6) & 0x3F] : '=');
443 b64.push_back(i + 2 < len ? b64Chars[b & 0x3F] : '=');
444 }
445 ss << " <image x=\"" << img->worldX << "\" y=\"" << img->worldY
446 << "\" width=\"" << img->worldWidth << "mm\" height=\"" << img->worldHeight
447 << "mm\" href=\"data:image/png;base64," << b64 << "\" />\n";
448 }
449 }
450 }
451 }
452
453 ss << "</svg>\n";
454 return ss.str();
455}

457std::vector<uint8_t> ClipboardManager::RenderToPng(const std::vector<std::shared_ptr<CanvasObject>>& objects,
458 const AABB& bounds,
459 double dpi,
460 BLImage* outBlImage) {
461 double scale = dpi / 25.4; // pixels per millimeter
462 int pixelW = std::clamp(static_cast<int>(std::ceil(bounds.Width() * scale)), 1, 8192);
463 int pixelH = std::clamp(static_cast<int>(std::ceil(bounds.Height() * scale)), 1, 8192);
464
465 BLImage img(pixelW, pixelH, BL_FORMAT_PRGB32);
466 BLContext ctx(img);
467 if (!ctx.is_valid()) return {};
468
469 // Clear with transparent alpha
470 ctx.set_comp_op(BL_COMP_OP_SRC_COPY);
471 ctx.fill_all(BLRgba32(0, 0, 0, 0));
472 ctx.set_comp_op(BL_COMP_OP_SRC_OVER);
473
474 // Coordinate mapping: translate so bounds.min is at origin (0,0), then scale by dpi
475 ctx.scale(scale, scale);
476 ctx.translate(-bounds.minX, -bounds.minY);
477
478 // Dummy viewport matching bounding box
479 Viewport vp;
480 vp.bounds = bounds;
481 vp.zoom = 1.0;
482 vp.pixelsPerMm = scale;
483
484 for (const auto& obj : objects) {
485 if (obj) {
486 obj->Render(ctx, vp);
487 }
488 }
489
490 ctx.end();
491
492 if (outBlImage) {
493 *outBlImage = img;
494 }
495
496 // Encode to PNG bytes
497 BLArray<uint8_t> pngBuffer;
498 BLImageCodec codec;
499 codec.find_by_name("PNG");
500 if (codec.is_valid()) {
501 img.write_to_data(pngBuffer, codec);
502 }
503
504 std::vector<uint8_t> result(pngBuffer.size());
505 if (!pngBuffer.is_empty()) {
506 std::memcpy(result.data(), pngBuffer.data(), pngBuffer.size());
507 }
508 return result;
509}

511std::string ClipboardManager::ExtractPlainText(const std::vector<std::shared_ptr<CanvasObject>>& objects) {
512 std::string text;
513 for (const auto& obj : objects) {
514 if (!obj) continue;
515
516 if (auto tb = std::dynamic_pointer_cast<TextBoxObject>(obj)) {
517 if (!text.empty()) text += "\n";
518 text += tb->text;
519 } else if (auto vid = std::dynamic_pointer_cast<VideoObject>(obj)) {
520 if (!text.empty()) text += "\n";
521 text += vid->sourceUrl;
522 } else if (auto aud = std::dynamic_pointer_cast<AudioObject>(obj)) {
523 if (!text.empty()) text += "\n";
524 text += aud->filePath;
525 }
526 }
527 return text;
528}
