18 inline std::string FindFontPath(const std::vector<std::string>& candidates) {
19 std::error_code ec;
20 const char* home = std::getenv("HOME");
21 for (const auto& path : candidates) {
22 std::string resolved = path;
23 if (!resolved.empty() && resolved[0] == '~' && home) {
24 resolved = std::string(home) + resolved.substr(1);
25 }
26 if (std::filesystem::exists(resolved, ec) && !ec) {
27 return resolved;
28 }
29 }
30 return "";
31 }
