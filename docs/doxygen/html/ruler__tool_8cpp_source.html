1/**
2 * =========================================================================================
3 * @file core/canvas_engine/tools/ruler_tool.cpp
4 * @brief Implementation of the Interactive Digital Straightedge Ruler
5 * =========================================================================================
6 *
7 * MATHEMATICAL PROCESS & TESSELLATION DETAILS:
8 * 1. World-to-Screen Affine Mapping:
9 * All ruler geometry (center, ticks, straightedge limits) is anchored in physical
10 * canvas millimeters (mm). When rendering, vertices are projected into screen pixels:
11 * S = WorldToScreen(W) + canvasOrigin
12 * This ensures that when the user pans or zooms the canvas, the ruler stays accurately
13 * registered against the real-world scale of the page.
14 *
15 * 2. Orthogonal Snapping Projection:
16 * Given stylus position P:
17 * t = (P - C) · u (distance along ruler axis)
18 * d_v = (P - C) · v (transverse distance across ruler)
19 * The pen position is snapped to the nearest straightedge if |d_v ± (W/2)| <= snapDistance.
20 *
21 * 3. Rotational Kinematics & Angular Snapping:
22 * Mouse drags on the center dial evaluate angle = atan2(my - cy, mx - cx).
23 * Angles are automatically snapped to 15° increments when within 2.0° of a major drafting angle.
24 */

52double RulerTool::NormalizeAngleDeg(double deg) noexcept {
53 while (deg >= 180.0) deg -= 360.0;
54 while (deg < -180.0) deg += 360.0;
55 return deg;
56}

58double RulerTool::SnapAngleToIncrements(double deg, double increment, double threshold) noexcept {
59 double snapped = std::round(deg / increment) * increment;
60 if (std::abs(deg - snapped) <= threshold) {
61 return snapped;
62 }
63 return deg;
64}

70RulerSnapResult RulerTool::SnapPoint(const Point2D& worldPoint) const {
71 RulerSnapResult result;
72 result.point = worldPoint;
73 result.isSnapped = false;
74
75 if (!m_enabled) {
76 return result;
77 }
78
79 const double rad = GetAngleRad();
80 const double ux = std::cos(rad);
81 const double uy = std::sin(rad);
82 const double vx = -uy;
83 const double vy = ux;
84
85 const double dx = worldPoint.x - m_centerWorld.x;
86 const double dy = worldPoint.y - m_centerWorld.y;
87
88 // Longitudinal projection (along ruler length)
89 const double t = dx * ux + dy * uy;
90 // Transverse projection (across ruler width)
91 const double dv = dx * vx + dy * vy;
92
93 const double halfLen = m_lengthMm * 0.5;
94 const double halfWidth = m_widthMm * 0.5;
95
96 // Distance to top straightedge (+halfWidth) and bottom straightedge (-halfWidth)
97 const double distTop = std::abs(dv - halfWidth);
98 const double distBot = std::abs(dv - (-halfWidth));
99
100 // Allow a generous 15mm run-off past ends so pen gestures along edges don't abruptly disconnect
101 const double marginMm = 15.0;
102 const bool withinLength = (t >= -halfLen - marginMm) && (t <= halfLen + marginMm);
103
104 if (!withinLength) {
105 return result;
106 }
107
108 if (distTop <= m_snapThresholdMm && distTop <= distBot) {
109 // Snap to top straightedge
110 result.point.x = m_centerWorld.x + t * ux + halfWidth * vx;
111 result.point.y = m_centerWorld.y + t * uy + halfWidth * vy;
112 result.isSnapped = true;
113 result.isTopEdge = true;
114 result.distanceAlongMm = t + halfLen;
115 } else if (distBot <= m_snapThresholdMm) {
116 // Snap to bottom straightedge
117 result.point.x = m_centerWorld.x + t * ux - halfWidth * vx;
118 result.point.y = m_centerWorld.y + t * uy - halfWidth * vy;
119 result.isSnapped = true;
120 result.isTopEdge = false;
121 result.distanceAlongMm = t + halfLen;
122 }
123
124 return result;
125}

146 // 1. Orientation & Direction Vectors in World Space
147 const double rad = GetAngleRad();
148 const double ux = std::cos(rad);
149 const double uy = std::sin(rad);
150 const double vx = -uy;
151 const double vy = ux;
152
153 const double halfLen = m_lengthMm * 0.5;
154 const double halfWidth = m_widthMm * 0.5;
155
156 // 2. Corner Vertices in World Space (mm)
157 // TL = Center - (L/2)*u + (W/2)*v (Top-Left)
158 // TR = Center + (L/2)*u + (W/2)*v (Top-Right)
159 // BR = Center + (L/2)*u - (W/2)*v (Bottom-Right)
160 // BL = Center - (L/2)*u - (W/2)*v (Bottom-Left)
161 const Point2D wTL = { m_centerWorld.x - halfLen * ux + halfWidth * vx, m_centerWorld.y - halfLen * uy + halfWidth * vy };
162 const Point2D wTR = { m_centerWorld.x + halfLen * ux + halfWidth * vx, m_centerWorld.y + halfLen * uy + halfWidth * vy };
163 const Point2D wBR = { m_centerWorld.x + halfLen * ux - halfWidth * vx, m_centerWorld.y + halfLen * uy - halfWidth * vy };
164 const Point2D wBL = { m_centerWorld.x - halfLen * ux - halfWidth * vx, m_centerWorld.y - halfLen * uy - halfWidth * vy };
165
166 // 3. Project to Screen Space (pixels)
167 const Point2D sTL_raw = transform.WorldToScreen(wTL.x, wTL.y);
168 const Point2D sTR_raw = transform.WorldToScreen(wTR.x, wTR.y);
169 const Point2D sBR_raw = transform.WorldToScreen(wBR.x, wBR.y);
170 const Point2D sBL_raw = transform.WorldToScreen(wBL.x, wBL.y);
171 const Point2D sC_raw = transform.WorldToScreen(m_centerWorld.x, m_centerWorld.y);
172
173 const ImVec2 sTL(canvasOrigin.x + static_cast<float>(sTL_raw.x), canvasOrigin.y + static_cast<float>(sTL_raw.y));
174 const ImVec2 sTR(canvasOrigin.x + static_cast<float>(sTR_raw.x), canvasOrigin.y + static_cast<float>(sTR_raw.y));
175 const ImVec2 sBR(canvasOrigin.x + static_cast<float>(sBR_raw.x), canvasOrigin.y + static_cast<float>(sBR_raw.y));
176 const ImVec2 sBL(canvasOrigin.x + static_cast<float>(sBL_raw.x), canvasOrigin.y + static_cast<float>(sBL_raw.y));
177 const ImVec2 sC (canvasOrigin.x + static_cast<float>(sC_raw.x), canvasOrigin.y + static_cast<float>(sC_raw.y));
178
179 // Screen Normalized Direction Vectors
180 const float lenPx = std::hypot(sTR.x - sTL.x, sTR.y - sTL.y);
181 const float widthPx = std::hypot(sTL.x - sBL.x, sTL.y - sBL.y);
182 if (lenPx < 10.0f || widthPx < 5.0f) {
183 return; // Too small to render meaningfully
184 }
185
186 const ImVec2 uScreen((sTR.x - sTL.x) / lenPx, (sTR.y - sTL.y) / lenPx);
187 const ImVec2 vScreen((sTL.x - sBL.x) / widthPx, (sTL.y - sBL.y) / widthPx);
188
189 // Left & Right Center Rotation Handles
190 const ImVec2 handleLeft ((sTL.x + sBL.x) * 0.5f, (sTL.y + sBL.y) * 0.5f);
191 const ImVec2 handleRight((sTR.x + sBR.x) * 0.5f, (sTR.y + sBR.y) * 0.5f);
192 const float handleRadiusPx = 14.0f;
193
194 // Center Protractor Dial Metrics
195 const float dialRadiusPx = 28.0f;

197 // 4. Point-in-Convex-Quad Hit Testing
198 auto PointInQuad = [](const ImVec2& p, const ImVec2& a, const ImVec2& b, const ImVec2& c, const ImVec2& d) -> bool {
199 auto Cross = [](const ImVec2& p1, const ImVec2& p2, const ImVec2& pt) -> float {
200 return (p2.x - p1.x) * (pt.y - p1.y) - (p2.y - p1.y) * (pt.x - p1.x);
201 };
202 const float c1 = Cross(a, b, p);
203 const float c2 = Cross(b, c, p);
204 const float c3 = Cross(c, d, p);
205 const float c4 = Cross(d, a, p);
206 const bool hasNeg = (c1 < 0) || (c2 < 0) || (c3 < 0) || (c4 < 0);
207 const bool hasPos = (c1 > 0) || (c2 > 0) || (c3 > 0) || (c4 > 0);
208 return !(hasNeg && hasPos);
209 };
210
211 const Point2D mouseWorld = transform.ScreenToWorld(mousePos.x - canvasOrigin.x, mousePos.y - canvasOrigin.y);
212 const double mouseDx = mouseWorld.x - m_centerWorld.x;
213 const double mouseDy = mouseWorld.y - m_centerWorld.y;
214 const double mouseDv = mouseDx * vx + mouseDy * vy;
215 const bool isCenterMoveZone = std::abs(mouseDv) < (halfWidth - 8.0);
216
217 const bool isHoveredBody = PointInQuad(mousePos, sTL, sTR, sBR, sBL);
218 const bool isHoveredBodyCenter = isHoveredBody && isCenterMoveZone;
219 const float distToDial = std::hypot(mousePos.x - sC.x, mousePos.y - sC.y);
220 const float distToLeftHandle = std::hypot(mousePos.x - handleLeft.x, mousePos.y - handleLeft.y);
221 const float distToRightHandle = std::hypot(mousePos.x - handleRight.x, mousePos.y - handleRight.y);
222
223 const bool isHoveredDial = (distToDial <= dialRadiusPx);
224 const bool isHoveredHandleLeft = (distToLeftHandle <= handleRadiusPx);
225 const bool isHoveredHandleRight = (distToRightHandle <= handleRadiusPx);
226
227 // Capture mouse in ImGui when interacting with the ruler body or handles to prevent stray canvas ink

229 io.WantCaptureMouse = true;
230 }

232 if (isHoveredDial || isHoveredHandleLeft || isHoveredHandleRight ||

234 ImGui::SetMouseCursor(ImGuiMouseCursor_ResizeAll);

240 if (io.MouseClicked[0]) {
241 if (isHoveredDial) {

243 m_dragStartMouse = mousePos;
244 m_dragStartAngleDeg = m_angleDeg;
245 m_dragInitialMouseAngle = std::atan2(mousePos.y - sC.y, mousePos.x - sC.x) * (180.0 / M_PI);
246 } else if (isHoveredHandleLeft) {
247 m_dragMode = DragMode::RotateHandleLeft;
248 m_dragStartMouse = mousePos;
249 m_dragStartAngleDeg = m_angleDeg;
250 m_dragInitialMouseAngle = std::atan2(mousePos.y - sC.y, mousePos.x - sC.x) * (180.0 / M_PI);
251 } else if (isHoveredHandleRight) {
252 m_dragMode = DragMode::RotateHandleRight;
253 m_dragStartMouse = mousePos;
254 m_dragStartAngleDeg = m_angleDeg;
255 m_dragInitialMouseAngle = std::atan2(mousePos.y - sC.y, mousePos.x - sC.x) * (180.0 / M_PI);

258 m_dragStartMouse = mousePos;
259 m_dragStartCenter = m_centerWorld;

263 if (m_dragMode != DragMode::None) {
264 if (io.MouseDown[0]) {
265 if (m_dragMode == DragMode::MoveBody) {

267 const float dxPx = mousePos.x - m_dragStartMouse.x;
268 const float dyPx = mousePos.y - m_dragStartMouse.y;
269 const Point2D dWorld = transform.ScreenDeltaToWorldDelta(dxPx, dyPx);

285 m_dragMode = DragMode::None;
286 }

293 if (!io.KeyShift) {
294 newAngle = SnapAngleToIncrements(newAngle, 15.0, 2.0);
