91void UISidebarComponent::RenderItem(const SidebarItem& item) {
92 ImGuiWindow* window = ImGui::GetCurrentWindow();
93 if (window->SkipItems) return;
94
95 const auto& theme = ThemeManager::Instance();
96 bool isDark = (theme.colorBg.x < 0.5f);
97
98 float itemH = 36.0f;
99 float rowW = ImGui::GetContentRegionAvail().x;
100 ImVec2 pos = window->DC.CursorPos;
101 ImRect bb(pos, ImVec2(pos.x + rowW, pos.y + itemH));
102
103 ImGuiID itemId = window->GetID(item.id.c_str());
104 ImGui::ItemSize(bb, 0.0f);
105 if (!ImGui::ItemAdd(bb, itemId)) return;
106
107 bool hovered = false;
108 bool held = false;
109 bool clicked = ImGui::ButtonBehavior(bb, itemId, &hovered, &held);
110
111 if (clicked && item.onClick) {
112 item.onClick();
113 }
114
115 if (hovered && ImGui::IsMouseClicked(ImGuiMouseButton_Right) && item.onContextMenu) {
116 item.onContextMenu();
117 }
118
119 // Smooth hover/active transition alpha
120 float hoverAlpha = 0.0f;
121 float activeAlpha = 0.0f;
122 UIAnimationManager::Instance().GetInteractionAlphas(itemId, hovered || item.isSelected, held, hoverAlpha, activeAlpha);
123
124 // Render interactive surface
125 ImDrawList* drawList = window->DrawList;
126 ImU32 restingFill = item.isSelected ? (isDark ? IM_COL32(45, 52, 68, 240) : IM_COL32(230, 235, 244, 255)) : IM_COL32(0, 0, 0, 0);
127 ImU32 hoverFill = isDark ? IM_COL32(52, 60, 78, 255) : IM_COL32(220, 226, 238, 255);
128 ImU32 currentFill = ColorUtils::Lerp(restingFill, hoverFill, hoverAlpha);
129
130 if ((currentFill & IM_COL32_A_MASK) != 0) {
131 drawList->AddRectFilled(bb.Min, bb.Max, currentFill, Radii::xs);
132 }
133
134 // Selected indicator left accent pill
135 if (item.isSelected) {
136 ImVec2 indMin(bb.Min.x + 2.0f, bb.Min.y + 4.0f);
137 ImVec2 indMax(bb.Min.x + 5.0f, bb.Max.y - 4.0f);
138 drawList->AddRectFilled(indMin, indMax, IM_COL32(242, 115, 38, 255), 2.0f);
139 }
140
141 // Render item contents
142 float fontSize = ImGui::GetFontSize();
143 float textY = bb.Min.y + (itemH - fontSize) * 0.5f;
144
145 if (currentWidth_ > 120.0f) {
146 // Expanded Mode: Icon + Label + Badge
147 float iconX = bb.Min.x + Spacing::md;
148 if (!item.iconText.empty()) {
149 ImU32 iconCol = item.isSelected ? IM_COL32(242, 115, 38, 255) : (isDark ? IM_COL32(190, 198, 212, 230) : IM_COL32(85, 92, 105, 230));
150 drawList->AddText(ImVec2(iconX, textY), iconCol, item.iconText.c_str());
151 iconX += ImGui::CalcTextSize(item.iconText.c_str()).x + Spacing::sm;
152 }
153
154 ImU32 textCol = item.isSelected ? (isDark ? IM_COL32(255, 255, 255, 255) : IM_COL32(15, 18, 25, 255)) : (isDark ? IM_COL32(215, 222, 235, 230) : IM_COL32(35, 40, 50, 230));
155 drawList->AddText(ImVec2(iconX, textY), textCol, item.label.c_str());
156
157 if (!item.badgeText.empty()) {
158 ImVec2 badgeSize = ImGui::CalcTextSize(item.badgeText.c_str());
159 float badgeX = bb.Max.x - badgeSize.x - Spacing::lg;
160 ImU32 badgeBg = isDark ? IM_COL32(255, 255, 255, 24) : IM_COL32(0, 0, 0, 18);
161 ImU32 badgeTextCol = isDark ? IM_COL32(180, 186, 200, 220) : IM_COL32(100, 105, 116, 220);
162 UIShapeManager::DrawPillBadge(drawList, ImVec2(badgeX + badgeSize.x * 0.5f, textY + fontSize * 0.5f),
163 item.badgeText.c_str(), badgeBg, badgeTextCol);
164 }
165 } else {
166 // Compact Rail Mode: Centered icon with tooltip flyout
167 ImVec2 iconSize = ImGui::CalcTextSize(item.iconText.c_str());
168 float iconX = bb.Min.x + (rowW - iconSize.x) * 0.5f;
169 ImU32 iconCol = item.isSelected ? IM_COL32(242, 115, 38, 255) : (isDark ? IM_COL32(200, 208, 222, 230) : IM_COL32(75, 82, 95, 230));
170 drawList->AddText(ImVec2(iconX, textY), iconCol, item.iconText.c_str());
171
172 if (hovered) {
173 ImGui::SetTooltip("%s", item.label.c_str());
174 }
175 }
