2/**
3 * =========================================================================================
4 * @file core/clipboard/platform/win32_clipboard.hpp
5 * @brief Native Windows Platform Clipboard Provider for High-Fidelity OLE Interoperability
6 * =========================================================================================
7 *
8 * ARCHITECTURAL DESIGN & INTEROPERABILITY:
9 * ----------------------------------------
10 * Standard Windows applications (Microsoft OneNote, Word, PowerPoint, Paint, Discord,
11 * Slack, web browsers) do not consume Linux/SDL-style MIME clipboard formats. Instead,
12 * they communicate over the Win32 OLE / User32 clipboard subsystem using:
13 *
14 * 1. "Ink Serialized Format" (ISF):
15 * Microsoft's native binary ink format. When OneNote copies ink strokes, it writes
16 * ISF to the clipboard. By reading ISF via COM (IInkDisp), FolioNote converts
17 * OneNote strokes into native vector InkContainer objects with millimeter coordinates.
18 * Conversely, when FolioNote copies ink strokes, it encodes them to ISF so OneNote
19 * pastes them as live, fully editable vector ink!
20 *
21 * 2. "PNG" (Registered Clipboard Format):
22 * Raw PNG image data with full 32-bit alpha transparency. Supported by Office, OneNote,
23 * Discord, Slack, Photoshop, and modern browsers for lossless transparent graphics.
24 *
25 * 3. CF_DIB / CF_DIBV5 (Device Independent Bitmap):
26 * Standard Win32 raster image format supported universally by all Windows applications
27 * including Microsoft Paint and legacy software.
28 *
29 * 4. "image/svg+xml":
30 * Standard W3C vector graphics format for modern vector editors (Illustrator, Inkscape).
31 *
32 * 5. CF_UNICODETEXT:
33 * Standard UTF-16 wide string for text, links, and URLs.
34 *
35 * MATHEMATICAL FOUNDATIONS:
36 * -------------------------
37 * 1. Microsoft Ink Coordinates (HIMETRIC to Millimeters):
38 * - In Microsoft Tablet PC COM (IInkDisp / IInkStrokeDisp), all coordinates and widths
39 * are stored in HIMETRIC units:
40 * 1 HIMETRIC unit = 0.01 mm = 10 micrometers = (1 / 2540) inch.
41 * - Conversion from HIMETRIC to FolioNote millimeter world space:
42 * worldMm = himetricCoord * 0.01
43 * - Conversion from FolioNote millimeter world space to HIMETRIC:
44 * himetricCoord = round(worldMm * 100.0)
45 *
46 * 2. Color Conversion (COLORREF vs FolioColor ARGB):
47 * - Windows COLORREF format is 0x00BBGGRR:
48 * R = c & 0xFF, G = (c >> 8) & 0xFF, B = (c >> 16) & 0xFF
49 * - FolioColor format is 0xAARRGGBB:
50 * FolioColor(R, G, B, 255)
51 * - Converting FolioColor to COLORREF:
52 * COLORREF = RGB(folio.r(), folio.g(), folio.b())
53 *
54 * 3. DIB Bitmap Header & Memory Layout (CF_DIB / CF_DIBV5):
55 * - Windows CF_DIB is bottom-up (first scanline in memory is the bottom row of the image).
56 * - Row byte alignment: DWORD-aligned (multiple of 4 bytes):
57 * rowBytes = ((width * 32 + 31) / 32) * 4 = width * 4 (for 32-bit BGRA).
58 * - To convert CF_DIB into a standard .bmp memory buffer for Blend2D decoding,
59 * a 14-byte BITMAPFILEHEADER (bfType = 0x4D42, bfOffBits) is prepended.
60 */

166/**
167 * @brief Creates a global memory handle containing a CF_DIB (BITMAPINFOHEADER + BGRA pixels).
168 *
169 * Mathematical Process:
170 * 1. Reads dimensions (w, h) and stride from the Blend2D surface.
171 * 2. Allocates GlobalAlloc(GMEM_MOVEABLE) for BITMAPINFOHEADER (40 bytes) + (w * 4 * h).
172 * 3. Populates standard BITMAPINFOHEADER with biCompression = BI_RGB and biBitCount = 32.
173 * 4. Inverts vertical orientation: Blend2D is top-down (row 0 at top), whereas CF_DIB is
174 * bottom-up (row 0 at bottom), so scanlines are copied in reverse order:
175 * dstRow(y) = srcRow(h - 1 - y).
176 *
177 * @param blImg Source Blend2D raster surface.
178 * @return HGLOBAL handle ready for SetClipboardData(CF_DIB, ...), or nullptr on failure.
179 */
180inline HGLOBAL CreateDibFromBlImage(const BLImage& blImg) {
181 if (blImg.is_empty()) return nullptr;
182
183 BLImageData imgData;
184 if (blImg.get_data(&imgData) != BL_SUCCESS) return nullptr;
185
186 const int w = imgData.size.w;
187 const int h = imgData.size.h;
188 if (w <= 0 || h <= 0) return nullptr;
189
190 const DWORD rowBytes = static_cast<DWORD>(w * 4);
191 const DWORD imageSize = rowBytes * static_cast<DWORD>(h);
192 const DWORD totalSize = sizeof(BITMAPINFOHEADER) + imageSize;
193
194 HGLOBAL hGlobal = ::GlobalAlloc(GMEM_MOVEABLE, totalSize);
195 if (!hGlobal) return nullptr;
196
197 auto* dest = static_cast<uint8_t*>(::GlobalLock(hGlobal));
198 if (!dest) {
199 ::GlobalFree(hGlobal);
200 return nullptr;
201 }
202
203 auto* bih = reinterpret_cast<BITMAPINFOHEADER*>(dest);
204 std::memset(bih, 0, sizeof(BITMAPINFOHEADER));
205 bih->biSize = sizeof(BITMAPINFOHEADER);
206 bih->biWidth = w;
207 bih->biHeight = h; // positive height = standard bottom-up DIB
208 bih->biPlanes = 1;
209 bih->biBitCount = 32;
210 bih->biCompression = BI_RGB;
211 bih->biSizeImage = imageSize;
212
213 uint8_t* pixelDest = dest + sizeof(BITMAPINFOHEADER);
214 const auto* srcPixels = static_cast<const uint8_t*>(imgData.pixel_data);
215
216 for (int y = 0; y < h; ++y) {
217 const uint8_t* srcRow = srcPixels + (h - 1 - y) * imgData.stride;
218 uint8_t* dstRow = pixelDest + y * rowBytes;
219 std::memcpy(dstRow, srcRow, rowBytes);
220 }
221
222 ::GlobalUnlock(hGlobal);
223 return hGlobal;
224}

296/**
297 * @brief Serializes all InkContainer objects in the selection into Microsoft Ink Serialized Format (ISF).
298 *
299 * Mathematical & COM Integration:
300 * 1. Instantiates COM CLSID_InkDisp (Microsoft Tablet PC Ink Object).
301 * 2. For each Stroke in each InkContainer:
302 * - Maps local stroke centerline points through the object's affine transform:
303 * pt_world = transform.map_point(pt_local.x, pt_local.y)
304 * - Converts millimeter world coordinates to HIMETRIC units:
305 * himetric_x = round(pt_world.x * 100.0)
306 * himetric_y = round(pt_world.y * 100.0)
307 * - Creates a 1D SAFEARRAY of VT_I4 containing interleaved [x0, y0, x1, y1, ...].
308 * - Calls pInk->CreateStroke(...) to register the vector path.
309 * - Sets drawing attributes: COLORREF (RGB) and stroke width in HIMETRIC:
310 * himetric_w = round(baseWidth * scale * 100.0)
311 * 3. Calls pInk->Save(IPF_InkSerializedFormat) to produce the compressed ISF byte stream.
312 *
313 * @param objects Canvas objects to scan for ink.
314 * @param outIsfBytes Destination vector populated with ISF binary data.
315 * @return true if at least one stroke was successfully converted; false otherwise.
316 */
317inline bool SerializeStrokesToIsf(const std::vector<std::shared_ptr<CanvasObject>>& objects,
318 std::vector<uint8_t>& outIsfBytes) {
319 IInkDisp* pInk = nullptr;
320 HRESULT hr = ::CoCreateInstance(kClsidInkDisp, nullptr, CLSCTX_INPROC_SERVER, kIidIInkDisp, reinterpret_cast<void**>(&pInk));
321 if (FAILED(hr) || !pInk) {
322 return false;
323 }
324
325 bool hasStrokes = false;
326
327 for (const auto& obj : objects) {
328 if (!obj) continue;
329 auto ink = std::dynamic_pointer_cast<InkContainer>(obj);
330 if (!ink) continue;
331
332 for (const auto& stroke : ink->strokes) {
333 std::vector<Point2D> pts;
334 if (!stroke.points.empty()) {
335 pts.reserve(stroke.points.size());
336 for (const auto& pt : stroke.points) {
337 pts.push_back({ pt.x, pt.y });
338 }
339 } else if (!stroke.segments.empty()) {
340 pts.reserve(stroke.segments.size() + 1);
341 pts.push_back(stroke.segments[0].p0);
342 for (const auto& seg : stroke.segments) {
343 pts.push_back(seg.p1);
344 }
345 }
346
347 if (pts.size() < 2) continue;
348
349 const auto numPoints = static_cast<long>(pts.size());
350 SAFEARRAYBOUND bound;
351 bound.cElements = static_cast<ULONG>(numPoints * 2);
352 bound.lLbound = 0;
353
354 SAFEARRAY* psa = ::SafeArrayCreate(VT_I4, 1, &bound);
355 if (!psa) continue;
356
357 LONG* coords = nullptr;
358 ::SafeArrayAccessData(psa, reinterpret_cast<void**>(&coords));
359 for (long p = 0; p < numPoints; ++p) {
360 BLPoint tp = ink->transform.map_point(pts[p].x, pts[p].y);
361 coords[p * 2] = static_cast<LONG>(std::round(tp.x * 100.0));
362 coords[p * 2 + 1] = static_cast<LONG>(std::round(tp.y * 100.0));
363 }
364 ::SafeArrayUnaccessData(psa);
365
366 VARIANT varPoints;
367 ::VariantInit(&varPoints);
368 varPoints.vt = VT_ARRAY | VT_I4;
369 varPoints.parray = psa;
370
371 VARIANT varPacketDesc;
372 ::VariantInit(&varPacketDesc);
373 varPacketDesc.vt = VT_ERROR;
374 varPacketDesc.scode = DISP_E_PARAMNOTFOUND;
375
376 IInkStrokeDisp* pStroke = nullptr;
377 hr = pInk->CreateStroke(varPoints, varPacketDesc, &pStroke);
378 ::VariantClear(&varPoints);
379
380 if (SUCCEEDED(hr) && pStroke) {
381 IInkDrawingAttributes* pDA = nullptr;
382 if (SUCCEEDED(pStroke->get_DrawingAttributes(&pDA)) && pDA) {
383 COLORREF winColor = RGB(stroke.color.r(), stroke.color.g(), stroke.color.b());
384 pDA->put_Color(winColor);
385
386 const double scale = std::hypot(ink->transform.m00, ink->transform.m01);
387 const auto wHimetric = static_cast<long>(std::round(stroke.baseWidthMm * (scale > 1e-4 ? scale : 1.0) * 100.0));
388 pDA->put_Width(static_cast<float>(wHimetric > 0 ? wHimetric : 50));
389 pDA->Release();
390 }
391 pStroke->Release();
392 hasStrokes = true;
393 }
394 }
395 }
396
397 if (!hasStrokes) {
398 pInk->Release();
399 return false;
400 }
401
402 VARIANT varIsf;
403 ::VariantInit(&varIsf);
404 hr = pInk->Save(InkPersistenceFormat::IPF_InkSerializedFormat,
405 InkPersistenceCompressionMode::IPCM_Default,
406 &varIsf);
407 pInk->Release();
408
409 if (SUCCEEDED(hr) && (varIsf.vt & VT_ARRAY) && varIsf.parray) {
410 LONG lBound = 0, uBound = 0;
411 ::SafeArrayGetLBound(varIsf.parray, 1, &lBound);
412 ::SafeArrayGetUBound(varIsf.parray, 1, &uBound);
413 const long isfSize = uBound - lBound + 1;
414 void* pData = nullptr;
415 ::SafeArrayAccessData(varIsf.parray, &pData);
416 outIsfBytes.resize(isfSize);
417 std::memcpy(outIsfBytes.data(), pData, isfSize);
418 ::SafeArrayUnaccessData(varIsf.parray);
419 ::VariantClear(&varIsf);
420 return true;
421 }
422
423 ::VariantClear(&varIsf);
424 return false;
425}

427/**
428 * @brief Parses raw Microsoft ISF bytes from OneNote into a native FolioNote InkContainer.
429 *
430 * Mathematical & Geometric Process:
431 * 1. Instantiates COM CLSID_InkDisp and calls pInk->Load(varData) with the ISF byte array.
432 * 2. Enumerates IInkStrokes:
433 * - Retrieves points in HIMETRIC units via pStroke->GetPoints(0, -1, &varPoints).
434 * - Converts HIMETRIC coordinates to world millimeters:
435 * px = coords[p * 2] * 0.01
436 * py = coords[p * 2 + 1] * 0.01
437 * - Extracts COLORREF and converts 0x00BBGGRR to FolioColor(R, G, B, 255).
438 * - Extracts stroke width in HIMETRIC and converts to millimeters: widthMm = winWidth * 0.01.
439 * - Builds discrete segments and calls StrokeOutlineBuilder::BuildOutline to produce the BLPath.
440 * 3. Centers the resulting InkContainer at (targetWorldX, targetWorldY) and bakes the transform.
441 *
442 * @param isfData Pointer to raw ISF bytes.
443 * @param isfSize Size in bytes.
444 * @param targetWorldX Target X coordinate in millimeters.
445 * @param targetWorldY Target Y coordinate in millimeters.
446 * @return std::shared_ptr<InkContainer> with imported vector strokes, or nullptr on failure.
447 */
448inline std::shared_ptr<InkContainer> ParseIsfToInkContainer(const uint8_t* isfData,
449 size_t isfSize,
450 double targetWorldX,
451 double targetWorldY) {
452 if (!isfData || isfSize == 0) return nullptr;
453
454 SAFEARRAYBOUND bound;
455 bound.cElements = static_cast<ULONG>(isfSize);
456 bound.lLbound = 0;
457
458 SAFEARRAY* psa = ::SafeArrayCreate(VT_UI1, 1, &bound);
459 if (!psa) return nullptr;
460
461 void* dest = nullptr;
462 ::SafeArrayAccessData(psa, &dest);
463 std::memcpy(dest, isfData, isfSize);
464 ::SafeArrayUnaccessData(psa);
465
466 VARIANT varData;
467 ::VariantInit(&varData);
468 varData.vt = VT_ARRAY | VT_UI1;
469 varData.parray = psa;
470
471 IInkDisp* pInk = nullptr;
472 HRESULT hr = ::CoCreateInstance(kClsidInkDisp, nullptr, CLSCTX_INPROC_SERVER, kIidIInkDisp, reinterpret_cast<void**>(&pInk));
473 if (FAILED(hr) || !pInk) {
474 ::VariantClear(&varData);
475 return nullptr;
476 }
477
478 hr = pInk->Load(varData);
479 ::VariantClear(&varData);
480
481 if (FAILED(hr)) {
482 pInk->Release();
483 return nullptr;
484 }
485
486 IInkStrokes* pStrokes = nullptr;
487 if (FAILED(pInk->get_Strokes(&pStrokes)) || !pStrokes) {
488 pInk->Release();
489 return nullptr;
490 }
491
492 long strokeCount = 0;
493 pStrokes->get_Count(&strokeCount);
494 if (strokeCount <= 0) {
495 pStrokes->Release();
496 pInk->Release();
497 return nullptr;
498 }
499
500 auto inkContainer = std::make_shared<InkContainer>();
501
502 for (long i = 0; i < strokeCount; ++i) {
503 IInkStrokeDisp* pStroke = nullptr;
504 if (FAILED(pStrokes->Item(i, &pStroke)) || !pStroke) continue;
505
506 COLORREF winColor = RGB(0, 0, 0);
507 long winColorLong = 0;
508 float winWidthFloat = 50.0f; // default in HIMETRIC units
509 IInkDrawingAttributes* pDA = nullptr;
510 if (SUCCEEDED(pStroke->get_DrawingAttributes(&pDA)) && pDA) {
511 if (SUCCEEDED(pDA->get_Color(&winColorLong))) {
512 winColor = static_cast<COLORREF>(winColorLong);
513 }
514 pDA->get_Width(&winWidthFloat);
515 pDA->Release();
516 }
517
518 VARIANT varPoints;
519 ::VariantInit(&varPoints);
520 if (SUCCEEDED(pStroke->GetPoints(0, -1, &varPoints))) {
521 if ((varPoints.vt & VT_ARRAY) && (varPoints.vt & VT_I4) && varPoints.parray) {
522 LONG lBound = 0, uBound = 0;
523 ::SafeArrayGetLBound(varPoints.parray, 1, &lBound);
524 ::SafeArrayGetUBound(varPoints.parray, 1, &uBound);
525 const long numCoords = uBound - lBound + 1;
526 const long numPoints = numCoords / 2;
527
528 LONG* coords = nullptr;
529 ::SafeArrayAccessData(varPoints.parray, reinterpret_cast<void**>(&coords));
530 if (coords && numPoints >= 2) {
531 Stroke s;
532 const uint8_t r = GetRValue(winColor);
533 const uint8_t g = GetGValue(winColor);
534 const uint8_t b = GetBValue(winColor);
535 s.color = BLRgba32(r, g, b, 255);
536
537 const double widthMm = (winWidthFloat > 0.0f) ? (static_cast<double>(winWidthFloat) * 0.01) : 0.5;
538 s.baseWidthMm = static_cast<float>(widthMm);
539
540 s.segments.reserve(numPoints - 1);
541 s.points.reserve(numPoints);
542
543 for (long p = 0; p < numPoints; ++p) {
544 const double px = coords[p * 2] * 0.01; // HIMETRIC -> mm
545 const double py = coords[p * 2 + 1] * 0.01; // HIMETRIC -> mm
546 s.points.push_back({ px, py, static_cast<float>(widthMm * 0.5) });
547 if (p > 0) {
548 Segment1D seg;
549 seg.p0 = Point2D{ s.points[p - 1].x, s.points[p - 1].y };
550 seg.p1 = Point2D{ px, py };
551 seg.width = static_cast<float>(widthMm);
552 s.segments.push_back(seg);
553 }
554 }
555
556 // Build closed polygon outline contour (BLPath)
557 std::vector<StrokeOutlineBuilder::InputPoint> pts;
558 pts.reserve(s.points.size());
559 for (const auto& sp : s.points) {
560 pts.push_back({ static_cast<float>(sp.x), static_cast<float>(sp.y), static_cast<float>(widthMm), 1.0f });
561 }
562 s.outlinePath = StrokeOutlineBuilder::BuildOutline(pts, CapType::Round);
563
564 inkContainer->AddStroke(s);
565 }
566 ::SafeArrayUnaccessData(varPoints.parray);
567 }
568 ::VariantClear(&varPoints);
569 }
570 pStroke->Release();
571 }
572
573 pStrokes->Release();
574 pInk->Release();
575
576 if (inkContainer->strokes.empty()) {
577 return nullptr;
578 }
579
580 inkContainer->UpdateBounds();
581
582 // Center container at (targetWorldX, targetWorldY)
583 const double cx = (inkContainer->bounds.minX + inkContainer->bounds.maxX) * 0.5;
584 const double cy = (inkContainer->bounds.minY + inkContainer->bounds.maxY) * 0.5;
585 const double dx = targetWorldX - cx;
586 const double dy = targetWorldY - cy;
587
588 BLMatrix2D trans = BLMatrix2D::make_translation(dx, dy);
589 inkContainer->ApplyTransform(trans);
590 inkContainer->BakeTransform();
591
592 return inkContainer;
593}

601 *
602 * Formats Published:
603 * 1. "Ink Serialized Format" (ISF) -> for OneNote vector ink.
604 * 2. "PNG" -> for Office, Discord, Slack, Photoshop (lossless 32-bit alpha).
605 * 3. CF_DIB / CF_DIBV5 -> for Microsoft Paint and universal Win32 paste.
606 * 4. "image/svg+xml" -> for vector applications.
607 * 5. CF_UNICODETEXT -> for plain text / URLs.
608 *
609 * @param objects Original selected canvas objects.
610 * @param svgXml Serialized SVG XML string.
611 * @param pngBytes Rendered PNG byte buffer.
612 * @param blImg Blend2D rendered raster surface.
613 * @param plainText Extracted plaintext string.
614 * @param[out] outSeqNumber Captured Windows clipboard sequence number after publishing.
615 * @return true if clipboard opened and data was set successfully.
616 */
617inline bool PublishToWindowsClipboard(const std::vector<std::shared_ptr<CanvasObject>>& objects,
618 const std::string& svgXml,
619 const std::vector<uint8_t>& pngBytes,
620 const BLImage& blImg,
621 const std::string& plainText,
622 uint32_t& outSeqNumber) {
623 if (!::OpenClipboard(nullptr)) {
624 LOG_WARN(General, "PublishToWindowsClipboard: Failed to open Win32 clipboard.");
625 return false;
626 }
627
628 ::EmptyClipboard();
629
630 // 1. Publish "Ink Serialized Format" (ISF) if ink strokes are present
631 std::vector<uint8_t> isfBytes;
632 if (SerializeStrokesToIsf(objects, isfBytes) && !isfBytes.empty()) {
633 const UINT cfIsf = GetIsfClipboardFormat();
634 HGLOBAL hIsf = ::GlobalAlloc(GMEM_MOVEABLE, isfBytes.size());
635 if (hIsf) {
636 void* p = ::GlobalLock(hIsf);
637 if (p) {
638 std::memcpy(p, isfBytes.data(), isfBytes.size());
639 ::GlobalUnlock(hIsf);
640 ::SetClipboardData(cfIsf, hIsf);
641 LOG_INFO(General, "Published Ink Serialized Format (ISF: " + std::to_string(isfBytes.size()) + " B) to Windows clipboard.");
642 } else {
643 ::GlobalFree(hIsf);
644 }
645 }
646 }
647
648 // 2. Publish "PNG" format (lossless 32-bit alpha for OneNote, Discord, Office)
649 if (!pngBytes.empty()) {
650 const UINT cfPng = GetPngClipboardFormat();
651 HGLOBAL hPng = ::GlobalAlloc(GMEM_MOVEABLE, pngBytes.size());
652 if (hPng) {
653 void* p = ::GlobalLock(hPng);
654 if (p) {
655 std::memcpy(p, pngBytes.data(), pngBytes.size());
656 ::GlobalUnlock(hPng);
657 ::SetClipboardData(cfPng, hPng);
658 } else {
659 ::GlobalFree(hPng);
660 }
661 }
662 }
663
664 // 3. Publish CF_DIB & CF_DIBV5 (universal bitmap for Paint, Word, etc.)
665 if (!blImg.is_empty()) {
666 HGLOBAL hDibV5 = CreateDibV5FromBlImage(blImg);
667 if (hDibV5) {
668 ::SetClipboardData(CF_DIBV5, hDibV5);
669 }
670 HGLOBAL hDib = CreateDibFromBlImage(blImg);
671 if (hDib) {
672 ::SetClipboardData(CF_DIB, hDib);
673 }
674 }
675
676 // 4. Publish "image/svg+xml"
677 if (!svgXml.empty()) {
678 const UINT cfSvg = GetSvgClipboardFormat();
679 HGLOBAL hSvg = ::GlobalAlloc(GMEM_MOVEABLE, svgXml.size() + 1);
680 if (hSvg) {
681 void* p = ::GlobalLock(hSvg);
682 if (p) {
683 std::memcpy(p, svgXml.c_str(), svgXml.size() + 1);
684 ::GlobalUnlock(hSvg);
685 ::SetClipboardData(cfSvg, hSvg);
686 } else {
687 ::GlobalFree(hSvg);
688 }
689 }
690 }
691
692 // 5. Publish CF_UNICODETEXT (plain text / URLs)
693 if (!plainText.empty()) {
694 const int wlen = ::MultiByteToWideChar(CP_UTF8, 0, plainText.c_str(), -1, nullptr, 0);
695 if (wlen > 0) {
696 HGLOBAL hText = ::GlobalAlloc(GMEM_MOVEABLE, static_cast<size_t>(wlen) * sizeof(wchar_t));
697 if (hText) {
698 auto* pW = static_cast<wchar_t*>(::GlobalLock(hText));
699 if (pW) {
700 ::MultiByteToWideChar(CP_UTF8, 0, plainText.c_str(), -1, pW, wlen);
701 ::GlobalUnlock(hText);
702 ::SetClipboardData(CF_UNICODETEXT, hText);
703 } else {
704 ::GlobalFree(hText);
705 }
706 }
707 }
708 }
709
710 ::CloseClipboard();
711 outSeqNumber = ::GetClipboardSequenceNumber();
712 return true;
713}

732/**
733 * @brief Attempts to ingest clipboard content from the native Windows OS clipboard.
734 *
735 * Ingestion Hierarchy:
736 * Priority 1: "Ink Serialized Format" (ISF) -> native editable InkContainer with vector strokes.
737 * Priority 2: "PNG" -> native ImageObject with full alpha transparency.
738 * Priority 3: CF_DIBV5 / CF_DIB -> synthesized BMP decoded to native ImageObject.
739 * Priority 4: CF_UNICODETEXT -> VideoObject (YouTube URL) or TextBoxObject.
740 *
741 * @param session Active DocumentSession.
742 * @param engine Active CanvasEngine.
743 * @param targetWorldX Target X coordinate in millimeters.
744 * @param targetWorldY Target Y coordinate in millimeters.
745 * @param[out] outCreated Newly instantiated and placed CanvasObjects.
746 * @return true if data was successfully ingested and added to the page; false otherwise.
747 */
748inline bool IngestFromWindowsClipboard(DocumentSession& session,
749 CanvasEngine& engine,
750 double targetWorldX,
751 double targetWorldY,
752 std::vector<std::shared_ptr<CanvasObject>>& outCreated) {
753 if (!::OpenClipboard(nullptr)) {
754 return false;
755 }
756
757 auto activePage = session.GetActivePage();
758 if (!activePage) {
759 ::CloseClipboard();
760 return false;
761 }
762
763 const double screenDpi = engine.transform.pixelsPerMm * 25.4;
764
765 // --- Priority 1: Ink Serialized Format (ISF from OneNote) ---
766 const UINT cfIsf = GetIsfClipboardFormat();
767 if (::IsClipboardFormatAvailable(cfIsf)) {
768 HANDLE hData = ::GetClipboardData(cfIsf);
769 if (hData) {
770 const void* pData = ::GlobalLock(hData);
771 const size_t dataSize = ::GlobalSize(hData);
772 if (pData && dataSize > 0) {
773 auto inkContainer = ParseIsfToInkContainer(static_cast<const uint8_t*>(pData), dataSize, targetWorldX, targetWorldY);
774 ::GlobalUnlock(hData);
775 if (inkContainer) {
776 inkContainer->uid = UIDGenerator::Next();
777 inkContainer->guuid = GUIDGenerator::GenerateV4();
778 inkContainer->isSelected = 1;
779
780 session.AddObject(inkContainer);
781 outCreated.push_back(inkContainer);
782 engine.selectionGizmo.SetSelectedObjects({inkContainer});
783 engine.needsFullRebake = true;
784 engine.isDirty = true;
785 ::CloseClipboard();
786 LOG_INFO(General, "Ingested live vector ink from OneNote (ISF: " +
787 std::to_string(inkContainer->strokes.size()) + " strokes) at (" +
788 std::to_string(targetWorldX) + ", " + std::to_string(targetWorldY) + ") mm");
789 return true;
790 }
791 } else if (pData) {
792 ::GlobalUnlock(hData);
793 }
794 }
795 }
796
797 // --- Priority 2: PNG format (from OneNote, Office, Discord, web browsers) ---
798 const UINT cfPng = GetPngClipboardFormat();
799 if (::IsClipboardFormatAvailable(cfPng)) {
800 HANDLE hData = ::GetClipboardData(cfPng);
801 if (hData) {
802 const void* pData = ::GlobalLock(hData);
803 const size_t dataSize = ::GlobalSize(hData);
804 if (pData && dataSize > 0) {
805 std::string relPath = CanvasEngine::DeduplicateAndSaveImage("", pData, dataSize, &session, ".png");
806 auto img = std::make_shared<ImageObject>(static_cast<const uint8_t*>(pData), dataSize, relPath, screenDpi);
807 ::GlobalUnlock(hData);
808
809 if (img->isLoaded) {
810 img->uid = UIDGenerator::Next();
811 img->guuid = GUIDGenerator::GenerateV4();
812 img->worldX = targetWorldX - img->worldWidth * 0.5;
813 img->worldY = targetWorldY - img->worldHeight * 0.5;
814 img->UpdateBounds();
815 img->isSelected = 1;
816
817 session.AddImage(img);
818 outCreated.push_back(img);
819 engine.selectionGizmo.SetSelectedObjects({img});
820 engine.needsFullRebake = true;
821 engine.isDirty = true;
822 ::CloseClipboard();
823 LOG_INFO(General, "Ingested PNG image from Windows clipboard (" +
824 std::to_string(img->naturalWidth) + "x" + std::to_string(img->naturalHeight) + " px)");
825 return true;
826 }
827 } else if (pData) {
828 ::GlobalUnlock(hData);
829 }
830 }
831 }
832
833 // --- Priority 3: CF_DIB / CF_DIBV5 (from Paint or legacy tools) ---
834 if (::IsClipboardFormatAvailable(CF_DIB) || ::IsClipboardFormatAvailable(CF_DIBV5)) {
835 const UINT dibFmt = ::IsClipboardFormatAvailable(CF_DIBV5) ? CF_DIBV5 : CF_DIB;
836 HANDLE hData = ::GetClipboardData(dibFmt);
837 if (hData) {
838 const auto* pData = static_cast<const uint8_t*>(::GlobalLock(hData));
839 const size_t dataSize = ::GlobalSize(hData);
840 if (pData && dataSize >= sizeof(BITMAPINFOHEADER)) {
841 const auto* bih = reinterpret_cast<const BITMAPINFOHEADER*>(pData);
842
843 // Synthesize 14-byte BITMAPFILEHEADER
844 BITMAPFILEHEADER bfh;
845 std::memset(&bfh, 0, sizeof(BITMAPFILEHEADER));
846 bfh.bfType = 0x4D42; // "BM"
847 bfh.bfSize = static_cast<DWORD>(sizeof(BITMAPFILEHEADER) + dataSize);
848
849 DWORD colorTableSize = 0;
850 if (bih->biBitCount <= 8) {
851 const DWORD numColors = bih->biClrUsed ? bih->biClrUsed : (1U << bih->biBitCount);
852 colorTableSize = numColors * sizeof(RGBQUAD);
853 } else if (bih->biCompression == BI_BITFIELDS) {
854 colorTableSize = 3 * sizeof(DWORD);
855 }
856 bfh.bfOffBits = static_cast<DWORD>(sizeof(BITMAPFILEHEADER) + bih->biSize + colorTableSize);
857
858 std::vector<uint8_t> bmpBuffer(sizeof(BITMAPFILEHEADER) + dataSize);
859 std::memcpy(bmpBuffer.data(), &bfh, sizeof(BITMAPFILEHEADER));
860 std::memcpy(bmpBuffer.data() + sizeof(BITMAPFILEHEADER), pData, dataSize);
861 ::GlobalUnlock(hData);
862
863 // Decode BMP buffer via Blend2D (auto-detects BMP header)
864 BLImage decodedBmp;
865 if (decodedBmp.read_from_data(bmpBuffer.data(), bmpBuffer.size()) == BL_SUCCESS && !decodedBmp.is_empty()) {
866 BLArray<uint8_t> pngBytes;
867 BLImageCodec pngCodec;
868 pngCodec.find_by_name("PNG");
869 decodedBmp.write_to_data(pngBytes, pngCodec);
870
871 if (!pngBytes.is_empty()) {
872 std::string relPath = CanvasEngine::DeduplicateAndSaveImage("", pngBytes.data(), pngBytes.size(), &session, ".png");
873 auto img = std::make_shared<ImageObject>(pngBytes.data(), pngBytes.size(), relPath, screenDpi);
874 if (img->isLoaded) {
875 img->uid = UIDGenerator::Next();
876 img->guuid = GUIDGenerator::GenerateV4();
877 img->worldX = targetWorldX - img->worldWidth * 0.5;
878 img->worldY = targetWorldY - img->worldHeight * 0.5;
879 img->UpdateBounds();
880 img->isSelected = 1;
881
882 session.AddImage(img);
883 outCreated.push_back(img);
884 engine.selectionGizmo.SetSelectedObjects({img});
885 engine.needsFullRebake = true;
886 engine.isDirty = true;
887 ::CloseClipboard();
888 LOG_INFO(General, "Ingested DIB bitmap from Windows clipboard (" +
889 std::to_string(img->naturalWidth) + "x" + std::to_string(img->naturalHeight) + " px)");
890 return true;
891 }
892 }
893 }
894 } else if (pData) {
895 ::GlobalUnlock(hData);
896 }
897 }
898 }
899
900 // --- Priority 4: CF_UNICODETEXT (Plain text, URLs, paths) ---
901 if (::IsClipboardFormatAvailable(CF_UNICODETEXT)) {
902 HANDLE hData = ::GetClipboardData(CF_UNICODETEXT);
903 if (hData) {
904 const auto* pWText = static_cast<const wchar_t*>(::GlobalLock(hData));
905 if (pWText) {
906 const int utf8Len = ::WideCharToMultiByte(CP_UTF8, 0, pWText, -1, nullptr, 0, nullptr, nullptr);
907 std::string clipText;
908 if (utf8Len > 0) {
909 clipText.resize(utf8Len - 1);
910 ::WideCharToMultiByte(CP_UTF8, 0, pWText, -1, clipText.data(), utf8Len, nullptr, nullptr);
911 }
912 ::GlobalUnlock(hData);
913
914 if (!clipText.empty()) {
915 const bool isHttp = (clipText.rfind("http://", 0) == 0 || clipText.rfind("https://", 0) == 0);
916 const bool isYouTube = (clipText.find("youtube.com") != std::string::npos || clipText.find("youtu.be") != std::string::npos);
917
918 if (isYouTube) {
919 const double cardW = 160.0;
920 const double cardH = 95.0;
921 auto vid = std::make_shared<VideoObject>(
922 clipText, "YouTube Video", cardW, cardH
923 );
924 vid->uid = UIDGenerator::Next();
925 vid->guuid = GUIDGenerator::GenerateV4();
926 vid->worldX = targetWorldX - cardW * 0.5;
927 vid->worldY = targetWorldY - cardH * 0.5;
928 vid->UpdateBounds();
929 vid->isSelected = 1;
930
931 session.AddObject(vid);
932 outCreated.push_back(vid);
933 engine.selectionGizmo.SetSelectedObjects({vid});
934 engine.needsFullRebake = true;
935 engine.isDirty = true;
936 ::CloseClipboard();
937 LOG_INFO(General, "Ingested YouTube URL from Windows clipboard as VideoObject: " + clipText);
938 return true;
939 }
940
941 // Fallback: Create TextBoxObject for text
942 auto tb = std::make_shared<TextBoxObject>();
943 tb->uid = UIDGenerator::Next();
944 tb->guuid = GUIDGenerator::GenerateV4();
945 tb->text = clipText;
946 tb->worldX = targetWorldX - 40.0;
947 tb->worldY = targetWorldY - 15.0;
948 tb->worldWidth = 80.0;
949 tb->worldHeight = 30.0;
950 tb->UpdateBounds();
951 tb->isSelected = 1;
952
953 session.AddTextBox(tb);
954 outCreated.push_back(tb);
955 engine.selectionGizmo.SetSelectedObjects({tb});
956 engine.needsFullRebake = true;
957 engine.isDirty = true;
958 ::CloseClipboard();
959 LOG_INFO(General, "Ingested text from Windows clipboard into TextBoxObject (" + std::to_string(clipText.size()) + " chars)");
960 return true;
961 }
962 }
963 }
964 }
965
966 ::CloseClipboard();
967 return false;
968}
