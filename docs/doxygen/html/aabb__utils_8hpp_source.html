23/**
24 * @brief Computes the enclosing world-space AABB of a transformed rectangle.
25 *
26 * MATHEMATICAL FOUNDATION & WORKING PROCESS:
27 * ------------------------------------------
28 * 1. Coordinate Normalization:
29 * Handles negative width/height (which occur during inverted user resizing/mirroring):
30 * minX_local = min(x, x + width), maxX_local = max(x, x + width)
31 * minY_local = min(y, y + height), maxY_local = max(y, y + height)
32 *
33 * 2. 2D Affine Transformation:
34 * Maps each of the 4 normalized corners through the 2D affine matrix M:
35 * [ x' ] [ m00 m01 m02 ] [ x ] [ m00*x + m01*y + m02 ]
36 * [ y' ] = [ m10 m11 m12 ] [ y ] = [ m10*x + m11*y + m12 ]
37 * [ 1 ] [ 0 0 1 ] [ 1 ] [ 1 ]
38 *
39 * Corners transformed:
40 * C0 = M * (minX, minY)
41 * C1 = M * (maxX, minY)
42 * C2 = M * (maxX, maxY)
43 * C3 = M * (minX, maxY)
44 *
45 * 3. Enclosing World-Space Bounds Calculation:
46 * Finds the minimum and maximum X and Y across all 4 mapped vertices:
47 * worldMinX = min(C0.x, C1.x, C2.x, C3.x)
48 * worldMinY = min(C0.y, C1.y, C2.y, C3.y)
49 * worldMaxX = max(C0.x, C1.x, C2.x, C3.x)
50 * worldMaxY = max(C0.y, C1.y, C2.y, C3.y)
51 *
52 * @param x Origin X in local millimeters.
53 * @param y Origin Y in local millimeters.
54 * @param width Extent width in local millimeters (may be negative).
55 * @param height Extent height in local millimeters (may be negative).
56 * @param transform 2D affine transformation matrix.
57 * @return AABB Tight world-space bounding box enclosing the transformed rectangle.
58 */
59[[nodiscard]] inline AABB ComputeTransformedBounds(double x, double y, double width, double height, const BLMatrix2D& transform) noexcept {
60 double minX = (std::min)(x, x + width);
61 double maxX = (std::max)(x, x + width);
62 double minY = (std::min)(y, y + height);
63 double maxY = (std::max)(y, y + height);
64
65 BLPoint corners[4] = {
66 transform.map_point(minX, minY),
67 transform.map_point(maxX, minY),
68 transform.map_point(maxX, maxY),
69 transform.map_point(minX, maxY)
70 };
71
72 return AABB{
73 (std::min)({corners[0].x, corners[1].x, corners[2].x, corners[3].x}),
74 (std::min)({corners[0].y, corners[1].y, corners[2].y, corners[3].y}),
75 (std::max)({corners[0].x, corners[1].x, corners[2].x, corners[3].x}),
76 (std::max)({corners[0].y, corners[1].y, corners[2].y, corners[3].y})
77 };
78}

94/**
95 * @brief Computes the enclosing world-space AABB of an arbitrary set of 2D points transformed by a matrix.
96 *
97 * @param points Pointer to array of BLPoint vertices.
98 * @param count Number of vertices in the array.
99 * @param transform 2D affine transformation matrix.
100 * @return AABB Tight world-space bounding box enclosing all transformed vertices.
101 */
102[[nodiscard]] inline AABB ComputeTransformedBounds(const BLPoint* points, size_t count, const BLMatrix2D& transform) noexcept {
103 if (!points || count == 0) {
104 return AABB{};
105 }
106
107 BLPoint first = transform.map_point(points[0].x, points[0].y);
108 double minX = first.x, maxX = first.x;
109 double minY = first.y, maxY = first.y;
110
111 for (size_t i = 1; i < count; ++i) {
112 BLPoint pt = transform.map_point(points[i].x, points[i].y);
113 minX = (std::min)(minX, pt.x);
114 maxX = (std::max)(maxX, pt.x);
115 minY = (std::min)(minY, pt.y);
116 maxY = (std::max)(maxY, pt.y);
117 }
118
119 return AABB(minX, minY, maxX, maxY);
120}

122/**
123 * @brief Commits an accumulated 2D affine transform matrix into intrinsic rectangular coordinates (X, Y, Width, Height)
124 * and resets the matrix to the identity state.
125 *
126 * MATHEMATICAL FOUNDATION & WORKING PROCESS:
127 * ------------------------------------------
128 * 1. Axis-Aligned Guard:
129 * Only axis-aligned scale + translation transforms (where shear/rotation m01 ≈ 0 and m10 ≈ 0)
130 * can be committed cleanly into an axis-aligned rectangle (worldX, worldY, worldWidth, worldHeight).
131 * Rotated objects must preserve their rotation matrix.
132 *
133 * 2. Identity Fast-Path:
134 * If the matrix is already identity (m00=1, m11=1, m20=0, m21=0), returns false (nothing to bake).
135 *
136 * 3. 2D Affine Transformation of Diagonal Corners:
137 * P0 = (inOutX, inOutY)
138 * P1 = (inOutX + inOutWidth, inOutY + inOutHeight)
139 *
140 * p0' = (m00 * P0.x + m20, m11 * P0.y + m21)
141 * p1' = (m00 * P1.x + m20, m11 * P1.y + m21)
142 *
143 * 4. Normalization and Bounds Protection:
144 * Handles handle dragging across opposite axes (horizontal or vertical flipping):
145 * inOutX = min(p0'.x, p1'.x)
146 * inOutY = min(p0'.y, p1'.y)
147 * inOutWidth = max(minSize, |p1'.x - p0'.x|)
148 * inOutHeight = max(minSize, |p1'.y - p0'.y|)
149 *
150 * 5. Matrix Reset:
151 * Resets inOutTransform to BLMatrix2D::make_identity().
152 *
153 * @param[in,out] inOutX Origin X coordinate in millimeters.
154 * @param[in,out] inOutY Origin Y coordinate in millimeters.
155 * @param[in,out] inOutWidth Width in millimeters.
156 * @param[in,out] inOutHeight Height in millimeters.
157 * @param[in,out] inOutTransform 2D affine matrix; committed and reset to identity on success.
158 * @param[in] minSize Minimum allowed width and height to prevent degenerate zero-area collapse.
159 * @return true if the transform was non-identity, axis-aligned, and successfully baked; false otherwise.
160 */
161inline bool BakeTransformedRect(double& inOutX, double& inOutY,
162 double& inOutWidth, double& inOutHeight,
163 BLMatrix2D& inOutTransform,
164 double minSize = 1.0) noexcept {
165 if (std::abs(inOutTransform.m01) < 1e-6 && std::abs(inOutTransform.m10) < 1e-6) {
166 if (inOutTransform.m00 == 1.0 && inOutTransform.m11 == 1.0 &&
167 inOutTransform.m20 == 0.0 && inOutTransform.m21 == 0.0) {
168 return false; // Already identity — nothing to bake
169 }
170
171 double p0x = (inOutTransform.m00 * inOutX) + inOutTransform.m20;
172 double p0y = (inOutTransform.m11 * inOutY) + inOutTransform.m21;
173 double p1x = (inOutTransform.m00 * (inOutX + inOutWidth)) + inOutTransform.m20;
174 double p1y = (inOutTransform.m11 * (inOutY + inOutHeight)) + inOutTransform.m21;
175
176 inOutX = (std::min)(p0x, p1x);
177 inOutY = (std::min)(p0y, p1y);
178 inOutWidth = (std::max)(minSize, std::abs(p1x - p0x));
179 inOutHeight = (std::max)(minSize, std::abs(p1y - p0y));
180
181 inOutTransform = BLMatrix2D::make_identity();
182 return true;
183 }
184 return false;
185}

187/**
188 * @brief Computes the closest point on an AABB to an external point (px, py).
189 *
190 * Mathematical Process:
191 * Clamps the query coordinates to the box bounds:
192 * Q_x = clamp(px, minX, maxX)
193 * Q_y = clamp(py, minY, maxY)
194 *
195 * @param box Target bounding box.
196 * @param px Query point X coordinate in millimeters.
197 * @param py Query point Y coordinate in millimeters.
198 * @return std::pair<double, double> Coordinates (Q_x, Q_y) of the closest point on the AABB.
199 */
200[[nodiscard]] inline std::pair<double, double> ClosestPoint(const AABB& box, double px, double py) noexcept {
201 if (box.IsEmpty()) {
202 return {px, py};
203 }
204 double qx = (std::max)(box.minX, (std::min)(px, box.maxX));
205 double qy = (std::max)(box.minY, (std::min)(py, box.maxY));
206 return {qx, qy};
207}

209/**
210 * @brief Computes the squared Euclidean distance from a point to the closest point on the AABB.
211 * Returns 0.0 if the point is strictly inside the box.
212 *
213 * Mathematical Formulation:
214 * dist^2 = (px - clamp(px, minX, maxX))^2 + (py - clamp(py, minY, maxY))^2
215 * Using squared distance avoids expensive std::sqrt() calls in tight inner collision loops.
216 *
217 * @param box Target bounding box.
218 * @param px Query point X coordinate in millimeters.
219 * @param py Query point Y coordinate in millimeters.
220 * @return Squared Euclidean distance as a positive double.
221 */
222[[nodiscard]] inline double DistanceSquared(const AABB& box, double px, double py) noexcept {
223 if (box.IsEmpty()) {
224 return std::numeric_limits<double>::infinity();
225 }
226 double clampedX = (std::max)(box.minX, (std::min)(px, box.maxX));
227 double clampedY = (std::max)(box.minY, (std::min)(py, box.maxY));
228 double dx = px - clampedX;
229 double dy = py - clampedY;
230 return (dx * dx) + (dy * dy);
231}

233/**
234 * @brief Evaluates whether a circle (center at cx, cy with given radius) touches or overlaps an AABB.
235 *
236 * Working Process:
237 * 1. Broad phase: Checks if circle center is inside the box (instant hit).
238 * 2. Distance test: Evaluates if squared distance from circle center to closest point on AABB <= radius^2.
239 *
240 * @param box Target bounding box.
241 * @param cx Circle center X in millimeters.
242 * @param cy Circle center Y in millimeters.
243 * @param radius Radius of circle in millimeters.
244 * @return true if the circle overlaps or touches the AABB; false otherwise.
245 */
246[[nodiscard]] inline bool IntersectsCircle(const AABB& box, double cx, double cy, double radius) noexcept {
247 if (box.IsEmpty() || radius < 0.0) {
248 return false;
249 }
250 // Direct interior containment fast-path
251 if (box.Contains(cx, cy)) {
252 return true;
253 }
254 return DistanceSquared(box, cx, cy) <= (radius * radius);
255}

257/**
258 * @brief Evaluates whether a continuous swept capsule segment (from (x0, y0) to (x1, y1) with radius) touches an AABB.
259 * Prevents high-speed tunneling in eraser strokes and continuous raycast queries.
260 *
261 * Mathematical Process:
262 * 1. Broad phase: Check if swept segment AABB intersects target box.
263 * 2. Endpoint tests: IntersectsCircle(box, x0, y0, radius) or IntersectsCircle(box, x1, y1, radius).
264 * 3. Segment projection: Projects box center C onto segment S(t) = w0 + t * (w1 - w0), t in [0, 1].
265 * 4. Circle test at closest segment point: IntersectsCircle(box, projX, projY, radius).
266 *
267 * @param box Target bounding box.
268 * @param x0 Segment start X in millimeters.
269 * @param y0 Segment start Y in millimeters.
270 * @param x1 Segment end X in millimeters.
271 * @param y1 Segment end Y in millimeters.
272 * @param radius Capsule radius in millimeters.
273 * @return true if the swept volume intersects the AABB; false otherwise.
274 */
275[[nodiscard]] inline bool IntersectsSwept(const AABB& box, double x0, double y0, double x1, double y1, double radius) noexcept {
276 if (box.IsEmpty() || radius < 0.0) {
277 return false;
278 }
279
280 // 1. Broad-phase query box check
281 AABB sweptBox(
282 (std::min)(x0, x1) - radius,
283 (std::min)(y0, y1) - radius,
284 (std::max)(x0, x1) + radius,
285 (std::max)(y0, y1) + radius
286 );
287 if (!box.Intersects(sweptBox)) {
288 return false;
289 }
290
291 // 2. Check segment endpoints
292 if (IntersectsCircle(box, x0, y0, radius) || IntersectsCircle(box, x1, y1, radius)) {
293 return true;
294 }
295
296 // 3. Project box center onto the segment
297 double centerX = (box.minX + box.maxX) * 0.5;
298 double centerY = (box.minY + box.maxY) * 0.5;
299
300 double vx = x1 - x0;
301 double vy = y1 - y0;
302 double lenSq = (vx * vx) + (vy * vy);
303
304 if (lenSq > 1e-6) {
305 double t = ((centerX - x0) * vx + (centerY - y0) * vy) / lenSq;
306 t = (std::max)(0.0, (std::min)(1.0, t));
307 double projX = x0 + t * vx;
308 double projY = y0 + t * vy;
309 return IntersectsCircle(box, projX, projY, radius);
310 }
311
312 return false;
313}
