2/**
3 * =========================================================================================
4 * @file utils/physics_model.hpp
5 * @brief High-performance computational physics models and numerical integrators
6 * =========================================================================================
7 *
8 * ARCHITECTURAL ROLE:
9 * -------------------
10 * Provides real-time physical simulation models for creative canvas interactions:
11 * - Second-order Damped Harmonic Oscillators (Spring-Damper systems)
12 * - Inertial velocity tracking and kinetic friction damping
13 * - Velocity-coupled dynamic radius physics specifically engineered for dynamic erasers
14 * - High-order numerical integrators (Semi-Implicit Euler, Runge-Kutta 4)
15 *
16 * MATHEMATICAL FOUNDATIONS & NUMERICAL INTEGRATION:
17 * -------------------------------------------------
18 * 1. Second-Order Damped Harmonic Oscillator (Mass-Spring-Damper):
19 * Governing linear ordinary differential equation of motion:
20 * m * (d²x / dt²) + c * (dx / dt) + k * (x - x_target) = 0
21 *
22 * Dividing by mass m yields the canonical equation:
23 * (d²x / dt²) + 2*ζ*ω_n * (dx / dt) + (ω_n)² * (x - x_target) = 0
24 *
25 * Physical Parameters:
26 * m : Virtual inertial mass (kg)
27 * k : Restoring spring stiffness coefficient (N/m)
28 * c : Viscous damping coefficient (N*s/m)
29 * ω_n : Natural undamped angular frequency = sqrt(k / m) [rad/s]
30 * ζ (zeta) : Dimensionless damping ratio = c / (2 * sqrt(k * m))
31 *
32 * Oscillator Regimes:
33 * • ζ = 1.0 (Critically Damped):
34 * Fastest asymptotic convergence to target position with zero overshoot or ringing.
35 * Ideal for UI panels, camera tracking, and smooth cursor convergence.
36 * • ζ < 1.0 (Underdamped):
37 * Produces decaying harmonic oscillations around target before settling.
38 * Useful for playful rubber-band bounces and elastic feedback.
39 * • ζ > 1.0 (Overdamped):
40 * Sluggish, non-oscillatory asymptotic exponential decay.
41 * Useful for heavy inertial sliders and viscous fluid feel.
42 *
43 * 2. Numerical Discretization (Symplectic / Semi-Implicit Euler):
44 * Standard Explicit (Forward) Euler is energy-increasing and unstable for oscillatory
45 * systems. This engine implements Semi-Implicit Euler, which preserves the symplectic
46 * 2-form in phase space and maintains unconditional orbital stability:
47 *
48 * a_n = [ -k * (x_n - x_target) - c * v_n ] / m
49 * v_n+1 = v_n + a_n * dt
50 * x_n+1 = x_n + v_n+1 * dt (Note: uses next velocity v_n+1, NOT v_n)
51 *
52 * 3. Sub-stepping & Numerical Stability:
53 * To prevent numerical divergence during large frame deltas (e.g. frame hitch dt > 16ms),
54 * Step() sub-steps iterations with a maximum slice:
55 * dt_sub <= 8.33 ms (equivalent to >= 120 Hz internal physics tick rate).
56 *
57 * 4. Velocity-Coupled Dynamic Eraser Physics:
58 * Couples raw pointer speed v (px/sec) to a dynamic virtual reticle radius r (mm):
59 * - Kinetic Energy / Target Mapping:
60 * v_norm = clamp((v - v_min) / (v_max - v_min), 0.0, 1.0)
61 * r_target = r_min + sqrt(v_norm) * (r_max - r_min)
62 * - Asymmetric Dynamic Response:
63 * • Rapid Expansion (v increases):
64 * High stiffness (k_expand) ensures immediate reticle growth on fast strokes.
65 * • Smooth Settling (v decreases):
66 * Gentle stiffness and critical damping (c_contract) prevent jarring visual popping
67 * when pointer abruptly decelerates.
68 * - Bounds Clamping: Physical boundary preservation ensuring r remains in [r_min, r_max].
69 */

78 * @brief Physical parameters configuring a harmonic oscillator.

80struct SpringConfig {
81 float mass = 1.0f; ///< Mass m (kg, > 0)
82 float stiffness = 120.0f; ///< Spring stiffness coefficient k (N/m)
83 float damping = 18.0f; ///< Viscous damping coefficient c (N*s/m)
84 float minPosition = 0.0f; ///< Hard lower bound clamp
85 float maxPosition = 1000.0f; ///< Hard upper bound clamp
86
87 /**
88 * @brief Factory creating a critically damped spring configuration given frequency.
89 * @param naturalFrequencyHz Desired response oscillation frequency in Hz.
90 * @param m System mass (default 1.0f).
91 */
92 static SpringConfig CriticallyDamped(float naturalFrequencyHz, float m = 1.0f) noexcept {
93 float omega = 2.0f * 3.14159265358979323846f * naturalFrequencyHz;
94 float k = m * omega * omega;
95 float c = 2.0f * std::sqrt(k * m); // zeta = 1.0
96 return { m, k, c, -1e9f, 1e9f };
97 }
98
99 /**
100 * @brief Computes the natural undamped angular frequency omega_n.
101 */
102 [[nodiscard]] float NaturalFrequency() const noexcept {
103 return (mass > 0.0f && stiffness > 0.0f) ? std::sqrt(stiffness / mass) : 0.0f;
104 }
105
106 /**
107 * @brief Computes the dimensionless damping ratio zeta.
108 */
109 [[nodiscard]] float DampingRatio() const noexcept {
110 float denom = 2.0f * std::sqrt(stiffness * mass);
111 return (denom > 0.0f) ? (damping / denom) : 0.0f;
112 }
113};

172 /**
173 * @brief Advances physical simulation by dt seconds using Semi-Implicit Euler integration.
174 * @param dt Elapsed timestep in seconds (> 0). Substepped if dt > maxSubstepDt.
175 */
176 void Step(float dt, float maxSubstepDt = 0.008333f) noexcept {
177 if (dt <= 0.0f) return;
178
179 // Substep large frames to preserve numerical stability
180 while (dt > 0.0f) {
181 float stepDt = std::min(dt, maxSubstepDt);
182 IntegrateStep(stepDt);
183 dt -= stepDt;
184 }
185
186 // Clamp to configured bounds
187 m_position = std::clamp(m_position, m_config.minPosition, m_config.maxPosition);
188 }
189
190private:
191 SpringConfig m_config;
192 float m_position = 0.0f;
193 float m_velocity = 0.0f;
194 float m_target = 0.0f;
195
196 void IntegrateStep(float dt) noexcept {
197 float displacement = m_position - m_target;
198 float springForce = -m_config.stiffness * displacement;
199 float dampingForce = -m_config.damping * m_velocity;
200 float netForce = springForce + dampingForce;
201 float acceleration = (m_config.mass > 0.0f) ? (netForce / m_config.mass) : 0.0f;
202
203 // Semi-Implicit (Symplectic) Euler:
204 // v_{n+1} = v_n + a * dt
205 // x_{n+1} = x_n + v_{n+1} * dt
206 m_velocity += acceleration * dt;
207 m_position += m_velocity * dt;
208 }
209};

223 m_mass = std::max(0.0001f, mass);

225 }
226

228 m_velocity += impulse / m_mass;
229 }
230
231 void ApplyForce(float force) noexcept {
232 m_forceAccum += force;
233 }
234
235 void Step(float dt) noexcept {
236 if (dt <= 0.0f) return;
237 float frictionForce = -m_friction * m_velocity;
238 float netForce = m_forceAccum + frictionForce;
239 float acc = netForce / m_mass;
240
241 m_velocity += acc * dt;
242 m_position += m_velocity * dt;
243 m_forceAccum = 0.0f;
244 }
245
246 [[nodiscard]] float GetPosition() const noexcept { return m_position; }
247 [[nodiscard]] float GetVelocity() const noexcept { return m_velocity; }
248 void Reset(float pos = 0.0f, float vel = 0.0f) noexcept {
249 m_position = pos;
250 m_velocity = vel;
251 m_forceAccum = 0.0f;
252 }
253
254private:
255 float m_mass = 1.0f;
256 float m_friction = 5.0f;
257 float m_position = 0.0f;
258 float m_velocity = 0.0f;
259 float m_forceAccum = 0.0f;
260};

262/**
263 * @class DynamicEraserPhysicsModel
264 * @brief High-precision physical model for velocity-reactive dynamic eraser sizing.
265 *
266 * Couples instantaneous pointer speed to a virtual mass-spring-damper:
267 * - Speed v in [v_min, v_max] maps through a smooth concave square-root transfer curve
268 * to an equilibrium target radius R_target in [R_min, R_max].
269 * - Asymmetric dynamics:
270 * * Acceleration / Expansion: Driven by high-stiffness spring with fast kinetic coupling.
271 * * Deceleration / Contraction: Damped smoothly with critical damping to eliminate reticle popping.
272 */
273class DynamicEraserPhysicsModel {
274public:
275 struct Parameters {
276 float minRadiusMm = 1.5f; ///< Minimum fine precision radius (at zero/slow speed)
277 float maxRadiusMm = 16.0f; ///< Maximum broad wiping radius (at high scrubbing speed)
278 float minSpeedPxPerSec = 100.0f;///< Velocity activation floor
279 float maxSpeedPxPerSec = 2000.0f;///< Velocity ceiling for maximum expansion
280
281 // Physical spring-damper tuning:
282 float mass = 1.0f; ///< Virtual reticle mass
283 float expandStiffness = 180.0f; ///< Spring stiffness during expansion (responsive)
284 float expandDamping = 22.0f; ///< Damping during expansion
285 float contractStiffness = 70.0f;///< Spring stiffness during contraction (gentle)
286 float contractDamping = 18.0f; ///< Damping during contraction (jitter-free)
287 };

330 /**
331 * @brief Updates the physical model given pointer speed and timestep.
332 * @param speedPxPerSec Instantaneous pointer velocity in pixels per second.
333 * @param dt Elapsed timestep in seconds.
334 * @return Simulated dynamic radius in physical millimeters.
335 */
336 float Update(float speedPxPerSec, float dt) noexcept {
337 // Map pointer velocity to target equilibrium radius via concave root transfer curve
338 float speedRange = m_params.maxSpeedPxPerSec - m_params.minSpeedPxPerSec;
339 float normSpeed = (speedRange > 0.0f) ?
340 std::clamp((speedPxPerSec - m_params.minSpeedPxPerSec) / speedRange, 0.0f, 1.0f) : 0.0f;
341
342 float responseCurve = std::sqrt(normSpeed);
343 float targetRadiusMm = m_params.minRadiusMm + responseCurve * (m_params.maxRadiusMm - m_params.minRadiusMm);
344
345 // Asymmetric physical tuning based on expansion vs. contraction
346 SpringConfig cfg = m_spring.GetConfig();
347 if (targetRadiusMm > m_spring.GetPosition()) {
348 cfg.stiffness = m_params.expandStiffness;
349 cfg.damping = m_params.expandDamping;
350 } else {
351 cfg.stiffness = m_params.contractStiffness;
352 cfg.damping = m_params.contractDamping;
353 }
354 m_spring.SetConfig(cfg);
355 m_spring.SetTarget(targetRadiusMm);
356
357 // Advance simulation
358 m_spring.Step(dt);
359 m_currentRadiusMm = m_spring.GetPosition();
360
361 return m_currentRadiusMm;
362 }

368 return m_currentRadiusMm;
369 }

376 }

382};
