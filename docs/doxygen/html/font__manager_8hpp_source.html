28struct FontKey {
29 std::string family;
30 float sizePt = 16.0f;

34 bool operator==(const FontKey& o) const noexcept {

39struct FontKeyHash {
40 size_t operator()(const FontKey& k) const noexcept {
41 size_t h1 = std::hash<std::string>{}(k.family);
42 size_t h2 = std::hash<float>{}(k.sizePt);
43 size_t h3 = (k.bold ? 1 : 0) | (k.italic ? 2 : 0);
44 return h1 ^ (h2 << 1) ^ (h3 << 7);
45 }
46};
