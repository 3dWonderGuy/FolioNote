1/**
2 * @file input_state_machine_touch.cpp
3 * @brief Implementation of Capacitive Multi-Touch dispatching: Touch Paint (single-finger ink),
4 * two-finger pinch-pan, tap selection, long-press holds, and multi-touch shortcuts (Undo/Redo).
5 *
6 * MULTI-TOUCH ARCHITECTURE:
7 * -------------------------
8 * Touch interactions on canvas follow a clear modal separation:
9 * 1. Touch Paint Disabled (Default):
10 * - 1 Finger Drag: Panning / scrolling the canvas.
11 * - 2 Fingers: Pinch-to-zoom and simultaneous two-finger panning around the focal centroid.
12 * - 2 Finger Tap: Shortcut for Undo.
13 * - 3 Finger Tap: Shortcut for Redo.
14 * - 1 Finger Tap: Direct object hit-testing / selection or popup dismissal.
15 * - 1 Finger Hold: Triggers context / radial menu.
16 * 2. Touch Paint Enabled (`touchTool.savedTool == Inking`):
17 * - 1 Finger Drag: Inks strokes on canvas with active pen preset.
18 * - 2+ Fingers: Always navigates (pinch-pan) regardless of mode.
19 */

26void InputStateMachine::DispatchTouch(CanvasEngine& canvas, DocumentSession& session, bool imguiWantsInput) {
27 size_t fingerCount = GetActiveFingerCount();
28 uint64_t now = SDL_GetTicks();
29
30 bool touchDown = fingerCount > 0;
31 bool justDown = touchDown && !wasTouchDown;
32 bool justUp = !touchDown && wasTouchDown;
33
34 // -------------------------------------------------------------------------
35 // 1. UI CAPTURE CHECK
36 // -------------------------------------------------------------------------
37 if (justDown) {
38 uiCapturedTouch = imguiWantsInput;
39 }
40 if (uiCapturedTouch) {
41 if (justUp) uiCapturedTouch = false;
42 wasTouchDown = touchDown;
43 return;
44 }
45
46 wasTouchDown = touchDown;
47
48 // -------------------------------------------------------------------------
49 // 2. TOUCH INKING MODE (Touch Paint: 1 Finger Inks, 2+ Fingers Navigate)
50 // -------------------------------------------------------------------------
51 if (touchTool.savedTool == InteractionState::Inking && fingerCount == 1) {
52 const TouchSlot* slot = nullptr;
53 for (const auto& s : activeFingers) {
54 if (s.fingerID != -1) { slot = &s; break; }
55 }
56 if (slot) {
57 const float canvasLocalX = slot->point.x - canvasOriginX;
58 const float canvasLocalY = slot->point.y - canvasOriginY;
59 const float pressure = (slot->point.pressure > 0.0f) ? slot->point.pressure : 1.0f;
60
61 if (justDown) {
62 canvas.OnPointerDown(canvasLocalX, canvasLocalY, pressure, latestEventTimeSec, palette.GetActivePen(), 0.0f, 0.0f);
63 } else if (touchDown && wasTouchDown) {
64 canvas.OnPointerMove(canvasLocalX, canvasLocalY, pressure, latestEventTimeSec, 0.0f, 0.0f);
65 } else if (justUp) {
66 canvas.OnPointerUp(session, palette.GetActivePen());
67 }
68 }
69 return; // Suppress gesture recognizer while single-finger inking
70 }
71
72 // -------------------------------------------------------------------------
73 // 3. GESTURE RECOGNITION (Navigation & Special Action Shortcuts)
74 // -------------------------------------------------------------------------
75 TouchGestureEvent gesture = gestureRecognizer.Evaluate(activeFingers, fingerCount, now, false);
76
77 const float localFocusX = gesture.focusX - canvasOriginX;
78 const float localFocusY = gesture.focusY - canvasOriginY;
79
80 switch (gesture.type) {
81 // Single finger scroll / pan
82 case TouchGestureType::SingleFingerScroll: {
83 canvas.Pan(gesture.deltaX, gesture.deltaY);
84 break;
85 }
86
87 // Two-finger pinch-to-zoom and simultaneous panning
88 case TouchGestureType::TwoFingerPinchPan: {
89 canvas.Pan(gesture.deltaX, gesture.deltaY);
90 if (std::abs(gesture.zoomDelta - 1.0f) > config.zoomDeltaEpsilon) {
91 canvas.ZoomAt(localFocusX, localFocusY, gesture.zoomDelta);
92 }
93 break;
94 }
95
96 // Multi-touch digital ink shortcut: 2-Finger Tap = UNDO
97 case TouchGestureType::TwoFingerTap: {
98 LOG_INFO(InputStateMachine, "2-Finger Tap gesture recognized -> Triggering UNDO");
99 specialActions.TriggerAction(FolioInput::SpecialActionType::Undo, localFocusX, localFocusY, canvas, session);
100 break;
101 }
102
103 // Multi-touch digital ink shortcut: 3-Finger Tap = REDO
104 case TouchGestureType::ThreeFingerTap: {
105 LOG_INFO(InputStateMachine, "3-Finger Tap gesture recognized -> Triggering REDO");
106 specialActions.TriggerAction(FolioInput::SpecialActionType::Redo, localFocusX, localFocusY, canvas, session);
107 break;
108 }
109
110 // Single finger tap: select object under contact or clear selection
111 case TouchGestureType::Tap: {
112 LOG_INFO(InputStateMachine, "Touch Tap gesture recognized at (" + std::to_string(localFocusX) + ", " + std::to_string(localFocusY) + ")");
113 specialActions.TriggerAction(FolioInput::SpecialActionType::SelectAtPoint, localFocusX, localFocusY, canvas, session);
114 break;
115 }
116
117 // Stationary long-press hold: open context menu or radial menu
118 case TouchGestureType::PressAndHold: {
119 LOG_INFO(InputStateMachine, "Touch Press-and-Hold recognized -> Triggering Context Menu");
120 specialActions.TriggerAction(FolioInput::SpecialActionType::OpenContextMenu, localFocusX, localFocusY, canvas, session);
121 break;
122 }
123
124 default: break;
125 }
126}
