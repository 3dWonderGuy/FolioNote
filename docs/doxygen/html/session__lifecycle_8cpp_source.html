106/**
107 * @brief Opens or activates a notebook by its persistent GUID or package directory path.
108 *
109 * ARCHITECTURAL PROCESS:
110 * 1. Flushes dirty state of outgoing active notebook asynchronously to prevent data loss.
111 * 2. Locates the notebook in workspace.notebooks (or loads it on demand if a disk path was provided).
112 * 3. Updates workspace.activeNotebookIndex.
113 * 4. Broadcasts NotifyActiveNotebookChanged, NotifyActiveSectionChanged, and NotifyActivePageChanged.
114 *
115 * @param guidOrPath Persistent UUID v4 of notebook or filesystem path to .notebook directory.
116 * @return true if notebook was found and activated.
117 */
118bool DocumentSession::OpenNotebook(const std::string& guidOrPath) {
119 if (guidOrPath.empty()) return false;
120 auto oldNb = GetActiveNotebook();
121 if (oldNb && (oldNb->guid == guidOrPath || oldNb->filePath == guidOrPath)) {
122 return true; // Already active
123 }
124
125 // Save unsaved changes before switching notebooks
126 if (oldNb) {
127 workspace.FlushActiveNotebookAsync();
128 }
129
130 for (size_t i = 0; i < workspace.notebooks.size(); ++i) {
131 const auto& nb = workspace.notebooks[i];
132 if (nb && (nb->guid == guidOrPath || nb->filePath == guidOrPath)) {
133 workspace.activeNotebookIndex = i;
134 auto newNb = nb;
135 NotifyActiveNotebookChanged(newNb, oldNb);
136 auto newSec = newNb->GetActiveSection();
137 NotifyActiveSectionChanged(newSec, nullptr);
138 auto newPage = newNb->GetActivePage();
139 NotifyActivePageChanged(newPage, nullptr);
140 NotifyHistoryChanged();
141 LOG_INFO(DocumentSession, "OpenNotebook: Activated notebook '" + newNb->name + "' [" + newNb->guid + "]");
142 return true;
143 }
144 }
145
146 // If not in resident list, check if it's a valid filesystem path to a notebook
147 std::error_code ec;
148 if (std::filesystem::exists(guidOrPath, ec) && std::filesystem::is_directory(guidOrPath, ec)) {
149 if (auto loaded = workspace.repository.LoadNotebookHierarchy(guidOrPath)) {
150 workspace.notebooks.push_back(loaded);
151 workspace.activeNotebookIndex = workspace.notebooks.size() - 1;
152 NotifyActiveNotebookChanged(loaded, oldNb);
153 NotifyActiveSectionChanged(loaded->GetActiveSection(), nullptr);
154 NotifyActivePageChanged(loaded->GetActivePage(), nullptr);
155 NotifyHistoryChanged();
156 LOG_INFO(DocumentSession, "OpenNotebook: Loaded and activated notebook from path: " + guidOrPath);
157 return true;
158 }
159 }
160
161 LOG_WARN(DocumentSession, "OpenNotebook failed: Could not find or load notebook for: " + guidOrPath);
162 return false;
163}

165/**
166 * @brief Creates a new notebook package inside a target library and activates it.
167 *
168 * ARCHITECTURAL PROCESS:
169 * 1. Formulates target directory path with collision resolution.
170 * 2. Flushes any currently active notebook.
171 * 3. Instantiates Notebook with initial default "General" section and "New Untitled Page".
172 * 4. Schedules initial persistence and registers inside workspace.notebooks.
173 * 5. Notifies observers.
174 *
175 * @param name Display title of the new notebook.
176 * @param libraryPath Target library directory path (defaults to Default library).
177 * @return Shared pointer to the newly created Notebook, or nullptr on failure.
178 */
179std::shared_ptr<Notebook> DocumentSession::CreateNotebook(const std::string& name, const std::string& libraryPath) {
180 std::string libDir = libraryPath;
181 if (libDir.empty()) {
182 std::filesystem::path appRoot(workspace.workspaceDirectory);
183 libDir = (appRoot / "Libraries" / "Default").string();
184 }
185
186 std::string safeName = name.empty() ? "Untitled Notebook" : name;
187 std::filesystem::path targetDir = std::filesystem::path(libDir) / (safeName + ".notebook");
188
189 // Disambiguate directory name if collision
190 std::error_code ec;
191 int counter = 1;
192 while (std::filesystem::exists(targetDir, ec)) {
193 targetDir = std::filesystem::path(libDir) / (safeName + " (" + std::to_string(counter++) + ").notebook");
194 }
195
196 auto oldNb = GetActiveNotebook();
197 if (oldNb) {
198 workspace.FlushActiveNotebookAsync();
199 }
200
201 auto newNb = std::make_shared<Notebook>(safeName);
202 newNb->filePath = targetDir.string();
203
204 // Create default initial section and page
205 auto defaultSec = std::make_shared<Section>("General");
206 defaultSec->notebookGuid = newNb->guid;
207 auto defaultPage = std::make_shared<CanvasPage>("New Untitled Page");
208 defaultSec->AddPage(defaultPage);
209 newNb->AddSection(defaultSec);
210
211 // Persist structure to disk
212 workspace.repository.SaveNotebookAsync(newNb);
213
214 workspace.notebooks.push_back(newNb);
215 workspace.activeNotebookIndex = workspace.notebooks.size() - 1;
216
217 NotifyActiveNotebookChanged(newNb, oldNb);
218 NotifyActiveSectionChanged(defaultSec, nullptr);
219 NotifyActivePageChanged(defaultPage, nullptr);
220 NotifyHistoryChanged();
221 LOG_INFO(DocumentSession, "CreateNotebook: Created and activated new notebook '" + safeName + "' at: " + targetDir.string());
222 return newNb;
223}

225/**
226 * @brief Closes the currently active notebook, persisting changes and switching to another notebook if available.
227 * @return true if closed successfully.
228 */
229bool DocumentSession::CloseActiveNotebook() {
230 auto activeNb = GetActiveNotebook();
231 if (!activeNb) return false;
232
233 workspace.FlushActiveNotebookAsync();
234 std::string closedGuid = activeNb->guid;
235
236 // Remove from resident notebooks if more than 1 notebook is open
237 if (workspace.notebooks.size() > 1) {
238 workspace.notebooks.erase(workspace.notebooks.begin() + workspace.activeNotebookIndex);
239 if (workspace.activeNotebookIndex >= workspace.notebooks.size()) {
240 workspace.activeNotebookIndex = workspace.notebooks.size() - 1;
241 }
242 auto nextNb = GetActiveNotebook();
243 NotifyActiveNotebookChanged(nextNb, activeNb);
244 if (nextNb) {
245 NotifyActiveSectionChanged(nextNb->GetActiveSection(), nullptr);
246 NotifyActivePageChanged(nextNb->GetActivePage(), nullptr);
247 }
248 }
249 LOG_INFO(DocumentSession, "CloseActiveNotebook: Closed notebook [" + closedGuid + "]");
250 return true;
251}
