106 curLine.endChar = 0;

108 curLine.right = chars[0].right;

133 // Disallow extreme backward wrap on the same vertical span (e.g. multi-column layouts)
134 bool horizWrap = (c.left < curLine.left - 5.0);

136 if (vertOverlap && !horizWrap) {
137 curLine.endChar = static_cast<int>(i);
138 curLine.left = std::min(curLine.left, c.left);
139 curLine.right = std::max(curLine.right, c.right);
140 curLine.top = std::min(curLine.top, c.top);
141 curLine.bottom = std::max(curLine.bottom, c.bottom);
142 } else {
143 lines.push_back(curLine);
144 curLine.startChar = static_cast<int>(i);
145 curLine.endChar = static_cast<int>(i);
146 curLine.left = c.left;
147 curLine.right = c.right;
148 curLine.top = c.top;
149 curLine.bottom = c.bottom;
150 }
151 }
152 lines.push_back(curLine);
153 }

155 FPDFText_ClosePage(textPage);
156 return true;
157#else
158 return false;
159#endif

162int PdfTextLayer::HitTestChar(double localX, double localY, double toleranceMm) const {
163 if (chars.empty()) return -1;
164
165 // 1. Direct bounding box check
166 for (const auto& c : chars) {
167 if (localX >= c.left && localX <= c.right && localY >= c.top && localY <= c.bottom) {
168 return c.charIndex;
169 }
170 }
171
172 // 2. Tolerance-padded proximity check
173 for (const auto& c : chars) {
174 if (localX >= (c.left - toleranceMm) && localX <= (c.right + toleranceMm) &&
175 localY >= (c.top - toleranceMm) && localY <= (c.bottom + toleranceMm)) {
176 return c.charIndex;
177 }
178 }
179 return -1;
180}

182int PdfTextLayer::FindNearestChar(double localX, double localY) const {
183 if (chars.empty()) return -1;
184 if (lines.empty()) return 0;
185
186 // 1. Find matching line vertically
187 int bestLineIdx = 0;
188 double bestLineDist = 1e18;
189
190 for (int l = 0; l < static_cast<int>(lines.size()); ++l) {
191 const auto& line = lines[l];
192 if (localY >= line.top && localY <= line.bottom) {
193 bestLineIdx = l;
194 bestLineDist = 0.0;
195 break;
196 }
197 double lineMidY = (line.top + line.bottom) * 0.5;
198 double distY = std::abs(localY - lineMidY);
199 if (distY < bestLineDist) {
200 bestLineDist = distY;
201 bestLineIdx = l;
202 }
203 }
204
205 const auto& targetLine = lines[bestLineIdx];
206
207 // 2. If mouse is to the left of the line, smoothly clamp to line start
208 if (localX <= targetLine.left) {
209 return targetLine.startChar;
210 }
211
212 // 3. If mouse is to the right of the line, smoothly clamp to line end
213 if (localX >= targetLine.right) {
214 return targetLine.endChar;
215 }
216
217 // 4. Find exact or closest character on this line
218 int bestChar = targetLine.startChar;
219 double bestCharDist = 1e18;
220 for (int c = targetLine.startChar; c <= targetLine.endChar; ++c) {
221 const auto& ch = chars[c];
222 if (localX >= ch.left && localX <= ch.right) {
223 return c;
224 }
225 double midX = (ch.left + ch.right) * 0.5;
226 double d = std::abs(localX - midX);
227 if (d < bestCharDist) {
228 bestCharDist = d;
229 bestChar = c;
230 }
231 }
232 return bestChar;
233}

235std::vector<AABB> PdfTextLayer::GetSelectionBoxes(const PdfTextSelection& sel) const {
236 std::vector<AABB> boxes;
237 if (!sel.hasSelection || chars.empty()) return boxes;
238
239 int minI = std::clamp(sel.MinIdx(), 0, static_cast<int>(chars.size()) - 1);
240 int maxI = std::clamp(sel.MaxIdx(), 0, static_cast<int>(chars.size()) - 1);
241
242 AABB currentLineBox;
243 bool hasLine = false;
244
245 for (int i = minI; i <= maxI; ++i) {
246 const auto& c = chars[i];
247 if (c.unicode == '\r' || c.unicode == '\n') {
248 if (hasLine) {
249 boxes.push_back(currentLineBox);
250 hasLine = false;
251 }
252 continue;
253 }
254
255 if (c.right <= c.left || c.bottom <= c.top) {
256 continue; // Skip zero-dimension glyphs
257 }
258
259 AABB charBox(c.left, c.top, c.right, c.bottom);
260 if (!hasLine) {
261 currentLineBox = charBox;
262 hasLine = true;
263 } else {
264 // Check vertical overlap with current line
265 double overlapMin = std::max(c.top, currentLineBox.minY);
266 double overlapMax = std::min(c.bottom, currentLineBox.maxY);
267 double charH = c.bottom - c.top;
268 double lineH = currentLineBox.maxY - currentLineBox.minY;
269 bool sameLine = (overlapMax > overlapMin) &&
270 ((overlapMax - overlapMin) >= std::min(charH, lineH) * 0.35);
271
272 if (sameLine) {
273 currentLineBox.minX = std::min(currentLineBox.minX, charBox.minX);
274 currentLineBox.maxX = std::max(currentLineBox.maxX, charBox.maxX);
275 currentLineBox.minY = std::min(currentLineBox.minY, charBox.minY);
276 currentLineBox.maxY = std::max(currentLineBox.maxY, charBox.maxY);
277 } else {
278 boxes.push_back(currentLineBox);
279 currentLineBox = charBox;
280 }
281 }
282 }
283
284 if (hasLine) {
285 boxes.push_back(currentLineBox);
286 }
287
288 return boxes;
289}
