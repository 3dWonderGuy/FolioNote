18void UIActionRegistry::RegisterAction(UIAction action) {
19 if (action.id.empty()) {
20 // Fallback: If no explicit ID provided, synthesize one from category and label
21 action.id = (action.category.empty() ? "action" : action.category) + "." + action.label;
22 }
23
24 // Register or update shortcut mapping if a valid keyboard shortcut is declared
25 if (!action.shortcut.empty()) {
26 m_shortcutMap[action.shortcut] = action.id;
27 }
28
29 m_actions[action.id] = std::move(action);
30}

32bool UIActionRegistry::UnregisterAction(const std::string& id) {
33 auto it = m_actions.find(id);
34 if (it == m_actions.end()) {
35 return false;
36 }
37
38 // Remove associated keyboard shortcut mapping if present
39 if (!it->second.shortcut.empty()) {
40 auto scIt = m_shortcutMap.find(it->second.shortcut);
41 if (scIt != m_shortcutMap.end() && scIt->second == id) {
42 m_shortcutMap.erase(scIt);
43 }
44 }
45
46 m_actions.erase(it);
47 return true;
48}

62bool UIActionRegistry::Execute(const std::string& id) const {
63 const UIAction* act = FindAction(id);
64 if (!act) {
65 return false;
66 }
67
68 if (!act->IsVisible() || !act->IsEnabled()) {
69 return false;
70 }
71
72 act->Execute();
73 return true;
74}

76bool UIActionRegistry::DispatchShortcut(const std::string& shortcut) const {
77 if (shortcut.empty()) {
78 return false;
79 }
80
81 auto scIt = m_shortcutMap.find(shortcut);
82 if (scIt == m_shortcutMap.end()) {
83 return false;
84 }
85
86 return Execute(scIt->second);
87}

89std::vector<UIAction> UIActionRegistry::GetActionsByCategory(const std::string& category) const {
90 std::vector<UIAction> result;
91 for (const auto& [id, action] : m_actions) {
92 if (action.category == category && action.IsVisible()) {
93 result.push_back(action);
94 }
95 }
96
97 // Deterministic sorting by `order` weight ascending
98 std::stable_sort(result.begin(), result.end(), [](const UIAction& a, const UIAction& b) {
99 return a.order < b.order;
100 });
101
102 return result;
103}

105std::vector<UIAction> UIActionRegistry::GetAllActions() const {
106 std::vector<UIAction> result;
107 result.reserve(m_actions.size());
108 for (const auto& [id, action] : m_actions) {

110 result.push_back(action);

113
114 std::stable_sort(result.begin(), result.end(), [](const UIAction& a, const UIAction& b) {
115 return a.order < b.order;
116 });
117

131std::vector<UIAction> UIActionRegistry::GetContextActions(const std::string& contextName) const {
132 auto it = m_contextProviders.find(contextName);
133 if (it != m_contextProviders.end() && it->second) {
134 std::vector<UIAction> actions = it->second();
135 std::stable_sort(actions.begin(), actions.end(), [](const UIAction& a, const UIAction& b) {
136 return a.order < b.order;
137 });
138 return actions;
139 }
140 return {};
