12 if (first == std::string::npos) return "";
13 size_t last = str.find_last_not_of(" \t\r\n");
14 return str.substr(first, (last - first + 1));

21std::vector<std::string> SplitTableCells(const std::string& line) {
22 std::vector<std::string> cells;
23 size_t start = 0;
24 if (!line.empty() && line.front() == '|') start = 1;
25 size_t end = line.size();
26 if (!line.empty() && line.back() == '|') end = line.size() - 1;
27
28 std::string content = line.substr(start, end - start);
29 std::string cell;
30 bool inEscape = false;
31 for (char c : content) {
32 if (inEscape) {
33 cell.push_back(c);
34 inEscape = false;
35 } else if (c == '\\') {
36 inEscape = true;
37 } else if (c == '|') {
38 cells.push_back(Trim(cell));
39 cell.clear();
40 } else {
41 cell.push_back(c);
42 }
43 }
44 cells.push_back(Trim(cell));
45 return cells;
46}

48bool IsTableSeparatorRow(const std::string& line) {
49 auto cells = SplitTableCells(line);
50 if (cells.empty()) return false;
51 for (const auto& c : cells) {
52 std::string t = Trim(c);
53 if (t.empty()) return false;
54 for (char ch : t) {
55 if (ch != '-' && ch != ':' && ch != ' ' && ch != '\t') return false;
56 }
57 }
58 return true;
59}

64 std::vector<MdSpan> spans;

66
67 size_t i = 0;
68 size_t n = text.size();
69 size_t plainStart = 0;
70
71 auto flushPlain = [&](size_t endIdx) {
72 if (endIdx > plainStart) {
73 MdSpan span;
74 span.type = MdSpanType::Text;
75 span.text = text.substr(plainStart, endIdx - plainStart);
76 span.sourceOffset = baseOffset + plainStart;
77 span.sourceLength = endIdx - plainStart;

79 }
80 };
81
82 while (i < n) {
83 // 1. Escaped characters: \* \_ \` etc.
84 if (text[i] == '\\' && i + 1 < n) {
85 i += 2;
86 continue;
87 }
88
89 // 2. Inline Code: `...`
90 if (text[i] == '`') {
91 size_t close = text.find('`', i + 1);
92 if (close != std::string::npos) {
93 flushPlain(i);
94 MdSpan span;
95 span.type = MdSpanType::Code;

97 span.sourceOffset = baseOffset + i;
98 span.sourceLength = close - i + 1;
99 spans.push_back(std::move(span));
100 i = close + 1;
101 plainStart = i;
102 continue;
103 }
104 }
105
106 // 3. Inline Math: $...$

108 size_t close = text.find('$', i + 1);
109 if (close != std::string::npos) {
110 flushPlain(i);

112 span.type = MdSpanType::MathInline;

114 span.sourceOffset = baseOffset + i;

118 plainStart = i;
119 continue;
120 }
121 }
122

125 size_t close = text.find("==", i + 2);
126 if (close != std::string::npos) {
127 flushPlain(i);
128 MdSpan span;
129 span.type = MdSpanType::Highlight;

143 if (close != std::string::npos) {

159 (text[i] == '_' && text[i + 1] == '_' && text[i + 2] == '_'))) {
160 std::string delim(3, text[i]);
161 size_t close = text.find(delim, i + 3);
162 if (close != std::string::npos) {
163 flushPlain(i);
164 MdSpan span;

166 span.text = text.substr(i + 3, close - i - 3);
167 span.sourceOffset = baseOffset + i;
168 span.sourceLength = close - i + 3;
169 spans.push_back(std::move(span));
170 i = close + 3;
171 plainStart = i;
172 continue;
173 }
174 }
175
176 if (i + 1 < n && ((text[i] == '*' && text[i + 1] == '*') ||
177 (text[i] == '_' && text[i + 1] == '_'))) {
178 std::string delim(2, text[i]);
179 size_t close = text.find(delim, i + 2);
180 if (close != std::string::npos) {
181 flushPlain(i);
182 MdSpan span;
183 span.type = MdSpanType::Bold;
184 span.text = text.substr(i + 2, close - i - 2);
185 span.sourceOffset = baseOffset + i;

187 spans.push_back(std::move(span));

191 }

194 if (text[i] == '*' || (text[i] == '_' && (i == 0 || !std::isalnum(static_cast<unsigned char>(text[i - 1]))))) {
195 char delim = text[i];
196 size_t close = text.find(delim, i + 1);
197 if (close != std::string::npos) {
198 flushPlain(i);
199 MdSpan span;
200 span.type = MdSpanType::Italic;

208 }

210
211 // 7. Links: [text](url)
212 if (text[i] == '[') {
213 size_t closeBracket = text.find(']', i + 1);
214 if (closeBracket != std::string::npos && closeBracket + 1 < n && text[closeBracket + 1] == '(') {
215 size_t closeParen = text.find(')', closeBracket + 2);

222 span.sourceOffset = baseOffset + i;
223 span.sourceLength = closeParen - i + 1;
224 spans.push_back(std::move(span));
225 i = closeParen + 1;
226 plainStart = i;
227 continue;
228 }
229 }
230 }
231
232 ++i;
233 }
234
235 flushPlain(n);
236 return spans;

239std::vector<MdBlock> MdParser::ParseDocument(const std::string& markdown) {
240 std::vector<MdBlock> blocks;
241 if (markdown.empty()) return blocks;
242
243 std::vector<std::string> lines;
244 std::vector<size_t> lineOffsets;
245 {
246 size_t offset = 0;
247 size_t n = markdown.size();
248 while (offset < n) {
249 lineOffsets.push_back(offset);
250 size_t nextNewline = markdown.find('\n', offset);
251 if (nextNewline == std::string::npos) {
252 lines.push_back(markdown.substr(offset));
253 break;
254 } else {
255 size_t len = nextNewline - offset;
256 if (len > 0 && markdown[offset + len - 1] == '\r') {
257 lines.push_back(markdown.substr(offset, len - 1));
258 } else {
259 lines.push_back(markdown.substr(offset, len));
260 }
261 offset = nextNewline + 1;
262 }
263 }
264 }
265
266 size_t lineIdx = 0;
267 size_t totalLines = lines.size();
268
269 while (lineIdx < totalLines) {
270 const std::string& line = lines[lineIdx];
271 std::string trimmed = Trim(line);
272 size_t lineStartOffset = lineOffsets[lineIdx];
273
274 // Blank line
275 if (trimmed.empty()) {
276 ++lineIdx;
277 continue;
278 }
279
280 // 1. Fenced Code Block: ```[lang] ... ```
281 if (StartsWith(trimmed, "```") || StartsWith(trimmed, "~~~")) {
282 std::string fence = trimmed.substr(0, 3);
283 std::string lang = Trim(trimmed.substr(3));
284 MdBlock block;
285 block.type = MdBlockType::CodeBlock;
286 block.language = lang;
287 block.sourceStartOffset = lineStartOffset;
288
289 std::string codeBody;
290 ++lineIdx;
291 while (lineIdx < totalLines) {
292 if (StartsWith(Trim(lines[lineIdx]), fence)) {
293 block.sourceEndOffset = lineOffsets[lineIdx] + lines[lineIdx].size();
294 ++lineIdx;
295 break;
296 }
297 codeBody += lines[lineIdx] + "\n";
298 ++lineIdx;
299 }
300 if (block.sourceEndOffset == 0 && lineIdx == totalLines) {
301 block.sourceEndOffset = markdown.size();
302 }
303 if (!codeBody.empty() && codeBody.back() == '\n') codeBody.pop_back();
304 block.codeContent = codeBody;
305 blocks.push_back(std::move(block));
306 continue;
307 }
308
309 // 2. Math Block: $$ ... $$
310 if (StartsWith(trimmed, "$$")) {
311 MdBlock block;
312 block.type = MdBlockType::MathBlock;
313 block.sourceStartOffset = lineStartOffset;
314
315 // Check single line math block: $$ math $$
316 if (trimmed.size() > 4 && trimmed.rfind("$$") == trimmed.size() - 2) {
317 block.codeContent = Trim(trimmed.substr(2, trimmed.size() - 4));
318 block.sourceEndOffset = lineStartOffset + line.size();
319 blocks.push_back(std::move(block));
320 ++lineIdx;
321 continue;
322 }
323
324 std::string mathBody;
325 ++lineIdx;
326 while (lineIdx < totalLines) {
327 if (StartsWith(Trim(lines[lineIdx]), "$$")) {
328 block.sourceEndOffset = lineOffsets[lineIdx] + lines[lineIdx].size();
329 ++lineIdx;
330 break;
331 }
332 mathBody += lines[lineIdx] + "\n";
333 ++lineIdx;
334 }
335 if (!mathBody.empty() && mathBody.back() == '\n') mathBody.pop_back();
336 block.codeContent = mathBody;
337 blocks.push_back(std::move(block));
338 continue;
339 }
340
341 // 3. Headings: #, ##, ###, ####, #####, ######
342 if (trimmed[0] == '#') {
343 size_t hCount = 0;
344 while (hCount < trimmed.size() && trimmed[hCount] == '#') ++hCount;
345 if (hCount <= 6 && hCount < trimmed.size() && trimmed[hCount] == ' ') {
346 std::string headerText = Trim(trimmed.substr(hCount + 1));
347 MdBlock block;
348 block.type = MdBlockType::Heading;
349 block.level = static_cast<int>(hCount);
350 block.sourceStartOffset = lineStartOffset;
351 block.sourceEndOffset = lineStartOffset + line.size();
352 size_t spanOffset = lineStartOffset + (line.find(headerText));
353 block.spans = ParseInlineSpans(headerText, spanOffset);
354 blocks.push_back(std::move(block));
355 ++lineIdx;
356 continue;
357 }
358 }
359
360 // 4. Horizontal Rule: ---, ***, ___
361 if ((trimmed == "---" || trimmed == "***" || trimmed == "___") ||
362 (trimmed.size() >= 3 && std::all_of(trimmed.begin(), trimmed.end(), [](char c) { return c == '-'; }))) {
363 MdBlock block;
364 block.type = MdBlockType::HorizontalRule;
365 block.sourceStartOffset = lineStartOffset;
366 block.sourceEndOffset = lineStartOffset + line.size();
367 blocks.push_back(std::move(block));
368 ++lineIdx;
369 continue;
370 }
371
372 // 5. Blockquote: > ...
373 if (trimmed[0] == '>') {
374 std::string quoteText;
375 size_t startOff = lineStartOffset;
376 size_t endOff = lineStartOffset + line.size();
377
378 while (lineIdx < totalLines && !Trim(lines[lineIdx]).empty() && Trim(lines[lineIdx])[0] == '>') {
379 std::string curTrim = Trim(lines[lineIdx]);
380 size_t qIdx = 0;
381 while (qIdx < curTrim.size() && curTrim[qIdx] == '>') ++qIdx;
382 quoteText += Trim(curTrim.substr(qIdx)) + " ";
383 endOff = lineOffsets[lineIdx] + lines[lineIdx].size();
384 ++lineIdx;
385 }
386 if (!quoteText.empty() && quoteText.back() == ' ') quoteText.pop_back();
387
388 MdBlock block;
389 block.type = MdBlockType::Blockquote;
390 block.sourceStartOffset = startOff;
391 block.sourceEndOffset = endOff;
392 block.spans = ParseInlineSpans(quoteText, startOff);
393 blocks.push_back(std::move(block));
394 continue;
395 }
396
397 // 6. Task List: - [ ] or - [x]
398 if ((StartsWith(trimmed, "- [ ] ") || StartsWith(trimmed, "- [x] ") || StartsWith(trimmed, "- [X] ") ||
399 StartsWith(trimmed, "* [ ] ") || StartsWith(trimmed, "* [x] ") || StartsWith(trimmed, "* [X] "))) {
400 bool isChecked = (trimmed[3] == 'x' || trimmed[3] == 'X');
401 std::string taskText = trimmed.substr(6);
402
403 MdBlock block;
404 block.type = MdBlockType::TaskList;
405 block.checked = isChecked;
406 block.sourceStartOffset = lineStartOffset;
407 block.sourceEndOffset = lineStartOffset + line.size();
408 size_t spanOffset = lineStartOffset + (line.find(taskText));
409 block.spans = ParseInlineSpans(taskText, spanOffset);
410 blocks.push_back(std::move(block));
411 ++lineIdx;
412 continue;
413 }
414
415 // 7. Bullet List: - , * , +
416 if (StartsWith(trimmed, "- ") || StartsWith(trimmed, "* ") || StartsWith(trimmed, "+ ")) {
417 std::string listText = trimmed.substr(2);
418 MdBlock block;
419 block.type = MdBlockType::BulletList;
420 block.sourceStartOffset = lineStartOffset;
421 block.sourceEndOffset = lineStartOffset + line.size();
422 size_t spanOffset = lineStartOffset + (line.find(listText));
423 block.spans = ParseInlineSpans(listText, spanOffset);
424 blocks.push_back(std::move(block));
425 ++lineIdx;
426 continue;
427 }
428
429 // 8. Numbered List: 1. , 2.
430 if (std::isdigit(static_cast<unsigned char>(trimmed[0]))) {
431 size_t dotPos = trimmed.find(". ");
432 if (dotPos != std::string::npos && dotPos <= 5) {
433 bool allDigits = true;
434 for (size_t d = 0; d < dotPos; ++d) {
435 if (!std::isdigit(static_cast<unsigned char>(trimmed[d]))) {
436 allDigits = false;
437 break;
438 }
439 }
440 if (allDigits) {
441 int itemIdx = 1;
442 try { itemIdx = std::stoi(trimmed.substr(0, dotPos)); } catch (...) {}
443 std::string listText = trimmed.substr(dotPos + 2);
444 MdBlock block;
445 block.type = MdBlockType::NumberedList;
446 block.listIndex = itemIdx;
447 block.sourceStartOffset = lineStartOffset;
448 block.sourceEndOffset = lineStartOffset + line.size();
449 size_t spanOffset = lineStartOffset + (line.find(listText));
450 block.spans = ParseInlineSpans(listText, spanOffset);
451 blocks.push_back(std::move(block));
452 ++lineIdx;
453 continue;
454 }
455 }
456 }
457
458 // 9. Table: line with '|' and next line is separator
459 if (trimmed.find('|') != std::string::npos && lineIdx + 1 < totalLines && IsTableSeparatorRow(lines[lineIdx + 1])) {
460 MdBlock block;
461 block.type = MdBlockType::Table;
462 block.sourceStartOffset = lineStartOffset;
463
464 // Headers
465 block.table.headers = SplitTableCells(trimmed);
466 // Alignments
467 auto sepCells = SplitTableCells(lines[lineIdx + 1]);
468 for (const auto& sc : sepCells) {
469 std::string s = Trim(sc);
470 if (s.size() >= 2 && s.front() == ':' && s.back() == ':') {
471 block.table.alignments.push_back(1); // Center
472 } else if (!s.empty() && s.back() == ':') {
473 block.table.alignments.push_back(2); // Right
474 } else {
475 block.table.alignments.push_back(0); // Left
476 }
477 }
478
479 lineIdx += 2; // Skip header and separator
480 while (lineIdx < totalLines && !Trim(lines[lineIdx]).empty() && lines[lineIdx].find('|') != std::string::npos) {
481 block.table.rows.push_back(SplitTableCells(lines[lineIdx]));
482 block.sourceEndOffset = lineOffsets[lineIdx] + lines[lineIdx].size();
483 ++lineIdx;
484 }
485 blocks.push_back(std::move(block));
486 continue;
487 }
488
489 // 10. Standard Paragraph: accumulate text lines until empty line or special token
490 {
491 std::string paraText;
492 size_t pStart = lineStartOffset;
493 size_t pEnd = lineStartOffset + line.size();
494
495 while (lineIdx < totalLines) {
496 const std::string& cur = lines[lineIdx];
497 std::string curTrim = Trim(cur);
498 if (curTrim.empty() || curTrim[0] == '#' || StartsWith(curTrim, "```") || StartsWith(curTrim, "$$") ||
499 StartsWith(curTrim, "> ") || StartsWith(curTrim, "- ") || StartsWith(curTrim, "* ") ||
500 curTrim == "---") {
501 break;
502 }
503 paraText += (paraText.empty() ? "" : " ") + curTrim;
504 pEnd = lineOffsets[lineIdx] + cur.size();
505 ++lineIdx;
506 }
507
508 if (!paraText.empty()) {
509 MdBlock block;
510 block.type = MdBlockType::Paragraph;
511 block.sourceStartOffset = pStart;
512 block.sourceEndOffset = pEnd;
513 block.spans = ParseInlineSpans(paraText, pStart);
514 blocks.push_back(std::move(block));
515 }
516 }
517 }
518
519 return blocks;
520}
