38/**
39 * @class FileLogger
40 * @brief Thread-safe file logger maintaining an active session file and a ring cache.
41 */
42class FileLogger {
43private:
44 std::ofstream logFile;
45 std::mutex fileMutex;
46 bool isInitialized = false;
47
48 // Fixed-capacity ring cache for live UI debug overlays (O(1) pop_front)
49 std::deque<FolioLogEntry> history;
50 const size_t maxHistorySize = 300;
51
52 // Default constructor does NOT perform filesystem I/O.
53 // This allows safe logging during translation-unit static initialization
54 // before SDL, platform directories, or Android sandboxes are initialized.
55 FileLogger() = default;
56
57 ~FileLogger() {
58 std::lock_guard<std::mutex> lock(fileMutex);
59 if (isInitialized && logFile.is_open()) {
60 logFile << "=================================================\n";
61 logFile << " FolioNote Clean Engine Shutdown \n";
62 logFile << "=================================================\n";
63 logFile.flush();
64 logFile.close();
65 }
66 }
67
68 // Internal helper: builds export string assuming fileMutex is already locked
69 std::string ExportLogsToStringUnlocked() const {
70 std::ostringstream ss;
71 ss << "=================================================\n";
72 ss << " FolioNote Diagnostic Log Export \n";
73 ss << " Total Cached Records: " << history.size() << "\n";
74 ss << "=================================================\n";
75 for (const auto& entry : history) {
76 ss << "[" << entry.timestamp << "] [" << entry.level << "] [" << entry.source << "] " << entry.message << "\n";
77 }
78 return ss.str();
79 }
80
81public:
82 static FileLogger& Instance() {
83 static FileLogger s_fileLogger;
84 return s_fileLogger;
85 }

87 /**
88 * @brief Initializes the physical session log file on disk.
89 * Safe to invoke after the application document/logs directory has been resolved.
90 * Drains any prior log entries cached in memory to the physical file.
91 */
92 void Initialize(const std::string& customLogsDir = "") {
93 std::lock_guard<std::mutex> lock(fileMutex);
94 if (isInitialized && logFile.is_open()) return;
95
96 std::string logsDir = customLogsDir.empty() ? FileManager::GetLogsDirectory() : customLogsDir;
97 if (logsDir.empty()) return;
98
99 if (!FileManager::CreateDirectories(logsDir)) {
100 return;
101 }
102
103 std::string logFilePath = FileManager::JoinPath(logsDir, "folionote_session.log");
104 logFile.open(std::filesystem::path(logFilePath), std::ios::out | std::ios::trunc);
105 if (logFile.is_open()) {
106 isInitialized = true;
107 logFile << "=================================================\n";
108 logFile << " FolioNote Session Started \n";
109 logFile << "=================================================\n";
110
111 // Drain pre-boot log records recorded in memory before file initialization
112 for (const auto& entry : history) {
113 logFile << "[" << entry.timestamp << "] [" << entry.level << "] [" << entry.source << "] " << entry.message << "\n";
114 }
115 logFile.flush();
116 }
117 }
118
119 /**
120 * @brief Writes a structured log entry to the in-memory cache and disk file if initialized.
121 * Flushes immediately only on ERROR to preserve 60/120 FPS frame timing.
122 */
123 void WriteLog(const FolioLogEntry& entry) {
124 std::lock_guard<std::mutex> lock(fileMutex);
125
126 // 1. Always maintain bounded in-memory ring buffer
127 history.push_back(entry);
128 if (history.size() > maxHistorySize) {
129 history.pop_front();
130 }
131
132 // 2. Stream formatted message to physical log if file handle is active
133 if (isInitialized && logFile.is_open()) {
134 logFile << "[" << entry.timestamp << "] [" << entry.level << "] [" << entry.source << "] " << entry.message << "\n";
135 if (entry.level == "ERROR") {
136 logFile.flush();
137 }
138 }
139 }

157 /**
158 * @brief Exports the in-memory log history into a single formatted string.

164
165 /**
166 * @brief Exports the log history to a designated or timestamped disk file.
167 * @param customPath Target file path, or empty string to use default logs folder.
168 * @param outPathUsed Absolute or relative path to the generated export.
169 * @return true on success, false on failure.
170 */
171 bool ExportLogsToFile(const std::string& customPath, std::string& outPathUsed) {
172 std::string exportText;
173 {
174 std::lock_guard<std::mutex> lock(fileMutex);
175 exportText = ExportLogsToStringUnlocked();
176 }
177
178 std::string targetPath = customPath;
179 if (targetPath.empty()) {
180 auto now = std::chrono::system_clock::now();
181 auto timer = std::chrono::system_clock::to_time_t(now);
182 std::tm bt{};
183#if defined(_WIN32)
184 localtime_s(&bt, &timer);
185#else
186 localtime_r(&timer, &bt);
187#endif
188 std::ostringstream nameStream;
189 nameStream << "folionote_export_"
190 << std::put_time(&bt, "%Y%m%d_%H%M%S")
191 << ".log";
192 targetPath = FileManager::JoinPath(FileManager::GetLogsDirectory(), nameStream.str());
193 }
194
195 std::string parentDir = FileManager::GetParentPath(targetPath);
196 if (!parentDir.empty()) {
197 FileManager::CreateDirectories(parentDir);
198 }
199
200 bool ok = FileManager::WriteTextAtomic(targetPath, exportText);
201 if (ok) {
202 outPathUsed = targetPath;
203 }
204 return ok;
205 }
