76/**
77 * @brief Executes a system command silently in a hidden background process and captures its stdout.
78 *
79 * GENERAL WORKING PROCESS & SECURITY:
80 * -----------------------------------
81 * On Windows, this creates an anonymous pipe with security attributes set to allow handle
82 * inheritance, configures STARTUPINFOA with STARTF_USESTDHANDLES and STARTF_USESHOWWINDOW (SW_HIDE),
83 * and creates the child process with CREATE_NO_WINDOW so no black console window flashes
84 * in front of the user. Output is streamed into a buffer and returned as a std::string.
85 *
86 * @param[in] cmd Shell command line string to execute.
87 * @return Standard output text emitted by the child process.
88 */
89static std::string RunCommandAndCaptureStdout(const std::string& cmd) {
90#if defined(_WIN32)
91 HANDLE hReadPipe = NULL;
92 HANDLE hWritePipe = NULL;
93
94 SECURITY_ATTRIBUTES sa;
95 sa.nLength = sizeof(SECURITY_ATTRIBUTES);
96 sa.bInheritHandle = TRUE;
97 sa.lpSecurityDescriptor = NULL;
98
99 if (!CreatePipe(&hReadPipe, &hWritePipe, &sa, 0)) {
100 return "";
101 }
102 SetHandleInformation(hReadPipe, HANDLE_FLAG_INHERIT, 0);
103
104 STARTUPINFOA si;
105 ZeroMemory(&si, sizeof(si));
106 si.cb = sizeof(si);
107 si.hStdOutput = hWritePipe;
108 si.hStdError = hWritePipe;
109 si.dwFlags |= STARTF_USESTDHANDLES | STARTF_USESHOWWINDOW;
110 si.wShowWindow = SW_HIDE;
111
112 PROCESS_INFORMATION pi;
113 ZeroMemory(&pi, sizeof(pi));
114
115 std::string fullCmd = "cmd.exe /c " + cmd;
116 std::vector<char> cmdBuf(fullCmd.begin(), fullCmd.end());
117 cmdBuf.push_back('\0');
118
119 if (!CreateProcessA(NULL, cmdBuf.data(), NULL, NULL, TRUE, CREATE_NO_WINDOW, NULL, NULL, &si, &pi)) {
120 CloseHandle(hReadPipe);
121 CloseHandle(hWritePipe);
122 return "";
123 }
124
125 // Close write handle in parent so ReadFile terminates when child exits
126 CloseHandle(hWritePipe);
127
128 std::string result;
129 char buffer[1024];
130 DWORD bytesRead = 0;
131
132 while (ReadFile(hReadPipe, buffer, sizeof(buffer) - 1, &bytesRead, NULL) && bytesRead > 0) {
133 buffer[bytesRead] = '\0';
134 result.append(buffer, bytesRead);
135 }
136
137 // Wait up to 15 seconds for process termination
138 WaitForSingleObject(pi.hProcess, 15000);
139
140 CloseHandle(hReadPipe);
141 CloseHandle(pi.hProcess);
142 CloseHandle(pi.hThread);
143
144 return result;
145#else
146 std::string result;
147 FILE* pipe = popen(cmd.c_str(), "r");
148 if (!pipe) return "";
149 char buffer[512];
150 while (fgets(buffer, sizeof(buffer), pipe) != nullptr) {
151 result += buffer;
152 }
153 pclose(pipe);
154 return result;
155#endif
156}

197std::string VideoObject::GetYouTubeId() const {
198 // 1. Short URL: youtu.be/<ID>
199 auto pos = sourceUrl.find("youtu.be/");
200 if (pos != std::string::npos) {
201 auto id = sourceUrl.substr(pos + 9);
202 auto end = id.find_first_of("?&#");
203 return (end != std::string::npos) ? id.substr(0, end) : id;
204 }
205
206 // 2. Embed URL: youtube.com/embed/<ID>
207 pos = sourceUrl.find("/embed/");
208 if (pos != std::string::npos) {
209 auto id = sourceUrl.substr(pos + 7);
210 auto end = id.find_first_of("?&#/");
211 return (end != std::string::npos) ? id.substr(0, end) : id;
212 }
213
214 // 3. Watch URL: youtube.com/watch?v=<ID>
215 pos = sourceUrl.find("v=");
216 if (pos != std::string::npos) {
217 auto id = sourceUrl.substr(pos + 2);
218 auto end = id.find_first_of("&");
219 return (end != std::string::npos) ? id.substr(0, end) : id;
220 }
221
222 return "";
223}

235/**
236 * @brief Asynchronously fetches and decodes the high-resolution YouTube poster thumbnail.
237 *
238 * GENERAL WORKING PROCESS:
239 * -------------------------
240 * 1. Checks if thumbnail is already being fetched or already loaded; returns immediately if so.
241 * 2. Queries YouTube CDN for the standard hqdefault (or maxresdefault) image.
242 * 3. Caches the downloaded JPEG into the OS temporary directory (%TEMP%/FolioNote/yt_thumbs/<id>.jpg).
243 * 4. Decodes the cached file using ImageDecoder::DecodeFromFile into a Blend2D BLImage.
244 * 5. Safely updates thumbnailImg and triggers onDirty() to repaint the canvas.
245 *
246 * @param[in] onDirty Callback to signal canvas repaint.
247 */
248void VideoObject::FetchYouTubeThumbnailAsync(std::function<void()> onDirty) {
249 if (!IsYouTube()) return;
250 std::string videoId = GetYouTubeId();
251 if (videoId.empty()) return;
252
253 if (isFetchingThumbnail.exchange(true)) {
254 return; // Already in flight
255 }
256
257 LOG_INFO(CanvasObject, "Fetching YouTube thumbnail poster for video ID: " + videoId);
258
259 std::thread([this, videoId, onDirty]() {
260 std::string thumbUrl = "https://img.youtube.com/vi/" + videoId + "/hqdefault.jpg";
261
262 std::filesystem::path cacheDir = std::filesystem::temp_directory_path() / "FolioNote" / "yt_thumbs";
263 std::error_code ec;
264 std::filesystem::create_directories(cacheDir, ec);
265 std::filesystem::path cacheFile = cacheDir / (videoId + ".jpg");
266
267 bool downloaded = false;
268 if (std::filesystem::exists(cacheFile, ec)) {
269 downloaded = true;
270 } else {
271#if defined(_WIN32)
272 HRESULT hr = URLDownloadToFileA(nullptr, thumbUrl.c_str(), cacheFile.string().c_str(), 0, nullptr);
273 downloaded = (hr == S_OK && std::filesystem::exists(cacheFile, ec));
274 if (!downloaded) {
275 // Secondary fallback via Windows curl
276 std::string curlCmd = "curl.exe -s -L -o \"" + cacheFile.string() + "\" \"" + thumbUrl + "\"";
277 int ret = std::system(curlCmd.c_str());
278 downloaded = (ret == 0 && std::filesystem::exists(cacheFile, ec));
279 }
280#else
281 std::string curlCmd = "curl -s -L -o \"" + cacheFile.string() + "\" \"" + thumbUrl + "\"";
282 int ret = std::system(curlCmd.c_str());
283 downloaded = (ret == 0 && std::filesystem::exists(cacheFile, ec));
284#endif
285 }
286
287 if (downloaded) {
288 auto decoded = ImageDecoder::DecodeFromFile(cacheFile.string());
289 if (!decoded.image.is_empty()) {
290 thumbnailImg = std::move(decoded.image);
291 thumbnailLoaded = true;
292 LOG_INFO(CanvasObject, "Successfully loaded YouTube thumbnail poster for: " + videoId +
293 " (" + std::to_string(thumbnailImg.width()) + "x" + std::to_string(thumbnailImg.height()) + ")");
294 if (onDirty) onDirty();
295 }
296 } else {
297 LOG_WARN(CanvasObject, "Failed to download YouTube thumbnail for ID: " + videoId);
298 }
299
300 isFetchingThumbnail.store(false);
301 }).detach();
302}

304/**
305 * @brief Resolves YouTube stream URL into a direct playable stream via yt-dlp.
306 *
307 * GENERAL WORKING PROCESS & FALLBACK:
308 * ------------------------------------
309 * 1. Spawns an asynchronous worker thread so the main render/UI loop never blocks.
310 * 2. Sets isResolvingStream = true so Render() displays a visual loading badge.
311 * 3. Probes python -m yt_dlp or standalone yt-dlp to extract the direct .googlevideo.com media stream URL.
312 * 4. If direct stream is extracted:
313 * - Caches URL in resolvedStreamUrl.
314 * - Opens media in libVLC player instance.
315 * - Sets isPlaying = true and marks canvas dirty.
316 * 5. If yt-dlp is not available or resolution fails:
317 * - Seamlessly falls back to opening the video in the default system web browser (FileManager::OpenWithDefaultApp).
318 *
319 * @param[in] onDirty Callback to signal canvas repaint.
320 */
321void VideoObject::ResolveYouTubeStreamAsync(std::function<void()> onDirty) {
322 if (!IsYouTube()) return;
323 if (isResolvingStream.exchange(true)) {
324 LOG_INFO(CanvasObject, "YouTube stream resolution already in progress for: " + sourceUrl);
325 return;
326 }
327
328 LOG_INFO(CanvasObject, "Initiating YouTube stream resolution via yt-dlp for: " + sourceUrl);
329 if (onDirty) onDirty();
330
331 std::thread([this, onDirty]() {
332 std::string streamUrl;
333
334 // Sequence of commands to resolve direct media stream (multiplexed H.264 + AAC audio preferred)
335 const std::vector<std::string> commands = {
336 "python -m yt_dlp --no-warnings --remote-components ejs:github --js-runtimes node -g -f \"best[vcodec^=avc1][acodec!=none]/best[ext=mp4]/18/22/b\" \"" + sourceUrl + "\"",
337 "yt-dlp --no-warnings --remote-components ejs:github --js-runtimes node -g -f \"best[vcodec^=avc1][acodec!=none]/best[ext=mp4]/18/22/b\" \"" + sourceUrl + "\"",
338 "python -m yt_dlp --no-warnings -g -f \"best[vcodec^=avc1][acodec!=none]/best[ext=mp4]/18/22/b\" \"" + sourceUrl + "\"",
339 "yt-dlp --no-warnings -g -f \"best[vcodec^=avc1][acodec!=none]/best[ext=mp4]/18/22/b\" \"" + sourceUrl + "\"",
340 "python -m yt_dlp --no-warnings -g -f \"best[ext=mp4]/best\" \"" + sourceUrl + "\"",
341 "yt-dlp --no-warnings -g -f \"best[ext=mp4]/best\" \"" + sourceUrl + "\""
342 };
343
344 for (const auto& cmd : commands) {
345 std::string out = RunCommandAndCaptureStdout(cmd);
346 size_t httpPos = out.find("http://");
347 if (httpPos == std::string::npos) {
348 httpPos = out.find("https://");
349 }
350 if (httpPos != std::string::npos) {
351 size_t endPos = out.find_first_of("\r\n", httpPos);
352 if (endPos != std::string::npos) {
353 streamUrl = out.substr(httpPos, endPos - httpPos);
354 } else {
355 streamUrl = out.substr(httpPos);
356 }
357 LOG_INFO(CanvasObject, "Resolved direct YouTube stream URL via: " + cmd.substr(0, 45) + "...");
358 break;
359 }
360 }
361
362 if (!streamUrl.empty()) {
363 resolvedStreamUrl = streamUrl;
364 if (!player) {
365 player = std::make_unique<VideoPlayerInstance>();
366 if (onDirty) player->SetFrameReadyCallback(onDirty);
367 }
368 LOG_INFO(CanvasObject, "Opening resolved YouTube stream URL in libVLC");
369 if (player->Open(resolvedStreamUrl, false)) {
370 isPlaying = true;
371 } else {
372 LOG_ERROR(CanvasObject, "Failed to open resolved YouTube stream: " + player->errorMessage);
373 FileManager::OpenWithDefaultApp(sourceUrl);
374 isPlaying = false;
375 }
376 } else {
377 LOG_WARN(CanvasObject, "yt-dlp unavailable or stream resolution failed for: " + sourceUrl +
378 " — Falling back to default system web browser");
379 FileManager::OpenWithDefaultApp(sourceUrl);
380 isPlaying = false;
381 }
382
383 isResolvingStream.store(false);
384 if (onDirty) onDirty();
385 }).detach();
386}

392void VideoObject::Play(std::function<void()> onDirty) {
393 // -------------------------------------------------------------------------
394 // Lazy-initialize the VideoPlayerInstance on first Play() call.
395 // This avoids allocating a VLC instance for every canvas VideoObject at load time.
396 // -------------------------------------------------------------------------
397 if (!player) {
398 player = std::make_unique<VideoPlayerInstance>();
399 if (onDirty) {
400 player->SetFrameReadyCallback(onDirty);
401 }
402 } else if (onDirty) {
403 player->SetFrameReadyCallback(onDirty);
404 }
405
406 // If the player is already opened and currently paused, resume playback directly!
407 // This avoids tearing down VLC and re-opening the media file from scratch.
408 if (player->GetState() == VideoPlayerState::Paused) {
409 LOG_INFO(CanvasObject, "VideoObject::Play: Resuming paused playback for uid=" + std::to_string(uid) + " (" + displayName + ")");
410 player->Play();
411 isPlaying = true;
412 return;
413 }
414
415 if (player->GetState() == VideoPlayerState::Playing) {
416 LOG_INFO(CanvasObject, "VideoObject::Play: Already playing for uid=" + std::to_string(uid));
417 isPlaying = true;
418 return;
419 }
420
421 // -------------------------------------------------------------------------
422 // YouTube stream resolution path
423 // -------------------------------------------------------------------------
424 if (IsYouTube()) {
425 if (!resolvedStreamUrl.empty()) {
426 LOG_INFO(CanvasObject, "VideoObject::Play: Playing previously resolved YouTube stream for uid=" + std::to_string(uid));
427 if (!player->Open(resolvedStreamUrl, false)) {
428 LOG_ERROR(CanvasObject, "Failed to open cached stream: " + player->errorMessage + " — re-resolving");
429 resolvedStreamUrl.clear();
430 ResolveYouTubeStreamAsync(onDirty);
431 } else {
432 isPlaying = true;
433 }
434 return;
435 }
436
437 ResolveYouTubeStreamAsync(onDirty);
438 return;
439 }
440
441 // Standard media file / direct network stream
442 LOG_INFO(CanvasObject, "VideoObject::Play: Opening media stream for uid=" + std::to_string(uid) + " (" + sourceUrl + ")");
443 if (!player->Open(sourceUrl, false)) {
444 LOG_ERROR(CanvasObject, "Failed to open video: " + sourceUrl + " — " + player->errorMessage);
445 }
446 isPlaying = true;
447}

465bool VideoObject::HandleCanvasClick(double wx, double wy, bool isDoubleClick,
466 std::function<void()> onDirty)
467{
468 if (!isVisible) return false;
469
470 // Check if the click lies inside the video container bounds
471 if (wx < worldX || wx > worldX + worldWidth ||
472 wy < worldY || wy > worldY + worldHeight) {
473 return false;
474 }
475
476 LOG_INFO(CanvasObject, "VideoObject::HandleCanvasClick: uid=" + std::to_string(uid) +
477 " name='" + displayName + "' isPlaying=" + (isPlaying ? "true" : "false") +
478 " isDoubleClick=" + (isDoubleClick ? "true" : "false"));
479
480 // Case 0: YouTube badge click in top-right corner opens external video in browser
481 if (IsYouTube()) {
482 const double bw = std::min(worldWidth * 0.18, 16.0);
483 const double bh = std::min(worldHeight * 0.08, 6.0);
484 const double bx = worldX + worldWidth - bw - 1.5;
485 const double by = worldY + 1.5;
486 if (wx >= bx && wx <= bx + bw && wy >= by && wy <= by + bh) {
487 LOG_INFO(CanvasObject, "Clicked YouTube badge — opening in external browser: " + sourceUrl);
488 FileManager::OpenWithDefaultApp(sourceUrl);
489 return true;
490 }
491 }
492
493 // Case 1: If the video is NOT currently playing, clicking anywhere on the container starts it!
494 if (!isPlaying) {
495 Play(onDirty);
496 if (onDirty) onDirty();
497 return true;
498 }
499
500 // Case 2: Double-click anywhere on an active video pauses it
501 if (isDoubleClick) {
502 Pause();
503 if (onDirty) onDirty();
504 return true;
505 }
506
507 // Case 3: Transport bar interaction (bottom strip)
508 const double barH = std::max(3.0, std::min(worldHeight * 0.18, 8.0));
509 const double barY = worldY + worldHeight - barH;
510
511 if (wy >= barY && wy <= worldY + worldHeight) {
512 const double pad = std::max(0.8, worldWidth * 0.015);
513 const double btnW = barH * 0.85;
514
515 // Transport Play/Pause button (left side)
516 if (wx >= worldX && wx <= worldX + pad + btnW + pad) {
517 TogglePlay(onDirty);
518 if (onDirty) onDirty();
519 return true;
520 }
521
522 // Transport Mute button (right side)
523 const double volX = worldX + worldWidth - pad - barH * 0.8;
524 if (wx >= volX - pad) {
525 ToggleMute();
526 if (onDirty) onDirty();
527 return true;
528 }
529
530 // Transport Scrubber seek bar
531 const double scrubX = worldX + pad + btnW + pad;
532 const double scrubW = worldWidth - (pad + btnW + pad) - (pad + barH * 2.0);
533 if (scrubW > 5.0 && wx >= scrubX && wx <= scrubX + scrubW) {
534 float dur = player ? player->durationSec.load(std::memory_order_relaxed) : 0.0f;
535 if (dur > 0.0f) {
536 double progress = std::clamp((wx - scrubX) / scrubW, 0.0, 1.0);
537 float targetSec = static_cast<float>(progress * dur);
538 LOG_INFO(CanvasObject, "Scrubber seek to: " + std::to_string(targetSec) + "s / " + std::to_string(dur) + "s");
539 SeekTo(targetSec);
540 if (onDirty) onDirty();
541 return true;
542 }
543 }
544 }
545
546 // Case 4: Center play/pause area
547 const double cx = worldX + worldWidth * 0.5;
548 const double cy = worldY + worldHeight * 0.5;
549 const double tSize = std::min(worldWidth, worldHeight) * 0.18;
550 double distSq = (wx - cx) * (wx - cx) + (wy - cy) * (wy - cy);
551 if (distSq <= (tSize * 1.6) * (tSize * 1.6)) {
552 TogglePlay(onDirty);
553 if (onDirty) onDirty();
554 return true;
555 }
556
557 return false;
558}

599void VideoObject::BakeTransform() {

618void VideoObject::Render(BLContext& ctx, const Viewport& viewport) const {
619 if (!isVisible) return;
620
621 if (worldWidth <= 0.0 || worldHeight <= 0.0) return;
622
623 ctx.save();
624 // Apply object's affine transform (for rotation / free-form resize history)
625 ctx.apply_transform(transform);
626
627 // Canvas coordinate space:
628 // ctx has already been transformed into screen space by staticCtx.set_transform(renderMatrix).
629 // Therefore, all drawing operations inside Render() take place directly in canvas world coordinates (mm),
630 // matching the behavior of ImageObject, TextBoxObject, and InkContainer.
631 const double x = worldX;
632 const double y = worldY;
633 const double w = worldWidth;
634 const double h = worldHeight;
635
636 // Hairline stroke scaling: keeps borders at consistent physical/screen pixel weight under zooming
637 // Math: strokeScale = 1.0 / zoom
638 const double strokeScale = (viewport.zoom > 0.001) ? (1.0 / viewport.zoom) : 1.0;
639
640 // ─────────────────────────────────────────────────────────────────────────
641 // 2. BLIT DECODED VIDEO FRAME (if player has a new frame ready or valid frame cached)
642 // ─────────────────────────────────────────────────────────────────────────
643 bool frameBlitted = false;
644
645 if (player && (player->HasNewFrame() || player->HasValidFrame())) {
646 int fw = 0, fh = 0;
647 const uint8_t* px = player->LockFrame(fw, fh);
648
649 if (px && fw > 0 && fh > 0) {
650 // Update native dimensions if this is the first frame
651 if (nativeVideoW != fw || nativeVideoH != fh) {
652 const_cast<VideoObject*>(this)->nativeVideoW = fw;
653 const_cast<VideoObject*>(this)->nativeVideoH = fh;
654 const_cast<VideoObject*>(this)->aspectRatio = static_cast<double>(fw) / static_cast<double>(fh);
655 }
656
657 // Zero-copy: wrap the BGRA32 pixel buffer in a BLImage (external data reference)
658 // BL_FORMAT_PRGB32 = pre-multiplied BGRA32, matching libVLC's "BGRA" sink output.
659 BLImage frame;
660 BLResult res = frame.create_from_data(fw, fh, BL_FORMAT_PRGB32,
661 const_cast<uint8_t*>(px),
662 static_cast<intptr_t>(fw) * 4);
663
664 if (res == BL_SUCCESS) {
665 BLRect destRect(x, y, w, h);
666 ctx.blit_image(destRect, frame);
667 frameBlitted = true;
668 }
669
670 player->UnlockFrame();
671 } else if (px == nullptr) {
672 player->UnlockFrame();
673 }
674 }
675
676 // ─────────────────────────────────────────────────────────────────────────
677 // 3. RENDER PLACEHOLDER / POSTER THUMBNAIL if no frame is decoded
678 // ─────────────────────────────────────────────────────────────────────────
679 if (!frameBlitted) {
680 if (thumbnailLoaded && !thumbnailImg.is_empty()) {
681 BLRect destRect(x, y, w, h);
682 ctx.blit_image(destRect, thumbnailImg);
683
684 // Overlay dark circular glow and center play triangle
685 const double cx = x + w * 0.5;
686 const double cy = y + h * 0.5;
687 const double tSize = std::min(w, h) * 0.16;
688
689 ctx.set_fill_style(BLRgba32(0x00, 0x00, 0x00, 120));
690 ctx.fill_circle(cx, cy, tSize * 1.5);
691
692 if (!isPlaying && !isResolvingStream.load()) {
693 BLPath tri;
694 tri.move_to(cx - tSize * 0.35, cy - tSize * 0.8);
695 tri.line_to(cx + tSize * 0.9, cy);
696 tri.line_to(cx - tSize * 0.35, cy + tSize * 0.8);
697 tri.close();
698 ctx.set_fill_style(BLRgba32(0xFF, 0xFF, 0xFF, 240));
699 ctx.fill_path(tri);
700 }
701 } else {
702 RenderPlaceholderCard(ctx, x, y, w, h);
703 }
704
705 // Background stream resolution progress indicator badge
706 if (isResolvingStream.load()) {
707 const double rw = std::min(w * 0.85, 45.0);
708 const double rh = std::min(h * 0.22, 9.0);
709 const double rx = x + (w - rw) * 0.5;
710 const double ry = y + (h - rh) * 0.5;
711 ctx.set_fill_style(BLRgba32(0x18, 0x1A, 0x22, 235));
712 ctx.fill_round_rect(BLRoundRect(rx, ry, rw, rh, 3.0, 3.0));
713 ctx.set_stroke_style(BLRgba32(0xFF, 0x00, 0x00, 220));
714 ctx.set_stroke_width(0.6 * strokeScale);
715 ctx.stroke_round_rect(BLRoundRect(rx, ry, rw, rh, 3.0, 3.0));
716 }
717 }
718
719 // ─────────────────────────────────────────────────────────────────────────
720 // 4. TRANSPORT CONTROLS OVERLAY
721 // ─────────────────────────────────────────────────────────────────────────
722 uint64_t nowMs = SDL_GetTicks();
723 bool showControls = (nowMs - lastHoverMs) < CONTROLS_FADE_DELAY_MS;
724
725 if (showControls && w >= 15.0) {
726 float timeSec = player ? player->timeSec.load(std::memory_order_relaxed) : 0.0f;
727 float durSec = player ? player->durationSec.load(std::memory_order_relaxed) : 0.0f;
728 RenderTransportBar(ctx, x, y, w, h, timeSec, durSec);
729 }
730
731 // ─────────────────────────────────────────────────────────────────────────
732 // 5. BORDER OUTLINE
733 // ─────────────────────────────────────────────────────────────────────────
734 ctx.set_stroke_style(BLRgba32(0x3E, 0x44, 0x55, 180));
735 ctx.set_stroke_width(0.8 * strokeScale);
736 ctx.stroke_rect(BLRect(x, y, w, h));
737
738 // ─────────────────────────────────────────────────────────────────────────
739 // 6. YOUTUBE BADGE (red pill with white play triangle in top-right corner)
740 // ─────────────────────────────────────────────────────────────────────────
741 if (IsYouTube() && w > 10.0) {
742 const double bw = std::min(w * 0.18, 14.0);
743 const double bh = std::min(h * 0.08, 5.0);
744 const double bx = x + w - bw - 1.2;
745 const double by = y + 1.2;
746
747 ctx.set_fill_style(BLRgba32(0xFF, 0x00, 0x00, 235));
748 ctx.fill_round_rect(BLRoundRect(bx, by, bw, bh, bh * 0.35, bh * 0.35));
749
750 // Draw YouTube white play triangle inside the badge
751 const double triH = bh * 0.45;
752 const double triW = triH * 0.85;
753 const double tcx = bx + (bw - triW) * 0.5;
754 const double tcy = by + (bh - triH) * 0.5;
755 BLPath miniTri;
756 miniTri.move_to(tcx, tcy);
757 miniTri.line_to(tcx + triW, tcy + triH * 0.5);
758 miniTri.line_to(tcx, tcy + triH);
759 miniTri.close();
760 ctx.set_fill_style(BLRgba32(0xFF, 0xFF, 0xFF, 255));
761 ctx.fill_path(miniTri);
762 }
763
764 ctx.restore();
765}

816// =============================================================================
817// PRIVATE: Transport Control Bar (Play/Pause, Scrubber, Timecode, Volume)
818// =============================================================================
819
820void VideoObject::RenderTransportBar(BLContext& ctx, double sx, double sy,
821 double sw, double sh,
822 float timeSec, float durSec) const
823{
824 // Transport bar sits at the bottom, 18% of video height (max 8mm)
825 const double barH = std::max(3.0, std::min(sh * 0.18, 8.0));
826 const double barY = sy + sh - barH;
827 const double pad = std::max(0.8, sw * 0.015);
828
829 // Semi-transparent background scrim
830 ctx.set_fill_style(BLRgba32(0x00, 0x00, 0x00, 185));
831 ctx.fill_rect(BLRect(sx, barY, sw, barH));
832
833 // ── Play/Pause button ────────────────────────────────────────────────────
834 const double btnW = barH * 0.85;
835 const double btnCx = sx + pad + btnW * 0.5;
836 const double btnCy = barY + barH * 0.5;
837
838 if (!isPlaying) {
839 // Play triangle
840 double ts = barH * 0.22;
841 BLPath tri;
842 tri.move_to(btnCx - ts * 0.4, btnCy - ts);
843 tri.line_to(btnCx + ts, btnCy);
844 tri.line_to(btnCx - ts * 0.4, btnCy + ts);
845 tri.close();
846 ctx.set_fill_style(BLRgba32(0xFF, 0xFF, 0xFF, 240));
847 ctx.fill_path(tri);
848 } else {
849 // Pause bars
850 double bw2 = barH * 0.07;
851 double bh2 = barH * 0.36;
852 ctx.set_fill_style(BLRgba32(0xFF, 0xFF, 0xFF, 240));
853 ctx.fill_rect(btnCx - bw2 * 1.5, btnCy - bh2 * 0.5, bw2, bh2);
854 ctx.fill_rect(btnCx + bw2 * 0.5, btnCy - bh2 * 0.5, bw2, bh2);
855 }
856
857 // ── Scrubber / Progress Bar ───────────────────────────────────────────────
858 const double scrubX = sx + pad + btnW + pad;
859 const double scrubW = sw - (pad + btnW + pad) - (pad + barH * 2.0);
860 const double scrubY = barY + barH * 0.5 - barH * 0.06;
861 const double scrubH = std::max(0.6, barH * 0.12);
862
863 if (scrubW > 5.0) {
864 // Track background
865 ctx.set_fill_style(BLRgba32(0xFF, 0xFF, 0xFF, 60));
866 ctx.fill_round_rect(BLRoundRect(scrubX, scrubY, scrubW, scrubH, scrubH * 0.5, scrubH * 0.5));
867
868 // Filled portion (progress)
869 if (durSec > 0.0f) {
870 // Math: fillWidth = scrubW * (timeSec / durSec) ∈ [0, scrubW]
871 double progress = std::max(0.0, std::min(1.0,
872 static_cast<double>(timeSec) / static_cast<double>(durSec)));
873 double fillW = scrubW * progress;
874 if (fillW > 0.1) {
875 ctx.set_fill_style(BLRgba32(0xFF, 0x40, 0x40, 255));
876 ctx.fill_round_rect(BLRoundRect(scrubX, scrubY, fillW, scrubH, scrubH * 0.5, scrubH * 0.5));
877 }
878
879 // Playhead circle
880 ctx.set_fill_style(BLRgba32(0xFF, 0xFF, 0xFF, 255));
881 ctx.fill_circle(scrubX + fillW, scrubY + scrubH * 0.5, scrubH * 1.2);
882 }
883 }
884
885 // ── Volume Icon ──────────────────────────────────────────────────────────
886 if (sw > 15.0) {
887 double volX = sx + sw - pad - barH * 0.8;
888 double volCy = barY + barH * 0.5;
889 double vs = barH * 0.18;
890
891 if (isMuted) {
892 ctx.set_stroke_style(BLRgba32(0xFF, 0xFF, 0xFF, 200));
893 ctx.set_stroke_width(0.3);
894 ctx.stroke_line(volX, volCy - vs, volX + vs * 1.5, volCy + vs);
895 ctx.stroke_line(volX + vs * 1.5, volCy - vs, volX, volCy + vs);
896 } else {
897 BLPath spk;
898 spk.move_to(volX, volCy - vs * 0.5);
899 spk.line_to(volX + vs, volCy - vs * 0.5);
900 spk.line_to(volX + vs * 1.8, volCy - vs);
901 spk.line_to(volX + vs * 1.8, volCy + vs);
902 spk.line_to(volX + vs, volCy + vs * 0.5);
903 spk.line_to(volX, volCy + vs * 0.5);
904 spk.close();
905 ctx.set_fill_style(BLRgba32(0xFF, 0xFF, 0xFF, 200));
906 ctx.fill_path(spk);
907 }
908 }
909}

912// PRIVATE: Timecode Formatter
913// =============================================================================
914
915std::string VideoObject::FormatTimecode(float seconds) {
916 if (seconds < 0.0f) seconds = 0.0f;
917 int totalSec = static_cast<int>(seconds);
918 int h = totalSec / 3600;
919 int m = (totalSec % 3600) / 60;
920 int s = totalSec % 60;
921
922 char buf[16];
923 if (h > 0) {
924 std::snprintf(buf, sizeof(buf), "%d:%02d:%02d", h, m, s);
925 } else {
926 std::snprintf(buf, sizeof(buf), "%02d:%02d", m, s);
927 }
928 return buf;
929}
