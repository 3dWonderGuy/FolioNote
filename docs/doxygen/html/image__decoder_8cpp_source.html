26/**
27 * UNIFIED CROSS-PLATFORM DECODING ARCHITECTURE:
28 * ---------------------------------------------
29 * FolioNote avoids OS-dependent decoders (such as Windows WIC, Apple ImageIO, or Android Bitmap)
30 * to guarantee identical rendering fidelity, deterministic color blitting, and zero platform
31 * branching across Windows, Linux, macOS, and Android.
32 *
33 * The decoding pipeline executes across 4 cross-platform layers:
34 * 1. LunaSVG Pipeline:
35 * Parses vector SVG documents and rasterizes them into Blend2D PRGB32 pixel buffers.
36 * 2. SDL3_image Multi-Frame Animation Pipeline:
37 * Parses multi-frame GIF and animated WebP containers via IMG_LoadAnimation_IO.
38 * Automatically resolves inter-frame disposal methods into standalone frames.
39 * 3. Blend2D Native SIMD Pipeline:
40 * JIT-accelerated (AVX2/NEON) decoding for high-frequency formats: PNG, JPEG, BMP, and QOI.
41 * 4. SDL3_image Extended Raster Pipeline:
42 * Fallback decoder for static WebP, TIFF, TGA, ICO, AVIF, and PCX.
43 *
44 * All pipelines output directly to Blend2D BL_FORMAT_PRGB32 (Premultiplied 32-bit ARGB/BGRA),
45 * matching the compositor memory layout for zero-copy blitting.
46 */

54BLImage ImageDecoder::ConvertRgbaToBLImage(const uint8_t* rgbaData, uint32_t width, uint32_t height) {
55 if (!rgbaData || width == 0 || height == 0) {
56 return BLImage();
57 }
58
59 BLImage img;
60 if (img.create(static_cast<int>(width), static_cast<int>(height), BL_FORMAT_PRGB32) != BL_SUCCESS) {
61 return BLImage();
62 }
63
64 BLImageData data;
65 if (img.make_mutable(&data) != BL_SUCCESS) {
66 return BLImage();
67 }
68
69 uint8_t* dst = static_cast<uint8_t*>(data.pixel_data);
70 const size_t dstStride = data.stride;
71
72 // Convert straight RGBA to Premultiplied ARGB32 (BL_FORMAT_PRGB32)
73 for (uint32_t y = 0; y < height; ++y) {
74 uint32_t* dstRow = reinterpret_cast<uint32_t*>(dst + y * dstStride);
75 const uint8_t* srcRow = rgbaData + y * width * 4;
76
77 for (uint32_t x = 0; x < width; ++x) {
78 const uint8_t r = srcRow[x * 4 + 0];
79 const uint8_t g = srcRow[x * 4 + 1];
80 const uint8_t b = srcRow[x * 4 + 2];
81 const uint8_t a = srcRow[x * 4 + 3];
82
83 // Alpha premultiplication formula: c_premul = (c * a + 127) / 255
84 const uint32_t pr = (static_cast<uint32_t>(r) * a + 127) / 255;
85 const uint32_t pg = (static_cast<uint32_t>(g) * a + 127) / 255;
86 const uint32_t pb = (static_cast<uint32_t>(b) * a + 127) / 255;
87
88 // In Blend2D on little-endian x86/ARM, PRGB32 is 0xAARRGGBB in uint32_t
89 dstRow[x] = (static_cast<uint32_t>(a) << 24) |
90 (pr << 16) |
91 (pg << 8) |
92 pb;
93 }
94 }
95
96 return img;
97}

103DecodedImageResult ImageDecoder::DecodeFromMemory(
104 const uint8_t* data,
105 size_t size,
106 std::string_view filenameHint)
107{
108 DecodedImageResult result;
109
110 if (!data || size == 0) {
111 result.errorMessage = "Input buffer is empty";
112 return result;
113 }
114
115 const auto& config = ObjectConfig::Get();
116 if (size > config.maxImageFileSizeBytes) {
117 result.errorMessage = "Image buffer size (" + std::to_string(size) +
118 " bytes) exceeds maximum configured limit (" +
119 std::to_string(config.maxImageFileSizeBytes) + " bytes)";
120 return result;
121 }
122
123 const uint32_t maxDecodedDim = config.maxDecodedPixelDimension;
124
125 ImageFormat detectedFormat = ImageFormatFromExtension(filenameHint);
126
127 // Sniff magic bytes if filename extension is absent, inaccurate, or mismatched
128 if (size >= 12 && data[0] == 'R' && data[1] == 'I' && data[2] == 'F' && data[3] == 'F' &&
129 data[8] == 'W' && data[9] == 'E' && data[10] == 'B' && data[11] == 'P')
130 {
131 detectedFormat = ImageFormat::WebP;
132 } else if (size >= 4 && data[0] == 'G' && data[1] == 'I' && data[2] == 'F' && data[3] == '8') {
133 detectedFormat = ImageFormat::GIF;
134 } else if (size >= 4 && ((data[0] == 'I' && data[1] == 'I' && data[2] == 0x2A && data[3] == 0x00) ||
135 (data[0] == 'M' && data[1] == 'M' && data[2] == 0x00 && data[3] == 0x2A)))
136 {
137 detectedFormat = ImageFormat::TIFF;
138 } else if (size >= 8 && data[0] == 0x89 && data[1] == 'P' && data[2] == 'N' && data[3] == 'G') {
139 detectedFormat = ImageFormat::PNG;
140 } else if (size >= 3 && data[0] == 0xFF && data[1] == 0xD8 && data[2] == 0xFF) {
141 detectedFormat = ImageFormat::JPEG;
142 }
143
144 // =========================================================================
145 // 1. SVG VECTOR PIPELINE (LunaSVG)
146 // =========================================================================
147 bool isSvg = (detectedFormat == ImageFormat::SVG);
148 if (!isSvg && size >= 5) {
149 // Sniff for XML/SVG header signature
150 std::string_view head(reinterpret_cast<const char*>(data), (std::min)(size, size_t(256)));
151 if (head.find("<svg") != std::string_view::npos ||
152 (head.find("<?xml") != std::string_view::npos && head.find("<svg") != std::string_view::npos))
153 {
154 isSvg = true;
155 }
156 }
157
158 if (isSvg) {
159 auto doc = lunasvg::Document::loadFromData(reinterpret_cast<const char*>(data), size);
160 if (doc) {
161 double naturalW = doc->width();
162 double naturalH = doc->height();
163 if (naturalW <= 0.0 || naturalH <= 0.0) {
164 naturalW = 512.0;
165 naturalH = 512.0;
166 }
167
168 // High-DPI Vector Rasterization:
169 // SVG is resolution-independent vector geometry. If an SVG's native viewBox is small
170 // (e.g. 24x24 or 32x32 for UI icons), rasterizing at viewBox size causes severe pixelation
171 // and blurriness when scaled up on canvas.
172 // Mathematical working process:
173 // maxNatural = max(naturalW, naturalH)
174 // targetDim = clamp(max(2048.0, maxNatural), 512.0, maxDecodedDim)
175 // scale = targetDim / maxNatural
176 // renderW = round(naturalW * scale)
177 // renderH = round(naturalH * scale)
178 // This ensures a 24x24 icon renders into a razor-sharp 2048x2048 PRGB32 surface.
179 const double maxDim = static_cast<double>(maxDecodedDim);
180 const double maxNatural = (std::max)(naturalW, naturalH);
181 const double targetCrispDim = (std::min)(maxDim, (std::max)(2048.0, maxNatural));
182 const double scale = (maxNatural > 0.0) ? (targetCrispDim / maxNatural) : 1.0;
183
184 const uint32_t renderW = (std::max)(1u, static_cast<uint32_t>(std::round(naturalW * scale)));
185 const uint32_t renderH = (std::max)(1u, static_cast<uint32_t>(std::round(naturalH * scale)));
186
187 lunasvg::Bitmap bitmap = doc->renderToBitmap(renderW, renderH);
188 if (bitmap.valid()) {
189 BLImage img;
190 if (img.create(static_cast<int>(renderW), static_cast<int>(renderH), BL_FORMAT_PRGB32) == BL_SUCCESS) {
191 BLImageData imgData;
192 if (img.make_mutable(&imgData) == BL_SUCCESS) {
193 const uint8_t* srcPixels = bitmap.data();
194 const int srcStride = bitmap.stride();
195 uint8_t* dstPixels = static_cast<uint8_t*>(imgData.pixel_data);
196 const int dstStride = static_cast<int>(imgData.stride);
197 const size_t bytesPerRow = renderW * 4;
198
199 // LunaSVG generates ARGB32 Premultiplied, matching Blend2D PRGB32 directly
200 for (uint32_t y = 0; y < renderH; ++y) {
201 std::memcpy(dstPixels + y * dstStride, srcPixels + y * srcStride, bytesPerRow);
202 }
203
204 result.image = std::move(img);
205 result.naturalWidth = static_cast<uint32_t>(naturalW);
206 result.naturalHeight = static_cast<uint32_t>(naturalH);
207 result.format = ImageFormat::SVG;
208 result.success = true;
209 return result;
210 }
211 }
212 }
213 }
214 }
215
216 // =========================================================================
217 // 2. MULTI-FRAME ANIMATION PIPELINE (GIF, Animated WebP via SDL3_image)
218 // =========================================================================
219 // Working Process & Math:
220 // GIF files (and animated WebP) encapsulate multiple raster frames with:
221 // - Individual frame durations (delayMs)
222 // - Disposal modes (restore background, restore previous, leave in place)
223 // - Transparency index masking and local color palettes
224 // SDL3_image's IMG_LoadAnimation_IO automatically composites disposal modes
225 // into complete, standalone 32-bit surfaces for each frame.
226 //
227 // Timing & Delay Clamping:
228 // GIF spec defines delay in 1/100ths of a second (10ms increments).
229 // Ancient encoders wrote 0 or 1 for delay. Standard browsers clamp delays
230 // <= 10ms to 100ms (10 FPS) to prevent runaway CPU spinlocks.
231 //
232 // RAM Protection:
233 // If natural dimensions exceed s_maxDecodedDimension, every frame is
234 // downsampled proportionally using Blend2D bilinear blitting:
235 // scale = s_maxDecodedDimension / max(naturalWidth, naturalHeight)
236 // =========================================================================
237 bool isGifOrWebp = (detectedFormat == ImageFormat::GIF || detectedFormat == ImageFormat::WebP);
238 if (!isGifOrWebp && size >= 4 && data[0] == 'G' && data[1] == 'I' && data[2] == 'F' && data[3] == '8') {
239 isGifOrWebp = true;
240 detectedFormat = ImageFormat::GIF;
241 }
242
243 if (isGifOrWebp) {
244 SDL_IOStream* animIo = SDL_IOFromConstMem(data, size);
245 if (animIo) {
246 IMG_Animation* anim = IMG_LoadAnimation_IO(animIo, true);
247 if (anim && anim->count > 0 && anim->w > 0 && anim->h > 0) {
248 result.naturalWidth = static_cast<uint32_t>(anim->w);
249 result.naturalHeight = static_cast<uint32_t>(anim->h);
250 result.format = (detectedFormat != ImageFormat::Unknown) ? detectedFormat : ImageFormat::GIF;
251 result.isAnimated = (anim->count > 1);
252
253 // RAM Protection: Compute uniform downsample factor if frames exceed maxDecodedDim
254 double scale = 1.0;
255 if (result.naturalWidth > maxDecodedDim || result.naturalHeight > maxDecodedDim) {
256 scale = static_cast<double>(maxDecodedDim) /
257 static_cast<double>((std::max)(result.naturalWidth, result.naturalHeight));
258 }
259 const int downW = (std::max)(1, static_cast<int>(std::round(result.naturalWidth * scale)));
260 const int downH = (std::max)(1, static_cast<int>(std::round(result.naturalHeight * scale)));
261
262 result.frames.reserve(anim->count);
263
264 for (int i = 0; i < anim->count; ++i) {
265 SDL_Surface* surf = anim->frames[i];
266 if (!surf) continue;
267
268 SDL_Surface* rgbaSurf = (surf->format == SDL_PIXELFORMAT_RGBA32)
269 ? surf
270 : SDL_ConvertSurface(surf, SDL_PIXELFORMAT_RGBA32);
271
272 if (rgbaSurf) {
273 BLImage converted = ConvertRgbaToBLImage(
274 static_cast<const uint8_t*>(rgbaSurf->pixels),
275 static_cast<uint32_t>(rgbaSurf->w),
276 static_cast<uint32_t>(rgbaSurf->h)
277 );
278
279 if (rgbaSurf != surf) {
280 SDL_DestroySurface(rgbaSurf);
281 }
282
283 if (!converted.is_empty()) {
284 BLImage finalImg;
285 if (scale < 1.0) {
286 if (finalImg.create(downW, downH, BL_FORMAT_PRGB32) == BL_SUCCESS) {
287 BLContext ctx(finalImg);
288 ctx.set_comp_op(BL_COMP_OP_SRC_COPY);
289 ctx.blit_image(BLRect(0, 0, downW, downH), converted);
290 ctx.end();
291 } else {
292 finalImg = std::move(converted);
293 }
294 } else {
295 finalImg = std::move(converted);
296 }
297
298 // GIF delay in milliseconds. Clamps <= 10ms to 100ms (10 FPS)
299 uint32_t delayMs = (anim->delays && anim->delays[i] > 10)
300 ? static_cast<uint32_t>(anim->delays[i])
301 : 100u;
302
303 result.frames.push_back(ImageFrame{ std::move(finalImg), delayMs });
304 }
305 }
306 }
307
308 // Note: IMG_FreeAnimation automatically frees anim->frames[i], anim->delays, and anim
309 IMG_FreeAnimation(anim);
310
311 if (!result.frames.empty()) {
312 result.image = result.frames[0].image;
313 result.success = true;
314 return result;
315 }
316 } else if (anim) {
317 IMG_FreeAnimation(anim);
318 }
319 }
320 }
321
322 // =========================================================================
323 // 3. BLEND2D NATIVE SIMD PIPELINE (PNG, JPEG, BMP, QOI)
324 // =========================================================================
325 BLImage blImg;
326 if (blImg.read_from_data(data, size) == BL_SUCCESS && !blImg.is_empty()) {
327 result.naturalWidth = static_cast<uint32_t>(blImg.width());
328 result.naturalHeight = static_cast<uint32_t>(blImg.height());
329 result.format = (detectedFormat != ImageFormat::Unknown) ? detectedFormat : ImageFormat::PNG;
330
331 // RAM Protection: If image exceeds maxDecodedDim, downsample display proxy
332 if (result.naturalWidth > maxDecodedDim || result.naturalHeight > maxDecodedDim) {
333 const double scale = static_cast<double>(maxDecodedDim) /
334 static_cast<double>((std::max)(result.naturalWidth, result.naturalHeight));
335 const int downW = (std::max)(1, static_cast<int>(result.naturalWidth * scale));
336 const int downH = (std::max)(1, static_cast<int>(result.naturalHeight * scale));
337
338 BLImage downsampled;
339 if (downsampled.create(downW, downH, BL_FORMAT_PRGB32) == BL_SUCCESS) {
340 BLContext ctx(downsampled);
341 ctx.set_comp_op(BL_COMP_OP_SRC_COPY);
342 ctx.blit_image(BLRect(0, 0, downW, downH), blImg);
343 ctx.end();
344 result.image = std::move(downsampled);
345 } else {
346 result.image = std::move(blImg);
347 }
348 } else {
349 result.image = std::move(blImg);
350 }
351
352 result.success = true;
353 return result;
354 }
355
356 // =========================================================================
357 // 4. EXTENDED FORMAT FALLBACK PIPELINE (WebP, TIFF, TGA, ICO, etc. via SDL3_image)
358 // =========================================================================
359 SDL_IOStream* io = SDL_IOFromConstMem(data, size);
360 if (io) {
361 SDL_Surface* surf = IMG_Load_IO(io, true);
362 if (surf && surf->w > 0 && surf->h > 0) {
363 result.naturalWidth = static_cast<uint32_t>(surf->w);
364 result.naturalHeight = static_cast<uint32_t>(surf->h);
365 if (detectedFormat != ImageFormat::Unknown) {
366 result.format = detectedFormat;
367 } else {
368 result.format = ImageFormat::WebP;
369 }
370
371 SDL_Surface* rgbaSurf = (surf->format == SDL_PIXELFORMAT_RGBA32)
372 ? surf
373 : SDL_ConvertSurface(surf, SDL_PIXELFORMAT_RGBA32);
374
375 if (rgbaSurf) {
376 BLImage converted = ConvertRgbaToBLImage(
377 static_cast<const uint8_t*>(rgbaSurf->pixels),
378 static_cast<uint32_t>(rgbaSurf->w),
379 static_cast<uint32_t>(rgbaSurf->h)
380 );
381
382 if (rgbaSurf != surf) {
383 SDL_DestroySurface(rgbaSurf);
384 }
385 SDL_DestroySurface(surf);
386
387 if (!converted.is_empty()) {
388 // RAM Protection: Downsample if exceeds max threshold
389 if (result.naturalWidth > maxDecodedDim || result.naturalHeight > maxDecodedDim) {
390 const double scale = static_cast<double>(maxDecodedDim) /
391 static_cast<double>((std::max)(result.naturalWidth, result.naturalHeight));
392 const int downW = (std::max)(1, static_cast<int>(result.naturalWidth * scale));
393 const int downH = (std::max)(1, static_cast<int>(result.naturalHeight * scale));
394
395 BLImage downsampled;
396 if (downsampled.create(downW, downH, BL_FORMAT_PRGB32) == BL_SUCCESS) {
397 BLContext ctx(downsampled);
398 ctx.set_comp_op(BL_COMP_OP_SRC_COPY);
399 ctx.blit_image(BLRect(0, 0, downW, downH), converted);
400 ctx.end();
401 result.image = std::move(downsampled);
402 } else {
403 result.image = std::move(converted);
404 }
405 } else {
406 result.image = std::move(converted);
407 }
408
409 result.success = true;
410 return result;
411 }
412 } else {
413 SDL_DestroySurface(surf);
414 }
415 }
416 }
417
418 const char* sdlErr = SDL_GetError();
419 if (sdlErr && *sdlErr) {
420 result.errorMessage = "Unsupported or corrupted image format (" + std::string(sdlErr) + ")";
421 SDL_ClearError();
422 } else {
423 result.errorMessage = "Unsupported or corrupted image format (unrecognized magic signature)";
424 }
425 return result;
426}

432DecodedImageResult ImageDecoder::DecodeFromFile(const std::string& filePath) {
433 DecodedImageResult result;
434
435 if (filePath.empty()) {
436 result.errorMessage = "File path is empty";
437 return result;
438 }
439
440 // Delegate path resolution to FileManager (automatically checks active package root or absolute disk paths)
441 std::string resolvedPath = FileManager::ResolveAssetPath(filePath);
442
443 if (!FileManager::Exists(resolvedPath)) {
444 result.errorMessage = "File does not exist: " + resolvedPath;
445 return result;
446 }
447
448 std::vector<uint8_t> fileBuffer;
449 if (!FileManager::ReadBinary(resolvedPath, fileBuffer) || fileBuffer.empty()) {
450 result.errorMessage = "Failed to read file bytes: " + resolvedPath;
451 return result;
452 }
453
454 return DecodeFromMemory(fileBuffer.data(), fileBuffer.size(), resolvedPath);
455}
