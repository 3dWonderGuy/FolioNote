2 * =========================================================================================
3 * @file package_importer.cpp
4 * @brief Implementation of High-Performance Package & Archive Importer for FolioNote
5 * =========================================================================================
6 *
7 * ARCHITECTURAL IMPLEMENTATION DETAILS:
8 * 1. Multi-Threaded 7-Zip Decompression:
9 * Invokes `7z.exe x -y -o"<dest>" "<src>"` to decompress LZMA2 / 7z archives with full CPU
10 * utilization. Falls back to Windows system `tar.exe` if 7-Zip is not installed.
11 * 2. Structural Inspection & Validation:
12 * Inspects decompressed contents to identify whether the archive contains a nested `.notebook`
13 * package directory or root-level notebook files (`structure.db` / `pages/`).
14 * 3. Atomic Collision-Free Relocation:
15 * Moves decompressed notebook packages into the target library root using conflict-free naming
16 * `Notebook_Name (Imported N).notebook` without overwriting existing user data.
17 * 4. Staging Isolation & Cleanup:
18 * Performs all extractions in an isolated ephemeral staging folder and guarantees complete
19 * removal of temporary files upon success or failure.
20 */

39bool ExecuteProcess(const std::string& commandLine) {
40#if defined(_WIN32)
41 STARTUPINFOA si;
42 PROCESS_INFORMATION pi;
43 ZeroMemory(&si, sizeof(si));
44 si.cb = sizeof(si);
45 si.dwFlags |= STARTF_USESHOWWINDOW;
46 si.wShowWindow = SW_HIDE; // Run background extraction silently
47 ZeroMemory(&pi, sizeof(pi));
48
49 std::vector<char> cmdBuf(commandLine.begin(), commandLine.end());
50 cmdBuf.push_back('\0');
51
52 if (!CreateProcessA(
53 NULL,
54 cmdBuf.data(),
55 NULL,
56 NULL,
57 FALSE,
58 CREATE_NO_WINDOW,
59 NULL,
60 NULL,
61 &si,
62 &pi)) {
63 return false;
64 }
65
66 // Wait up to 5 minutes for decompression of large notebook libraries
67 WaitForSingleObject(pi.hProcess, 300000);
68
69 DWORD exitCode = 1;
70 GetExitCodeProcess(pi.hProcess, &exitCode);
71
72 CloseHandle(pi.hProcess);
73 CloseHandle(pi.hThread);
74
75 return (exitCode == 0);
76#else
77 int res = std::system(commandLine.c_str());
78 return (res == 0);
79#endif
80}

82std::string GenerateUniqueDestinationPath(const std::filesystem::path& destDir, const std::string& baseStem) {
83 std::error_code ec;
84 std::filesystem::path target = destDir / (baseStem + ".notebook");
85 int counter = 1;
86 while (std::filesystem::exists(target, ec)) {
87 target = destDir / (baseStem + " (Imported " + std::to_string(counter++) + ").notebook");
88 }
89 return target.string();
90}

116 lower.rfind(".foliolib.7z") != std::string::npos ||

118 lower.rfind(".zip") != std::string::npos);

127bool PackageImporter::DecompressArchive(const std::string& archivePath, const std::string& targetDirectory) {
128 if (!FileManager::Exists(archivePath)) {
129 LOG_ERROR(FileManager, "PackageImporter::DecompressArchive: Archive file does not exist: " + archivePath);
130 return false;
131 }
132
133 FileManager::CreateDirectories(targetDirectory);
134
135 std::string z7Path = Find7ZipExecutable();
136 if (!z7Path.empty()) {
137 std::string cmd = "\"" + z7Path + "\" x -y -o\"" + targetDirectory + "\" \"" + archivePath + "\"";
138 LOG_INFO(FileManager, "PackageImporter: Decompressing with 7-Zip: " + archivePath);
139 if (ExecuteProcess(cmd)) {
140 return true;
141 }
142 LOG_WARN(FileManager, "PackageImporter: 7-Zip extraction returned non-zero, trying tar fallback.");
143 }
144
145 // Windows native tar fallback
146 std::string tarCmd = "tar.exe -xzf \"" + archivePath + "\" -C \"" + targetDirectory + "\"";
147 LOG_INFO(FileManager, "PackageImporter: Decompressing with tar: " + archivePath);
148 return ExecuteProcess(tarCmd);
149}

151std::string PackageImporter::ImportPackage(
152 const std::string& sourcePath,
153 const std::string& destinationLibraryPath
154) {
155 std::error_code ec;
156 std::filesystem::path src(sourcePath);
157 if (!std::filesystem::exists(src, ec)) {
158 LOG_ERROR(FileManager, "PackageImporter: Source package path does not exist: " + sourcePath);
159 return "";
160 }
161
162 std::filesystem::path destDir(destinationLibraryPath);
163 if (!std::filesystem::exists(destDir, ec)) {
164 std::filesystem::create_directories(destDir, ec);
165 }
166
167 // Case 1: Uncompressed .notebook folder
168 if (std::filesystem::is_directory(src, ec)) {
169 std::string stem = src.stem().string();
170 std::string targetPkg = GenerateUniqueDestinationPath(destDir, stem);
171
172 LOG_INFO(FileManager, "PackageImporter: Copying uncompressed package to: " + targetPkg);
173 std::filesystem::copy(src, targetPkg, std::filesystem::copy_options::recursive | std::filesystem::copy_options::overwrite_existing, ec);
174 if (ec) {
175 LOG_ERROR(FileManager, "PackageImporter: Failed to copy directory: " + ec.message());
176 return "";
177 }
178 return targetPkg;
179 }
180
181 // Case 2: Compressed archive (.folionb.7z / .fnpack / .7z / .zip)
182 if (IsArchiveFile(sourcePath)) {
183 std::string tempStaging = FileManager::JoinPath(
184 FileManager::GetTempDirectory(),
185 "_stage_import_" + GUIDGenerator::GenerateV4().substr(0, 8)
186 );
187
188 if (!DecompressArchive(sourcePath, tempStaging)) {
189 LOG_ERROR(FileManager, "PackageImporter: Extraction failed for: " + sourcePath);
190 std::filesystem::remove_all(tempStaging, ec);
191 return "";
192 }
193
194 // Inspect staging directory to find .notebook directory or root files
195 std::string discoveredNotebookDir;
196 for (const auto& entry : std::filesystem::directory_iterator(tempStaging, ec)) {
197 if (entry.is_directory(ec) && entry.path().extension() == ".notebook") {
198 discoveredNotebookDir = entry.path().string();
199 break;
200 }
201 }
202
203 std::string finalDest;
204 if (!discoveredNotebookDir.empty()) {
205 std::string stem = std::filesystem::path(discoveredNotebookDir).stem().string();
206 finalDest = GenerateUniqueDestinationPath(destDir, stem);
207 std::filesystem::rename(discoveredNotebookDir, finalDest, ec);
208 if (ec) {
209 // Fallback to recursive copy if on different drives
210 std::filesystem::copy(discoveredNotebookDir, finalDest, std::filesystem::copy_options::recursive, ec);
211 }
212 } else {
213 // Root-level archive files: move the staging folder itself as the .notebook package
214 std::string stem = src.stem().string();
215 // If stem ends with .folionb, strip it
216 if (stem.size() > 8 && stem.rfind(".folionb") == stem.size() - 8) {
217 stem = stem.substr(0, stem.size() - 8);
218 }
219 finalDest = GenerateUniqueDestinationPath(destDir, stem);
220 std::filesystem::rename(tempStaging, finalDest, ec);
221 if (ec) {
222 std::filesystem::copy(tempStaging, finalDest, std::filesystem::copy_options::recursive, ec);
223 }
224 }
225
226 // Clean up temporary staging
227 std::filesystem::remove_all(tempStaging, ec);
228
229 LOG_INFO(FileManager, "PackageImporter: Successfully imported archive to: " + finalDest);
230 return finalDest;
231 }
232
233 return "";
234}

236std::vector<std::string> PackageImporter::ImportLibraryPackage(
237 const std::string& sourceArchivePath,
238 const std::string& destinationLibraryPath
239) {
240 std::vector<std::string> importedPackages;
241 std::error_code ec;
242
243 if (!FileManager::Exists(sourceArchivePath)) {
244 LOG_ERROR(FileManager, "PackageImporter: Library archive does not exist: " + sourceArchivePath);
245 return importedPackages;
246 }
247
248 std::filesystem::path destDir(destinationLibraryPath);
249 std::filesystem::create_directories(destDir, ec);
250
251 std::string tempStaging = FileManager::JoinPath(
252 FileManager::GetTempDirectory(),
253 "_stage_lib_" + GUIDGenerator::GenerateV4().substr(0, 8)
254 );
255
256 if (!DecompressArchive(sourceArchivePath, tempStaging)) {
257 LOG_ERROR(FileManager, "PackageImporter: Failed to decompress library archive: " + sourceArchivePath);
258 std::filesystem::remove_all(tempStaging, ec);
259 return importedPackages;
260 }
261
262 // Traverse staging directory to discover all .notebook packages
263 for (const auto& entry : std::filesystem::recursive_directory_iterator(tempStaging, ec)) {
264 if (entry.is_directory(ec) && entry.path().extension() == ".notebook") {
265 std::string stem = entry.path().stem().string();
266 std::string targetPkg = GenerateUniqueDestinationPath(destDir, stem);
267
268 std::filesystem::rename(entry.path(), targetPkg, ec);
269 if (ec) {
270 std::filesystem::copy(entry.path(), targetPkg, std::filesystem::copy_options::recursive, ec);
271 }
272
273 if (!ec) {
274 importedPackages.push_back(targetPkg);
275 LOG_INFO(FileManager, "PackageImporter: Mounted library notebook: " + targetPkg);
276 }
277 }
278 }
279
280 std::filesystem::remove_all(tempStaging, ec);
281 return importedPackages;
282}
