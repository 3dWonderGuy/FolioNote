1/**
2 * =========================================================================================
3 * @file core/layers/baked_canvas_layer.cpp
4 * @brief Implementation of Layer 1 High-Performance Cached Rasterizer for Committed Objects
5 * =========================================================================================
6 *
7 * GENERAL ARCHITECTURAL PROCESS:
8 * ------------------------------
9 * This translation unit manages the caching lifecycle of all committed vector canvas entities.
10 *
11 * 1. Surface Allocation:
12 * - Uses Blend2D's 32-bit premultiplied ARGB format (BL_FORMAT_PRGB32) for fast SIMD
13 * rasterization and hardware blits.
14 * - The backing context (`m_bakedContext`) is detached (`end()`) before any re-allocation
15 * to ensure memory safety.
16 *
17 * 2. Invalidation Discipline:
18 * - Path A (Full Frustum Rebake):
19 * When the camera pans, zooms, or window resizes, the entire visible frustum is cleared
20 * and repainted. The page R-Tree spatial index queries all objects touching `viewport.visibleWorldBounds`.
21 * - Path B (Partial Dirty Region Patch):
22 * When an existing object moves, is erased, or is inserted, the damage bounds `m_dirtyWorldRegion`
23 * are merged. The raster context sets a screen scissor-clip (`clip_to_rect`), repaints the paper
24 * background only within the clipped rect, queries only objects touching the damage box,
25 * and redraws them in ascending z-order.
26 *
27 * 3. Deterministic Painter's Layering:
28 * - All candidates retrieved from spatial queries are sorted by ascending `zOrder` (`a->zOrder < b->zOrder`),
29 * guaranteeing identical visual stacking regardless of R-Tree node insertion order.
30 */

49 *
50 * @param[in] width New viewport width in screen pixels.
51 * @param[in] height New viewport height in screen pixels.
52 */
53void BakedCanvasLayer::Resize(int width, int height) {
54 if (m_surfaceWidth == width && m_surfaceHeight == height) {
55 return;
56 }
57
58 m_surfaceWidth = width;
59 m_surfaceHeight = height;
60
61 // Detach context before destroying surface memory to avoid dangling handles
62 m_bakedContext.end();
63 m_bakedSurface.create(width, height, BL_FORMAT_PRGB32);
64
65 Invalidate();
66}

88/**
89 * @brief Marks a specific world-space bounding box dirty for localized partial re-baking.
90 *
91 * MATHEMATICAL PROCESS:
92 * - If a full rebake is already pending (`m_needsFullRebake == true`), any partial damage
93 * is inherently covered by the full pass; early-exit to avoid redundant AABB expansion.
94 * - Otherwise, merges `dirtyBounds` into `m_dirtyWorldRegion`:
95 * minX = min(minX, dirtyBounds.minX), maxX = max(maxX, dirtyBounds.maxX)
96 * minY = min(minY, dirtyBounds.minY), maxY = max(maxY, dirtyBounds.maxY)
97 *
98 * @param[in] dirtyBounds The AABB in world millimeters requiring re-rasterization.
99 */
100void BakedCanvasLayer::InvalidateRect(const AABB& dirtyBounds) {
101 if (m_needsFullRebake) {
102 return; // Full rebake already queued; ignore sub-region
103 }
104
105 m_dirtyWorldRegion.Merge(dirtyBounds);
106 m_isDirty = true;
107}

109/**
110 * @brief Renders the scene into the backing cache surface (handles both full & partial rebakes).
111 *
112 * MATHEMATICAL & COMPOSITING PROCESS:
113 * 1. Surface Allocation: Verifies or instantiates a 32-bit premultiplied ARGB backing store (BL_FORMAT_PRGB32)
114 * matching current viewport dimensions (m_surfaceWidth x m_surfaceHeight).
115 * 2. Background Pass: Fills paper color across the active region.
116 * 3. Coordinate Transformation:
117 * - Applies viewport.worldToScreenMatrix mapping world coordinates (mm) to screen buffer pixels:
118 * [ x_screen ] [ m00 m01 dx ] [ x_world ]
119 * [ y_screen ] = [ m10 m11 dy ] [ y_world ]
120 * [ 1 ] [ 0 0 1 ] [ 1 ]
121 * 4. Spatial Culling & Z-Ordering:
122 * - In Path A (Full Rebake): Queries page spatialIndex (R-Tree) using viewport.visibleWorldBounds.
123 * - In Path B (Partial Rebake): Clips the context to screen-transformed dirty rect,
124 * repaints background, and queries only objects intersecting m_dirtyWorldRegion.
125 * - Resolves object UIDs via page->FindObjectByUid(uid) and sorts candidates strictly by zOrder ascending
126 * (min to max) to guarantee back-to-front painter's algorithm rasterization.
127 * 5. Lifecycle Flush: Resets dirty flags and bounding regions upon completion.
128 *
129 * @param[in] page The active CanvasPage containing spatialIndex and document entities.
130 * @param[in] viewport The current camera viewport containing matrices and visibility bounds.
131 */
132void BakedCanvasLayer::Render(CanvasPage* page, const Viewport& viewport) {
133 if (!m_isDirty || m_surfaceWidth <= 0 || m_surfaceHeight <= 0 || !page) {
134 return;
135 }
136
137 if (m_bakedSurface.is_empty()) {
138 m_bakedSurface.create(m_surfaceWidth, m_surfaceHeight, BL_FORMAT_PRGB32);
139 }
140
141 m_bakedContext.begin(m_bakedSurface);
142
143 if (m_needsFullRebake) {
144 // --- Path A: Full Viewport Rebake (Camera pan/zoom/resize) ---
145 DrawBackground(page, viewport);
146
147 // 1. Spatial query visible entities via R-Tree frustum culling
148 auto uids = page->spatialIndex.Query(viewport.visibleWorldBounds);
149
150 // 2. Collect visible entities and resolve shared ownership
151 std::vector<std::shared_ptr<CanvasObject>> candidates;
152 candidates.reserve(uids.size());
153 for (uint32_t uid : uids) {
154 auto obj = page->FindObjectByUid(uid);
155 if (obj && obj->isVisible) {
156 candidates.push_back(std::move(obj));
157 }
158 }
159
160 // 3. Sort strictly by ascending z-order and document pageIndex to preserve back-to-front visual layering
161 std::sort(candidates.begin(), candidates.end(), [](const auto& a, const auto& b) {
162 if (a->zOrder != b->zOrder) return a->zOrder < b->zOrder;
163 return a->pageIndex < b->pageIndex;
164 });
165
166 // 4. Apply camera world-to-screen matrix transform and rasterize objects
167 m_bakedContext.save();
168 m_bakedContext.set_transform(viewport.worldToScreenMatrix);
169
170 for (const auto& obj : candidates) {
171 obj->Render(m_bakedContext, viewport);
172 }
173
174 m_bakedContext.restore();
175 }
176 else {
177 // --- Path B: Partial Region Patch (Single object mutation) ---
178 if (m_dirtyWorldRegion.IsValid() && !m_dirtyWorldRegion.IsEmpty()) {
179 m_bakedContext.save();
180
181 // 1. Scissor-clip in screen space using the projected dirty bounds
182 BLRect screenClip = viewport.WorldToScreenRect(m_dirtyWorldRegion);
183 m_bakedContext.clip_to_rect(screenClip);
184
185 // 2. Repaint paper backdrop inside the clipped screen patch
186 DrawBackground(page, viewport);
187
188 // 3. Apply camera world-to-screen transform for vector object rendering
189 m_bakedContext.set_transform(viewport.worldToScreenMatrix);
190
191 // 4. Query only objects touching the dirty bounding box via R-Tree
192 auto uids = page->spatialIndex.Query(m_dirtyWorldRegion);
193
194 std::vector<std::shared_ptr<CanvasObject>> candidates;
195 candidates.reserve(uids.size());
196 for (uint32_t uid : uids) {
197 auto obj = page->FindObjectByUid(uid);
198 if (obj && obj->isVisible) {
199 candidates.push_back(std::move(obj));
200 }
201 }
202
203 // 5. Sort strictly by ascending z-order and document pageIndex
204 std::sort(candidates.begin(), candidates.end(), [](const auto& a, const auto& b) {
205 if (a->zOrder != b->zOrder) return a->zOrder < b->zOrder;
206 return a->pageIndex < b->pageIndex;
207 });
208
209 for (const auto& obj : candidates) {
210 obj->Render(m_bakedContext, viewport);
211 }
212
213 m_bakedContext.restore();
214 }
215 }
216
217 m_bakedContext.end();
218
219 // Reset tracking flags for the next frame
220 m_isDirty = false;
221 m_needsFullRebake = false;
222 m_dirtyWorldRegion.Reset();
223}

225/**
226 * @brief Renders the paper color backdrop, page margins, and grid/dot patterns.
227 *
228 * @param[in] activePage The active canvas page with theme settings.
229 * @param[in] viewport Active viewport for coordinate reference.
230 */
231void BakedCanvasLayer::DrawBackground(CanvasPage* activePage, const Viewport& viewport) {
232 if (!activePage) {
233 m_bakedContext.fill_all(m_theme.bgColor);
234 return;
235 }
236
237 const auto infinityMode = activePage->infinityMode;
238 const auto paperStyle = activePage->paperStyle;
239 const double gridSpacingMm = (activePage->gridSpacingMm > 0.5) ? activePage->gridSpacingMm : 5.0;
240
241 double pageWMm = (activePage->pageWidthMm > 0.0) ? activePage->pageWidthMm : 215.9;
242 double pageHMm = (activePage->pageHeightMm > 0.0) ? activePage->pageHeightMm : 279.4;
243 if (activePage->pageIsLandscape) {
244 std::swap(pageWMm, pageHMm);
245 }
246
247 Point2D originScreen = viewport.WorldToScreen(0.0, 0.0);
248 Point2D cornerScreen = viewport.WorldToScreen(pageWMm, pageHMm);
249 double rectScreenW = cornerScreen.x - originScreen.x;
250 double rectScreenH = cornerScreen.y - originScreen.y;
251
252 if (infinityMode == CanvasInfinityMode::VerticalScroll) {
253 BLRgba32 deskCol = (m_theme.bgColor.r() > 128)
254 ? BLRgba32(static_cast<uint8_t>(m_theme.bgColor.r() * 0.93),
255 static_cast<uint8_t>(m_theme.bgColor.g() * 0.93),
256 static_cast<uint8_t>(m_theme.bgColor.b() * 0.94))
257 : BLRgba32(static_cast<uint8_t>(std::min(255, (int)(m_theme.bgColor.r() * 1.25 + 10))),
258 static_cast<uint8_t>(std::min(255, (int)(m_theme.bgColor.g() * 1.25 + 10))),
259 static_cast<uint8_t>(std::min(255, (int)(m_theme.bgColor.b() * 1.25 + 12))));
260 m_bakedContext.fill_all(deskCol);
261 m_bakedContext.fill_rect(originScreen.x, std::max(0.0, originScreen.y), rectScreenW, m_surfaceHeight - std::max(0.0, originScreen.y), m_theme.bgColor);
262 } else if (infinityMode == CanvasInfinityMode::HorizontalScroll) {
263 BLRgba32 deskCol = (m_theme.bgColor.r() > 128)
264 ? BLRgba32(static_cast<uint8_t>(m_theme.bgColor.r() * 0.93),
265 static_cast<uint8_t>(m_theme.bgColor.g() * 0.93),
266 static_cast<uint8_t>(m_theme.bgColor.b() * 0.94))
267 : BLRgba32(static_cast<uint8_t>(std::min(255, (int)(m_theme.bgColor.r() * 1.25 + 10))),
268 static_cast<uint8_t>(std::min(255, (int)(m_theme.bgColor.g() * 1.25 + 10))),
269 static_cast<uint8_t>(std::min(255, (int)(m_theme.bgColor.b() * 1.25 + 12))));
270 m_bakedContext.fill_all(deskCol);
271 m_bakedContext.fill_rect(std::max(0.0, originScreen.x), originScreen.y, m_surfaceWidth - std::max(0.0, originScreen.x), rectScreenH, m_theme.bgColor);
272 } else {
273 m_bakedContext.fill_all(m_theme.bgColor);
274 }
275
276 if (activePage->showPageBorder) {
277 m_bakedContext.save();
278 m_bakedContext.set_stroke_style(m_theme.borderColor);
279 double strokePx = std::max(1.0, activePage->pageBorderWidth * viewport.zoom);
280 m_bakedContext.set_stroke_width(strokePx);
281
282 double bx = originScreen.x;
283 double by = originScreen.y;
284 double bw = rectScreenW;
285 double bh = rectScreenH;
286
287 if (activePage->pageBorderStyle == PageBorderStyle::Dashed) {
288 BLArray<double> dashArray;
289 dashArray.append(8.0);
290 dashArray.append(6.0);
291 m_bakedContext.set_stroke_dash_array(dashArray);
292 }
293
294 if (activePage->pageBorderStyle == PageBorderStyle::Corners) {
295 double arm = std::min(std::min(bw, bh) * 0.25, 20.0 * viewport.zoom);
296 if (arm > 2.0) {
297 m_bakedContext.stroke_line(bx, by + arm, bx, by);
298 m_bakedContext.stroke_line(bx, by, bx + arm, by);
299 m_bakedContext.stroke_line(bx + bw - arm, by, bx + bw, by);
300 m_bakedContext.stroke_line(bx + bw, by, bx + bw, by + arm);
301 m_bakedContext.stroke_line(bx, by + bh - arm, bx, by + bh);
302 m_bakedContext.stroke_line(bx, by + bh, bx + arm, by + bh);
303 m_bakedContext.stroke_line(bx + bw - arm, by + bh, bx + bw, by + bh);
304 m_bakedContext.stroke_line(bx + bw, by + bh, bx + bw, by + bh - arm);
305 }
306 } else {
307 m_bakedContext.stroke_rect(bx, by, bw, bh);
308 }
309 m_bakedContext.restore();
310 }
311
312 if (paperStyle == PaperStyle::Blank) {
313 return;
314 }
315
316 const double stepMm = gridSpacingMm;
317 m_bakedContext.save();
318 double startX = std::floor(viewport.bounds.minX / stepMm) * stepMm;
319 double endX = std::ceil(viewport.bounds.maxX / stepMm) * stepMm;
320 double startY = std::floor(viewport.bounds.minY / stepMm) * stepMm;
321 double endY = std::ceil(viewport.bounds.maxY / stepMm) * stepMm;
322
323 m_bakedContext.set_stroke_style(m_theme.gridColor);
324 m_bakedContext.set_stroke_width(1.0);
325
326 if (paperStyle == PaperStyle::Dotted) {
327 m_bakedContext.set_fill_style(m_theme.gridColor);
328 for (double wy = startY; wy <= endY; wy += stepMm) {
329 if (infinityMode != CanvasInfinityMode::FullInfinity && wy < 0.0) continue;
330 if (infinityMode == CanvasInfinityMode::HorizontalScroll && wy > pageHMm) continue;
331 for (double wx = startX; wx <= endX; wx += stepMm) {
332 if (infinityMode != CanvasInfinityMode::FullInfinity && wx < 0.0) continue;
333 if (infinityMode == CanvasInfinityMode::VerticalScroll && wx > pageWMm) continue;
334 Point2D pt = viewport.WorldToScreen(wx, wy);
335 m_bakedContext.fill_circle(pt.x, pt.y, 1.2);
336 }
337 }
338 }
339 else {
340 if (paperStyle == PaperStyle::Grid) {
341 for (double wx = startX; wx <= endX; wx += stepMm) {
342 if (infinityMode != CanvasInfinityMode::FullInfinity && wx < 0.0) continue;
343 if (infinityMode == CanvasInfinityMode::VerticalScroll && wx > pageWMm) continue;
344 double minY = viewport.bounds.minY;
345 double maxY = viewport.bounds.maxY;
346 if (infinityMode != CanvasInfinityMode::FullInfinity) minY = std::max(0.0, minY);
347 if (infinityMode == CanvasInfinityMode::HorizontalScroll) maxY = std::min(pageHMm, maxY);
348 if (minY < maxY) {
349 Point2D sTop = viewport.WorldToScreen(wx, minY);
350 Point2D sBot = viewport.WorldToScreen(wx, maxY);
351 m_bakedContext.stroke_line(sTop.x, sTop.y, sBot.x, sBot.y);
352 }
353 }
354 }
355
356 // Horizontal lines for Grid, Lined, and Cornell
357 for (double wy = startY; wy <= endY; wy += stepMm) {
358 if (infinityMode != CanvasInfinityMode::FullInfinity && wy < 0.0) continue;
359 if (infinityMode == CanvasInfinityMode::HorizontalScroll && wy > pageHMm) continue;
360 double minX = viewport.bounds.minX;
361 double maxX = viewport.bounds.maxX;
362 if (infinityMode != CanvasInfinityMode::FullInfinity) minX = std::max(0.0, minX);
363 if (infinityMode == CanvasInfinityMode::VerticalScroll) maxX = std::min(pageWMm, maxX);
364 if (minX < maxX) {
365 Point2D sLeft = viewport.WorldToScreen(minX, wy);
366 Point2D sRight = viewport.WorldToScreen(maxX, wy);
367 m_bakedContext.stroke_line(sLeft.x, sLeft.y, sRight.x, sRight.y);
368 }
369 }
370
371 if (paperStyle == PaperStyle::Cornell) {
372 // Cornell summary margin vertical line at 50mm from left
373 double marginXMm = 50.0;
374 if (viewport.bounds.minX <= marginXMm && viewport.bounds.maxX >= marginXMm) {
375 double minY = (infinityMode != CanvasInfinityMode::FullInfinity) ? std::max(0.0, viewport.bounds.minY) : viewport.bounds.minY;
376 double maxY = (infinityMode == CanvasInfinityMode::HorizontalScroll) ? std::min(pageHMm, viewport.bounds.maxY) : viewport.bounds.maxY;
377 if (minY < maxY) {
378 m_bakedContext.set_stroke_style(BLRgba32(0xFF, 0x52, 0x52, 0x88));
379 m_bakedContext.set_stroke_width(1.5);
380 Point2D sTop = viewport.WorldToScreen(marginXMm, minY);
381 Point2D sBot = viewport.WorldToScreen(marginXMm, maxY);
382 m_bakedContext.stroke_line(sTop.x, sTop.y, sBot.x, sBot.y);
383 }
384 }
385 }
386 }
387
388 m_bakedContext.restore();
389}
