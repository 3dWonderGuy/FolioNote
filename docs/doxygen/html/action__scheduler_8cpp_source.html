35void ActionScheduler::Schedule(std::unique_ptr<ICanvasAction> action, ActionContext& ctx) {
36 if (!action) return;
37
38 if (isUpdating) {
39 // Enqueue to prevent iterator invalidation during hot update loop
40 pendingQueue.push_back(std::move(action));
41 return;
42 }
43
44 const uint32_t reqResources = action->GetRequiredResources();
45
46 // Preempt and cancel conflicting active actions
47 if (reqResources != 0) {
48 for (auto it = activeActions.begin(); it != activeActions.end(); ) {
49 if ((it->resources & reqResources) != 0) {
50 it->action->OnEnd(ActionEndReason::Interrupted, ctx);
51 it = activeActions.erase(it);
52 } else {
53 ++it;
54 }
55 }
56 }
57
58 // Initialize the incoming action
59 action->OnStart(ctx);
60
61 // If incoming action claims Camera, Selection, or DocumentStructure, instantly mark layer dirty
62 if ((reqResources & (ToResourceMask(CanvasResource::Camera) |
63 ToResourceMask(CanvasResource::Selection) |
64 ToResourceMask(CanvasResource::DocumentStructure))) != 0) {
65 ctx.InvalidateLayer();
66 }
67
68 // If action completed immediately during OnStart (e.g. InstantAction)
69 if (action->IsFinished(ctx)) {
70 action->OnEnd(ActionEndReason::Finished, ctx);
71 ctx.InvalidateLayer();
72 RecalculateResourceMask();
73 return;
74 }
75
76 // Register into active pool
77 activeActions.push_back(RunningAction{std::move(action), reqResources});
78 claimedResourceMask |= reqResources;
79}

97void ActionScheduler::CancelAll(ActionContext& ctx) {

115void ActionScheduler::Update(double dt, ActionContext& ctx) {
116 isUpdating = true;
117
118 // Process all running actions for this frame
119 for (auto it = activeActions.begin(); it != activeActions.end(); ) {
120 it->action->OnUpdate(dt, ctx);
121
122 // Instantly mark layer dirty during active camera, selection, or document actions
123 if ((it->resources & (ToResourceMask(CanvasResource::Camera) |
124 ToResourceMask(CanvasResource::Selection) |
125 ToResourceMask(CanvasResource::DocumentStructure))) != 0) {
126 ctx.InvalidateLayer();
127 }
128
129 if (it->action->IsFinished(ctx)) {
130 it->action->OnEnd(ActionEndReason::Finished, ctx);
131 ctx.InvalidateLayer();
132 it = activeActions.erase(it);
133 } else {
134 ++it;
135 }
136 }
137
138 isUpdating = false;
139
140 // Flush any pending actions that were queued during update
141 if (!pendingQueue.empty()) {
142 auto queued = std::move(pendingQueue);
143 pendingQueue.clear();
144 for (auto& action : queued) {
145 Schedule(std::move(action), ctx);
146 }
147 }
148
149 RecalculateResourceMask();
150}
