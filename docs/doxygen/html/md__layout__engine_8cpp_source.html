12double MeasureSpanWidth(const std::string& text, float fontSizePt, bool isCode = false) {
13 if (text.empty()) return 0.0;
14 double fontHeightMm = fontSizePt * PT_TO_MM;
15 double avgCharWidth = isCode ? (fontHeightMm * 0.60) : (fontHeightMm * 0.52);
16
17 double total = 0.0;
18 for (size_t idx = 0; idx < text.size(); ++idx) {
19 uint8_t c = static_cast<uint8_t>(text[idx]);
20 // Skip UTF-8 continuation bytes to avoid counting multibyte codepoints multiple times
21 if ((c & 0xC0) == 0x80) continue;
22
23 if (c == ' ' || c == '\t') {
24 total += avgCharWidth * 0.65;
25 } else if (c == 'i' || c == 'l' || c == '.' || c == ',' || c == '!') {
26 total += avgCharWidth * (isCode ? 1.0 : 0.45);
27 } else if (c == 'm' || c == 'w' || c == 'M' || c == 'W') {
28 total += avgCharWidth * (isCode ? 1.0 : 1.40);
29 } else if (c >= 0x80) {
30 // Non-ASCII codepoint (CJK, Cyrillic, Greek, accented latin)
31 total += avgCharWidth * (isCode ? 1.0 : 1.15);
32 } else {
33 total += avgCharWidth;
34 }
35 }
36 return total;
37}

156 currentY += block.height_mm + config.paragraphSpacingMm;
157 break;
158 }
159
160 case MdBlockType::MathBlock: {
161 block.height_mm = 16.0;
162 currentY += block.height_mm + config.paragraphSpacingMm;
163 break;
164 }
165
166 case MdBlockType::HorizontalRule: {
167 block.height_mm = 6.0;
168 currentY += block.height_mm + 2.0;
169 break;
170 }
171 }
172 }

177size_t MdLayoutEngine::HitTestOffset(
178 double localX_mm,
179 double localY_mm,
180 const std::vector<MdBlock>& blocks,
181 const MdStyleConfig& config
182) {
183 if (blocks.empty()) return 0;
184
185 for (const auto& block : blocks) {
186 if (localY_mm >= block.localY_mm && localY_mm <= block.localY_mm + block.height_mm) {
187 if (block.spans.empty()) {
188 return block.sourceStartOffset;
189 }
190
191 double indentX = config.marginLeftMm;
192 if (block.type == MdBlockType::BulletList || block.type == MdBlockType::NumberedList || block.type == MdBlockType::TaskList) {
193 indentX += (config.listIndentMm * block.level + 6.0);
194 } else if (block.type == MdBlockType::Blockquote) {
195 indentX += (config.blockquoteIndentMm + 4.0);
196 }
197
198 double targetX = localX_mm - indentX;
199 if (targetX <= 0.0) {
200 return block.sourceStartOffset;
201 }
202
203 double curX = 0.0;
204 for (const auto& sp : block.spans) {
205 if (targetX <= curX + sp.widthMm) {
206 double relX = std::max(0.0, targetX - curX);
207 double ratio = (sp.widthMm > 0.0) ? (relX / sp.widthMm) : 0.0;
208 size_t charOffset = static_cast<size_t>(std::round(ratio * sp.sourceLength));
209 return sp.sourceOffset + std::min(charOffset, sp.sourceLength);
210 }
211 curX += sp.widthMm;
212 }
213
214 return block.spans.back().sourceOffset + block.spans.back().sourceLength;
215 }
216 }
217
218 if (localY_mm < blocks.front().localY_mm) {
219 return blocks.front().sourceStartOffset;
220 }
221 return blocks.back().sourceEndOffset;
222}

224MdCaret MdLayoutEngine::GetCaretForOffset(
225 size_t offset,
226 const std::vector<MdBlock>& blocks,
227 const MdStyleConfig& config
228) {
229 MdCaret caret;
230 caret.offset = offset;
231 caret.worldPosMm = Point2D{ config.marginLeftMm, config.marginTopMm };
232 caret.heightMm = config.bodyFontSizePt * PT_TO_MM;
233 caret.visible = true;
234
235 if (blocks.empty()) return caret;
236
237 for (const auto& block : blocks) {
238 if (offset >= block.sourceStartOffset && offset <= block.sourceEndOffset) {
239 caret.worldPosMm.y = block.localY_mm;
240 caret.worldPosMm.x = config.marginLeftMm;
241
242 if (block.type == MdBlockType::BulletList || block.type == MdBlockType::NumberedList || block.type == MdBlockType::TaskList) {
243 caret.worldPosMm.x += (config.listIndentMm * block.level + 6.0);
244 } else if (block.type == MdBlockType::Blockquote) {
245 caret.worldPosMm.x += config.blockquoteIndentMm + 4.0;
246 }
247
248 if (!block.spans.empty()) {
249 caret.heightMm = block.spans.front().sizePt * PT_TO_MM;
250 }
251
252 double spanAccX = 0.0;
253 for (const auto& sp : block.spans) {
254 if (offset <= sp.sourceOffset) {
255 break;
256 } else if (offset >= sp.sourceOffset + sp.sourceLength) {
257 spanAccX += sp.widthMm;
258 } else {
259 size_t relOffset = offset - sp.sourceOffset;
260 double ratio = (sp.sourceLength > 0) ? (static_cast<double>(relOffset) / sp.sourceLength) : 0.0;
261 spanAccX += sp.widthMm * ratio;
262 break;
263 }
264 }
265 caret.worldPosMm.x += spanAccX;
266 break;
267 }
268 }
269
270 return caret;
271}
