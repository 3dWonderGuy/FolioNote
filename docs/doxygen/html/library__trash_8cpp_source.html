28/**
29 * @brief Moves a .notebook package into its library's .trash/ directory for safe recovery.
30 *
31 * GENERAL WORKING PROCESS & INVARIANTS:
32 * 1. Identifies which library bundle owns the target notebook by checking `libraries` or
33 * climbing to the parent path.
34 * 2. Creates the hidden `.trash/` subdirectory inside that library root if it does not already exist.
35 * 3. Disambiguates destination filename if an identically named notebook was previously trashed.
36 * 4. Atomically relocates the package folder via `FileManager::Move`.
37 * 5. Refreshes the library catalogs so active views immediately reflect package removal.
38 *
39 * @param notebookPath Absolute filesystem path to the .notebook directory.
40 * @return bool True if successfully moved to trash; false on validation error or I/O failure.
41 */
42bool LibraryManager::MoveNotebookToTrash(const std::string& notebookPath) {
43 if (notebookPath.empty() || !FileManager::IsDirectory(notebookPath)) {
44 LOG_ERROR(LibraryManager, "MoveNotebookToTrash rejected: Package not found: " + notebookPath);
45 return false;
46 }
47
48 // Step 1: Discover owning library
49 std::string libRoot;
50 for (const auto& lib : libraries) {
51 if (lib.ContainsNotebook(notebookPath)) {
52 libRoot = lib.rootPath;
53 break;
54 }
55 }
56 if (libRoot.empty()) {
57 libRoot = FileManager::GetParentPath(notebookPath);
58 }
59 if (libRoot.empty()) {
60 libRoot = defaultLibraryPath;
61 }
62
63 // Step 2: Ensure .trash/ directory exists inside the library root
64 std::string trashDir = FileManager::JoinPath(libRoot, ".trash");
65 if (!FileManager::CreateDirectories(trashDir)) {
66 LOG_ERROR(LibraryManager, "MoveNotebookToTrash failed to create trash directory: " + trashDir);
67 return false;
68 }
69
70 // Step 3: Determine collision-free destination path inside .trash
71 std::string nbStem = FileManager::GetStem(notebookPath);
72 std::string targetTrashPath = FileManager::DisambiguatePath(trashDir, nbStem, FOLIO_NOTEBOOK_EXTENSION);
73
74 // Step 4: Perform move
75 if (!FileManager::Move(notebookPath, targetTrashPath)) {
76 LOG_ERROR(LibraryManager, "MoveNotebookToTrash failed to move " + notebookPath + " -> " + targetTrashPath);
77 return false;
78 }
79
80 LOG_INFO(LibraryManager, "Moved notebook package to library trash: " + targetTrashPath);
81
82 // Step 5: Refresh catalog
83 RefreshAll();
84 return true;
85}

87/**
88 * @brief Restores a notebook from .trash/ back to an active library directory.
89 *
90 * GENERAL WORKING PROCESS:
91 * 1. Resolves destination library: if targetLibraryPath is unspecified, defaults to the
92 * parent library owning the `.trash/` folder.
93 * 2. Generates a collision-safe destination filename in the library root.
94 * 3. Relocates package from `.trash/` to the active library via `FileManager::Move`.
95 * 4. Refreshes library notebook catalogs.
96 *
97 * @param trashNotebookPath Absolute path to the trashed .notebook package.
98 * @param targetLibraryPath Optional destination library path (defaults to owning library).
99 * @return bool True if successfully restored; false on failure.
100 */
101bool LibraryManager::RestoreNotebookFromTrash(const std::string& trashNotebookPath, const std::string& targetLibraryPath) {
102 if (trashNotebookPath.empty() || !FileManager::IsDirectory(trashNotebookPath)) {
103 LOG_ERROR(LibraryManager, "RestoreNotebookFromTrash rejected: Trashed notebook not found: " + trashNotebookPath);
104 return false;
105 }
106
107 std::string destLib = targetLibraryPath;
108 if (destLib.empty()) {
109 // Parent of .trash is the library directory
110 std::string trashDir = FileManager::GetParentPath(trashNotebookPath);
111 destLib = FileManager::GetParentPath(trashDir);
112 }
113 if (destLib.empty() || !FileManager::IsDirectory(destLib)) {
114 destLib = defaultLibraryPath;
115 }
116
117 std::string nbStem = FileManager::GetStem(trashNotebookPath);
118 std::string destPath = FileManager::DisambiguatePath(destLib, nbStem, FOLIO_NOTEBOOK_EXTENSION);
119
120 if (!FileManager::Move(trashNotebookPath, destPath)) {
121 LOG_ERROR(LibraryManager, "RestoreNotebookFromTrash failed to move " + trashNotebookPath + " -> " + destPath);
122 return false;
123 }
124
125 LOG_INFO(LibraryManager, "Restored notebook from trash to active library: " + destPath);
126 RefreshAll();
127 return true;
128}

130/**
131 * @brief Discovers all trashed .notebook folders inside a library's .trash/ subfolder.
132 *
133 * @param libraryPath Absolute path to the library bundle.
134 * @return std::vector<std::string> List of absolute paths to discovered trashed packages.
135 */
136std::vector<std::string> LibraryManager::GetTrashNotebooks(const std::string& libraryPath) const {
137 std::vector<std::string> result;
138 std::string trashDir = FileManager::JoinPath(libraryPath, ".trash");
139 if (!FileManager::IsDirectory(trashDir)) {
140 return result;
141 }
142
143 auto entries = FileManager::ListEntries(trashDir, FOLIO_NOTEBOOK_EXTENSION);
144 for (const auto& entry : entries) {
145 if (entry.type == FileType::Directory) {
146 result.push_back(entry.fullPath);
147 }
148 }
149 return result;
150}

152/**
153 * @brief Permanently purges a trashed .notebook package from disk.
154 *
155 * SAFETY INVARIANT:
156 * Explicitly verifies that the target path contains `.trash` to guarantee
157 * that an active, non-trashed notebook package is NEVER accidentally destroyed.
158 *
159 * @param trashNotebookPath Absolute path to the trashed notebook directory.
160 * @return bool True if deleted permanently; false on validation error or I/O failure.
161 */
162bool LibraryManager::PermanentlyDeleteNotebookFromTrash(const std::string& trashNotebookPath) {
163 if (trashNotebookPath.empty() || !FileManager::IsDirectory(trashNotebookPath)) {
164 return false;
165 }
166
167 // Safety guard: ensure the folder resides inside a .trash directory
168 std::string normalized = FileManager::NormalizeSeparators(trashNotebookPath);
169 if (normalized.find("/.trash/") == std::string::npos && normalized.find("\\.trash\\") == std::string::npos) {
170 LOG_ERROR(LibraryManager, "PermanentlyDeleteNotebookFromTrash rejected: Path is not inside a .trash directory: " + trashNotebookPath);
171 return false;
172 }
173
174 bool success = FileManager::RemoveDirectoryRecursive(trashNotebookPath);
175 if (success) {
176 LOG_INFO(LibraryManager, "Permanently purged notebook from trash: " + trashNotebookPath);
177 } else {
178 LOG_ERROR(LibraryManager, "Failed to permanently purge notebook from trash: " + trashNotebookPath);
179 }
180 return success;
181}

183/**
184 * @brief Permanently purges all contents of a library's .trash/ directory.
185 *
186 * @param libraryPath Absolute path to the library bundle.
187 * @return size_t Count of trashed notebooks permanently erased.
188 */
189size_t LibraryManager::EmptyLibraryTrash(const std::string& libraryPath) {
190 auto trashed = GetTrashNotebooks(libraryPath);
191 size_t count = 0;
192 for (const auto& path : trashed) {
193 if (PermanentlyDeleteNotebookFromTrash(path)) {
194 count++;
195 }
196 }
197 LOG_INFO(LibraryManager, "Emptied library trash for " + libraryPath + ". Erased " + std::to_string(count) + " notebooks.");
198 return count;
199}
