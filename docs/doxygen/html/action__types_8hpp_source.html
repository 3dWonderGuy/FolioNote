23/**
24 * @brief Resource flags for mutual exclusion arbitration across active canvas operations.
25 * Actions declare required resources; when an incoming action conflicts with active ones,
26 * the scheduler arbitrates preemption or cancellation.
27 */
28enum class CanvasResource : uint32_t {

30 Selection = 1 << 0, ///< Manipulating active selected objects (drag, rotate, scale)
31 Camera = 1 << 1, ///< Viewport navigation, smooth pan/zoom easing

33 BackgroundCulling = 1 << 3, ///< Spatial index R-Tree rebuilding or geometry decimation
34 DocumentStructure = 1 << 4, ///< Section/page hierarchy mutations or layer modifications
35 All = 0xFFFFFFFF
36};
