2/**
3 * =========================================================================================
4 * @file core/objects/object_config.hpp
5 * @brief Centralized Configuration and Runtime Constraints for Canvas Objects
6 * =========================================================================================
7 *
8 * ARCHITECTURAL DESIGN & WORKING PROCESS:
9 * ---------------------------------------
10 * ObjectConfig serves as the single source of truth for runtime limits, physical canvas
11 * boundaries, and safety constraints across all canvas object types (images, attachments,
12 * hyperlinks, and interactive chips).
13 *
14 * Key Design Principles:
15 * 1. Decoupled Core:
16 * Lives within `src/core/objects/` without dependencies on UI frameworks (ImGui),
17 * JSON serializers, or application lifecycle singletons.
18 * 2. Instant Zero-Overhead Access:
19 * Objects query `ObjectConfig::Get()` on the rendering hot path (up to 120 FPS)
20 * with direct POD member reads.
21 * 3. Bidirectional Sync with SettingsManager:
22 * SettingsManager loads user configurations from `settings.json`, updates its
23 * internal ObjectConfig instance, and synchronizes the global `ObjectConfig::Get()`
24 * singleton during startup and when the user modifies settings.
25 */
26

79 /**
80 * @brief Resolves the card background color based on active theme mode and opacity.
81 * @param alphaMultiplier Global opacity multiplier [0.0, 1.0].
82 * @return BLRgba32 evaluated color.
83 */
84 [[nodiscard]] BLRgba32 GetCardBackgroundColor(float alphaMultiplier = 1.0f) const noexcept {
85 uint32_t c = isDarkMode ? cardBgColorDark : cardBgColorLight;
86 uint8_t baseAlpha = static_cast<uint8_t>((c >> 24) & 0xFF);
87 uint8_t a = static_cast<uint8_t>(baseAlpha * std::clamp(alphaMultiplier, 0.0f, 1.0f));
88 return BLRgba32((c >> 16) & 0xFF, (c >> 8) & 0xFF, c & 0xFF, a);
89 }
90
91 /**
92 * @brief Resolves the card stroke border color with error state handling.
93 * @param isBroken True if file cannot be found on disk (renders warning red).
94 * @param alphaMultiplier Global opacity multiplier [0.0, 1.0].
95 * @return BLRgba32 evaluated stroke color.
96 */
97 [[nodiscard]] BLRgba32 GetCardBorderColor(bool isBroken, float alphaMultiplier = 1.0f) const noexcept {
98 if (isBroken) {
99 uint8_t a = static_cast<uint8_t>(230.0f * std::clamp(alphaMultiplier, 0.0f, 1.0f));
100 return BLRgba32(0xE8, 0x11, 0x23, a);
101 }
102 uint32_t c = isDarkMode ? cardBorderColorDark : cardBorderColorLight;
103 uint8_t baseAlpha = static_cast<uint8_t>((c >> 24) & 0xFF);
104 uint8_t a = static_cast<uint8_t>(baseAlpha * std::clamp(alphaMultiplier, 0.0f, 1.0f));
105 return BLRgba32((c >> 16) & 0xFF, (c >> 8) & 0xFF, c & 0xFF, a);
106 }

108 /**
109 * @brief Resolves typography label color considering theme and validity.
110 * @param isBroken True if file path is missing or invalid.
111 * @param alphaMultiplier Global opacity multiplier [0.0, 1.0].
112 * @return BLRgba32 evaluated text color.
113 */
114 [[nodiscard]] BLRgba32 GetCardTextColor(bool isBroken, float alphaMultiplier = 1.0f) const noexcept {
115 if (isBroken) {
116 uint8_t a = static_cast<uint8_t>(255.0f * std::clamp(alphaMultiplier, 0.0f, 1.0f));
117 return BLRgba32(0xFF, 0x88, 0x88, a);
118 }
119 uint32_t c = isDarkMode ? cardTextColorDark : cardTextColorLight;
120 uint8_t baseAlpha = static_cast<uint8_t>((c >> 24) & 0xFF);
121 uint8_t a = static_cast<uint8_t>(baseAlpha * std::clamp(alphaMultiplier, 0.0f, 1.0f));
122 return BLRgba32((c >> 16) & 0xFF, (c >> 8) & 0xFF, c & 0xFF, a);
123 }

155 * Clamped to 4096px = 4096 * 4096 * 4 = 67.1 MB RAM (94% reduction).
156 */
