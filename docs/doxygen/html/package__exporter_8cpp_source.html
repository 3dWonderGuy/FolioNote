7 * ARCHITECTURAL IMPLEMENTATION DETAILS:
8 * 1. Safe SQLite WAL Flushing:

52bool ExecuteProcess(const std::string& commandLine) {
53#if defined(_WIN32)
54 STARTUPINFOA si;
55 PROCESS_INFORMATION pi;
56 ZeroMemory(&si, sizeof(si));
57 si.cb = sizeof(si);
58 si.dwFlags |= STARTF_USESHOWWINDOW;
59 si.wShowWindow = SW_HIDE; // Run background compression without popping up a console window
60 ZeroMemory(&pi, sizeof(pi));
61
62 std::string mutableCmd = commandLine;
63 if (!CreateProcessA(
64 NULL,
65 &mutableCmd[0],
66 NULL,
67 NULL,
68 FALSE,
69 CREATE_NO_WINDOW,
70 NULL,
71 NULL,
72 &si,
73 &pi
74 )) {
75 LOG_ERROR(FileManager, "PackageExporter: CreateProcess failed for: " + commandLine);
76 return false;
77 }
78
79 // Wait for compression process to finish (up to 5 minutes)
80 WaitForSingleObject(pi.hProcess, 300000);
81
82 DWORD exitCode = 1;
83 GetExitCodeProcess(pi.hProcess, &exitCode);
84
85 CloseHandle(pi.hProcess);
86 CloseHandle(pi.hThread);
87
88 return (exitCode == 0);
89#else
90 int res = std::system(commandLine.c_str());
91 return (res == 0);
92#endif
93}

97bool PackageExporter::ExportNotebookPackage(
98 const std::string& notebookPath,
99 const std::string& destinationArchive,
100 int compressionLevel
101) {
102 if (notebookPath.empty() || !FileManager::IsDirectory(notebookPath)) {
103 LOG_ERROR(FileManager, "ExportNotebookPackage: Invalid notebook path: " + notebookPath);
104 return false;
105 }
106
107 std::string destDir = FileManager::GetParentPath(destinationArchive);
108 if (!destDir.empty()) {
109 FileManager::CreateDirectories(destDir);
110 }
111
112 // Checkpoint SQLite WAL buffers so the archive contains a clean, consistent structure.db
113 BackupManager::FlushSqliteWal(notebookPath);
114
115 std::string z7Path = Find7ZipExecutable();
116 if (!z7Path.empty()) {
117 int level = std::clamp(compressionLevel, 1, 9);
118 std::string cmd = "\"" + z7Path + "\" a -t7z -mx" + std::to_string(level) +
119 " -mmt=on -y \"" + destinationArchive + "\" \"" + notebookPath + "\\*\"";
120
121 LOG_INFO(FileManager, "ExportNotebookPackage: Running 7-Zip compression: " + destinationArchive);
122 if (ExecuteProcess(cmd)) {
123 LOG_INFO(FileManager, "ExportNotebookPackage: Successfully exported archive: " + destinationArchive);
124 return true;
125 }
126 }
127
128 // Fallback to tar.exe if 7-Zip was not found or failed
129 std::string tarCmd = "tar.exe -czf \"" + destinationArchive + "\" -C \"" +
130 FileManager::GetParentPath(notebookPath) + "\" \"" +
131 FileManager::GetFileName(notebookPath) + "\"";
132 LOG_INFO(FileManager, "ExportNotebookPackage: Running tar fallback: " + destinationArchive);
133 return ExecuteProcess(tarCmd);
134}
