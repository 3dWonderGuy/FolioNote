1/**
2 * =========================================================================================
3 * @file pdf_vector_exporter.cpp
4 * @brief Implementation of Standalone High-Fidelity Vector PDF Exporter
5 * =========================================================================================
6 *
7 * ARCHITECTURAL IMPLEMENTATION DETAILS:
8 * 1. Native Vector Graphics Pipeline:
9 * Writes PDF drawing operators directly with zero third-party dependencies:
10 * - Coordinate translation: Converts canvas millimeters to PDF typography points ($1\text{ mm} \approx 2.83465\text{ pt}$).
11 * - Inverts the Y-axis to match the standard Cartesian bottom-left origin of PDF.
12 * - Sets round line caps (`1 J`) and round line joins (`1 j`) for pen stroke fidelity.
13 * 2. Searchable Text Operators:
14 * Encodes text boxes using PDF text blocks (`BT ... /F1 ... Tf (...) Tj ET`) with Helvetica
15 * standard Type 1 font metrics, ensuring all words are searchable (`Ctrl+F`) and copyable.
16 * 3. Interactive URI Link Annotations:
17 * Extracts hyperlinks and generates interactive `/Subtype /Link` annotations with `/S /URI`,
18 * preserving web references and desktop deep links (`folionote://page/...`).
19 */

40constexpr double MM_TO_PT = 72.0 / 25.4; // 2.83464567293

42std::string EscapePdfString(const std::string& input) {
43 std::string out;
44 for (char c : input) {
45 if (c == '(' || c == ')' || c == '\\') {
46 out += '\\';
47 }
48 out += c;
49 }
50 return out;
51}

60PdfPageContent GenerateSheetPdfContent(
61 const SheetTile& tile,
62 const std::shared_ptr<CanvasPage>& page,
63 const ExportOptions& options
64) {
65 PdfPageContent result;
66 result.widthPt = tile.widthMm * MM_TO_PT;
67 result.heightPt = tile.heightMm * MM_TO_PT;
68
69 std::ostringstream ss;
70 double invScale = tile.uniformScale * MM_TO_PT;
71
72 // 1. Vector Ink Strokes
73 for (const auto& obj : tile.intersectingObjects) {
74 if (auto ink = std::dynamic_pointer_cast<InkContainer>(obj)) {
75 for (const auto& stroke : ink->strokes) {
76 if (stroke.segments.empty()) continue;
77
78 uint32_t val = stroke.color.value;
79 float r = ((val >> 16) & 0xFF) / 255.0f;
80 float g = ((val >> 8) & 0xFF) / 255.0f;
81 float b = (val & 0xFF) / 255.0f;
82 double widthPt = std::max(0.5, stroke.baseWidth * invScale);
83
84 // Set stroke color, line width, round caps (1 J), and round joins (1 j)
85 ss << std::fixed << std::setprecision(2);
86 ss << r << " " << g << " " << b << " RG\n";
87 ss << widthPt << " w 1 J 1 j\n";
88
89 // First point
90 double p0x = (stroke.segments[0].p0.x - tile.worldX) * invScale;
91 double p0y = result.heightPt - ((stroke.segments[0].p0.y - tile.worldY) * invScale);
92 ss << p0x << " " << p0y << " m\n";
93
94 for (const auto& seg : stroke.segments) {
95 double p1x = (seg.p1.x - tile.worldX) * invScale;
96 double p1y = result.heightPt - ((seg.p1.y - tile.worldY) * invScale);
97 ss << p1x << " " << p1y << " l\n";
98 }
99 ss << "S\n";
100 }
101 }
102 }
103
104 // 2. Searchable Text Objects & Interactive Link Annotations
105 for (const auto& obj : tile.intersectingObjects) {
106 if (auto tb = std::dynamic_pointer_cast<Folio::TextBoxObject>(obj)) {
107 double txPt = (tb->worldX - tile.worldX) * invScale;
108 double fontSizePt = (tb->fontSize > 0.0f ? tb->fontSize : 12.0f) * static_cast<float>(tile.uniformScale);
109 double tyPt = result.heightPt - ((tb->worldY - tile.worldY) * invScale) - fontSizePt;
110
111 // Split text by lines
112 std::istringstream textStream(tb->text);
113 std::string line;
114 double currentY = tyPt;
115
116 ss << "BT\n/F1 " << fontSizePt << " Tf\n0.12 0.13 0.16 rg\n";
117
118 while (std::getline(textStream, line)) {
119 if (!line.empty()) {
120 ss << txPt << " " << currentY << " Td\n";
121 ss << "(" << EscapePdfString(line) << ") Tj\n";
122 ss << (-txPt) << " " << 0 << " Td\n"; // Return X to 0
123
124 // Check for hyperlinks in line
125 if (options.enableLinks) {
126 static const std::regex urlRegex(R"((https?://[^\s<>"]+|folionote://[^\s<>"]+))");
127 auto wordsBegin = std::sregex_iterator(line.begin(), line.end(), urlRegex);
128 auto wordsEnd = std::sregex_iterator();
129
130 for (auto it = wordsBegin; it != wordsEnd; ++it) {
131 std::smatch match = *it;
132 std::string url = match.str();
133
134 bool allowed = (url.rfind("folionote://", 0) == 0) ? options.enableAppDeepLinks : options.enableWebLinks;
135 if (allowed) {
136 double approxCharW = fontSizePt * 0.55;
137 double linkX1 = txPt + (match.position() * approxCharW);
138 double linkX2 = linkX1 + (match.length() * approxCharW);
139 double linkY1 = currentY - 2.0;
140 double linkY2 = currentY + fontSizePt + 2.0;
141
142 std::ostringstream annot;
143 annot << "<< /Type /Annot /Subtype /Link /Rect ["
144 << linkX1 << " " << linkY1 << " " << linkX2 << " " << linkY2
145 << "] /Border [0 0 0] /A << /S /URI /URI (" << EscapePdfString(url) << ") >> >>";
146 result.linkAnnotDicts.push_back(annot.str());
147 }
148 }
149 }
150 }
151 currentY -= (fontSizePt * 1.35); // Advance line height
152 }
153 ss << "ET\n";
154 }
155 }
156
157 result.streamData = ss.str();
158 return result;
159}

161bool WriteCompletePdfDocument(const std::vector<PdfPageContent>& pages, const std::string& destinationPdfPath) {
162 if (pages.empty()) return false;
163
164 std::string destDir = FileManager::GetParentPath(destinationPdfPath);
165 if (!destDir.empty()) {
166 FileManager::CreateDirectories(destDir);
167 }
168
169 std::ofstream pdf(destinationPdfPath, std::ios::binary | std::ios::trunc);
170 if (!pdf.is_open()) return false;
171
172 std::vector<size_t> xrefOffsets;
173 xrefOffsets.push_back(0); // Object 0 is special
174
175 auto RecordObject = [&](std::string_view) {
176 xrefOffsets.push_back(static_cast<size_t>(pdf.tellp()));
177 };
178
179 // PDF Header
180 pdf << "%PDF-1.4\n%\xE2\xE3\xCF\xD3\n";
181
182 // 1. Catalog Object (1 0 obj)
183 RecordObject("Catalog");
184 pdf << "1 0 obj\n<< /Type /Catalog /Pages 2 0 R >>\nendobj\n";
185
186 // 2. Pages Root Object (2 0 obj)
187 // Page objects start at object ID 4
188 size_t pageCount = pages.size();
189 RecordObject("Pages");
190 pdf << "2 0 obj\n<< /Type /Pages /Count " << pageCount << " /Kids [";
191 for (size_t i = 0; i < pageCount; ++i) {
192 pdf << (4 + i * 2) << " 0 R ";
193 }
194 pdf << "] >>\nendobj\n";
195
196 // 3. Shared Font Object (3 0 obj) - Standard Helvetica Type 1
197 RecordObject("Font");
198 pdf << "3 0 obj\n<< /Type /Font /Subtype /Type1 /BaseFont /Helvetica /Encoding /WinAnsiEncoding >>\nendobj\n";
199
200 // Write individual Page and Content stream objects
201 for (size_t i = 0; i < pageCount; ++i) {
202 const auto& pg = pages[i];
203 size_t pageObjId = 4 + i * 2;
204 size_t contentObjId = pageObjId + 1;
205
206 // Page Object
207 RecordObject("Page");
208 pdf << pageObjId << " 0 obj\n<< /Type /Page /Parent 2 0 R /MediaBox [0 0 "
209 << pg.widthPt << " " << pg.heightPt << "] /Contents " << contentObjId
210 << " 0 R /Resources << /Font << /F1 3 0 R >> >>";
211
212 if (!pg.linkAnnotDicts.empty()) {
213 pdf << " /Annots [";
214 for (const auto& a : pg.linkAnnotDicts) {
215 pdf << a << " ";
216 }
217 pdf << "]";
218 }
219 pdf << " >>\nendobj\n";
220
221 // Content Stream Object
222 RecordObject("Content");
223 pdf << contentObjId << " 0 obj\n<< /Length " << pg.streamData.size() << " >>\nstream\n"
224 << pg.streamData << "endstream\nendobj\n";
225 }
226
227 // Cross-Reference Table
228 size_t xrefPos = static_cast<size_t>(pdf.tellp());
229 size_t totalObjects = xrefOffsets.size();
230 pdf << "xref\n0 " << totalObjects << "\n";
231 pdf << "0000000000 65535 f \n";
232
233 for (size_t i = 1; i < totalObjects; ++i) {
234 std::ostringstream offsetStr;
235 offsetStr << std::setw(10) << std::setfill('0') << xrefOffsets[i];
236 pdf << offsetStr.str() << " 00000 n \n";
237 }
238
239 // Trailer
240 pdf << "trailer\n<< /Size " << totalObjects << " /Root 1 0 R >>\n";
241 pdf << "startxref\n" << xrefPos << "\n%%EOF\n";
242
243 pdf.close();
244 LOG_INFO(FileManager, "PdfVectorExporter: Successfully wrote vector PDF (" +
245 std::to_string(pageCount) + " pages) to: " + destinationPdfPath);
246 return true;
247}

251bool PdfVectorExporter::ExportPage(
252 const std::shared_ptr<CanvasPage>& page,
253 const std::string& destinationPdfPath,
254 const ExportOptions& options
255) {
256 if (!page) return false;
257 auto tiles = SheetTiler::TilePage(
258 *page, options.traversalOrder, options.scalingMode, options.skipEmptySheets
259 );
260
261 std::vector<PdfPageContent> pdfPages;
262 for (const auto& tile : tiles) {
263 pdfPages.push_back(GenerateSheetPdfContent(tile, page, options));
264 }
265
266 return WriteCompletePdfDocument(pdfPages, destinationPdfPath);
267}
