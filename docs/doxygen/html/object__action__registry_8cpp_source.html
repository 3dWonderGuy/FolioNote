1/**
2 * =========================================================================================
3 * @file core/objects/object_action_registry.cpp
4 * @brief Implementation of Standard Canvas Object Actions Registry and Menu Builder
5 * =========================================================================================
6 *
7 * ARCHITECTURAL DESIGN & INVERSION OF CONTROL:
8 * --------------------------------------------
9 * ObjectActionRegistry centralizes universal canvas object operations (deletion, z-ordering,
10 * clipboard interactions, duplication, and specialized media controls) within the core
11 * object hierarchy.
12 *
13 * Moving this implementation out of the header isolates heavy dependencies:
14 * - CanvasEngine (viewport rebaking, camera, selection gizmo)
15 * - DocumentSession (undo/redo command history, selection management)
16 * - CanvasPage (z-ordering, object lifetime, dirty notifications)
17 * - ClipboardManager (cross-subsystem copy/cut operations)
18 * - Specialized container types (AttachmentObject, ImageObject, VideoObject)
19 * - Platform subsystem APIs (<SDL3/SDL.h> for native external URL dispatch)
20 *
21 * Preventing these concrete subsystem headers from leaking into `object_action_registry.hpp`
22 * breaks compilation cascades across the UI and state machine subsystems.
23 */

40/**
41 * @brief Constructs a sorted sequence of interactive context actions for a canvas object.
42 *
43 * MATHEMATICAL & PIPELINE BEHAVIOR:
44 * ---------------------------------
45 * 1. Universal Actions Setup:
46 * - Delete (order: 0, destructive):
47 * Removes object via `page->RemoveObjectByUid(targetObj->uid)`, pushes an atomic
48 * `RemoveObjectsCommand` onto the undo/redo stack, and flags `needsFullRebake = true`.
49 * - Layering / Z-Ordering:
50 * - Bring to Front (order: 200): Shifts object to index `N - 1`.
51 * - Bring Forward (order: 210): Increments z-index by swapping with index `i + 1`.
52 * - Send Backward (order: 220): Decrements z-index by swapping with index `i - 1`.
53 * - Send to Back (order: 230): Shifts object to background plane (index 0).
54 * - Clipboard (Copy order: 280, Cut order: 290):
55 * Dispatches active selection (or target object fallback) to `ClipboardManager::Instance()`.
56 * - Duplication (order: 300):
57 * Calls `session.DuplicateSelection(10.0)`, translating duplicates by vector `[+10mm, +10mm]^T`.
58 *
59 * 2. Specialized Container Callbacks:
60 * - AttachmentObject: Injects undoable `RelinkAttachmentCommand` on relinking events.
61 * - ImageObject: Injects visual state synchronization (gizmo bounds recalculation + dirty flags).
62 * - VideoObject: Populates transport controls (Play/Pause, Stop, Mute, Loop, Reset Size, Native Player).
63 *
64 * 3. Inversion-of-Control Hook:
65 * Calls `obj->CustomizeActions(actions)` allowing subclasses to inject custom entries.
66 *
67 * 4. Deterministic Ordering:
68 * Applies `std::stable_sort` over `ContextMenuItem::order` ascending.
69 *
70 * @param[in] obj Target canvas object selected or right-clicked.
71 * @param[in] page Active canvas page hosting the object.
72 * @param[in,out] engine Engine holding viewport state, camera, and selection gizmo.
73 * @param[in,out] session Document session managing transaction history and dirty state.
74 * @return std::vector<ContextMenuItem> Ordered, interactive action list.
75 */
76std::vector<ContextMenuItem> ObjectActionRegistry::BuildActionsForObject(
77 const std::shared_ptr<CanvasObject>& obj,
78 std::shared_ptr<CanvasPage> page,
79 CanvasEngine& engine,
80 DocumentSession& session
81) {
82 if (!obj || !page) return {};
83
84 std::vector<ContextMenuItem> actions;
85 actions.reserve(16);
86
87 // =========================================================================
88 // 1. UNIVERSAL DEFAULT ACTIONS
89 // =========================================================================
90
91 // Action: Delete (Primary domain action, order: 0)
92 {
93 ContextMenuItem act;
94 act.label = "Delete";
95 act.shortcut = "Del";
96 act.icon = "🗑";
97 act.iconKey = "trash";
98 act.order = 0;
99 act.isSeparatorBefore = false;
100 act.isDestructive = true;
101 act.onTrigger = [page, &session, &engine, targetObj = obj]() {
102 page->RemoveObjectByUid(targetObj->uid);
103 session.RecordHistoryCommand(page, std::make_unique<Folio::RemoveObjectsCommand>(targetObj));
104 engine.selectionGizmo.ClearSelection();
105 engine.isDirty = true;
106 engine.needsFullRebake = true;
107 };
108 actions.push_back(std::move(act));
109 }
110
111 // Action: Order / Layering Cascading Submenu (order: 200)
112 {
113 ContextMenuItem layerMenu;
114 layerMenu.label = "Order";
115 layerMenu.icon = "📑";
116 layerMenu.iconKey = "layers";
117 layerMenu.order = 200;
118 layerMenu.isSeparatorBefore = true;
119
120 // Subaction: Bring to Front
121 ContextMenuItem front;
122 front.label = "Bring to Front";
123 front.shortcut = "Ctrl+Shift+]";
124 front.icon = "⬆";
125 front.iconKey = "bring_to_front";
126 front.order = 201;
127 front.onTrigger = [page, &engine, uid = obj->uid]() {
128 page->BringToFront(uid);
129 engine.isDirty = true;
130 engine.needsFullRebake = true;
131 };
132 layerMenu.AddSubAction(std::move(front));
133
134 // Subaction: Bring Forward (Single-step z-order increment)
135 ContextMenuItem forward;
136 forward.label = "Bring Forward";
137 forward.shortcut = "Ctrl+]";
138 forward.icon = "⇡";
139 forward.iconKey = "bring_forward";
140 forward.order = 202;
141 forward.onTrigger = [page, &engine, uid = obj->uid]() {
142 page->ShiftFront(uid);
143 engine.isDirty = true;
144 engine.needsFullRebake = true;
145 };
146 layerMenu.AddSubAction(std::move(forward));
147
148 // Subaction: Send Backward (Single-step z-order decrement)
149 ContextMenuItem backward;
150 backward.label = "Send Backward";
151 backward.shortcut = "Ctrl+[";
152 backward.icon = "⇣";
153 backward.iconKey = "send_backward";
154 backward.order = 203;
155 backward.onTrigger = [page, &engine, uid = obj->uid]() {
156 page->ShiftBack(uid);
157 engine.isDirty = true;
158 engine.needsFullRebake = true;
159 };
160 layerMenu.AddSubAction(std::move(backward));
161
162 // Subaction: Send to Back
163 ContextMenuItem back;
164 back.label = "Send to Back";
165 back.shortcut = "Ctrl+Shift+[";
166 back.icon = "⬇";
167 back.iconKey = "send_to_back";
168 back.order = 204;
169 back.onTrigger = [page, &engine, uid = obj->uid]() {
170 page->SendToBack(uid);
171 engine.isDirty = true;
172 engine.needsFullRebake = true;
173 };
174 layerMenu.AddSubAction(std::move(back));
175
176 actions.push_back(std::move(layerMenu));
177 }
178
179 // Action: Set as Background / Unlock from Background (order: 190)
180 // Background elements are marked non-selectable (isSelectable = 0) and sent to back,
181 // so pen/lasso operations draw over them smoothly without selecting or moving them.
182 {
183 ContextMenuItem bgAct;
184 if (obj->isSelectable) {
185 bgAct.label = "Set as Background";
186 bgAct.icon = "📌";
187 bgAct.iconKey = "pin_background";
188 bgAct.order = 190;
189 bgAct.isSeparatorBefore = true;
190 bgAct.onTrigger = [obj, page, &engine, &session]() {
191 obj->isSelectable = 0;
192 obj->isSelected = 0;
193 page->SendToBack(obj->uid);
194 engine.selectionGizmo.ClearSelection();
195 engine.isDirty = true;
196 engine.needsFullRebake = true;
197 page->isModified = true;
198 session.NotifyPageModified(page);
199 };
200 } else {
201 bgAct.label = "Unlock from Background";
202 bgAct.icon = "🔓";
203 bgAct.iconKey = "unpin_background";
204 bgAct.order = 190;
205 bgAct.isSeparatorBefore = true;
206 bgAct.onTrigger = [obj, page, &engine, &session]() {
207 obj->isSelectable = 1;
208 obj->isSelected = 1;
209 engine.selectionGizmo.SetSelectedObjects({obj});
210 engine.isDirty = true;
211 engine.needsFullRebake = true;
212 page->isModified = true;
213 session.NotifyPageModified(page);
214 };
215 }
216 actions.push_back(std::move(bgAct));
217 }
218
219 // Action: Copy (order: 280)
220 {
221 ContextMenuItem act;
222 act.label = "Copy";
223 act.shortcut = "Ctrl+C";
224 act.icon = "📋";
225 act.iconKey = "copy";
226 act.order = 280;
227 act.isSeparatorBefore = true;
228 act.onTrigger = [obj, &session]() {
229 auto selected = session.GetSelectedObjects();
230 if (selected.empty()) {
231 selected = { obj };
232 }
233 ClipboardManager::Instance().CopyObjects(selected);
234 };
235 actions.push_back(std::move(act));
236 }
237
238 // Action: Cut (order: 290)
239 {
240 ContextMenuItem act;
241 act.label = "Cut";
242 act.shortcut = "Ctrl+X";
243 act.icon = "✂";
244 act.iconKey = "cut";
245 act.order = 290;
246 act.onTrigger = [obj, &session, &engine]() {
247 auto selected = session.GetSelectedObjects();
248 if (selected.empty()) {
249 selected = { obj };
250 }
251 ClipboardManager::Instance().CutObjects(selected, session);
252 engine.selectionGizmo.ClearSelection();
253 engine.isDirty = true;
254 engine.needsFullRebake = true;
255 };
256 actions.push_back(std::move(act));
257 }
258
259 // Action: Duplicate (order: 300)
260 {
261 ContextMenuItem act;
262 act.label = "Duplicate";
263 act.shortcut = "Ctrl+D";
264 act.icon = "📄";
265 act.iconKey = "duplicate";
266 act.order = 300;
267 act.onTrigger = [&session, &engine]() {
268 session.DuplicateSelection(10.0);
269 engine.isDirty = true;
270 engine.needsFullRebake = true;
271 };
272 actions.push_back(std::move(act));
273 }
274
275 // Attach undo/redo and dirty hooks for specialized objects
276 if (obj->type == ObjectType::AttachmentFile) {
277 auto attach = std::static_pointer_cast<AttachmentObject>(obj);
278 attach->onRelinkCallback = [page, &session, &engine, uid = attach->uid](
279 const std::string& oldPath, const std::string& newPath,
280 const std::string& oldName, const std::string& newName
281 ) {
282 session.RecordHistoryCommand(page, std::make_unique<Folio::RelinkAttachmentCommand>(
283 uid, oldPath, newPath, oldName, newName
284 ));
285 page->isModified = true;
286 engine.isDirty = true;
287 engine.needsFullRebake = true;
288 };
289 }
290
291 if (obj->type == ObjectType::Image) {
292 auto img = std::static_pointer_cast<ImageObject>(obj);
293 img->onVisualStateChanged = [img, page, &session, &engine]() {
294 page->UpdateObject(img);
295 if (img->isSelected) {
296 engine.selectionGizmo.RecalculateBounds();
297 }
298 engine.isDirty = true;
299 engine.needsFullRebake = true;
300 session.NotifyPageModified(page);
301 };
302 }
303
304 // =========================================================================
305 // 2. VIRTUAL HOOK: Allow object subclass to customize domain-specific actions
306 // =========================================================================
307 obj->CustomizeActions(actions);
308
309 // =========================================================================
310 // 3. DETERMINISTIC SORTING
311 // =========================================================================
312 std::stable_sort(actions.begin(), actions.end(), [](const ContextMenuItem& a, const ContextMenuItem& b) {
313 return a.order < b.order;
314 });
315
316 return actions;
317}
