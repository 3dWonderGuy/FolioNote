18/**
19 * @brief Recursively flattens an action hierarchy into a flat list of executable actions.
20 * @param source Source action tree.
21 * @param[out] outFlat Flattened list of leaf/executable actions.
22 */
23void CollectExecutableActions(const std::vector<UIAction>& source, std::vector<UIAction>& outFlat) {
24 for (const auto& act : source) {
25 if (!act.IsVisible()) {
26 continue;
27 }
28
29 if (act.HasSubActions()) {
30 CollectExecutableActions(act.subActions, outFlat);
31 } else {
32 outFlat.push_back(act);
33 }
34 }
35}

60bool ActionCommandPalette::SubstringMatch(const std::string& str, const std::string& query) {
61 if (query.empty()) return true;
62 if (str.empty()) return false;
63
64 auto it = std::search(
65 str.begin(), str.end(),
66 query.begin(), query.end(),
67 [](char ch1, char ch2) {
68 return std::tolower(static_cast<unsigned char>(ch1)) ==
69 std::tolower(static_cast<unsigned char>(ch2));
70 }
71 );
72 return it != str.end();
73}
