88 /**
89 * @brief Multiplatform stack-allocated timestamp generator formatted as "HH:MM:SS.mmm".
90 * Uses thread-safe platform variants (localtime_s on Windows, localtime_r on Linux/macOS/Android).
91 *
92 * @return Formatted timestamp string by value.
93 */
94 inline std::string GetCurrentTimestamp() {
95 auto now = std::chrono::system_clock::now();
96 auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(now.time_since_epoch()) % 1000;
97 auto timer = std::chrono::system_clock::to_time_t(now);
98 std::tm bt{};
99
100#if defined(_WIN32)
101 localtime_s(&bt, &timer);
102#else
103 localtime_r(&timer, &bt);
104#endif
105
106 char buf[16];
107 std::snprintf(buf, sizeof(buf), "%02d:%02d:%02d.%03d",
108 bt.tm_hour, bt.tm_min, bt.tm_sec, static_cast<int>(ms.count()));
109 return std::string(buf, 12);
110 }

143 /**
144 * @brief Central multiplatform logging dispatch function.
145 *
146 * Working Process:
147 * 1. Checks verbose filter: if level is INFO and verbose logging is disabled, drops immediately.
148 * 2. Formats timestamp and severity string ("INFO", "WARN", "ERROR").
149 * 3. Console / Terminal Output:
150 * - On Windows/Linux/macOS: Writes to std::cout or std::cerr (flushed on error).
151 * - On Android: Forwarded to SDL_Log for native LogCat capture.
152 * 4. Disk File Persistence (FileLogger):
153 * - Every entry (including INFO) is written to `folionote_session.log`.
154 * - Buffered I/O: INFO writes to an OS memory buffer (~1-2 microseconds) without blocking the thread.
155 * - ERROR entries trigger an immediate physical disk flush to guarantee persistence across crashes.
156 * 5. Real-Time Streaming: Broadcasts structured entry to any registered CLI/IPC listeners.
157 */
158 inline void LogConsole(LogLevel level, LogSource source, const std::string& msg) {
159 if (level == LogLevel::Info && !enableVerboseLogging) {
160 return;
161 }
162
163 std::string timestamp = GetCurrentTimestamp();
164
165 const char* levelStr = "INFO";
166 if (level == LogLevel::Error) {
167 levelStr = "ERROR";
168 } else if (level == LogLevel::Warn) {
169 levelStr = "WARN";
170 }
171
172 const char* sourceStr = LogSourceToString(source);
173
174 // Terminal / Console output
175 {
176 std::lock_guard<std::mutex> lock(GetConsoleLogMutex());
177#if defined(__ANDROID__)
178 // On Android, std::cout is redirected to /dev/null by default; route to logcat via SDL
179 if (level == LogLevel::Error) {
180 SDL_LogError(SDL_LOG_CATEGORY_APPLICATION, "[%s] [%s] %s", sourceStr, levelStr, msg.c_str());
181 } else if (level == LogLevel::Warn) {
182 SDL_LogWarn(SDL_LOG_CATEGORY_APPLICATION, "[%s] [%s] %s", sourceStr, levelStr, msg.c_str());
183 } else {
184 SDL_LogInfo(SDL_LOG_CATEGORY_APPLICATION, "[%s] [%s] %s", sourceStr, levelStr, msg.c_str());
185 }
186#else
187 std::ostream& out = (level == LogLevel::Error) ? std::cerr : std::cout;
188 out << '[' << timestamp << "] [" << levelStr << "] [" << sourceStr << "] " << msg << '\n';
189 out.flush();
190#endif
191 }
192
193 // Structured entry for disk and CLI streaming
194 FolioLogEntry entry{timestamp, levelStr, sourceStr, msg};
195
196 // Local file backup (writes INFO, WARN, and ERROR to folionote_session.log)
197 ::Folio::FileLogger::Instance().WriteLog(entry);
198
199 // Broadcast to CLI listener if registered
200 {
201 std::lock_guard<std::mutex> lock(GetListenerMutex());
202 for (const auto& listener : GetStreamListeners()) {
203 if (listener) {
204 listener(entry);
205 }
206 }
207 }
208 }
