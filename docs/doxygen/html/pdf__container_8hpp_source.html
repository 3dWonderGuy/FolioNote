59 /**
60 * @brief Ensures that the underlying PDF page raster surface is decoded and cached in memory.
61 *
62 * --- RESOLUTION WORKFLOW ---
63 * 1. Checks `resolvedDiskPath` first (direct absolute path on disk).
64 * 2. If empty or missing, falls back to `pdfPath`.
65 * 3. If missing and `fallbackDir` (e.g. notebook package root) is supplied, attempts
66 * resolving relative path via FileManager::JoinPath(fallbackDir, pdfPath).
67 * 4. Once verified, invokes Folio::PdfRenderer::RenderPage to decode the vector page.
68 *
69 * @param fallbackDir Optional directory (such as notebook root folder) to locate relative assets.
70 */
71 void EnsurePageLoaded(const std::string& fallbackDir = "") const {
72 if (isLoaded) return;
73 if (loadAttempted && cachedPageImage.is_empty() && fallbackDir.empty()) return;
74 loadAttempted = true;
75
76 std::string diskPath = resolvedDiskPath;
77 if (diskPath.empty()) {
78 diskPath = pdfPath;
79 }
80
81 // 1. If direct path fails, attempt fallback directory resolution
82 if (!diskPath.empty() && !FileManager::Exists(diskPath) && !fallbackDir.empty()) {
83 std::string resolved = FileManager::JoinPath(fallbackDir, pdfPath);
84 if (FileManager::Exists(resolved)) {
85 diskPath = resolved;
86 const_cast<PdfContainer*>(this)->resolvedDiskPath = diskPath;
87 }
88 }
89
90 // 2. Final existence check before pushing to renderer
91 if (diskPath.empty() || !FileManager::Exists(diskPath)) {
92 return;
93 }
94
95 // 3. Render and cache
96 auto res = Folio::PdfRenderer::RenderPage(diskPath, pageIndex, 150.0);
97 if (res.success && !res.image.is_empty()) {
98 cachedPageImage = res.image;
99 isLoaded = true;
100 if (res.widthMm > 10.0 && res.heightMm > 10.0) {
101 auto* mutableThis = const_cast<PdfContainer*>(this);
102 mutableThis->worldWidth = res.widthMm;
103 mutableThis->worldHeight = res.heightMm;
104 mutableThis->UpdateBounds();
105 }
106 }
107 }

134 bool HitTest(double queryWorldX, double queryWorldY) const override {
135 if (!isVisible || !isSelectable) return false;
136 if (isBackground) return false; // In background mode, let clicks pass through to canvas
137 if (!bounds.Contains(queryWorldX, queryWorldY)) return false;
138
139 // Fast-path: already selected object allows immediate interaction inside bounds
140 if (isSelected) return true;
141
142 BLMatrix2D inv;
143 BLMatrix2D::invert(inv, transform);
144 BLPoint local = inv.map_point(queryWorldX, queryWorldY);
145
146 double minX = std::min(worldX, worldX + worldWidth);
147 double maxX = std::max(worldX, worldX + worldWidth);
148 double minY = std::min(worldY, worldY + worldHeight);
149 double maxY = std::max(worldY, worldY + worldHeight);
150
151 return (local.x >= minX && local.x <= maxX && local.y >= minY && local.y <= maxY);
152 }

154 /**
155 * @brief Circular proximity hit-test.
156 *
157 * @note When `isBackground` is true, the PDF container acts as an immutable paper backdrop;
158 * all selection probes and clicks are rejected so freehand drawing and lasso selection
159 * pass freely over the page without selecting or moving it.

162 if (!isVisible || !isSelectable) return false;

168 void Render(BLContext& ctx, const Viewport& viewport) const override {
169 if (!isVisible || opacity <= 0.0f) return;
170 if (!bounds.Intersects(viewport.bounds)) return;
171
172 EnsurePageLoaded();
173
174 ctx.save();
175 ctx.apply_transform(transform);
176
177 double rx = std::min(worldX, worldX + worldWidth);
178 double ry = std::min(worldY, worldY + worldHeight);
179 double rw = std::abs(worldWidth);
180 double rh = std::abs(worldHeight);
181
182 // 1. Drop shadow behind the page
183 ctx.set_fill_style(BLRgba32(0x00, 0x00, 0x00, 0x22));
184 ctx.fill_round_rect(rx + 1.5, ry + 2.0, rw, rh, 2.0);
185
186 // 2. White Paper Surface
187 ctx.set_fill_style(BLRgba32(0xFF, 0xFF, 0xFF, 0xFF));
188 ctx.fill_round_rect(rx, ry, rw, rh, 1.5);
189
190 // 3. Render cached page surface if available
191 if (isLoaded && !cachedPageImage.is_empty()) {
192 BLRect dst(rx, ry, rw, rh);
193 ctx.blit_image(dst, cachedPageImage);
194 } else {
195 // High-fidelity CAD / architectural vector placeholder
196 // Subtle page border
197 ctx.set_stroke_style(BLRgba32(0xCC, 0xD1, 0xD9, 0xFF));
198 ctx.set_stroke_width(0.6);
199 ctx.stroke_round_rect(rx, ry, rw, rh, 1.5);
200
201 // Red PDF Banner / Header
202 ctx.set_fill_style(BLRgba32(0xEA, 0x43, 0x35, 0xEE)); // Adobe/Google Red
203 ctx.fill_round_rect(rx + 8.0, ry + 8.0, 42.0, 16.0, 3.0);
204
205 // Document Header line placeholder
206 ctx.set_fill_style(BLRgba32(0x5F, 0x63, 0x68, 0xCC));
207 ctx.fill_round_rect(rx + 56.0, ry + 12.0, std::min(rw - 70.0, 120.0), 8.0, 2.0);
208
209 // Watermark grid / document lines
210 ctx.set_stroke_style(BLRgba32(0xE8, 0xEA, 0xED, 0xBB));
211 ctx.set_stroke_width(0.5);
212 double lineY = ry + 40.0;
213 while (lineY < ry + rh - 20.0) {
214 ctx.stroke_line(rx + 12.0, lineY, rx + rw - 12.0, lineY);
215 lineY += 12.0;
216 }
217
218 if (isBackground) {
219 ctx.set_fill_style(BLRgba32(0x1A, 0x73, 0xE8, 0x33));
220 ctx.fill_round_rect(rx + 8.0, ry + rh - 24.0, 90.0, 14.0, 2.0);
221 }
222 }
223
224 ctx.restore();
225 }
