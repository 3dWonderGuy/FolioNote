2/**
3 * =========================================================================================
4 * @file canvas_command.hpp
5 * @brief Command interfaces and concrete action records for the Undo/Redo subsystem
6 * =========================================================================================
7 *
8 * ARCHITECTURAL DESIGN:
9 * Follows the Command Pattern to encapsulate canvas mutations into discrete, reversible units.
10 * Supports:
11 * 1. Single Object Insertion: Pen strokes, images, text boxes, shapes.
12 * 2. Multi-Object Deletion: Delete key, cut, selection removal.
13 * 3. Compound Continuous Eraser Actions: Groups multiple erased strokes and sliced fragments
14 * from a single continuous pen gesture into one single undoable action.
15 * 4. Multi-Object Transformation: Gizmo moving, scaling, and rotating.
16 */

31/**
32 * @class ICanvasCommand
33 * @brief Abstract base class for all reversible canvas mutations.
34 */
35class ICanvasCommand {
36public:
37 virtual ~ICanvasCommand() = default;
38
39 /// Apply or re-apply the mutation to the target page
40 virtual void Execute(CanvasPage& page, CanvasEngine* engine = nullptr) = 0;
41
42 /// Reverse the mutation on the target page
43 virtual void Undo(CanvasPage& page, CanvasEngine* engine = nullptr) = 0;
44
45 /// Bounding box in world coordinates affected by this command
46 virtual AABB GetTargetBounds() const = 0;
47
48 /// Human-readable label for debugging and history inspection
49 virtual std::string GetName() const = 0;
50};
