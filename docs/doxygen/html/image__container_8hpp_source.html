2/**
3 * =========================================================================================
4 * @file core/objects/media/images/image_container.hpp
5 * @brief Canvas Object Container Representing Placed Raster Images on the Infinite Canvas
6 * =========================================================================================
7 *
8 * ARCHITECTURAL DESIGN & UNIFIED COORDINATOR:
9 * -------------------------------------------
10 * ImageObject serves as the sole, high-level coordinator for image loading, decoding,
11 * format deduction, animation frame unpacking, and physical millimeter projection.
12 *
13 * Callers and import pipelines simply save the raw payload to the notebook package sidecar
14 * (e.g. "imports/images/<hash>.ext"), instantiate an ImageObject, and call:
15 * - LoadFromSource(relativePath, dpi) for files on disk, or
16 * - LoadFromMemory(data, size, filenameHint, dpi) for clipboard paste / embedded payloads.
17 *
18 * All codec delegation (Blend2D, SDL_image, LunaSVG), multi-frame extraction (GIF / animated WebP),
19 * RAM safety clamping, aspect-locked physical millimeter sizing, and AABB computation
20 * are handled internally.
21 */

180 * currentFrameIndex = (currentFrameIndex + 1) % frameCount
181 * cachedBlImage = frames[currentFrameIndex].image
182 * lastFrameTickMs = nowMs
183 * return true // Signal canvas rebake needed
184 *
185 * @param[in] nowMs Current application time in milliseconds (e.g. from SDL_GetTicks()).
186 * @return True if the frame was advanced (requiring static canvas rebake); false if unchanged.

245 /**
246 * @brief Converts native pixel resolution into world millimeters via screen DPI,
247 * enforcing hard maximum and minimum constraints while strictly preserving aspect ratio.
248 *
249 * Mathematical Derivation:
250 * mmPerPixel = 25.4 / effectiveDpi
251 * rawW = naturalWidth * mmPerPixel
252 * rawH = naturalHeight * mmPerPixel
253 *
254 * maxDimension = max(rawW, rawH)
255 * if maxDimension > s_maxDimensionLimitMm:
256 * downscale = s_maxDimensionLimitMm / maxDimension
257 * rawW *= downscale, rawH *= downscale
258 *
259 * minDimension = min(rawW, rawH)
260 * if minDimension < s_minDimensionLimitMm:
261 * upscale = s_minDimensionLimitMm / minDimension
262 * rawW *= upscale, rawH *= upscale
263 *
264 * @param[in] dpi Target display density in dots-per-inch (defaults to 96.0 fallback if <= 0).
