25std::string SystemDialogs::ShowOpenFileDialog(const std::string& title) {
26#if defined(_WIN32)
27 // 32KB dynamic wide character buffer prevents MAX_PATH buffer overflows on deep hierarchies

29 OPENFILENAMEW ofn = {};
30 ofn.lStructSize = sizeof(ofn);
31 ofn.lpstrFile = fileBuf.data();
32 ofn.nMaxFile = static_cast<DWORD>(fileBuf.size());
33 ofn.lpstrFilter = L"All Files (*.*)\0*.*\0\0";
34 ofn.Flags = OFN_FILEMUSTEXIST | OFN_PATHMUSTEXIST | OFN_EXPLORER;
35
36 std::filesystem::path nativeTitle = PathUtils::Utf8ToNativePath(title);
37 if (!title.empty()) {
38 ofn.lpstrTitle = nativeTitle.c_str();
39 }
40
41 if (GetOpenFileNameW(&ofn)) {
42 return PathUtils::NormalizeSeparators(PathUtils::NativePathToUtf8(fileBuf.data()));
43 }
44 return "";
45#elif defined(__APPLE__)
46 // Native macOS file picker via AppleScript NSOpenPanel modal dialog
47 std::string prompt = title.empty() ? "Select File" : title;
48 std::string cmd = "osascript -e 'POSIX path of (choose file with prompt \"" + prompt + "\")' 2>/dev/null";
49 FILE* pipe = popen(cmd.c_str(), "r");
50 if (!pipe) return "";
51 char buf[2048] = {};
52 std::string result;
53 if (fgets(buf, sizeof(buf), pipe)) {
54 result = buf;
55 while (!result.empty() && (result.back() == '\n' || result.back() == '\r')) {
56 result.pop_back();

58 }

61#elif defined(__linux__) && !defined(__ANDROID__)
62 // Native Linux file picker via standard desktop portal (Zenity or KDialog)
63 std::string cmd;
64 if (std::system("which zenity >/dev/null 2>&1") == 0) {
65 cmd = "zenity --file-selection --title=\"" + (title.empty() ? "Select File" : title) + "\" 2>/dev/null";

159bool SystemDialogs::OpenWithDefaultApp(const std::string& pathOrUrl) {
160 if (pathOrUrl.empty()) {
161 LOG_WARN_CODE(FileManager, FolioErrorCode::SysFileNotFound, "OpenWithDefaultApp rejected: path/URL string is empty.");
162 return false;
163 }
164
165 std::string targetUri = PathUtils::PathToFileUri(pathOrUrl);
166
167 // Cross-platform dispatch via SDL3 (Windows, macOS, Linux, Android)
168 if (SDL_OpenURL(targetUri.c_str())) {
169 return true;
170 }
171
172 LOG_WARN(FileManager, "SDL_OpenURL failed for (" + targetUri + "): " + SDL_GetError());
173
174#if defined(_WIN32)
175 // Native Win32 fallback via ShellExecuteW
176 std::filesystem::path nativeP = PathUtils::Utf8ToNativePath(pathOrUrl);
177 HINSTANCE res = ShellExecuteW(nullptr, L"open", nativeP.c_str(), nullptr, nullptr, SW_SHOWNORMAL);
178 if (reinterpret_cast<INT_PTR>(res) > 32) {
179 return true;
180 }
181 LOG_ERROR_CODE(FileManager, FolioErrorCode::SysFileAccessDenied, "ShellExecuteW fallback also failed for: " + pathOrUrl);
182#else
183 LOG_ERROR_CODE(FileManager, FolioErrorCode::SysFileAccessDenied, "Failed to launch default application for: " + pathOrUrl);
184#endif
185
186 return false;
187}
