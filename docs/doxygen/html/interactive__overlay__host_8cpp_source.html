14// COORDINATE PROJECTION
15// =============================================================================
16
17/**
18 * Mathematical Projection from World Millimeters to Screen Pixels:
19 *
20 * In FolioNote, the camera viewport defined by vp.bounds stores the visible
21 * window's top-left coordinates in world space (minX, minY):
22 * cameraOriginX = vp.bounds.minX
23 * cameraOriginY = vp.bounds.minY
24 *
25 * Screen resolution density:
26 * scale = (vp.pixelsPerMm > 0.0) ? (vp.pixelsPerMm * vp.zoom) : vp.zoom;
27 *
28 * Screen projection formulas:
29 * screenX = round((obj.worldX - cameraOriginX) * scale)
30 * screenY = round((obj.worldY - cameraOriginY) * scale)
31 * screenWidth = round(obj.worldWidth * scale)
32 * screenHeight = round(obj.worldHeight * scale)
33 *
34 * @param obj CanvasObject providing world physical position and dimensions.
35 * @param vp Viewport providing camera bounds and zoom factor.
36 * @return Screen-space integer pixel rectangle for overlay alignment.
37 */
38OverlayRect InteractiveOverlayHost::ComputeScreenRect(const CanvasObject& obj, const Viewport& vp) noexcept {
39 const double scale = (vp.pixelsPerMm > 0.0) ? (vp.pixelsPerMm * vp.zoom) : vp.zoom;
40 const double cameraOriginX = vp.bounds.minX;
41 const double cameraOriginY = vp.bounds.minY;
42
43 const double screenX = (obj.worldX - cameraOriginX) * scale;
44 const double screenY = (obj.worldY - cameraOriginY) * scale;
45 const double screenW = obj.worldWidth * scale;
46 const double screenH = obj.worldHeight * scale;
47
48 return OverlayRect{
49 static_cast<int>(std::round(screenX)),
50 static_cast<int>(std::round(screenY)),
51 static_cast<int>(std::round(std::max(0.0, screenW))),
52 static_cast<int>(std::round(std::max(0.0, screenH)))
53 };
54}

60void InteractiveOverlayHost::UpdateActiveOverlays(uint64_t nowMs, double deltaSec,
61 const std::vector<CanvasObject*>& visibleObjects)
62{
63 /**
64 * Working Process:
65 * Frustum-culled visibility check: only objects within the current camera frustum
66 * that own an active Layer 3 overlay subsystem receive frame tick updates.
67 */
68 for (CanvasObject* obj : visibleObjects) {
69 if (!obj || !obj->isVisible || !obj->HasLiveOverlay()) {
70 continue;
71 }
72
73 IInteractiveOverlay* overlay = obj->GetOverlay();
74 if (overlay) {
75 overlay->OnUpdate(nowMs, deltaSec);
76 }
77 }

99void InteractiveOverlayHost::RenderOverlays(BLContext& screenCtx, const Viewport& vp,
100 const std::vector<CanvasObject*>& visibleObjects)
101{
102 /**
103 * Layer 3 Live Presentation Pass:
104 * Renders active overlays onto the screen composite context.
105 * Screen coordinates are projected using ComputeScreenRect(vp), ensuring
106 * pixel-perfect registration over the Layer 1 static placeholder card.
107 */
108 for (CanvasObject* obj : visibleObjects) {
109 if (!obj || !obj->isVisible || !obj->HasLiveOverlay()) {
110 continue;
111 }
112
113 const OverlayRect screenRect = ComputeScreenRect(*obj, vp);
114 if (screenRect.IsEmpty()) {
115 continue;
116 }
117
118 IInteractiveOverlay* overlay = obj->GetOverlay();
119 if (overlay) {
120 overlay->OnRenderOverlay(screenCtx, screenRect);
121 }
122 }
123}

149bool InteractiveOverlayHost::HandleInput(const SDL_Event& event, const Viewport& vp,
150 CanvasObject* activeObject)
151{
152 if (!activeObject || !activeObject->HasLiveOverlay()) {
153 return false;
154 }
155
156 if (!activeObject->IsInteracting()) {
157 return false;
158 }
159
160 IInteractiveOverlay* overlay = activeObject->GetOverlay();
161 if (!overlay) {
162 return false;
163 }
164
165 const OverlayRect screenRect = ComputeScreenRect(*activeObject, vp);
166 return overlay->OnInputEvent(event, screenRect);
167}
