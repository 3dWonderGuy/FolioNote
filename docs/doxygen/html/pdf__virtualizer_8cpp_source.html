102 if (oldDocY <= curPageTopDocY + pageH_px + (PAGE_GAP_PX * 0.5f) || i == totalPages - 1) {
103 anchorPageIdx = i;
104 anchorLocalMmY = std::clamp(static_cast<double>(oldDocY - curPageTopDocY) / oldPxPerMm, 0.0, mmH);

106 }

108 curPageTopDocY += pageH_px + PAGE_GAP_PX;
109 }

126 scrollY = std::clamp(newScrollY, 0.0f, newMaxScrollY);
127
128 // 2. HORIZONTAL ANCHORING
129 float newMaxScrollX = std::max(0.0f, newMaxDocW_px + 30.0f - contentW);
130 float oldMaxScrollX = std::max(0.0f, oldMaxDocW_px + 30.0f - contentW);
131

133 scrollX = 0.0f;
134 } else {

136 float newScrollX = 0.0f;
137

139 float oldDocX = scrollX + anchorRelX;
140 float scalingX = oldDocX - 15.0f;
141 float newDocX = 15.0f + scalingX * (newZoom / oldZoom);
142 newScrollX = newDocX - anchorRelX;
143 } else {
144 float oldPageScreenLeft = contentX + (contentW - oldMaxDocW_px) * 0.5f;
145 float offsetFromPageLeft = anchorScreenX - oldPageScreenLeft;
146 float newOffsetFromPageLeft = offsetFromPageLeft * (newZoom / oldZoom);
147 newScrollX = 15.0f + newOffsetFromPageLeft - anchorRelX;
148 }
149

151 }

154 maxScrollX = newMaxScrollX;
155 maxScrollY = newMaxScrollY;
