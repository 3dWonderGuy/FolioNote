1/**
2 * =========================================================================================
3 * @file sheet_tiler.cpp
4 * @brief Implementation of RNote-Style Standard Sheet Tiling & Spatial Grid Slicing
5 * =========================================================================================
6 *
7 * ARCHITECTURAL IMPLEMENTATION DETAILS:
8 * 1. Physical Paper Dimension Resolution:
9 * Resolves ISO and ANSI standard paper sizes (A4, US Letter, A3, A5) accounting for
10 * portrait vs. landscape orientation.
11 * 2. Automatic Width Fit (Aspect-Ratio Preserved Zoom):
12 * Calculates $W_{\text{content}} = X_{\max} - X_{\min}$. When drawn space exceeds the
13 * physical sheet width, computes uniform scaling factor $s = W_{\text{sheet}} / W_{\text{content}}$.
14 * Virtual sheet dimensions in canvas coordinates become $V_W = W_{\text{sheet}} / s$ and
15 * $V_H = H_{\text{sheet}} / s$, ensuring the drawing fits standard paper with zero distortion.
16 * 3. Dual Traversal Slicing:
17 * - Column-Major: Column by column (top-to-bottom, advancing left-to-right).
18 * - Row-Major: Row by row (left-to-right, advancing top-to-bottom).
19 * 4. Spatial Binning & Bounding Intersection:
20 * Each sheet tile filters canvas objects whose AABBs intersect the tile's world rectangle.
21 */

32void SheetTiler::GetStandardSheetDimensions(const CanvasPage& page, double& outWidthMm, double& outHeightMm) {
33 double w = 210.0;
34 double h = 297.0;
35
36 switch (page.pageSizeFormat) {
37 case PageSizeFormat::Letter:
38 w = 215.9;
39 h = 279.4;
40 break;
41 case PageSizeFormat::A4:
42 w = 210.0;
43 h = 297.0;
44 break;
45 case PageSizeFormat::A3:
46 w = 297.0;
47 h = 420.0;
48 break;
49 case PageSizeFormat::A5:
50 w = 148.0;
51 h = 210.0;
52 break;
53 case PageSizeFormat::Custom:
54 w = page.pageWidthMm > 10.0 ? page.pageWidthMm : 210.0;
55 h = page.pageHeightMm > 10.0 ? page.pageHeightMm : 297.0;
56 break;
57 }
58
59 if (page.pageIsLandscape) {
60 std::swap(w, h);
61 }
62
63 outWidthMm = w;
64 outHeightMm = h;
65}

67AABB SheetTiler::ComputeContentAABB(const CanvasPage& page) {
68 if (page.objects.empty()) {
69 double w = 210.0, h = 297.0;
70 GetStandardSheetDimensions(page, w, h);
71 return AABB(0.0, 0.0, w, h);
72 }
73
74 bool hasValidAABB = false;
75 AABB totalAABB;
76
77 for (const auto& obj : page.objects) {
78 if (!obj) continue;
79 const AABB& box = obj->bounds;
80 if (box.Width() > 0.0 || box.Height() > 0.0) {
81 if (!hasValidAABB) {
82 totalAABB = box;
83 hasValidAABB = true;
84 } else {
85 totalAABB.Merge(box);
86 }
87 }
88 }
89
90 if (!hasValidAABB) {
91 double w = 210.0, h = 297.0;
92 GetStandardSheetDimensions(page, w, h);
93 return AABB(0.0, 0.0, w, h);
94 }
95
96 return totalAABB;
97}

99std::vector<SheetTile> SheetTiler::TilePage(
100 const CanvasPage& page,
101 SheetTraversalOrder traversalOrder,
102 PageScalingMode scalingMode,
103 bool skipEmptySheets
104) {
105 std::vector<SheetTile> tiles;
106
107 double sheetWidthMm = 210.0;
108 double sheetHeightMm = 297.0;
109 GetStandardSheetDimensions(page, sheetWidthMm, sheetHeightMm);
110
111 AABB contentAABB = ComputeContentAABB(page);
112
113 // Calculate uniform aspect-ratio scaling factor
114 double uniformScale = 1.0;
115 if (scalingMode == PageScalingMode::AutomaticWidthFit) {
116 double contentWidth = std::max(10.0, contentAABB.maxX - contentAABB.minX);
117 if (contentWidth > sheetWidthMm) {
118 uniformScale = sheetWidthMm / contentWidth;
119 }
120 }
121
122 // Virtual sheet dimensions in unscaled canvas millimeters
123 double virtualTileW = sheetWidthMm / uniformScale;
124 double virtualTileH = sheetHeightMm / uniformScale;
125
126 // Grid coordinates
127 int cMin = 0;
128 int cMax = 0;
129 int rMin = 0;
130 int rMax = 0;
131
132 if (!page.objects.empty()) {
133 cMin = std::max(0, static_cast<int>(std::floor(contentAABB.minX / virtualTileW)));
134 cMax = std::max(0, static_cast<int>(std::floor(contentAABB.maxX / virtualTileW)));
135 rMin = std::max(0, static_cast<int>(std::floor(contentAABB.minY / virtualTileH)));
136 rMax = std::max(0, static_cast<int>(std::floor(contentAABB.maxY / virtualTileH)));
137 }
138
139 // Traversal Order:
140 // 1. ColumnMajor_TopBottom_LeftRight: Print each column top-to-bottom, advancing left-to-right
141 // 2. RowMajor_LeftRight_TopBottom: Print each row left-to-right, advancing top-to-bottom
142 std::vector<std::pair<int, int>> gridSequence;
143
144 if (traversalOrder == SheetTraversalOrder::ColumnMajor_TopBottom_LeftRight) {
145 for (int c = cMin; c <= cMax; ++c) {
146 for (int r = rMin; r <= rMax; ++r) {
147 gridSequence.emplace_back(c, r);
148 }
149 }
150 } else {
151 for (int r = rMin; r <= rMax; ++r) {
152 for (int c = cMin; c <= cMax; ++c) {
153 gridSequence.emplace_back(c, r);
154 }
155 }
156 }
157
158 size_t sequenceIndex = 0;
159
160 for (const auto& [c, r] : gridSequence) {
161 double tileWorldX = static_cast<double>(c) * virtualTileW;
162 double tileWorldY = static_cast<double>(r) * virtualTileH;
163 AABB tileAABB(tileWorldX, tileWorldY, tileWorldX + virtualTileW, tileWorldY + virtualTileH);
164
165 // Spatial binning: discover objects intersecting this sheet tile
166 std::vector<std::shared_ptr<CanvasObject>> intersecting;
167 for (const auto& obj : page.objects) {
168 if (!obj) continue;
169 if (tileAABB.Intersects(obj->bounds)) {
170 intersecting.push_back(obj);
171 }
172 }
173
174 bool hasContent = !intersecting.empty();
175
176 // If skipEmptySheets is true, skip tiles with 0 intersecting objects (unless grid is only 1 sheet)
177 if (skipEmptySheets && !hasContent && gridSequence.size() > 1) {
178 continue;
179 }
180
181 SheetTile tile;
182 tile.columnIndex = c;
183 tile.rowIndex = r;
184 tile.sequenceIndex = sequenceIndex++;
185 tile.worldX = tileWorldX;
186 tile.worldY = tileWorldY;
187 tile.widthMm = sheetWidthMm;
188 tile.heightMm = sheetHeightMm;
189 tile.uniformScale = uniformScale;
190 tile.hasContent = hasContent;
191 tile.sheetAnchorId = "sheet_" + page.guid + "_c" + std::to_string(c) + "_r" + std::to_string(r);
192 tile.intersectingObjects = std::move(intersecting);
193
194 tiles.push_back(std::move(tile));
195 }
196
197 // Safety fallback: guarantee at least one sheet tile
198 if (tiles.empty()) {
199 SheetTile fallback;
200 fallback.columnIndex = 0;
201 fallback.rowIndex = 0;
202 fallback.sequenceIndex = 0;
203 fallback.worldX = 0.0;
204 fallback.worldY = 0.0;
205 fallback.widthMm = sheetWidthMm;
206 fallback.heightMm = sheetHeightMm;
207 fallback.uniformScale = 1.0;
208 fallback.hasContent = false;
209 fallback.sheetAnchorId = "sheet_" + page.guid + "_c0_r0";
210 tiles.push_back(std::move(fallback));
211 }
212
213 return tiles;
214}
