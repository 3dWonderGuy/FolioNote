350void MdEditorView::RebakePreviewSurface(int width, int height, bool invert) {
351 if (width <= 0 || height <= 0) return;
352
353 InitGL();
354
355 surfaceW = width;
356 surfaceH = height;
357
358 if (surfaceW > allocatedCapacityW || surfaceH > allocatedCapacityH) {
359 allocatedCapacityW = std::max(allocatedCapacityW * 2, surfaceW);
360 allocatedCapacityH = std::max(allocatedCapacityH * 2, surfaceH);
361
362 previewSurface.create(allocatedCapacityW, allocatedCapacityH, BL_FORMAT_PRGB32);
363
364 glBindTexture(GL_TEXTURE_2D, glTexture);
365 glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA8, allocatedCapacityW, allocatedCapacityH, 0, GL_RGBA, GL_UNSIGNED_BYTE, nullptr);
366 }
367
368 // Rasterize Paper via Blend2D
369 BLContext ctx(previewSurface);
370 ctx.clear_all();
371
372 // Background backdrop
373 BLRgba32 backdrop = invert ? BLRgba32(0x10, 0x12, 0x16, 0xFF) : BLRgba32(0xEB, 0xF0, 0xF5, 0xFF);
374 ctx.fill_rect(BLRect(0, 0, surfaceW, surfaceH), backdrop);
375
376 // Calculate Physical Paper placement in mm
377 double pxPerMm = (96.0 / 25.4) * previewZoom;
378 double pageWMm = engine.config.pageWidthMm;
379 double pageWPx = pageWMm * pxPerMm;
380
381 // Horizontally center paper in preview pane
382 double originX_mm = ((surfaceW - pageWPx) * 0.5) / pxPerMm;
383 originX_mm = std::max(originX_mm, 5.0 / pxPerMm);
384
385 // Vertical millimeter offset from scroll
386 double originY_mm = 10.0 - (previewScrollY / pxPerMm);
387
388 ctx.save();
389 ctx.scale(pxPerMm, pxPerMm);
390
391 // Tier 1: Committed paper & typography
392 engine.RenderToBlend2D(ctx, originX_mm, originY_mm, pageWMm, engine.totalDocumentHeightMm, invert);
393
394 // Tier 2: Live interaction overlay (selection highlights & blinking caret)
395 double currentSec = SDL_GetTicks() / 1000.0;
396 engine.RenderLiveLayer(ctx, originX_mm, originY_mm, currentSec);
397
398 ctx.restore();
399 ctx.end();
400
401 // Upload to OpenGL Texture
402 BLImageData imgData;
403 previewSurface.get_data(&imgData);
404 glBindTexture(GL_TEXTURE_2D, glTexture);
405
406#if defined(__ANDROID__)
407 glPixelStorei(GL_UNPACK_ALIGNMENT, 1);
408 glTexSubImage2D(GL_TEXTURE_2D, 0, 0, 0, surfaceW, surfaceH, GL_RGBA, GL_UNSIGNED_BYTE, imgData.pixel_data);
409 glPixelStorei(GL_UNPACK_ALIGNMENT, 4);
410#else
411 glPixelStorei(GL_UNPACK_ALIGNMENT, 1);
412 glPixelStorei(GL_UNPACK_ROW_LENGTH, static_cast<GLint>(imgData.stride / 4));
413 glTexSubImage2D(GL_TEXTURE_2D, 0, 0, 0, surfaceW, surfaceH, GL_BGRA, GL_UNSIGNED_BYTE, imgData.pixel_data);
414 glPixelStorei(GL_UNPACK_ROW_LENGTH, 0);
415 glPixelStorei(GL_UNPACK_ALIGNMENT, 4);
416#endif
417
418 needsRebake = false;
419}
