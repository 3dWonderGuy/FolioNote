1/**
2 * =========================================================================================
3 * @file input/stateMachine/pointer_icon_manager.hpp
4 * @brief Centralized Hardware & Canvas Pointer Cursor Management and Anti-Flicker Controller
5 * =========================================================================================
6 *
7 * ARCHITECTURAL DESIGN & PURPOSE:
8 * -------------------------------
9 * In desktop graphics engines (e.g., Photoshop, Illustrator, Figma, OneNote), cursor
10 * management requires strict statefulness and input capture:
11 *
12 * 1. Hardware Cursor Authority:
13 * Directly manages SDL3 hardware cursors (SDL_Cursor*) rather than relying exclusively
14 * on Dear ImGui's internal cursor stack. Dear ImGui resets cursor state at the start
15 * of each frame, which causes rapid alternating flicker if mouse-move events do not
16 * fire on every frame.
17 *
18 * 2. Drag & Interaction Locking (Anti-Flicker Mechanism):
19 * When resizing a gizmo handle, inking a stroke, or panning the viewport, the cursor
20 * is locked to a specific shape until the pointer is released (PointerUp). Transient
21 * hover tests and boundary crossings cannot override or flicker the cursor.
22 *
23 * 3. Handle Role Mapping:
24 * Polymorphically maps 8-point gizmo handles (TopLeft, RightCenter, Body, etc.) to
25 * appropriate bi-directional resize, move, or rotate hardware cursors.
26 *
27 * 4. Tool & Inking Reticles:
28 * Switches effortlessly between Precision Crosshair (Drawing/Shapes), Text I-Beam,
29 * Open/Grabbing Hand (Panning), and Hidden (Eraser with custom canvas reticle).
30 */

41/**

45enum class PointerCursorShape : uint8_t {
46 Arrow = 0, ///< Standard desktop pointer arrow
47 Crosshair, ///< Precision crosshair for inking, pen, and shape creation
48 TextInput, ///< Text I-beam for typing and text box editing
49 ResizeAll, ///< 4-way move cross (dragging selection body)
50 ResizeNS, ///< Vertical bi-directional resize (TopCenter, BottomCenter)
51 ResizeEW, ///< Horizontal bi-directional resize (LeftCenter, RightCenter)
52 ResizeNESW, ///< Diagonal bi-directional resize (TopRight, BottomLeft)
53 ResizeNWSE, ///< Diagonal bi-directional resize (TopLeft, BottomRight)
54 Hand, ///< Open hand (panning mode hover, clickable link)
55 Grabbing, ///< Closed grasping hand (active viewport pan drag)
56 NotAllowed, ///< Prohibited action
57 Hidden ///< Hardware cursor hidden (custom reticles, e.g. Eraser)
58};

60/**

64enum class CursorLockReason : uint8_t {

80class PointerIconManager {
81public:
82 PointerIconManager();
83 ~PointerIconManager();

85 // Prevent copying to protect raw SDL_Cursor* resource lifetimes
86 PointerIconManager(const PointerIconManager&) = delete;
87 PointerIconManager& operator=(const PointerIconManager&) = delete;
88
89 /**
90 * @brief Pre-allocates and caches all SDL system cursors.

92 void Initialize();
93
94 /**
95 * @brief Releases all allocated SDL cursor resources.
96 */
97 void Shutdown();

99 /**
100 * @brief Locks the cursor to a specific shape during modal dragging.
101 *
102 * While locked, transient hover updates are rejected to eliminate flicker.
103 *
104 * @param shape Desired cursor shape.
105 * @param reason The modal interaction taking ownership.
106 */
107 void Lock(PointerCursorShape shape, CursorLockReason reason) noexcept;
108
109 /**
110 * @brief Unlocks the cursor when a modal interaction completes.
111 *
112 * @param reason The modal interaction releasing ownership.
113 */
114 void Unlock(CursorLockReason reason) noexcept;
115
116 /**
117 * @brief Checks if a cursor lock is currently active.
118 */
119 [[nodiscard]] bool IsLocked() const noexcept { return currentLock != CursorLockReason::None; }
120
121 /**
122 * @brief Returns the reason for the current cursor lock.
123 */
124 [[nodiscard]] CursorLockReason GetCurrentLockReason() const noexcept { return currentLock; }
125
126 /**
127 * @brief Sets the transient hover cursor (e.g. hovering canvas, handles, or tools).
128 *
129 * Ignored if a cursor lock is currently active.
130 *
131 * @param shape Requested cursor shape.
132 */
133 void SetHoverShape(PointerCursorShape shape) noexcept;
134
135 /**
136 * @brief Helper to set hover cursor based on a gizmo HandleRole.

138 * @param role Manipulator handle role.
139 */
140 void SetHoverForHandleRole(HandleRole role) noexcept;
141
142 /**
143 * @brief Maps a HandleRole to its corresponding PointerCursorShape.
144 *
145 * @param role Manipulator handle role.
146 * @return PointerCursorShape Expected cursor shape.
147 */
148 [[nodiscard]] PointerCursorShape GetShapeForHandleRole(HandleRole role) const noexcept;
149
150 /**
151 * @brief Synchronizes and commits the cursor shape to SDL and ImGui for the current frame.
152 *
153 * GENERAL WORKING PROCESS:
154 * -------------------------
155 * 1. If ImGui UI chrome has focus and cursor is NOT locked by a canvas drag, delegates to
156 * ImGui's requested cursor.
157 * 2. If the cursor is locked (e.g. gizmo drag, inking), enforces the locked shape regardless
158 * of mouse position.
159 * 3. If over the canvas, enforces the active tool / hover shape.
160 * 4. Updates SDL_SetCursor only when the shape has actually changed to prevent OS IPC overhead.
161 *
162 * @param isOverCanvas Whether pointer is within the active canvas area.
163 * @param imguiHasFocus Whether ImGui UI chrome currently captures input.
164 */
165 void Apply(bool isOverCanvas, bool imguiHasFocus);
166
167 /**
168 * @brief Returns the currently active cursor shape.
169 */
170 [[nodiscard]] PointerCursorShape GetCurrentShape() const noexcept { return currentShape; }
171
172private:
173 void SetSdlCursor(PointerCursorShape shape);
174
175 PointerCursorShape currentShape = PointerCursorShape::Arrow;
176 PointerCursorShape hoverShape = PointerCursorShape::Arrow;
177 PointerCursorShape lockedShape = PointerCursorShape::Arrow;
178 CursorLockReason currentLock = CursorLockReason::None;
179
180 std::unordered_map<PointerCursorShape, SDL_Cursor*> cursors;
181 bool isInitialized = false;
182};
