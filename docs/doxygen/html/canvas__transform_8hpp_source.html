77 /**
78 * @brief Transforms an axis-aligned bounding box from world space (mm) to screen coordinates (px).
79 *
80 * MATHEMATICAL TRANSFORMATION:
81 * - Uses the 2D affine transformation matrix `worldToScreenMatrix`:
82 * [ x_s ] [ m00 m10 m20 ] [ x_w ]
83 * [ y_s ] = [ m01 m11 m21 ] [ y_w ]
84 * [ 1 ] [ 0 0 1 ] [ 1 ]
85 * - Projects all 4 corner vertices of `worldBox` to accurately enclose rotated/scaled geometry.
86 * - Evaluates extrema:
87 * x_min = min(x_0, x_1, x_2, x_3), x_max = max(x_0, x_1, x_2, x_3)
88 * y_min = min(y_0, y_1, y_2, y_3), y_max = max(y_0, y_1, y_2, y_3)
89 * - Returns screen rectangle: BLRect(x_min, y_min, x_max - x_min, y_max - y_min).
90 *
91 * @param worldBox Axis-aligned bounding box in world millimeters.
92 * @return BLRect Projected axis-aligned rectangle in screen pixels.
93 */
94 [[nodiscard]] BLRect WorldToScreenRect(const AABB& worldBox) const noexcept {
95 if (worldBox.IsEmpty() || !worldBox.IsFinite()) {
96 return BLRect(0.0, 0.0, 0.0, 0.0);
97 }
98
99 // Project all 4 corners through the affine matrix
100 const double px[4] = {
101 worldBox.minX * worldToScreenMatrix.m00 + worldBox.minY * worldToScreenMatrix.m10 + worldToScreenMatrix.m20,
102 worldBox.maxX * worldToScreenMatrix.m00 + worldBox.minY * worldToScreenMatrix.m10 + worldToScreenMatrix.m20,
103 worldBox.maxX * worldToScreenMatrix.m00 + worldBox.maxY * worldToScreenMatrix.m10 + worldToScreenMatrix.m20,
104 worldBox.minX * worldToScreenMatrix.m00 + worldBox.maxY * worldToScreenMatrix.m10 + worldToScreenMatrix.m20
105 };
106 const double py[4] = {
107 worldBox.minX * worldToScreenMatrix.m01 + worldBox.minY * worldToScreenMatrix.m11 + worldToScreenMatrix.m21,
108 worldBox.maxX * worldToScreenMatrix.m01 + worldBox.minY * worldToScreenMatrix.m11 + worldToScreenMatrix.m21,
109 worldBox.maxX * worldToScreenMatrix.m01 + worldBox.maxY * worldToScreenMatrix.m11 + worldToScreenMatrix.m21,
110 worldBox.minX * worldToScreenMatrix.m01 + worldBox.maxY * worldToScreenMatrix.m11 + worldToScreenMatrix.m21
111 };
112
113 const double minX = (std::min)({ px[0], px[1], px[2], px[3] });
114 const double maxX = (std::max)({ px[0], px[1], px[2], px[3] });
115 const double minY = (std::min)({ py[0], py[1], py[2], py[3] });
116 const double maxY = (std::max)({ py[0], py[1], py[2], py[3] });
117
118 return BLRect(minX, minY, (std::max)(0.0, maxX - minX), (std::max)(0.0, maxY - minY));
119 }

200 void ZoomAtScreenPoint(double cursorScreenX, double cursorScreenY, double factor) noexcept {
201 const double oldZoom = zoom;
202 const double newZoom = std::clamp(oldZoom * factor, 0.25, 8.0);
203 if (std::abs(newZoom - oldZoom) < 0.0001) return;
204
205 // Anchor world point under the cursor during zoom
206 const double oldScale = pixelsPerMm * oldZoom;
207 const double newScale = pixelsPerMm * newZoom;
208
209 const double worldAnchorX = (cursorScreenX / oldScale) - panXMm;
210 const double worldAnchorY = (cursorScreenY / oldScale) - panYMm;
211
212 zoom = newZoom;
213 panXMm = (cursorScreenX / newScale) - worldAnchorX;
214 panYMm = (cursorScreenY / newScale) - worldAnchorY;
215
216 ClampPan();
217 }

219 void ClampPan() noexcept {
220 if (infinityMode == CanvasInfinityMode::FullInfinity) {
221 // Unbounded pan in all directions
222 return;
223 }
224 if (infinityMode == CanvasInfinityMode::SemiInfinity) {
225 if (panXMm > 0.0) panXMm = 0.0;
226 if (panYMm > 0.0) panYMm = 0.0;
227 return;
228 }
229 if (infinityMode == CanvasInfinityMode::VerticalScroll) {
230 if (panYMm > 0.0) panYMm = 0.0;
231 return;
232 }
233 if (infinityMode == CanvasInfinityMode::HorizontalScroll) {
234 if (panXMm > 0.0) panXMm = 0.0;
235 return;
236 }
237 }
