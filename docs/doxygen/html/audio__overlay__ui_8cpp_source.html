25 * windowPos.x = screenX - (estimatedCapsuleW * 0.5f)

45/**
46 * @brief Formats fractional seconds into MM:SS timestamp format.
47 *
48 * @param sec Total elapsed duration in seconds.
49 * @param buf Output char buffer.
50 * @param bufSize Size of output buffer in bytes.
51 */
52void FormatMmSs(double sec, char* buf, size_t bufSize) {
53 if (sec < 0.0) sec = 0.0;
54 int totalSec = static_cast<int>(std::floor(sec));
55 int m = totalSec / 60;
56 int s = totalSec % 60;
57 std::snprintf(buf, bufSize, "%02d:%02d", m, s);
58}
