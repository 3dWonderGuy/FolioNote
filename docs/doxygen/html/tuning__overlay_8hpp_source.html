62 float EvaluatePressure(float rawPressure) const {
63 if (!enablePressureCurve)
64 return 1.0f;
65 if (rawPressure <= pressureDeadzoneMin)
66 return minRelativeThicknessFraction;
67 if (rawPressure >= pressureSaturationMax)
68 return 1.0f;
69
70 float norm = (rawPressure - pressureDeadzoneMin) /
71 (pressureSaturationMax - pressureDeadzoneMin);
72 float curved = std::pow(norm, pressureGamma);
73 return minRelativeThicknessFraction +
74 (1.0f - minRelativeThicknessFraction) *
75 std::clamp(curved, 0.0f, 1.0f);
76 }
