1/**
2 * @file input_manager.cpp
3 * @brief Implementation of the top-level InputManager coordinating SDL3 events,
4 * hardware device dispatch, and the semantic InputStateMachine.
5 *
6 * ARCHITECTURE OVERVIEW:
7 * ----------------------
8 * FolioNote supports rich multimodal input across diverse hardware configurations:
9 * 1. Active / EMR Stylus (Pressure, Tilt, Distance, Proximity, Barrel Buttons, Eraser)
10 * 2. Capacitive Multi-Touch (Multi-finger tracking, Gestures, Pinch-to-Zoom, Two-Finger Pan)
11 * 3. Mouse / Trackpad (Absolute position, Relative deltas, Buttons, Smooth Scroll Wheel)
12 * 4. Keyboard Modifiers (Ctrl, Shift, Alt, Space for transient navigation)
13 *
14 * InputManager acts as the primary hardware ingestion bridge. It consumes raw operating
15 * system events via SDL3, performs essential validation (window state stability, synthetic
16 * event filtering, coordinate normalization), updates the low-level hardware tracking
17 * structures in InputStateMachine, and triggers semantic evaluation and arbitration.
18 *
19 * DATA FLOW:
20 * ----------
21 * [OS / SDL3 Event Loop]
22 * |
23 * v
24 * InputManager::ProcessEvent()
25 * |---> WindowStateManager (guard against input during window resizes / moves)
26 * |---> Timestamp Normalization (convert SDL ns timestamps to seconds relative to launch)
27 * |---> Hardware Handlers:
28 * | * HandlePenEvent() --> stateMachine.pen
29 * | * HandleTouchEvent() --> stateMachine.activeFingers (normalized [0,1] -> px)
30 * | * HandleMouseEvent() --> stateMachine.mouse (synthetic touch/pen filtered)
31 * | * HandleKeyboardEvent() --> stateMachine.keyboard
32 * |
33 * v
34 * ImGui Focus Arbitration (differentiate canvas/PDF document hover from UI chrome capture)
35 * |
36 * v
37 * InputStateMachine::ProcessInputState()
38 * (Device arbitration Stylus > Touch > Mouse -> Semantic dispatch to CanvasEngine / DocumentSession)
39 */
40

51/**
52 * @brief Primary entry point for all SDL events destined for input handling.
53 *
54 * Executes a 4-stage pipeline:
55 * Stage 1: Window State Validation - Abort if window is in an unstable geometry state.
56 * Stage 2: Event Timestamp Tracking - Convert nanoseconds to application-relative seconds.
57 * Stage 3: Hardware-Specific Routing - Populate raw hardware telemetry structures.
58 * Stage 4: State Machine Evaluation - Arbitrate active device and dispatch semantic actions.
59 *
60 * @param event The raw SDL_Event received from SDL_PollEvent.
61 * @param canvas Reference to CanvasEngine (handles rendering, viewport camera, stroke mesh).
62 * @param session Reference to DocumentSession (manages committed objects, pages, undo/redo).
63 * @param windowSM Reference to WindowStateManager (tracks maximize/minimize/resize stability).
64 */
65void InputManager::ProcessEvent(const SDL_Event& event, CanvasEngine& canvas, DocumentSession& session, WindowStateManager& windowSM) {
66 // -----------------------------------------------------------------------------
67 // STAGE 1: WINDOW STATE VALIDATION
68 // -----------------------------------------------------------------------------
69 // Inform the window state manager about window events (SDL_EVENT_WINDOW_*).
70 // It maintains an internal state machine (e.g. Stable, Resizing, Moving).
71 windowSM.ProcessSDLEvent(event);
72
73 // If the window is currently being resized or moved by the user or OS, canvas
74 // dimensions, swapchain buffers, and projection matrices are in flux. Aborting
75 // input processing here prevents stray strokes, coordinate warping, and division
76 // by zero in aspect-ratio calculations.
77 static bool s_loggedUnstable = false;
78 if (windowSM.getCurrentState() != WindowState::Stable) {
79 if (!s_loggedUnstable) {
80 s_loggedUnstable = true;
81 LOG_WARN_CODE(InputManager, FolioErrorCode::InputWindowUnstable,
82 "Input suspended: Window geometry is actively resizing or moving.");
83 }
84 return;
85 }
86 s_loggedUnstable = false;
87
88 // -----------------------------------------------------------------------------
89 // STAGE 2: EVENT TIMESTAMP TRACKING
90 // -----------------------------------------------------------------------------
91 // SDL3 provides high-resolution 64-bit nanosecond event timestamps (event.common.timestamp).
92 // EventTimestampToSec converts this into floating-point seconds relative to application start:
93 // t_sec = (timestamp_ns - appStartTimeNs) * 1e-9
94 // This monotonically increasing timestamp is essential for Catmull-Rom spline smoothing,
95 // instantaneous drawing velocity calculation, and gesture flick detection.
96 stateMachine.latestEventTimeSec = stateMachine.EventTimestampToSec(event.common.timestamp);
97
98 // -----------------------------------------------------------------------------
99 // STAGE 2.5: LAYER 3 INTERACTIVE OVERLAY EVENT ROUTING
100 // -----------------------------------------------------------------------------
101 if (canvas.interactiveOverlayHost.HandleInput(event, canvas.GetViewport(), canvas.interactiveOverlayHost.GetFocusedObject())) {
102 canvas.needsFullRebake = true;
103 canvas.isDirty = true;
104 return;
105 }
106
107 // -----------------------------------------------------------------------------
108 // STAGE 3: HARDWARE-SPECIFIC INPUT ROUTING
109 // -----------------------------------------------------------------------------
110 // Distribute the raw event to dedicated hardware handlers.
111 // Each handler parses device-specific data packets and updates stateMachine telemetry.
112 switch (event.type) {
113 // --- TEXT INPUT FOR HEADLESS CONTROLLER ---
114 case SDL_EVENT_TEXT_INPUT: {
115 if (canvas.textEditor.IsActive()) {
116 if (stateMachine.hasPendingEmptyTextBox && stateMachine.pendingTextBoxUid != 0) {
117 auto activePage = session.GetActivePage();
118 if (activePage) {
119 auto pendingBox = activePage->FindObjectByUid(stateMachine.pendingTextBoxUid);
120 if (pendingBox) {
121 session.RecordHistoryCommand(activePage, std::make_unique<Folio::AddObjectCommand>(pendingBox));
122 }
123 }
124 stateMachine.hasPendingEmptyTextBox = false;
125 stateMachine.pendingTextBoxUid = 0;
126 }
127 canvas.textEditor.OnTextInput(event.text.text);
128 canvas.needsFullRebake = true;
129 canvas.isDirty = true;
130 } else if ((stateMachine.GetActiveDeviceTool() == InteractionState::Selecting ||
131 stateMachine.currentAction == InteractionState::Selecting) &&
132 !ImGui::GetIO().WantCaptureKeyboard) {
133 auto activePage = session.GetActivePage();
134 if (activePage) {
135 // Check if a TextBoxObject is currently selected: typing activates editing on it!
136 std::shared_ptr<Folio::TextBoxObject> selectedBox = nullptr;
137 for (auto& obj : activePage->objects) {
138 if (obj && obj->isSelected && obj->type == ObjectType::Text) {
139 selectedBox = std::dynamic_pointer_cast<Folio::TextBoxObject>(obj);
140 break;
141 }
142 }
143
144 if (selectedBox) {
145 canvas.ClearSelection(&session);
146 canvas.textEditor.Attach(selectedBox.get(), &session);
147 canvas.textEditor.OnTextInput(event.text.text);
148 canvas.needsFullRebake = true;
149 canvas.isDirty = true;
150 LOG_INFO(InputManager, "Typing activated text editor on selected text box uid=" +
151 std::to_string(selectedBox->uid));
152 } else {
153 // Auto-create text box if user starts typing on empty canvas while in Selecting mode
154 Point2D clickPos = stateMachine.lastCanvasClickWorldMm;
155 if (clickPos.x == 0.0 && clickPos.y == 0.0) {
156 auto vp = canvas.GetViewport();
157 double cx = (vp.bounds.minX + vp.bounds.maxX) * 0.5;
158 double cy = (vp.bounds.minY + vp.bounds.maxY) * 0.5;
159 clickPos = Point2D(cx - 25.0, cy - 10.0);
160 }
161 auto newBox = std::make_shared<Folio::TextBoxObject>(clickPos.x, clickPos.y);
162 newBox->textColor = canvas.defaultTextColor;
163 newBox->fontFamily = canvas.defaultTextFontFamily;
164 newBox->fontSize = canvas.defaultTextFontSize;
165 newBox->isBold = canvas.defaultTextBold;
166 newBox->isItalic = canvas.defaultTextItalic;
167 newBox->isUnderline = canvas.defaultTextUnderline;
168 newBox->isStrikethrough = canvas.defaultTextStrikethrough;
169 newBox->highlightColor = canvas.defaultTextHighlightColor;
170 newBox->alignment = canvas.defaultTextAlignment;
171 activePage->AddObject(newBox);
172 session.RecordHistoryCommand(activePage, std::make_unique<Folio::AddObjectCommand>(newBox));
173 canvas.textEditor.Attach(newBox.get(), &session);
174 canvas.textEditor.OnTextInput(event.text.text);
175 canvas.needsFullRebake = true;
176 canvas.isDirty = true;
177 LOG_INFO(InputManager, "Auto-created OneNote text box from keyboard input at (" +
178 std::to_string(clickPos.x) + ", " + std::to_string(clickPos.y) + ")");
179 }
180 }
181 }
182 break;
183 }
184
185 // --- KEYBOARD SHORTCUTS & MODIFIERS ---
186 case SDL_EVENT_KEY_DOWN: {
187 if (event.key.key == SDLK_ESCAPE) {
188 if (canvas.textEditor.IsActive()) {
189 auto target = canvas.textEditor.GetTarget();
190 canvas.textEditor.Detach(&session);
191 if (target && target->PlainText().empty()) {
192 auto activePage = session.GetActivePage();
193 if (activePage) activePage->RemoveObjectByUid(target->uid);
194 }
195 canvas.needsFullRebake = true;
196 canvas.isDirty = true;
197 }
198 } else if (canvas.textEditor.IsActive()) {
199 if (event.key.key == SDLK_Z && (event.key.mod & SDL_KMOD_CTRL)) {
200 canvas.textEditor.CommitTextEdit(&session);
201 if (event.key.mod & SDL_KMOD_SHIFT) {
202 session.Redo(&canvas);
203 } else {
204 session.Undo(&canvas);
205 }
206 canvas.needsFullRebake = true;
207 canvas.isDirty = true;
208 break;
209 }
210 if (event.key.key == SDLK_RETURN || event.key.key == SDLK_KP_ENTER) {
211 canvas.textEditor.OnKeyDown(event.key.key, event.key.mod);
212 canvas.textEditor.CommitTextEdit(&session);
213 } else {
214 canvas.textEditor.OnKeyDown(event.key.key, event.key.mod);
215 }
216 canvas.needsFullRebake = true;
217 canvas.isDirty = true;
218 } else if ((stateMachine.GetActiveDeviceTool() == InteractionState::Selecting ||
219 stateMachine.currentAction == InteractionState::Selecting) &&
220 !ImGui::GetIO().WantCaptureKeyboard) {
221 // If a TextBoxObject is selected and user hits Backspace, Enter, or navigation key, activate editor
222 if (event.key.key == SDLK_BACKSPACE || event.key.key == SDLK_DELETE ||
223 event.key.key == SDLK_RETURN || event.key.key == SDLK_KP_ENTER) {
224 auto activePage = session.GetActivePage();
225 if (activePage) {
226 for (auto& obj : activePage->objects) {
227 if (obj && obj->isSelected && obj->type == ObjectType::Text) {
228 auto selectedBox = std::dynamic_pointer_cast<Folio::TextBoxObject>(obj);
229 if (selectedBox) {
230 canvas.ClearSelection(&session);
231 canvas.textEditor.Attach(selectedBox.get(), &session);
232 canvas.textEditor.OnKeyDown(event.key.key, event.key.mod);
233 canvas.needsFullRebake = true;
234 canvas.isDirty = true;
235 break;
236 }
237 }
238 }
239 }
240 }
241 }
242 HandleKeyboardEvent(event);
243 break;
244 }
245 case SDL_EVENT_KEY_UP:
246 HandleKeyboardEvent(event);
247 break;
248
249 // --- ACTIVE / PASSIVE DIGITIZER (STYLUS / PEN) ---
250 case SDL_EVENT_PEN_PROXIMITY_IN:
251 case SDL_EVENT_PEN_PROXIMITY_OUT:
252 case SDL_EVENT_PEN_AXIS:
253 case SDL_EVENT_PEN_DOWN:
254 case SDL_EVENT_PEN_UP:
255 case SDL_EVENT_PEN_MOTION:
256 case SDL_EVENT_PEN_BUTTON_DOWN:
257 case SDL_EVENT_PEN_BUTTON_UP:
258 HandlePenEvent(event);
259 break;
260
261 // --- CAPACITIVE MULTI-TOUCH ---
262 case SDL_EVENT_FINGER_DOWN:
263 case SDL_EVENT_FINGER_UP:
264 case SDL_EVENT_FINGER_CANCELED:
265 case SDL_EVENT_FINGER_MOTION:
266 HandleTouchEvent(event);
267 break;
268
269 // --- MOUSE & TRACKPAD ---
270 case SDL_EVENT_MOUSE_BUTTON_DOWN:
271 case SDL_EVENT_MOUSE_BUTTON_UP:
272 case SDL_EVENT_MOUSE_MOTION:
273 case SDL_EVENT_MOUSE_WHEEL:
274 HandleMouseEvent(event);
275 break;
276
277 default:
278 // Other event types (controllers, sensors, drop events) are handled elsewhere or ignored.
279 break;
280 }
281
282 // -----------------------------------------------------------------------------
283 // STAGE 4: STATE MACHINE EVALUATION & UI ARBITRATION
284 // -----------------------------------------------------------------------------
285 // Determine whether Dear ImGui UI chrome has captured pointer focus.
286 //
287 // ARBITRATION LOGIC:
288 // ImGui's IO flag `WantCaptureMouse` is true whenever the pointer is over any ImGui
289 // window. However, both the main Infinite Canvas and the PDF Viewer render inside
290 // ImGui window viewports (as ImGui::Image). Therefore, if the cursor is hovering
291 // the document canvas (wasCanvasImageHovered) or the PDF page surface
292 // (stateMachine.isPdfCanvasHovered), we negate WantCaptureMouse so that the canvas
293 // receives interaction rather than the host ImGui window frame.
294 //
295 // Mathematical formulation:
296 // imguiHasFocus = WantCaptureMouse AND NOT (wasCanvasImageHovered OR isPdfCanvasHovered OR isCanvasDragging)
297 const bool isCanvasDragging = canvas.selectionGizmo.isDragging ||
298 canvas.marqueeBox.isActive ||
299 canvas.layerCompositor.GetLiveInteractionLayer().HasActiveInteraction() ||
300 (canvas.selectionMode == CanvasEngine::SelectionMode::Lasso && stateMachine.mouse.leftButton) ||
301 (stateMachine.currentAction == InteractionState::Panning && stateMachine.mouse.leftButton);
302
303 // Determine if cursor is hovering an overlay window (floating HUD, toolbar, menu, modal) or active widget
304 bool isOverOverlayOrWidget = false;
305 ImGuiContext* g = ImGui::GetCurrentContext();
306 if (g) {
307 if (g->ActiveId != 0) {
308 isOverOverlayOrWidget = true;
309 } else if (g->HoveredWindow) {
310 std::string_view hoveredName = g->HoveredWindow->Name;
311 if (hoveredName != "##CanvasPanel") {
312 isOverOverlayOrWidget = true;
313 }
314 }
315 }
316
317 bool imguiHasFocus = false;
318 if (isOverOverlayOrWidget) {
319 imguiHasFocus = !isCanvasDragging;
320 } else {
321 imguiHasFocus = (ImGui::GetIO().WantCaptureMouse && !(wasCanvasImageHovered || stateMachine.isPdfCanvasHovered) && !isCanvasDragging);
322 }
323
324 // Diagnostic log for genuine physical mouse clicks (excluding synthetic SDL touch/pen mouse events)
325 if (event.type == SDL_EVENT_MOUSE_BUTTON_DOWN) {
326 if (event.button.which != SDL_TOUCH_MOUSEID && event.button.which != SDL_PEN_MOUSEID) {
327 LOG_INFO(InputManager, "Mouse button " + std::to_string(event.button.button) + " DOWN at (" +
328 std::to_string(static_cast<int>(event.button.x)) + ", " +
329 std::to_string(static_cast<int>(event.button.y)) + ") [CanvasHovered=" +
330 (wasCanvasImageHovered ? "true" : "false") + ", ImGuiCaptured=" +
331 (imguiHasFocus ? "true" : "false") + "]");
332 }
333 }
334
335 // Hand off all consolidated telemetry to the InputStateMachine:
336 // 1. Device Priority Arbitration: Stylus (highest) > Touch > Mouse.
337 // 2. Interaction State Determination: Inking, Eraser, Panning, Selecting, Text.
338 // 3. Dispatch: Invokes CanvasEngine / DocumentSession APIs to draw strokes, pan, or erase.
339 stateMachine.ProcessInputState(canvas, session, imguiHasFocus);
340
341 // Ensure OS text input is active so key events yield character input
342 SDL_Window* focusedWin = SDL_GetKeyboardFocus();
343 if (focusedWin && !SDL_TextInputActive(focusedWin)) {
344 SDL_StartTextInput(focusedWin);
345 }
346}

352/**
353 * @brief Processes high-precision stylus and active digitizer telemetry.
354 *
355 * Captures all physical capabilities of modern styluses (Surface Pen, Apple Pencil, Wacom):
356 * - Proximity / Hover detection (sensor field entry/exit before physical contact).
357 * - Continuous axis updates: Pressure [0.0, 1.0], Distance, Tilt-X, Tilt-Y [-90.0, +90.0].
358 * - Physical tip contact (Down, Motion, Up).
359 * - Hardware barrel buttons (Barrel 1, 2, 3) and physical eraser tail tip detection.
360 *
361 * @param event The SDL_Event containing SDL_EVENT_PEN_* data.
362 */
363void InputManager::HandlePenEvent(const SDL_Event& event) {
364 // Record current system tick to establish recent pen activity for device arbitration timeouts
365 stateMachine.lastPenTimestampMs = SDL_GetTicks();
366 auto& pen = stateMachine.pen;
367
368 // -------------------------------------------------------------------------
369 // 1. PROXIMITY TELEMETRY
370 // Stylus enters or exits the digitizer's electromagnetic resonance (EMR) field
371 // (typically ~10-20mm above the display surface).
372 // -------------------------------------------------------------------------
373 if (event.type == SDL_EVENT_PEN_PROXIMITY_IN) {
374 pen.inProximity = true;
375 pen.isHovering = true;
376 LOG_INFO(InputManager, "PEN PROXIMITY_IN (PenID: " + std::to_string(event.pproximity.which) + ")");
377 return;
378 }
379 else if (event.type == SDL_EVENT_PEN_PROXIMITY_OUT) {
380 // Pen was lifted beyond the digitizer's sensing range; reset hover & contact flags
381 pen.inProximity = false;
382 pen.isHovering = false;
383 pen.isDown = false;
384 pen.eraserTip = false;
385 LOG_INFO(InputManager, "PEN PROXIMITY_OUT (PenID: " + std::to_string(event.pproximity.which) + ")");
386 return;
387 }
388
389 // -------------------------------------------------------------------------
390 // 2. HIGH-FREQUENCY AXIS TELEMETRY
391 // Digitizers report continuous physical parameters independently of contact events.
392 // -------------------------------------------------------------------------
393 if (event.type == SDL_EVENT_PEN_AXIS) {
394 // Pressure: Normalized [0.0, 1.0]. Clamped to avoid hardware driver overflows.
395 if (event.paxis.axis == SDL_PEN_AXIS_PRESSURE) {
396 pen.pressure = std::clamp(event.paxis.value, 0.0f, 1.0f);
397 }
398 // Distance: Normalized distance above surface while hovering in proximity.
399 else if (event.paxis.axis == SDL_PEN_AXIS_DISTANCE) {
400 pen.distance = event.paxis.value;
401 }
402 // Tilt X/Y: Stylus inclination angles in degrees relative to the screen normal (-90° to +90°).
403 // Used by rendering shaders to simulate angle-dependent brush calligraphics and pencil shading.
404 else if (event.paxis.axis == SDL_PEN_AXIS_XTILT) {
405 pen.tiltX = event.paxis.value;
406 }
407 else if (event.paxis.axis == SDL_PEN_AXIS_YTILT) {
408 pen.tiltY = event.paxis.value;
409 }
410
411 // Check if the physical tail-end eraser tip is currently active via hardware state bitmask
412 bool isEraser = (event.paxis.pen_state & SDL_PEN_INPUT_ERASER_TIP) != 0;
413 if (pen.eraserTip != isEraser) {
414 pen.eraserTip = isEraser;
415 LOG_INFO(InputManager, std::string("PEN ERASER TIP ") + (isEraser ? "ACTIVE" : "INACTIVE"));
416 }
417 return;
418 }
419
420 // -------------------------------------------------------------------------
421 // 3. CONTACT & MOTION (DOWN, MOTION, UP)
422 // -------------------------------------------------------------------------
423 if (event.type == SDL_EVENT_PEN_DOWN) {
424 // Pen tip touches the screen surface; start stroke / engagement
425 pen.isDown = true;
426 pen.isHovering = false;
427 pen.inProximity = true;
428 pen.x = event.ptouch.x;
429 pen.y = event.ptouch.y;
430
431 // Check eraser status either from the ptouch.eraser boolean or pen_state bitmask
432 bool isEraser = event.ptouch.eraser || ((event.ptouch.pen_state & SDL_PEN_INPUT_ERASER_TIP) != 0);
433 if (pen.eraserTip != isEraser) {
434 pen.eraserTip = isEraser;
435 LOG_INFO(InputManager, std::string("PEN ERASER TIP ") + (isEraser ? "ACTIVE" : "INACTIVE"));
436 }
437 }
438 else if (event.type == SDL_EVENT_PEN_MOTION) {
439 // Stylus position changed (either dragging while down or hovering above the glass)
440 pen.x = event.pmotion.x;
441 pen.y = event.pmotion.y;
442 pen.inProximity = true;
443 if (!pen.isDown) pen.isHovering = true;
444
445 bool isEraser = (event.pmotion.pen_state & SDL_PEN_INPUT_ERASER_TIP) != 0;
446 if (pen.eraserTip != isEraser) {
447 pen.eraserTip = isEraser;
448 LOG_INFO(InputManager, std::string("PEN ERASER TIP ") + (isEraser ? "ACTIVE" : "INACTIVE"));
449 }
450 }
451 else if (event.type == SDL_EVENT_PEN_UP) {
452 // Pen tip lifted off the surface; end stroke, transition back to hover state
453 pen.isDown = false;
454 pen.isHovering = true;
455
456 bool isEraser = event.ptouch.eraser || ((event.ptouch.pen_state & SDL_PEN_INPUT_ERASER_TIP) != 0);
457 if (pen.eraserTip != isEraser) {
458 pen.eraserTip = isEraser;
459 LOG_INFO(InputManager, std::string("PEN ERASER TIP ") + (isEraser ? "ACTIVE" : "INACTIVE"));
460 }
461 }
462 // -------------------------------------------------------------------------
463 // 4. HARDWARE BARREL BUTTONS
464 // Physical side buttons on the pen stylus barrel (mapped to eraser or lasso shortcuts)
465 // -------------------------------------------------------------------------
466 else if (event.type == SDL_EVENT_PEN_BUTTON_DOWN || event.type == SDL_EVENT_PEN_BUTTON_UP) {
467 bool down = (event.type == SDL_EVENT_PEN_BUTTON_DOWN);
468 LOG_INFO(InputManager, "PEN BUTTON " + std::to_string(event.pbutton.button) + (down ? " DOWN" : " UP"));
469
470 // SDL button indices: 1 = primary barrel, 2 = secondary barrel, 3 = tertiary
471 if (event.pbutton.button == 1) pen.barrel1 = down;
472 else if (event.pbutton.button == 2) pen.barrel2 = down;
473 else if (event.pbutton.button == 3) pen.barrel3 = down;
474 else {
475 LOG_WARN(InputManager, "Unknown pen button: " + std::to_string(event.pbutton.button));
476 }
477
478 bool isEraser = (event.pbutton.pen_state & SDL_PEN_INPUT_ERASER_TIP) != 0;
479 if (pen.eraserTip != isEraser) {
480 pen.eraserTip = isEraser;
481 LOG_INFO(InputManager, std::string("PEN ERASER TIP ") + (isEraser ? "ACTIVE" : "INACTIVE"));
482 }
483 }
484}

490/**
491 * @brief Processes standard mouse and trackpad interactions.
492 *
493 * SYNTHETIC EVENT SUPPRESSION:
494 * Operating systems (Windows Ink, macOS Cocoa) synthesize legacy mouse events
495 * whenever touch or pen contacts occur, ensuring legacy applications receive clicks.
496 * Because FolioNote has native, dedicated pipelines for Touch and Pen with higher
497 * precision, processing synthetic mouse events would cause double clicks, jitter,
498 * and conflicting device arbitration. We identify and drop synthetic events using
499 * SDL's sentinel device IDs: `SDL_TOUCH_MOUSEID` and `SDL_PEN_MOUSEID`.
500 *
501 * @param event The SDL_Event containing SDL_EVENT_MOUSE_* data.
502 */
503void InputManager::HandleMouseEvent(const SDL_Event& event) {
504 // Filter out synthetic mouse events generated by SDL from capacitive touch or pen input
505 if (event.type == SDL_EVENT_MOUSE_BUTTON_DOWN || event.type == SDL_EVENT_MOUSE_BUTTON_UP) {
506 if (event.button.which == SDL_TOUCH_MOUSEID || event.button.which == SDL_PEN_MOUSEID) {
507 return;
508 }
509 } else if (event.type == SDL_EVENT_MOUSE_MOTION) {
510 if (event.motion.which == SDL_TOUCH_MOUSEID || event.motion.which == SDL_PEN_MOUSEID) {
511 return;
512 }
513 }
514
515 // Timestamp genuine physical mouse activity
516 stateMachine.lastMouseTimestampMs = SDL_GetTicks();
517 auto& mouse = stateMachine.mouse;
518
519 // -------------------------------------------------------------------------
520 // BUTTON STATE TRACKING
521 // -------------------------------------------------------------------------
522 if (event.type == SDL_EVENT_MOUSE_BUTTON_DOWN || event.type == SDL_EVENT_MOUSE_BUTTON_UP) {
523 bool down = (event.type == SDL_EVENT_MOUSE_BUTTON_DOWN);
524 if (event.button.button == SDL_BUTTON_LEFT) mouse.leftButton = down;
525 else if (event.button.button == SDL_BUTTON_RIGHT) mouse.rightButton = down;
526 else if (event.button.button == SDL_BUTTON_MIDDLE) mouse.middleButton = down;
527
528 // Window-relative coordinates in pixels
529 mouse.x = event.button.x;
530 mouse.y = event.button.y;
531 }
532 // -------------------------------------------------------------------------
533 // MOTION & DELTAS
534 // -------------------------------------------------------------------------
535 else if (event.type == SDL_EVENT_MOUSE_MOTION) {
536 mouse.x = event.motion.x;
537 mouse.y = event.motion.y;
538 // Relative displacement deltas (useful for camera panning and drag vectors)
539 mouse.dx = event.motion.xrel;
540 mouse.dy = event.motion.yrel;
541 }
542 // -------------------------------------------------------------------------
543 // SCROLL WHEEL
544 // -------------------------------------------------------------------------
545 else if (event.type == SDL_EVENT_MOUSE_WHEEL) {
546 // wheelX: Positive = right, Negative = left (horizontal scroll / trackpad gesture)
547 // wheelY: Positive = forward/up (zoom in / scroll up), Negative = backward/down
548 mouse.wheelX = event.wheel.x;
549 mouse.wheelY = event.wheel.y;
550 }
551}

557/**
558 * @brief Tracks modifier keys and hotkeys relevant to canvas navigation.
559 *
560 * Monitors:
561 * - Ctrl: Multi-select modifier, zoom modifier, undo/redo shortcuts.
562 * - Shift: Constrained drawing axis (straight horizontal/vertical lines), aspect ratio lock.
563 * - Alt: Color picker eyedropper, alternate clone drag.
564 * - Space: Transient canvas panning hand tool (hold Space + drag mouse to pan).
565 *
566 * @param event The SDL_Event containing SDL_EVENT_KEY_DOWN or SDL_EVENT_KEY_UP.
567 */
568void InputManager::HandleKeyboardEvent(const SDL_Event& event) {
569 // Query the global modifier bitmask to capture both left and right variants of modifiers
570 SDL_Keymod mod = SDL_GetModState();
571 stateMachine.keyboard.ctrl = (mod & SDL_KMOD_CTRL) != 0;
572 stateMachine.keyboard.shift = (mod & SDL_KMOD_SHIFT) != 0;
573 stateMachine.keyboard.alt = (mod & SDL_KMOD_ALT) != 0;
574
575 // Spacebar triggers transient hand/panning tool when held down
576 stateMachine.keyboard.space = (event.key.key == SDLK_SPACE) ? (event.type == SDL_EVENT_KEY_DOWN) : stateMachine.keyboard.space;

583/**
584 * @brief Processes multi-touch capacitive screen events (up to 10 concurrent fingers).
585 *
586 * COORDINATE SPACE TRANSFORMATION:
587 * SDL3 touch coordinates (event.tfinger.x, y, dx, dy) are normalized floating-point
588 * values in the range [0.0, 1.0] relative to the touch device surface.
589 * To align them with the screen-space pixel coordinates used by the Mouse and Pen
590 * pipelines, we query the window dimensions (W, H) and transform:
591 *
592 * Pixel X: px = x_normalized * WindowWidth
593 * Pixel Y: py = y_normalized * WindowHeight
594 * Pixel Delta X: pdx = dx_normalized * WindowWidth
595 * Pixel Delta Y: pdy = dy_normalized * WindowHeight
596 *
597 * SLOT ALLOCATION STRATEGY:
598 * FolioNote uses a pre-allocated fixed-capacity array (std::array<TouchSlot, 10> activeFingers)
599 * to avoid heap allocations in the high-frequency touch path:
600 * - Sentinel: `fingerID == -1` designates an empty slot.
601 * - FINGER_DOWN: Locates the first empty slot (-1) and binds the finger ID, position, and pressure.
602 * - FINGER_MOTION: Locates the existing slot matching the finger ID and updates its position/delta.
603 * - FINGER_UP/CANC: Locates the active slot, resets `fingerID = -1`, and clears coordinates.
604 *
605 * Downstream, TouchGestureRecognizer evaluates these active slots to compute multi-finger gestures:
606 * - Pinch-to-Zoom: Euclidean distance ratio between two finger contact points:
607 * scale = ||p1_curr - p2_curr|| / ||p1_prev - p2_prev||
608 * - Two-Finger Pan: Centroid translation delta:
609 * delta_C = ((p1_curr + p2_curr) / 2) - ((p1_prev + p2_prev) / 2)
610 *
611 * @param event The SDL_Event containing SDL_EVENT_FINGER_* data.
612 */
613void InputManager::HandleTouchEvent(const SDL_Event& event) {
614 stateMachine.lastTouchTimestampMs = SDL_GetTicks();
615
616 // Query window dimensions in pixels for normalized -> pixel coordinate conversion
617 int w = 1920, h = 1080;
618 SDL_Window* win = SDL_GetWindowFromID(event.tfinger.windowID);
619 if (win) {
620 SDL_GetWindowSizeInPixels(win, &w, &h);
621 }
622 float fw = static_cast<float>(w > 0 ? w : 1);
623 float fh = static_cast<float>(h > 0 ? h : 1);
624
625 // SDL3 touch coordinates are normalized [0.0, 1.0]; scale to window pixel space
626 float px = event.tfinger.x * fw;
627 float py = event.tfinger.y * fh;
628 float pdx = event.tfinger.dx * fw;
629 float pdy = event.tfinger.dy * fh;
630
631 // -------------------------------------------------------------------------
632 // FINGER CONTACT DOWN: Allocate slot
633 // -------------------------------------------------------------------------
634 if (event.type == SDL_EVENT_FINGER_DOWN) {
635 for (auto& slot : stateMachine.activeFingers) {
636 // Find first unused slot marked by sentinel fingerID == -1
637 if (slot.fingerID == -1) {
638 slot.fingerID = event.tfinger.fingerID;
639 slot.point.x = px;
640 slot.point.y = py;
641 slot.point.dx = pdx;
642 slot.point.dy = pdy;
643 slot.point.pressure = event.tfinger.pressure;
644 break;
645 }
646 }
647 }
648 // -------------------------------------------------------------------------
649 // FINGER MOTION: Update allocated slot
650 // -------------------------------------------------------------------------
651 else if (event.type == SDL_EVENT_FINGER_MOTION) {
652 for (auto& slot : stateMachine.activeFingers) {
653 if (slot.fingerID == event.tfinger.fingerID) {
654 slot.point.x = px;
655 slot.point.y = py;
656 slot.point.dx = pdx;
657 slot.point.dy = pdy;
658 slot.point.pressure = event.tfinger.pressure;
659 break;
660 }
661 }
662 }
663 // -------------------------------------------------------------------------
664 // FINGER UP / CANCELED: Free slot
665 // -------------------------------------------------------------------------
666 else if (event.type == SDL_EVENT_FINGER_UP || event.type == SDL_EVENT_FINGER_CANCELED) {
667 for (auto& slot : stateMachine.activeFingers) {
668 if (slot.fingerID == event.tfinger.fingerID) {
669 // Return slot to pool by resetting sentinel to -1
670 slot.fingerID = -1;
671 slot.point = {};
672 break;
673 }
674 }
675 }
