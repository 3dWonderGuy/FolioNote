22// =============================================================================
23// DEVICE TYPE
24// Identifies which physical input source is currently driving interaction.
25// =============================================================================
26enum class DeviceType : uint8_t {
27 Unknown = 0,
28 Stylus, ///< Pressure-sensitive pen/stylus (highest priority)
29 Touch, ///< Capacitive multi-touch finger contacts
30 Mouse ///< Standard mouse / trackpad
31};
32
33// =============================================================================
34// INTERACTION STATE
35// High-level semantic action the active device is performing on the canvas.
36// =============================================================================
37enum class InteractionState : uint8_t {
38 Idle = 0, ///< Pointer navigation mode (no active canvas tool)
39 Inking, ///< Drawing a stroke with the active pen preset
40 Eraser, ///< Erasing strokes or segments
41 Selecting, ///< Lasso / freehand / box selection & gizmo manipulation
42 Panning, ///< Panning the canvas viewport
43 Transforming, ///< Moving / scaling selected objects
44 DrawingShape, ///< Drag-creating geometric vector shapes
45 Text ///< OneNote click-to-type & text editing
46};
47
48// =============================================================================
49// STYLUS CONTACT STATE
50// Tracks whether the pen is out of range, hovering, or pressed against the surface.
51// =============================================================================
52enum class StylusState : uint8_t {
53 OutOfRange = 0,
54 Hovering,
55 Engaged
56};
57
58// =============================================================================
59// STYLUS BUTTON STATE
60// Maps physical barrel button hardware to semantic button actions.
61// =============================================================================
62enum class StylusButtonState : uint8_t {
63 None = 0,
64 BarrelPressed, ///< Barrel button 1 (or eraser tip) — used for Eraser override
65 Barrel2Pressed ///< Barrel button 2 — used for Lasso/Select override
66};
67
68// =============================================================================
69// PER-DEVICE TOOL STATE
70// Remembers the last tool explicitly selected for a device.
71// =============================================================================
72struct DeviceToolState {
73 InteractionState savedTool = InteractionState::Inking;
74};

196 [[nodiscard]] InteractionState GetActiveDeviceTool() const noexcept {
197 switch (ActiveDevice) {
198 case DeviceType::Stylus: return stylusTool.savedTool;
199 case DeviceType::Touch: return touchTool.savedTool;
200 case DeviceType::Mouse: return mouseTool.savedTool;
201 default: return InteractionState::Idle;
202 }
203 }

205 [[nodiscard]] bool IsShapeDrawingActive() const noexcept {
206 return currentAction == InteractionState::DrawingShape;
207 }
208
209 void SetToolForDevice(DeviceType device, InteractionState tool) noexcept {
210 switch (device) {
211 case DeviceType::Stylus: stylusTool.savedTool = tool; break;
212 case DeviceType::Touch: touchTool.savedTool = tool; break;
213 case DeviceType::Mouse: mouseTool.savedTool = tool; break;
214 default: break;
215 }
216 if (device == ActiveDevice) {
217 currentAction = tool;
218 }
219 std::string toolName = (tool == InteractionState::Inking) ? "Inking" :
220 (tool == InteractionState::Eraser) ? "Eraser" :
221 (tool == InteractionState::Selecting) ? "Selecting" :
222 (tool == InteractionState::Panning) ? "Panning" :
223 (tool == InteractionState::DrawingShape) ? "DrawingShape" : "Idle";
224 LOG_INFO(InputStateMachine, "Set tool for device " + std::to_string(static_cast<int>(device)) + " to " + toolName);
225 }
