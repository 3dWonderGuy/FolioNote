26static constexpr std::array<uint32_t, 256> GenerateCRC32Table() {
27 std::array<uint32_t, 256> tbl{};
28 for (uint32_t i = 0; i < 256; ++i) {
29 uint32_t c = i;
30 for (int j = 0; j < 8; ++j) {
31 c = (c & 1) ? (0xEDB88320L ^ (c >> 1)) : (c >> 1);
32 }
33 tbl[i] = c;
34 }
35 return tbl;
36}

38uint32_t CRC32::Compute(const uint8_t* data, size_t length) noexcept {
39 static constexpr auto table = GenerateCRC32Table();
40 uint32_t crc = 0xFFFFFFFFL;
41 for (size_t i = 0; i < length; ++i) {
42 crc = table[(crc ^ data[i]) & 0xFF] ^ (crc >> 8);
43 }
44 return crc ^ 0xFFFFFFFFL;
45}

878bool BinarySerializer::SerializePage(const CanvasPage& page, std::vector<uint8_t>& outCompressedBlob, SerializationStats* outStats) {
879 ByteWriter writer;
880 SerializationStats stats;
881
882 size_t estimatedCapacity = EstimatePageByteSize(page);
883 writer.Reserve(estimatedCapacity);
884
885 writer.WriteU32(MAGIC_HEADER);
886 writer.WriteU32(FORMAT_VERSION);
887
888 // 1. Identity & Creation Timestamps
889 writer.WriteString(page.guid);
890 writer.WriteString(page.title);
891 writer.WriteString(page.createdDateStr);
892 writer.WriteString(page.createdTimeStr);
893
894 // 2. Per-Page Template, Layout, Boundary & Grid Configuration
895 writer.WriteU8(static_cast<uint8_t>(page.infinityMode));
896 writer.WriteU8(static_cast<uint8_t>(page.paperStyle));
897 writer.WriteDouble(page.gridSpacingMm);
898 writer.WriteU8(static_cast<uint8_t>(page.pageSizeFormat));
899 writer.WriteBool(page.pageIsLandscape);
900 writer.WriteDouble(page.pageWidthMm);
901 writer.WriteDouble(page.pageHeightMm);
902
903 // 3. Per-Page Border Configuration
904 writer.WriteBool(page.showPageBorder);
905 writer.WriteU8(static_cast<uint8_t>(page.pageBorderType));
906 writer.WriteU8(static_cast<uint8_t>(page.pageBorderStyle));
907 writer.WriteDouble(page.pageBorderWidth);
908
909 // 4. Dedicated Standalone PDF Document State
910 writer.WriteBool(page.isDedicatedPdf);
911 writer.WriteString(page.dedicatedPdfPath);
912 writer.WriteString(page.dedicatedPdfBookmarks);
913 writer.WriteString(page.dedicatedPdfHighlights);
914
915 // 5. Dedicated Standalone Markdown Document State (v6+)
916 writer.WriteBool(page.isDedicatedMd);
917 writer.WriteString(page.dedicatedMdContent);
918
919 // 6. Canvas Objects
920 writer.WriteU32(static_cast<uint32_t>(page.objects.size()));
921 stats.objectCount = static_cast<uint32_t>(page.objects.size());
922
923 for (const auto& obj : page.objects) {
924 SerializeObject(obj, writer, stats);
925 }
926
927 stats.rawSizeBytes = writer.buffer.size();
928
929 std::vector<uint8_t> compressedData;
930 if (!Compress(writer.buffer, compressedData)) {
931 LOG_ERROR(BinarySerializer, "Failed to compress page serialization stream for page GUID: " + page.guid);
932 return false;
933 }
934
935 uint32_t crc = CRC32::Compute(compressedData.data(), compressedData.size());
936 stats.crc32Checksum = crc;
937
938 outCompressedBlob.clear();
939 outCompressedBlob.reserve(8 + compressedData.size());
940
941 uint32_t rawSize = static_cast<uint32_t>(writer.buffer.size());
942 outCompressedBlob.push_back(static_cast<uint8_t>(rawSize & 0xFF));
943 outCompressedBlob.push_back(static_cast<uint8_t>((rawSize >> 8) & 0xFF));
944 outCompressedBlob.push_back(static_cast<uint8_t>((rawSize >> 16) & 0xFF));
945 outCompressedBlob.push_back(static_cast<uint8_t>((rawSize >> 24) & 0xFF));
946
947 outCompressedBlob.push_back(static_cast<uint8_t>(crc & 0xFF));
948 outCompressedBlob.push_back(static_cast<uint8_t>((crc >> 8) & 0xFF));
949 outCompressedBlob.push_back(static_cast<uint8_t>((crc >> 16) & 0xFF));
950 outCompressedBlob.push_back(static_cast<uint8_t>((crc >> 24) & 0xFF));
951
952 outCompressedBlob.insert(outCompressedBlob.end(), compressedData.begin(), compressedData.end());
953
954 stats.compressedSizeBytes = outCompressedBlob.size();
955 if (stats.rawSizeBytes > 0) {
956 stats.compressionRatio = (1.0 - (static_cast<double>(stats.compressedSizeBytes) / static_cast<double>(stats.rawSizeBytes))) * 100.0;
957 }
958
959 LOG_INFO(BinarySerializer, "Serialized page '" + page.title + "' (" + page.guid + ") -> " + stats.ToString());
960
961 if (outStats) {
962 *outStats = stats;
963 }
964 return true;
965}

967bool BinarySerializer::DeserializePage(const uint8_t* blobData, size_t blobSize, CanvasPage& outPage, SerializationStats* outStats) {
968 if (!blobData || blobSize < 8) {
969 LOG_ERROR(BinarySerializer, "DeserializePage failed: input blob is null or smaller than minimum container header (size: " + std::to_string(blobSize) + " B)");
970 return false;
971 }
972
973 SerializationStats stats;
974 stats.compressedSizeBytes = blobSize;
975
976 uint32_t uncompressedSize = static_cast<uint32_t>(blobData[0]) |
977 (static_cast<uint32_t>(blobData[1]) << 8) |
978 (static_cast<uint32_t>(blobData[2]) << 16) |
979 (static_cast<uint32_t>(blobData[3]) << 24);
980
981 stats.rawSizeBytes = uncompressedSize;
982
983 const uint8_t* payloadPtr = nullptr;
984 size_t payloadSize = 0;
985 uint32_t expectedCRC = 0;
986
987 if (blobSize >= 12) {
988 expectedCRC = static_cast<uint32_t>(blobData[4]) |
989 (static_cast<uint32_t>(blobData[5]) << 8) |
990 (static_cast<uint32_t>(blobData[6]) << 16) |
991 (static_cast<uint32_t>(blobData[7]) << 24);
992
993 uint32_t actualCRC = CRC32::Compute(blobData + 8, blobSize - 8);
994 if (actualCRC == expectedCRC) {
995 payloadPtr = blobData + 8;
996 payloadSize = blobSize - 8;
997 stats.crc32Checksum = actualCRC;
998 } else if (blobData[8] == 0x78) {
999 LOG_ERROR(BinarySerializer, "DeserializePage failed: CRC32 integrity check failed! Expected: 0x" +
1000 std::to_string(expectedCRC) + " | Actual: 0x" + std::to_string(actualCRC) +
1001 " (Data corruption detected in page payload)");
1002 return false;
1003 }
1004 }
1005
1006 if (!payloadPtr) {
1007 if (blobSize >= 4 && blobData[4] == 0x78) {
1008 payloadPtr = blobData + 4;
1009 payloadSize = blobSize - 4;
1010 stats.crc32Checksum = 0;
1011 } else {
1012 LOG_ERROR(BinarySerializer, "DeserializePage failed: unrecognized or corrupted container header");
1013 return false;
1014 }
1015 }
1016
1017 std::vector<uint8_t> uncompressed;
1018 if (!Decompress(payloadPtr, payloadSize, uncompressedSize, uncompressed)) {
1019 LOG_ERROR(BinarySerializer, "DeserializePage failed: zlib decompression error for blob size " + std::to_string(blobSize) + " B");
1020 return false;
1021 }
1022
1023 try {
1024 ByteReader reader(uncompressed.data(), uncompressed.size());
1025
1026 uint32_t magic = reader.ReadU32();
1027 if (magic != MAGIC_HEADER) {
1028 LOG_ERROR(BinarySerializer, "DeserializePage failed: Invalid Magic Header 0x" + std::to_string(magic) + " (Expected: 0x464E5047 / 'FNPG')");
1029 return false;
1030 }
1031
1032 uint32_t version = reader.ReadU32();
1033 if (version > FORMAT_VERSION) {
1034 LOG_ERROR(BinarySerializer, "DeserializePage failed: Unsupported binary format version " + std::to_string(version) + " (Current supported: " + std::to_string(FORMAT_VERSION) + ")");
1035 return false;
1036 }
1037
1038 // 1. Identity & Timestamps
1039 outPage.guid = reader.ReadString();
1040 outPage.title = reader.ReadString();
1041 outPage.createdDateStr = reader.ReadString();
1042 outPage.createdTimeStr = reader.ReadString();
1043
1044 // 2. Per-Page Template, Layout, Boundary & Grid Configuration
1045 outPage.infinityMode = static_cast<CanvasInfinityMode>(reader.ReadU8());
1046 outPage.paperStyle = static_cast<PaperStyle>(reader.ReadU8());
1047 outPage.gridSpacingMm = reader.ReadDouble();
1048 outPage.pageSizeFormat = static_cast<PageSizeFormat>(reader.ReadU8());
1049 outPage.pageIsLandscape = reader.ReadBool();
1050 outPage.pageWidthMm = reader.ReadDouble();
1051 outPage.pageHeightMm = reader.ReadDouble();
1052
1053 // 3. Per-Page Border Configuration
1054 outPage.showPageBorder = reader.ReadBool();
1055 outPage.pageBorderType = static_cast<PageBorderType>(reader.ReadU8());
1056 outPage.pageBorderStyle = static_cast<PageBorderStyle>(reader.ReadU8());
1057 outPage.pageBorderWidth = reader.ReadDouble();
1058
1059 // 4. Dedicated Standalone PDF Document State
1060 outPage.isDedicatedPdf = reader.ReadBool();
1061 outPage.dedicatedPdfPath = reader.ReadString();
1062 outPage.dedicatedPdfBookmarks = reader.ReadString();
1063 outPage.dedicatedPdfHighlights = reader.ReadString();
1064
1065 // 5. Dedicated Standalone Markdown Document State (v6+)
1066 if (version >= 6) {
1067 outPage.isDedicatedMd = reader.ReadBool();
1068 outPage.dedicatedMdContent = reader.ReadString();
1069 } else {
1070 outPage.isDedicatedMd = false;
1071 outPage.dedicatedMdContent = "";
1072 }
1073
1074 outPage.Clear();
1075
1076 uint32_t objectCount = reader.ReadU32();
1077 if (objectCount > ByteReader::MAX_SAFE_ARRAY_LEN) {
1078 LOG_ERROR(BinarySerializer, "DeserializePage failed: object count exceeds safe limit (" + std::to_string(objectCount) + ")");
1079 return false;
1080 }
1081
1082 outPage.objects.reserve(objectCount);
1083 stats.objectCount = objectCount;
1084
1085
1086 for (uint32_t i = 0; i < objectCount; ++i) {
1087 auto obj = DeserializeObject(reader, stats, version);
1088 if (obj) {
1089 outPage.AddObject(obj);
1090 }
1091 }
1092
1093 outPage.isModified = false;
1094 outPage.isLoaded = true;
1095 outPage.Touch();
1096
1097 if (stats.rawSizeBytes > 0) {
1098 stats.compressionRatio = (1.0 - (static_cast<double>(stats.compressedSizeBytes) / static_cast<double>(stats.rawSizeBytes))) * 100.0;
1099 }
1100
1101 LOG_INFO(BinarySerializer, "Successfully deserialized page '" + outPage.title + "' (" + outPage.guid + ") -> " + stats.ToString());
1102
1103 if (outStats) {
1104 *outStats = stats;
1105 }
1106 return true;
1107
1108 } catch (const std::exception& e) {
1109 LOG_ERROR(BinarySerializer, "Exception during page binary deserialization: " + std::string(e.what()));
1110 return false;
1111 }
1112}

1137bool BinarySerializer::ApplyJournalEntry(CanvasPage& page, const JournalEntry& entry) {
1138 page.Touch();
1139 page.isModified = true;
1140
1141 switch (entry.opType) {
1142 case JournalOpType::AddObject: {
1143 if (entry.payload.empty()) return false;
1144 ByteReader reader(entry.payload.data(), entry.payload.size());
1145 SerializationStats stats;
1146 auto obj = DeserializeObject(reader, stats, FORMAT_VERSION);
1147 if (obj) {
1148 page.AddObject(obj);
1149 return true;
1150 }
1151 return false;
1152 }
1153
1154 case JournalOpType::RemoveObject: {
1155 for (const auto& obj : page.objects) {
1156 if (obj && obj->guuid == entry.targetGuid) {
1157 page.RemoveObject(obj);
1158 return true;
1159 }
1160 }
1161 return false;
1162 }
1163
1164 case JournalOpType::TransformObject: {
1165 if (entry.payload.size() < 48 + 32) return false;
1166 ByteReader reader(entry.payload.data(), entry.payload.size());
1167 BLMatrix2D transform;
1168 transform.m00 = reader.ReadDouble();
1169 transform.m01 = reader.ReadDouble();
1170 transform.m10 = reader.ReadDouble();
1171 transform.m11 = reader.ReadDouble();
1172 transform.m20 = reader.ReadDouble();
1173 transform.m21 = reader.ReadDouble();
1174
1175 AABB bounds;
1176 bounds.minX = reader.ReadDouble();
1177 bounds.minY = reader.ReadDouble();
1178 bounds.maxX = reader.ReadDouble();
1179 bounds.maxY = reader.ReadDouble();
1180
1181 for (const auto& obj : page.objects) {
1182 if (obj && obj->guuid == entry.targetGuid) {
1183 obj->transform = transform;
1184 obj->bounds = bounds;
1185 page.UpdateObject(obj);
1186 return true;
1187 }
1188 }
1189 return false;
1190 }
1191
1192 case JournalOpType::ModifyStyle: {
1193 if (entry.payload.size() < 4 + 8 + 4) return false;
1194 ByteReader reader(entry.payload.data(), entry.payload.size());
1195 uint32_t colorVal = reader.ReadU32();
1196 double baseWidth = reader.ReadDouble();
1197 float opacity = reader.ReadFloat();
1198
1199 for (const auto& obj : page.objects) {
1200 if (obj && obj->guuid == entry.targetGuid) {
1201 obj->opacity = opacity;
1202 if (obj->type == ObjectType::InkContainer) {
1203 auto ink = std::static_pointer_cast<InkContainer>(obj);
1204 for (auto& st : ink->strokes) {
1205 st.color.value = colorVal;
1206 st.baseWidth = baseWidth;
1207 }
1208 ink->InvalidateCache();
1209 }
1210 page.UpdateObject(obj);
1211 return true;
1212 }
1213 }
1214 return false;
1215 }
1216
1217 case JournalOpType::ModifyZOrder: {
1218 if (entry.payload.size() < 4) return false;
1219 ByteReader reader(entry.payload.data(), entry.payload.size());
1220 int32_t zOrder = static_cast<int32_t>(reader.ReadU32());
1221
1222 for (const auto& obj : page.objects) {
1223 if (obj && obj->guuid == entry.targetGuid) {
1224 obj->zOrder = zOrder;
1225 return true;
1226 }
1227 }
1228 return false;
1229 }
1230
1231 case JournalOpType::ModifyText: {
1232 if (entry.payload.empty()) return false;
1233 ByteReader reader(entry.payload.data(), entry.payload.size());
1234 std::string updatedText = reader.ReadString();
