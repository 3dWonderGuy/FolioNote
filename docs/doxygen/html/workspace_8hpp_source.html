11/**
12 * =========================================================================================
13 * @file workspace.hpp
14 * @brief Represents the user's root workspace, managing open Notebooks and storage.
15 * =========================================================================================
16 *
17 * ARCHITECTURAL ROLE:
18 * - The Workspace is the top-level container for all open Notebooks in the application.
19 * - It owns the primary `Folio::PageRepository` instance, acting as the gateway to notebook
20 * directory storage (SQLite `structure.db` hierarchy and self-contained `.ink` page BLOBs).
21 *
22 * CORE LIFECYCLE RESPONSIBILITIES:
23 * 1. Workspace Discovery & Initialization (`LoadWorkspace`):
24 * - Scans the designated root directory (e.g. `FolioNote/`) and library directories for `.notebook` folders.
25 * - Automatically deserializes notebook hierarchies, sections, and page metadata via `PageRepository`.
26 * - If no notebooks exist on disk, it auto-generates a `DemoBook.notebook` directory with default schema.
27 * 2. On-Demand Lazy Loading:
28 * - In `GetActivePage()`, if the requested page is marked `!isLoaded` (its vector strokes, paper settings,
29 * and canvas objects have not yet been read from its `.ink` file or were evicted to save memory), the
30 * Workspace loads it on-demand from disk via `repository.LoadPage(page)`.
31 * 3. Asynchronous Persistence (`FlushActiveNotebookAsync`):
32 * - Dispatches save tasks to background worker threads via `PageRepository::SaveNotebookAsync`,
33 * preventing UI hitching while serializing self-contained `.ink` page files and committing
34 * hierarchy metadata to SQLite.
35 * 4. Dual-Axis LRU Working-Set Cache Eviction (`MaintainWorkingSetLRU`):
36 * - Periodically called to evict in-memory strokes of inactive pages that haven't been accessed
37 * for a configurable timeout or when resident pages exceed the capacity cap, keeping RAM usage
38 * strictly bounded regardless of notebook library size.
39 *
40 * POTENTIAL FUTURE ENHANCEMENTS:
41 * - Two-Tier Discovery vs. Open Working Set (OneNote Model):
42 * - Separate the Library Catalog (`LibraryManager` / Notebook Hub discovery) from the Active
43 * Working Set (`Workspace::notebooks`).
44 * - Instead of auto-mounting every discovered notebook on disk into the active sidebar on startup
45 * (which can bloat UI and memory if a user has 50+ notebooks), maintain a persistent list of
46 * "Opened Notebooks" in session state.
47 * - Provide a "Close Notebook" action that unmounts a notebook from `Workspace::notebooks`
48 * (hiding it from the active sidebar without deleting it from disk) and an "Open More Notebooks"
49 * dialog to mount notebooks from the library catalog into the working set on demand.
50 * - Session State Persistence: Remember which notebook, section, and page was open on last exit.
51 * - Multi-Notebook Tabs: Switch between multiple notebooks in separate UI tabs or split-views.
52 * - Cloud / Sync Bridge: Check for file updates or cloud sync notifications on active notebooks.
53 * - Background Auto-Save Timer: Automatically trigger `FlushActiveNotebookAsync` on a periodic schedule.
54 */

103 if (!std::filesystem::exists(libPath, ec) || !std::filesystem::is_directory(libPath, ec)) return;
104 LOG_INFO(General, "Workspace: Scanning library folder: " + libPath.string());

106 if (std::filesystem::is_directory(entry.status()) && entry.path().extension() == ".notebook") {

109 bool isDuplicate = false;

112 if (existing->guid == nb->guid) {
113 isDuplicate = true;

118 std::filesystem::equivalent(existing->filePath, nb->filePath, eqEc)) {
119 isDuplicate = true;

121 }

125 LOG_INFO(Notebook, "Workspace: Loaded notebook '" + nb->name + "' [" + nb->guid + "] from: " + entry.path().string());

127 foundAny = true;

130 }
131 } else {
132 LOG_ERROR_CODE(Notebook, FolioErrorCode::DocNotebookLoadFailed,

185 LOG_INFO(Notebook, "Workspace: Loaded migrated notebook '" + nb->name + "' [" + nb->guid + "]");

188 }
189 }
190 }
191 }

196 if (!foundAny) {
197 std::string demoPath = (defaultLib / "DemoBook.notebook").string();
198 LOG_INFO(Notebook, "Workspace: No existing notebooks found. Creating initial DemoBook at: " + demoPath);

200 auto demoNb = std::make_shared<Notebook>("DemoBook", ImVec4(0.20f, 0.48f, 0.92f, 1.0f));
201 demoNb->filePath = demoPath;

203 if (repository.OpenNotebookPackage(demoPath)) {
204 repository.SaveNotebookAsync(demoNb);
205 }
206
207 notebooks.push_back(demoNb);
208 }
209
210 activeNotebookIndex = 0;
211 LOG_INFO(General, "Workspace: Discovery complete. Total notebooks loaded: " + std::to_string(notebooks.size()));

261 for (const auto& sec : nb->sections) {
262 if (sec) {
263 if (auto p = sec->FindPageByGuid(pageGuid)) return p;
264 }
265 }
266 for (const auto& grp : nb->sectionGroups) {
267 if (!grp) continue;
268 for (const auto& sec : grp->sections) {
269 if (sec) {
270 if (auto p = sec->FindPageByGuid(pageGuid)) return p;
271 }
272 }
273 for (const auto& sub : grp->subGroups) {
274 if (!sub) continue;
275 for (const auto& sec : sub->sections) {
276 if (sec) {
277 if (auto p = sec->FindPageByGuid(pageGuid)) return p;
278 }
279 }
280 }

302 /**
303 * @brief Evaluates all inactive pages in the working set and evicts those exceeding timeoutMs or capacity cap.
304 *
305 * DUAL-AXIS LRU EVICTION ALGORITHM & WORKING PROCESS:
306 * 1. Active Page Immunity: The currently viewed page is pinned in RAM and never evicted.
307 * 2. Axis 1 (Inactivity Timeout): Inactive resident pages unaccessed for longer than `timeoutMs`
308 * qualify for eviction.
309 * 3. Axis 2 (Capacity Cap): When total loaded pages exceed `maxLoadedPages`, the oldest unaccessed
310 * pages are evicted first.
311 * 4. Zero Data Loss: Any dirty page is persisted to its .ink file before its RAM objects are cleared.
312 *
313 * Mathematical Bound:
314 * Total resident page RAM is strictly bounded by O(K * avgPageSize) where K = maxLoadedPages,
315 * guaranteeing predictable memory footprint regardless of library or notebook size.
316 *
317 * @param timeoutMs Milliseconds of inactivity before in-memory strokes are unloaded (default: 60s).
318 * @param maxLoadedPages Maximum resident pages allowed in RAM (default: 10).
319 */
320 void MaintainWorkingSetLRU(uint64_t timeoutMs = 60000, uint32_t maxLoadedPages = 10) {
321 auto nb = GetActiveNotebook();
322 if (nb) {
323 repository.MaintainLRUCache(nb, timeoutMs, maxLoadedPages);
324 }
325 }

345 /**
346 * @brief Closes a notebook by GUID and adjusts activeNotebookIndex.
347 * @return True if notebook was found and closed.
348 */
349 bool CloseNotebook(const std::string& guid) {
350 auto it = std::find_if(notebooks.begin(), notebooks.end(), [&](const std::shared_ptr<Notebook>& nb) {
351 return nb && nb->guid == guid;
352 });
353
354 if (it != notebooks.end()) {
355 std::string nbName = (*it)->name;
356 size_t index = std::distance(notebooks.begin(), it);
357 notebooks.erase(it);
358
359 if (notebooks.empty()) {
360 activeNotebookIndex = 0;
361 } else if (activeNotebookIndex >= notebooks.size() || activeNotebookIndex == index) {
362 activeNotebookIndex = (notebooks.size() > 0) ? std::min(index, notebooks.size() - 1) : 0;
363 }
364 LOG_INFO(Notebook, "Workspace: Closed notebook '" + nbName + "' [" + guid + "]. Remaining open: " + std::to_string(notebooks.size()));
365 return true;
366 }
367 LOG_WARN_CODE(Notebook, FolioErrorCode::DocNotebookNotFound,
368 "Workspace::CloseNotebook failed: Notebook GUID not found: " + guid);
369 return false;
370 }

398 "Workspace::OpenAndActivateNotebook: Failed to open notebook at: " + notebook->filePath);
399 return false;
