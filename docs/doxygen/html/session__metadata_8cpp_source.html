138/**
139 * @brief Checks whether the active page or any resident pages have unsaved modifications.
140 * @return true if unsaved changes exist in RAM, false otherwise.
141 */
142bool DocumentSession::HasUnsavedChanges() const {
143 auto activePage = GetActivePage();
144 if (activePage && activePage->isModified) return true;
145 auto nb = workspace.GetActiveNotebook();
146 if (!nb) return false;
147 for (const auto& s : nb->sections) {
148 if (!s) continue;
149 for (const auto& p : s->pages) {
150 if (p && p->isLoaded && p->isModified) return true;
151 }
152 }
153 for (const auto& g : nb->sectionGroups) {
154 if (!g) continue;
155 for (const auto& s : g->sections) {
156 if (!s) continue;
157 for (const auto& p : s->pages) {
158 if (p && p->isLoaded && p->isModified) return true;
159 }
160 }
161 }
162 return false;
163}

165/**
166 * @brief Flushes the active CanvasPage asynchronously to disk via atomic .ink staging.
167 */
168std::future<bool> DocumentSession::SaveActivePageAsync() {
169 auto activePage = GetActivePage();
170 if (!activePage) {
171 std::promise<bool> p;
172 p.set_value(false);
173 return p.get_future();
174 }
175 auto nb = workspace.GetActiveNotebook();
176 std::string secGuid;
177 int32_t sortOrder = activePage->sortOrder;
178 if (nb) {
179 auto sec = nb->GetActiveSection();
180 if (sec) secGuid = sec->guid;
181 }
182 return workspace.repository.SavePageAsync(activePage, secGuid, sortOrder);
183}

267/**
268 * @brief Broadcasts a page dirty/modified state event.
269 *
270 * AUTOMATIC SESSION VERSIONING:
271 * Tracks user page interactions in the current session. On the first modification to a page
272 * during this application run, captures an initial session checkpoint and enforces the
273 * configured retention cap (`maxPageVersionsToKeep`, where 0 = unlimited).
274 */
275void DocumentSession::NotifyPageModified(const std::shared_ptr<CanvasPage>& page) {
276 if (page && !page->guid.empty()) {
277 if (sessionInteractedPages.find(page->guid) == sessionInteractedPages.end()) {
278 sessionInteractedPages.insert(page->guid);
279 auto activeNb = GetActiveNotebook();
280 if (activeNb && !activeNb->filePath.empty()) {
281 Folio::PageVersionManager::CreateRevision(*page, activeNb->filePath, "Session Autosave", false);
282 int cap = SettingsManager::Instance().maxPageVersionsToKeep;
283 Folio::PageVersionManager::PruneRevisions(activeNb->filePath, page->guid, static_cast<size_t>(cap));
284 }
285 }
286 }
287
288 auto snapshot = observers;
289 for (auto* obs : snapshot) {
290 if (obs) obs->OnPageModified(page);
291 }
292}

318/**
319 * @brief Captures a point-in-time revision snapshot of the active page.
320 *
321 * WORKING PROCESS:
322 * 1. Resolves active page and containing notebook bundle path.
323 * 2. Delegates serialization to PageVersionManager.
324 * 3. Enforces unnamed revision retention cap from SettingsManager (0 = unlimited).
325 *
326 * @param versionName Optional descriptive milestone label (e.g. "Draft 1", "Before Midterm").
327 * @param isNamed If true, marks version as permanent milestone immune to auto-pruning.
328 * @return true on success; false if active page is invalid.
329 */
330bool DocumentSession::CreateActivePageRevision(const std::string& versionName, bool isNamed) {
331 auto activePage = GetActivePage();
332 auto activeNb = GetActiveNotebook();
333 if (!activePage || !activeNb || activeNb->filePath.empty()) return false;
334
335 Folio::PageRevisionInfo info;
336 bool success = Folio::PageVersionManager::CreateRevision(
337 *activePage, activeNb->filePath, versionName, isNamed, &info
338 );
339
340 if (success && !isNamed) {
341 int cap = SettingsManager::Instance().maxPageVersionsToKeep;
342 Folio::PageVersionManager::PruneRevisions(activeNb->filePath, activePage->guid, static_cast<size_t>(cap));
343 }
344 return success;
345}

358/**
359 * @brief Restores the active page to a specific historical revision.
360 *
361 * NON-DESTRUCTIVE ROLLBACK:
362 * If `createSafetySnapshot` is true, automatically captures a safety snapshot of current
363 * page state prior to overwriting it with the chosen historical revision.
364 */
365bool DocumentSession::RestoreActivePageRevision(const std::string& versionId, bool createSafetySnapshot) {
366 auto activePage = GetActivePage();
367 auto activeNb = GetActiveNotebook();
368 if (!activePage || !activeNb || activeNb->filePath.empty()) return false;
369
370 bool success = Folio::PageVersionManager::RestoreRevision(
371 *activePage, activeNb->filePath, versionId, createSafetySnapshot
372 );
373
374 if (success) {
375 NotifyPageModified(activePage);
376 NotifyHistoryChanged();
377 }
