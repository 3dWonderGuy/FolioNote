3/**
4 * =========================================================================================
5 * @file thread_pool.hpp
6 * @brief High-performance asynchronous worker thread pool for the Folio core engine.
7 *
8 * General Concept (How it works in Plain English):
9 * Think of the ThreadPool like a restaurant kitchen:
10 * - The "Worker Threads" are the cooks (usually 3 to 7 cooks based on your CPU cores).
11 * - The "Task Queue" is the order ticket carousel.
12 *
13 * Instead of hiring a brand new cook every time an order comes in (which in computer terms
14 * means spawning an OS thread—an expensive operation that consumes ~1MB of stack and thousands
15 * of CPU cycles), we keep a small team of cooks permanently waiting in the kitchen.
16 *
17 * When an order arrives (e.g. saving an ink file, generating a thumbnail, or indexing notes):
18 * 1. The caller drops the task onto the ticket carousel (tasks queue).
19 * 2. A sleeping worker cook is immediately woken up by a bell (condition_variable).
20 * 3. The cook grabs the task, cooks it, and goes right back to sleep until the next order.
21 * 4. If there are no tasks, all workers sleep at 0% CPU usage.
22 * =========================================================================================
23 */
24

45 /**
46 * @brief Constructs the ThreadPool and spawns background worker threads.
47 *
48 * CPU Allocation Math & Heuristic:
49 * - If `threads == 0`, we query the operating system for total logical CPU cores (`std::thread::hardware_concurrency()`).
50 * - Formula: `threads = (cores > 2) ? (cores - 1) : 2;`
51 * - Why `cores - 1`? We always reserve 1 entire core exclusively for the main UI rendering thread!
52 * This guarantees the interactive canvas stays silky smooth at 60–120 FPS even when background
53 * threads are busy saving huge files or indexing search databases.
54 *
55 * @param threads Number of worker threads to spawn (0 = auto-detect hardware).
56 */
57 explicit ThreadPool(size_t threads = 0)
58 : stop(false) {
59
60 // Step 1: Auto-calculate optimal worker count based on physical hardware
61 if (threads == 0) {
62 unsigned int hw = std::thread::hardware_concurrency();
63 threads = (hw > 2) ? static_cast<size_t>(hw - 1) : 2u;
64 }
65
66 workers.reserve(threads);
67
68 // Step 2: Spawn worker threads and set up their permanent loop
69 for (size_t i = 0; i < threads; ++i) {
70 workers.emplace_back([this, i] {
71 while (true) {
72 std::function<void()> task;
73
74 // --- Critical Section: Waiting for Work ---
75 {
76 // Acquire unique lock over the shared task queue
77 std::unique_lock<std::mutex> lock(this->queueMutex);
78
79 // Put thread to sleep until either:
80 // a) A new task is pushed into the queue (!tasks.empty())
81 // b) The pool is shutting down (stop == true)
82 // While sleeping, condition.wait releases the mutex and uses 0% CPU cycles.
83 this->condition.wait(lock, [this] {
84 return this->stop.load(std::memory_order_acquire) || !this->tasks.empty();
85 });
86
87 // If shutdown was requested and no tasks remain in queue, exit thread cleanly
88 if (this->stop.load(std::memory_order_acquire) && this->tasks.empty()) {
89 return;
90 }
91
92 // Pop the oldest task (FIFO order) from queue
93 task = std::move(this->tasks.front());
94 this->tasks.pop();
95 } // Mutex is released here so other workers can grab tasks simultaneously!
96
97 // --- Task Execution & Sandbox ---
98 if (task) {
99 try {
100 task(); // Execute the actual work
101 } catch (const std::exception& ex) {
102 // Catch standard C++ exceptions so one bad task doesn't kill the entire application
103 LOG_ERROR_CODE(General, FolioErrorCode::ThreadPoolWorkerException,
104 "Worker [" + std::to_string(i) + "] caught unhandled exception: " + ex.what());
105 } catch (...) {
106 // Catch non-standard exceptions
107 LOG_ERROR_CODE(General, FolioErrorCode::ThreadPoolWorkerException,
108 "Worker [" + std::to_string(i) + "] caught non-standard unknown exception");
109 }
110 }
111 }
112 });
113 }
114
115 LOG_INFO(General, "Initialized worker thread pool with " + std::to_string(threads) + " threads");
116 }

118 /**
119 * @brief Enqueues a callable task and returns a `std::future` to receive its return value.
120 *
121 * Use Case:
122 * Use `Enqueue` when you need to know WHEN the task finishes, or when you need its RETURN VALUE.
123 * Example:
124 * std::future<bool> result = pool.Enqueue([]() { return LoadHeavyFile(); });
125 * // Later...
126 * if (result.get()) { ... }
127 *
128 * Working Process:
129 * 1. Packages the function `f` and arguments into a `std::packaged_task`.
130 * 2. Extracts the `std::future` associated with the packaged task.
131 * 3. Locks `queueMutex` and checks if the pool has already been stopped.
132 * 4. Pushes the task into the queue.
133 * 5. Rings the bell (`condition.notify_one()`) to wake up 1 sleeping worker thread.
134 * 6. Returns the `future` to the caller.
135 *
136 * @tparam F Callable function, lambda, or member function pointer.
137 * @tparam Args Argument types to pass to the function.
138 * @return std::future holding the eventual return value of the function.
139 */
140 template<class F, class... Args>
141 auto Enqueue(F&& f, Args&&... args)
142 -> std::future<typename std::invoke_result<F, Args...>::type> {
143 using return_type = typename std::invoke_result<F, Args...>::type;
144
145 // Wrap callable into packaged task so it can fulfill a future
146 auto task = std::make_shared<std::packaged_task<return_type()>>(
147 std::bind(std::forward<F>(f), std::forward<Args>(args)...)
148 );
149
150 std::future<return_type> res = task->get_future();
151
152 {
153 std::unique_lock<std::mutex> lock(queueMutex);
154
155 // Rejection Guard: Do not accept new work if pool is already being shut down
156 if (stop.load(std::memory_order_acquire)) {
157 LOG_ERROR_CODE(General, FolioErrorCode::ThreadPoolShuttingDown,
158 "Enqueue called on a stopped thread pool");
159 throw std::runtime_error("Folio::ThreadPool: Enqueue called on a stopped thread pool.");
160 }
161
162 tasks.emplace([task]() {
163 (*task)();
164 });
165 }
166
167 // Wake up a sleeping worker thread to pick up this new task
168 condition.notify_one();
169 return res;
170 }

172 /**
173 * @brief Enqueues a "fire-and-forget" task without the overhead of creating a std::future.
174 *
175 * Use Case:
176 * Use `EnqueueDetached` when you don't care about the return value and don't need to block waiting for it.
177 * Examples: Auto-saving a dirty page to disk, pruning a cache, or logging metrics in the background.
178 *
179 * Working Process:
180 * 1. Locks `queueMutex`.
181 * 2. If shutting down, logs warning and drops task safely.
182 * 3. Pushes raw `std::function<void()>` into the FIFO task queue.
183 * 4. Rings the bell (`condition.notify_one()`) to wake up 1 sleeping worker thread.
184 *
185 * @param task Void lambda or callable to execute asynchronously.
186 */
187 void EnqueueDetached(std::function<void()> task) {
188 {
189 std::unique_lock<std::mutex> lock(queueMutex);
190
191 if (stop.load(std::memory_order_acquire)) {
192 LOG_WARN_CODE(General, FolioErrorCode::ThreadPoolShuttingDown,
193 "EnqueueDetached called on a stopped thread pool; dropping task");
194 return;
195 }
196
197 tasks.push(std::move(task));
198 }
199
200 // Wake up a sleeping worker thread
201 condition.notify_one();
202 }

211 /**
212 * @brief Cleanly shuts down the thread pool.
213 *
214 * Working Process:
215 * 1. Atomically sets `stop = true`.
216 * 2. Wakes up ALL sleeping worker threads simultaneously (`condition.notify_all()`).
217 * 3. Drains remaining tasks in the queue.
218 * 4. Calls `join()` on each OS worker thread, waiting for them to finish cleanly.
219 * 5. Clears worker list and logs shutdown confirmation.
220 */
221 void Shutdown() {
222 bool expected = false;
223 if (stop.compare_exchange_strong(expected, true, std::memory_order_acq_rel)) {
224 LOG_INFO(General, "Initiating shutdown; draining " + std::to_string(tasks.size()) +
225 " pending tasks across " + std::to_string(workers.size()) + " workers");
226
227 // Wake up all sleeping threads so they see stop == true
228 condition.notify_all();
229
230 // Wait for all threads to terminate
231 for (std::thread& worker : workers) {
232 if (worker.joinable()) {
233 worker.join();
234 }
235 }
236
237 workers.clear();
238 LOG_INFO(General, "All worker threads terminated successfully");
239 }
240 }
