63 /**
64 * @brief Builds a 2D closed ribbon polygon or patterned outline (solid, dashed, dotted) from an ordered list of points.
65 *
66 * @param rawPoints Input centerline vertices with thickness.
67 * @param capType Endcap style for terminal segments.
68 * @param pattern Stroke pattern (Solid, Dashed, Dotted, DashDot).
69 * @param isClosed When true, closes the loop from the last point to the first point seamlessly.
70 */
71 static BLPath BuildOutline(const std::vector<InputPoint>& rawPoints,
72 CapType capType = CapType::Round,
73 StrokePattern pattern = StrokePattern::Solid,
74 bool isClosed = false) {
75 BLPath path;
76 if (rawPoints.empty()) return path;
77
78 // If closed, ensure vertices form a complete loop by appending the origin if needed
79 std::vector<InputPoint> pointsCopy;
80 const std::vector<InputPoint>* ptsPtr = &rawPoints;
81 if (isClosed && rawPoints.size() >= 3) {
82 double dx = rawPoints.back().x - rawPoints.front().x;
83 double dy = rawPoints.back().y - rawPoints.front().y;
84 if (std::hypot(dx, dy) > 1e-5) {
85 pointsCopy = rawPoints;
86 pointsCopy.push_back(rawPoints.front());
87 ptsPtr = &pointsCopy;
88 }
89 }
90 const auto& pts = *ptsPtr;
91
92 if (pts.size() == 1) {
93 double r = std::max(0.05, (double)pts[0].width * 0.5);
94 path.add_circle(BLCircle{pts[0].x, pts[0].y, r});
95 return path;
96 }
97
98 if (pattern == StrokePattern::Dotted) {
99 double nextDotDist = 0.0;
100 double distAccum = 0.0;
101
102 for (size_t i = 0; i < pts.size() - 1; ++i) {
103 const auto& p0 = pts[i];
104 const auto& p1 = pts[i + 1];
105 double dx = p1.x - p0.x;
106 double dy = p1.y - p0.y;
107 double segLen = std::hypot(dx, dy);
108 if (segLen < 1e-7) continue;
109
110 while (nextDotDist <= distAccum + segLen) {
111 double t = (nextDotDist - distAccum) / segLen;
112 t = std::clamp(t, 0.0, 1.0);
113 double dotX = p0.x + t * dx;
114 double dotY = p0.y + t * dy;
115 float dotW = p0.width + static_cast<float>(t) * (p1.width - p0.width);
116 double r = std::max(0.05, (double)dotW * 0.5);
117 path.add_circle(BLCircle{dotX, dotY, r});
118
119 double spacing = std::max(0.6, (double)dotW * 2.2);
120 nextDotDist += spacing;
121 }
122 distAccum += segLen;
123 }
124 if (path.is_empty() && !pts.empty()) {
125 double r = std::max(0.05, (double)pts[0].width * 0.5);
126 path.add_circle(BLCircle{pts[0].x, pts[0].y, r});
127 }
128 return path;
129 }
130
131 if (pattern == StrokePattern::Dashed) {
132 const size_t N = pts.size();
133 std::vector<double> cumDist(N, 0.0);
134 for (size_t i = 1; i < N; ++i) {
135 cumDist[i] = cumDist[i - 1] + std::hypot(pts[i].x - pts[i - 1].x, pts[i].y - pts[i - 1].y);
136 }
137 double totalLen = cumDist.back();
138 if (totalLen < 1e-5) {
139 double r = std::max(0.05, (double)pts[0].width * 0.5);
140 path.add_circle(BLCircle{pts[0].x, pts[0].y, r});
141 return path;
142 }
143
144 float avgWidth = 0.0f;
145 for (const auto& pt : pts) avgWidth += pt.width;
146 avgWidth /= static_cast<float>(N);
147
148 double dashLen = std::max(1.8, (double)avgWidth * 4.0);
149 double gapLen = std::max(1.0, (double)avgWidth * 2.5);
150 double cycle = dashLen + gapLen;
151
152 auto samplePointAt = [&](double d) -> InputPoint {
153 d = std::clamp(d, 0.0, totalLen);
154 auto it = std::lower_bound(cumDist.begin(), cumDist.end(), d);
155 if (it == cumDist.begin()) return pts.front();
156 if (it == cumDist.end()) return pts.back();
157 size_t idx = std::distance(cumDist.begin(), it);
158 double d0 = cumDist[idx - 1];
159 double d1 = cumDist[idx];
160 double segL = d1 - d0;
161 double t = (segL > 1e-7) ? (d - d0) / segL : 0.0;
162 t = std::clamp(t, 0.0, 1.0);
163 const auto& p0 = pts[idx - 1];
164 const auto& p1 = pts[idx];
165 InputPoint res;
166 res.x = p0.x + t * (p1.x - p0.x);
167 res.y = p0.y + t * (p1.y - p0.y);
168 res.width = p0.width + static_cast<float>(t) * (p1.width - p0.width);
169 res.pressure = p0.pressure + static_cast<float>(t) * (p1.pressure - p0.pressure);
170 return res;
171 };
172
173 for (double s0 = 0.0; s0 < totalLen; s0 += cycle) {
174 double s1 = std::min(totalLen, s0 + dashLen);
175 if (s1 <= s0) break;
176
177 std::vector<InputPoint> dashPts;
178 dashPts.push_back(samplePointAt(s0));
179
180 for (size_t i = 0; i < N; ++i) {
181 if (cumDist[i] > s0 + 1e-5 && cumDist[i] < s1 - 1e-5) {
182 dashPts.push_back(pts[i]);
183 }
184 }
185 dashPts.push_back(samplePointAt(s1));
186
187 if (dashPts.size() <= 1 || std::hypot(dashPts.front().x - dashPts.back().x, dashPts.front().y - dashPts.back().y) < 1e-5) {
188 double r = std::max(0.05, (double)dashPts[0].width * 0.5);
189 path.add_circle(BLCircle{dashPts[0].x, dashPts[0].y, r});
190 } else {
191 BLPath subRibbon = BuildSolidRibbon(dashPts, capType);
192 path.add_path(subRibbon);
193 }
194 }
195 return path;
196 }
197
198 if (pattern == StrokePattern::DashDot) {
199 const size_t N = pts.size();
200 std::vector<double> cumDist(N, 0.0);
201 for (size_t i = 1; i < N; ++i) {
202 cumDist[i] = cumDist[i - 1] + std::hypot(pts[i].x - pts[i - 1].x, pts[i].y - pts[i - 1].y);
203 }
204 double totalLen = cumDist.back();
205 if (totalLen < 1e-5) {
206 double r = std::max(0.05, (double)pts[0].width * 0.5);
207 path.add_circle(BLCircle{pts[0].x, pts[0].y, r});
208 return path;
209 }
210
211 float avgWidth = 0.0f;
212 for (const auto& pt : pts) avgWidth += pt.width;
213 avgWidth /= static_cast<float>(N);
214
215 double dashLen = std::max(2.0, (double)avgWidth * 4.2);
216 double gap1 = std::max(1.0, (double)avgWidth * 2.0);
217 double dotLen = std::max(0.5, (double)avgWidth * 1.0);
218 double gap2 = std::max(1.0, (double)avgWidth * 2.0);
219 double cycle = dashLen + gap1 + dotLen + gap2;
220
221 auto samplePointAt = [&](double d) -> InputPoint {
222 d = std::clamp(d, 0.0, totalLen);
223 auto it = std::lower_bound(cumDist.begin(), cumDist.end(), d);
224 if (it == cumDist.begin()) return pts.front();
225 if (it == cumDist.end()) return pts.back();
226 size_t idx = std::distance(cumDist.begin(), it);
227 double d0 = cumDist[idx - 1];
228 double d1 = cumDist[idx];
229 double segL = d1 - d0;
230 double t = (segL > 1e-7) ? (d - d0) / segL : 0.0;
231 t = std::clamp(t, 0.0, 1.0);
232 const auto& p0 = pts[idx - 1];
233 const auto& p1 = pts[idx];
234 InputPoint res;
235 res.x = p0.x + t * (p1.x - p0.x);
236 res.y = p0.y + t * (p1.y - p0.y);
237 res.width = p0.width + static_cast<float>(t) * (p1.width - p0.width);
238 res.pressure = p0.pressure + static_cast<float>(t) * (p1.pressure - p0.pressure);
239 return res;
240 };
241
242 for (double s0 = 0.0; s0 < totalLen; s0 += cycle) {
243 // 1. Dash segment
244 double s1 = std::min(totalLen, s0 + dashLen);
245 if (s1 > s0) {
246 std::vector<InputPoint> dashPts;
247 dashPts.push_back(samplePointAt(s0));
248 for (size_t i = 0; i < N; ++i) {
249 if (cumDist[i] > s0 + 1e-5 && cumDist[i] < s1 - 1e-5) {
250 dashPts.push_back(pts[i]);
251 }
252 }
253 dashPts.push_back(samplePointAt(s1));
254 if (dashPts.size() <= 1 || std::hypot(dashPts.front().x - dashPts.back().x, dashPts.front().y - dashPts.back().y) < 1e-5) {
255 double r = std::max(0.05, (double)dashPts[0].width * 0.5);
256 path.add_circle(BLCircle{dashPts[0].x, dashPts[0].y, r});
257 } else {
258 BLPath subRibbon = BuildSolidRibbon(dashPts, capType);
259 path.add_path(subRibbon);
260 }
261 }
262
263 // 2. Dot segment
264 double dotCenter = s0 + dashLen + gap1 + dotLen * 0.5;
265 if (dotCenter <= totalLen) {
266 InputPoint dotPt = samplePointAt(dotCenter);
267 double r = std::max(0.05, (double)dotPt.width * 0.55);
268 path.add_circle(BLCircle{dotPt.x, dotPt.y, r});
269 }
270 }
271 return path;
272 }
273
274 return BuildSolidRibbon(pts, capType);

280 static BLPath BuildSolidRibbon(const std::vector<InputPoint>& pts, CapType capType = CapType::Round) {
281 BLPath path;
282 if (pts.empty()) return path;
283
284 if (pts.size() == 1) {
285 double r = std::max(0.05, (double)pts[0].width * 0.5);
286 path.add_circle(BLCircle{pts[0].x, pts[0].y, r});
287 return path;
288 }
289
290 const size_t N = pts.size();
291
292 struct Vec2 { double x = 0; double y = 0; };
293 std::vector<Vec2> left(N);
294 std::vector<Vec2> right(N);
295
296 for (size_t i = 0; i < N; ++i) {
297 double nx = 0.0, ny = 0.0;
298 if (i > 0) {
299 double dx = pts[i].x - pts[i - 1].x;
300 double dy = pts[i].y - pts[i - 1].y;
301 double len = std::hypot(dx, dy);
302 if (len > 1e-7) {
303 nx += -dy / len;
304 ny += dx / len;
305 }
306 }
307 if (i < N - 1) {
308 double dx = pts[i + 1].x - pts[i].x;
309 double dy = pts[i + 1].y - pts[i].y;
310 double len = std::hypot(dx, dy);
311 if (len > 1e-7) {
312 nx += -dy / len;
313 ny += dx / len;
314 }
315 }
316 double nLen = std::hypot(nx, ny);
317 if (nLen > 1e-7) {
318 nx /= nLen;
319 ny /= nLen;
320 } else {
321 nx = 1.0;
322 ny = 0.0;
323 }
324
325 double r = std::max(0.05, (double)pts[i].width * 0.5);
326 left[i] = { pts[i].x + nx * r, pts[i].y + ny * r };
327 right[i] = { pts[i].x - nx * r, pts[i].y - ny * r };
328 }
329
330 path.move_to(left[0].x, left[0].y);
331
332 for (size_t i = 1; i < N; ++i) {
333 path.line_to(left[i].x, left[i].y);
334 }
335
336 path.line_to(right[N - 1].x, right[N - 1].y);
337
338 for (size_t i = N - 1; i > 0; --i) {
339 path.line_to(right[i - 1].x, right[i - 1].y);
340 }
341
342 path.line_to(left[0].x, left[0].y);
343 path.close();
344 return path;
345 }
