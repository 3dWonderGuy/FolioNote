3/**
4 * @file ink_engine.hpp
5 * @brief Universal inking physics, geometry synthesis, and presentation data engine.
6 *
7 * ARCHITECTURAL DESIGN:
8 * ---------------------
9 * InkEngine is a centralized, modular data and execution engine responsible for all
10 * handwriting, dynamic width evaluation, predictive stroke modeling, and presentation
11 * laser trails.
12 *
13 * Decoupling Contract:
14 * - CanvasEngine, PdfEngine, or MarkdownEditor ingest pointer events and route them into InkEngine.
15 * - The Layer Compositor (e.g. LiveInteractionLayer) interacts with InkEngine to query live stroke
16 * geometry, render predictive nib tips, or execute dirty-rect redraws.
17 * - On pen lift (PointerUp), InkEngine finalizes the stroke and hands an immutable FinishedStrokeData
18 * payload to the consumer layer / document session for database and spatial index persistence.
19 */

40/**
41 * @struct EphemeralStroke
42 * @brief Transient presentation stroke (e.g. Laser Pointer trail) with temporal quadratic fade.
43 *
44 * Mathematical Decay Model:
45 * -------------------------
46 * Given stroke start timestamp t0 (ms) and total lifetime duration T (ms),
47 * at current time t:
48 * elapsed = clamp(t - t0, 0, T)
49 * progress = elapsed / T ∈ [0.0, 1.0]
50 * alphaFactor = (1.0 - progress)^2 (Quadratic ease-out fade)
51 *
52 * The stroke is automatically pruned once elapsed >= T.
53 */
54struct EphemeralStroke {
55 BLPath outlinePath;
56 BLRgba32 baseColor;
57 uint64_t startTimeMs = 0;
58 uint32_t durationMs = 2500;
59};

92 /**
93 * @brief Initiates a new live vector stroke gesture.
94 *
95 * Working Process:
96 * 1. Resets Google Ink modeler and transient point accumulation buffers.
97 * 2. Configures spring-mass, damping drag, and wobble suppression parameters from active configuration.
98 * 3. Seeds the first sample point and establishes the initial nominal stroke thickness.
99 *
100 * @param worldX Initial stylus X position in physical millimeters.
101 * @param worldY Initial stylus Y position in physical millimeters.
102 * @param pressure Initial contact pressure [0.0, 1.0].
103 * @param timeSec Monotonic event timestamp in seconds.
104 * @param tool Active pen styling, nib dimensions, cap geometry, and pigment.
105 * @param zoomScale Current viewport zoom factor (normalizes velocity thinning to screen speed).
106 * @param tiltX Stylus tilt angle along X axis in radians/degrees.
107 * @param tiltY Stylus tilt angle along Y axis in radians/degrees.
108 */

111 float tiltX = 0.0f, float tiltY = 0.0f);

113 /**
114 * @brief Ingests continuous motion telemetry from the stylus or mouse.
115 *
116 * Working Process:
117 * 1. Submits raw input sample (pos, time, pressure) to Google Ink's position and wobble filters.
118 * 2. Solves dynamic line width via pressure gamma mapping and velocity thinning equations.
119 * 3. Applies Exponential Moving Average (EMA) smoothing to eliminate single-frame thickness jitter.
120 * 4. Reconstructs confirmed 2D outline path via StrokeOutlineBuilder.
121 * 5. Solves predictive continuation frames to minimize perceived display latency at the pen tip.
122 *
123 * @param worldX Current stylus X position in millimeters.
124 * @param worldY Current stylus Y position in millimeters.
125 * @param pressure Current contact pressure [0.0, 1.0].
126 * @param timeSec Monotonic event timestamp in seconds.
127 * @param tiltX Current stylus tilt X.
128 * @param tiltY Current stylus tilt Y.
129 */
130 void AppendPoint(double worldX, double worldY, float pressure, double timeSec,
131 float tiltX = 0.0f, float tiltY = 0.0f);
132
133 /**
134 * @brief Finalizes the active stroke gesture upon pointer release.
135 *
136 * Working Process:
137 * 1. Submits a final kUp event to flush lingering spring inertia in Google Modeler.
138 * 2. Handles single-click / tap edge cases (synthesizes a dot if no lateral movement occurred).
139 * 3. Builds the final 2D closed polygon ribbon with end-caps.
140 * 4. Clears transient live buffers and returns the immutable FinishedStrokeData snapshot.
141 *
142 * @return FinishedStrokeData The completed stroke package ready for layer / document commitment.
143 */
144 FinishedStrokeData FinishStroke();

222 * @brief Evaluates target stroke thickness combining pressure response and velocity thinning.
223 *
224 * Mathematical Working Process:
225 * 1. Stage A (Pressure Ceiling):
226 * p_norm = clamp((pressure - deadzone) / (saturation - deadzone), 0, 1)
227 * p_curve = p_norm^gamma
228 * w_pressure = baseSize * (min_mult + p_curve * (max_mult - min_mult))

230 * 2. Stage B (Velocity Thinning):

235 * 3. Result:
