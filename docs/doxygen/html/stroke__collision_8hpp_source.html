42 /**
43 * @brief Computes point-to-line-segment distance with transient register-only mini-AABB rejection.
44 */
45 [[nodiscard]] static inline bool HitTestSegment(
46 const Point2D& p0, const Point2D& p1, float width,
47 double qx, double qy, double queryRadius,
48 float& outT, double& outDistSq) noexcept
49 {
50 const double maxR = static_cast<double>(width) * 0.5 + queryRadius;
51
52 // 1. Transient segment mini-AABB in CPU registers (zero memory allocation)
53 const double segMinX = std::min(p0.x, p1.x) - maxR;
54 const double segMaxX = std::max(p0.x, p1.x) + maxR;
55 const double segMinY = std::min(p0.y, p1.y) - maxR;
56 const double segMaxY = std::max(p0.y, p1.y) + maxR;
57
58 // Fast-out: culls out 90%+ of segments with simple register comparisons
59 if (qx < segMinX || qx > segMaxX || qy < segMinY || qy > segMaxY) {
60 return false;
61 }
62
63 // 2. Exact Euclidean Point-to-Segment Math
64 const double vx = p1.x - p0.x;
65 const double vy = p1.y - p0.y;
66 const double wx = qx - p0.x;
67 const double wy = qy - p0.y;
68
69 const double lenSq = vx * vx + vy * vy;
70 double t = 0.0;
71
72 if (lenSq > 1e-9) {
73 const double dot = wx * vx + wy * vy;
74 t = std::clamp(dot / lenSq, 0.0, 1.0);
75 }
76
77 const double projX = p0.x + t * vx;
78 const double projY = p0.y + t * vy;
79
80 const double dx = qx - projX;
81 const double dy = qy - projY;
82 const double distSq = dx * dx + dy * dy;
83
84 if (distSq <= maxR * maxR) {
85 outT = static_cast<float>(t);
86 outDistSq = distSq;
87 return true;
88 }
89
90 return false;
91 }

93 /**
94 * @brief Evaluates whether a query circle intersects any segment of a stroke.
95 */
96 [[nodiscard]] static SegmentHit HitTestStroke(
97 const std::vector<Segment1D>& segments,
98 double qx, double qy, double queryRadius) noexcept
99 {
100 SegmentHit bestHit;
101 bestHit.distSq = 1e30;
102
103 for (size_t i = 0; i < segments.size(); ++i) {
104 const auto& seg = segments[i];
105 float t = 0.0f;
106 double distSq = 0.0;
107 if (HitTestSegment(seg.p0, seg.p1, seg.width, qx, qy, queryRadius, t, distSq)) {
108 if (distSq < bestHit.distSq) {
109 bestHit.hit = true;
110 bestHit.segmentIndex = i;
111 bestHit.t = t;
112 bestHit.distSq = distSq;
113 }
114 }
115 }
116
117 return bestHit;
118 }

120 /**
121 * @brief Point hit test for single-point strokes (e.g. period / tap dots).

123 [[nodiscard]] static inline bool HitTestPoint(
124 const Point2D& p, float width,
125 double qx, double qy, double queryRadius) noexcept
126 {
127 const double maxR = std::max(0.2, static_cast<double>(width) * 0.5 + queryRadius);
128 const double dx = p.x - qx;
129 const double dy = p.y - qy;
130 return (dx * dx + dy * dy) <= (maxR * maxR);

133 /**
134 * @brief Computes minimum distance squared between two line segments in 2D space.
135 * Implements Dan Sunday's robust segment-to-segment algorithm with zero heap allocation.
136 */
137 [[nodiscard]] static inline double SegmentSegmentDistanceSq(
138 const Point2D& p0, const Point2D& p1,
139 const Point2D& q0, const Point2D& q1) noexcept
140 {
141 const double ux = p1.x - p0.x;
142 const double uy = p1.y - p0.y;
143 const double vx = q1.x - q0.x;
144 const double vy = q1.y - q0.y;
145 const double wx = p0.x - q0.x;
146 const double wy = p0.y - q0.y;
147
148 const double a = ux * ux + uy * uy;
149 const double b = ux * vx + uy * vy;
150 const double c = vx * vx + vy * vy;
151 const double d = ux * wx + uy * wy;
152 const double e = vx * wx + vy * wy;
153 const double D = a * c - b * b;
154
155 double sN, sD = D;
156 double tN, tD = D;
157
158 if (D < 1e-9) {
159 sN = 0.0;
160 sD = 1.0;
161 tN = e;
162 tD = c;
163 } else {
164 sN = (b * e - c * d);
165 tN = (a * e - b * d);
166 if (sN < 0.0) {
167 sN = 0.0;
168 tN = e;
169 tD = c;
170 } else if (sN > sD) {
171 sN = sD;
172 tN = e + b;
173 tD = c;
174 }
175 }
176
177 if (tN < 0.0) {
178 tN = 0.0;
179 if (-d < 0.0) sN = 0.0;
180 else if (-d > a) sN = sD;
181 else {
182 sN = -d;
183 sD = a;
184 }
185 } else if (tN > tD) {
186 tN = tD;
187 if ((-d + b) < 0.0) sN = 0.0;
188 else if ((-d + b) > a) sN = sD;
189 else {
190 sN = (-d + b);
191 sD = a;
192 }
193 }
194
195 const double sc = (std::abs(sN) < 1e-9 ? 0.0 : sN / sD);
196 const double tc = (std::abs(tN) < 1e-9 ? 0.0 : tN / tD);
197
198 const double dpx = wx + (sc * ux) - (tc * vx);
199 const double dpy = wy + (sc * uy) - (tc * vy);
200 return dpx * dpx + dpy * dpy;
201 }

203 /**
204 * @brief Continuous swept-capsule collision test against all segments of a stroke.
205 * Detects hits regardless of mouse movement speed (zero tunneling/skipping).
206 */
207 [[nodiscard]] static inline bool HitTestStrokeSwept(
208 const std::vector<Segment1D>& segments,
209 const Point2D& w0, const Point2D& w1, double queryRadius) noexcept
210 {
211 for (const auto& seg : segments) {
212 const double maxR = queryRadius + static_cast<double>(seg.width) * 0.5;
213 const double minX = std::min({ seg.p0.x, seg.p1.x, w0.x, w1.x }) - maxR;
214 const double maxX = std::max({ seg.p0.x, seg.p1.x, w0.x, w1.x }) + maxR;
215 const double minY = std::min({ seg.p0.y, seg.p1.y, w0.y, w1.y }) - maxR;
216 const double maxY = std::max({ seg.p0.y, seg.p1.y, w0.y, w1.y }) + maxR;
217
218 if (minX > maxX || minY > maxY) continue;
219
220 const double distSq = SegmentSegmentDistanceSq(seg.p0, seg.p1, w0, w1);
221 if (distSq <= maxR * maxR) {
222 return true;
223 }
224 }
225 return false;
226 }

234 /**
235 * @brief Linearly interpolates two centerline points.
236 */
237 [[nodiscard]] static inline Point2D LerpPoint(const Point2D& p0, const Point2D& p1, double t) noexcept {
238 Point2D p;
239 p.x = p0.x + t * (p1.x - p0.x);
240 p.y = p0.y + t * (p1.y - p0.y);
241 p.pressure = p0.pressure + static_cast<float>(t) * (p1.pressure - p0.pressure);
242 p.timeSeconds = p0.timeSeconds + t * (p1.timeSeconds - p0.timeSeconds);
243 p.tiltX = p0.tiltX + static_cast<float>(t) * (p1.tiltX - p0.tiltX);
244 p.tiltY = p0.tiltY + static_cast<float>(t) * (p1.tiltY - p0.tiltY);
245 return p;
246 }

248 /**
249 * @brief Slices a chain of segments against a circular eraser kernel.
250 * @param segments Original stroke segments.
251 * @param ex Eraser center X.
252 * @param ey Eraser center Y.
253 * @param eraserRadius Eraser radius.
254 * @param outSubStrokes Destination list of surviving segment chains.
255 * @return true if the stroke was modified or split.
256 */
257 static bool SliceSegments(
258 const std::vector<Segment1D>& segments,
259 double ex, double ey, double eraserRadius,
260 std::vector<std::vector<Segment1D>>& outSubStrokes)
261 {
262 if (segments.empty()) return false;
263
264 outSubStrokes.clear();
265 bool modified = false;
266 std::vector<Segment1D> currentChain;
267
268 auto PushChainIfValid = [&]() {
269 if (!currentChain.empty()) {
270 outSubStrokes.push_back(std::move(currentChain));
271 currentChain.clear();
272 }
273 };
274
275 auto PushSegmentIfNonDegenerate = [&](const Point2D& a, const Point2D& b, float w) {
276 if (std::hypot(b.x - a.x, b.y - a.y) > 1e-4) {
277 currentChain.push_back(Segment1D{ a, b, w });
278 }
279 };
280
281 for (const auto& seg : segments) {
282 const double effR = eraserRadius + static_cast<double>(seg.width) * 0.5;
283 const double effRSq = effR * effR;
284
285 const double dx0 = seg.p0.x - ex;
286 const double dy0 = seg.p0.y - ey;
287 const bool p0In = (dx0 * dx0 + dy0 * dy0) < effRSq;
288
289 const double dx1 = seg.p1.x - ex;
290 const double dy1 = seg.p1.y - ey;
291 const bool p1In = (dx1 * dx1 + dy1 * dy1) < effRSq;
292
293 // Solve quadratic intersection: || (p0 + t * V) - E ||^2 = effRSq
294 const double vx = seg.p1.x - seg.p0.x;
295 const double vy = seg.p1.y - seg.p0.y;
296 const double wx = seg.p0.x - ex;
297 const double wy = seg.p0.y - ey;
298
299 const double a = vx * vx + vy * vy;
300 const double b = 2.0 * (wx * vx + wy * vy);
301 const double c = (wx * wx + wy * wy) - effRSq;
302
303 double t1 = -1.0, t2 = -1.0;
304 bool hasIntersection = false;
305
306 if (a > 1e-9) {
307 const double discr = b * b - 4.0 * a * c;
308 if (discr >= 0.0) {
309 const double sqrtD = std::sqrt(discr);
310 t1 = (-b - sqrtD) / (2.0 * a);
311 t2 = (-b + sqrtD) / (2.0 * a);
312 if (t1 > t2) std::swap(t1, t2);
313 hasIntersection = true;
314 }
315 }
316
317 if (!p0In && !p1In) {
318 // Both endpoints outside eraser
319 if (hasIntersection && t1 > 0.001 && t2 < 0.999 && t1 < t2) {
320 // Segment punches entirely through the eraser: splits into 2 pieces!
321 modified = true;
322 Point2D cutIn = LerpPoint(seg.p0, seg.p1, t1);
323 Point2D cutOut = LerpPoint(seg.p0, seg.p1, t2);
324
325 // First part terminates current sub-stroke
326 PushSegmentIfNonDegenerate(seg.p0, cutIn, seg.width);
327 PushChainIfValid();
328
329 // Second part begins new sub-stroke
330 PushSegmentIfNonDegenerate(cutOut, seg.p1, seg.width);
331 } else {
332 // Segment completely outside
333 currentChain.push_back(seg);
334 }
335 } else if (!p0In && p1In) {
336 // Enters eraser: cut at t1 and finish current sub-stroke
337 modified = true;
338 double tCut = (hasIntersection && t1 >= 0.0 && t1 <= 1.0) ? t1 : 0.5;
339 Point2D cutPt = LerpPoint(seg.p0, seg.p1, tCut);
340
341 PushSegmentIfNonDegenerate(seg.p0, cutPt, seg.width);
342 PushChainIfValid();
343 } else if (p0In && !p1In) {
344 // Exits eraser: cut at t2 and start new sub-stroke
345 modified = true;
346 double tCut = (hasIntersection && t2 >= 0.0 && t2 <= 1.0) ? t2 : 0.5;
347 Point2D cutPt = LerpPoint(seg.p0, seg.p1, tCut);
348
349 PushSegmentIfNonDegenerate(cutPt, seg.p1, seg.width);
350 } else {
351 // Both endpoints inside: entirely deleted
352 modified = true;
353 PushChainIfValid();
354 }
355 }
356
357 PushChainIfValid();
358 return modified;
359 }
