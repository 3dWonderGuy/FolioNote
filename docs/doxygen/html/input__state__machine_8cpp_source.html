1/**
2 * @file input_state_machine.cpp
3 * @brief Core implementation of InputStateMachine: hardware state arbitration,
4 * device priority resolution, tool persistence, and dispatcher routing.
5 *
6 * ARBITRATION ARCHITECTURE:
7 * -------------------------
8 * FolioNote implements strict single-driver device arbitration. When a user writes on a
9 * touchscreen, multiple hardware sources report input simultaneously:
10 * - Stylus active digitizer (pen tip contact, pressure, tilt)
11 * - Capacitive multi-touch sensor (palm and fingers resting on glass)
12 * - Physical mouse / trackpad
13 *
14 * To prevent palm touches from creating stray strokes or interrupting drawing, the state
15 * machine enforces a priority hierarchy:
16 * Stylus (Priority 1) > Touch (Priority 2) > Mouse (Priority 3)
17 *
18 * Temporal Grace Windows:
19 * - Pen Grace: `config.penGracePeriodMs` (default 600ms).
20 * - Touch Grace: `config.touchGracePeriodMs` (default 250ms).
21 */

30/**
31 * @brief Counts the number of active capacitive finger contacts.
32 *
33 * Checks all slots in the fixed-capacity `activeFingers` array for valid finger IDs (fingerID != -1).
34 *
35 * @return size_t Count of active finger contacts (0 to 10).
36 */
37size_t InputStateMachine::GetActiveFingerCount() const noexcept {
38 size_t count = 0;
39 for (const auto& slot : activeFingers) {
40 if (slot.fingerID != -1) ++count;
41 }
42 return count;
43}

45/**
46 * @brief Resolves active input device and handles tool memory restoration & transient overrides.
47 *
48 * ARBITRATION MATHEMATICAL FORMULATION:
49 * -------------------------------------
50 * Let $t$ be the current timestamp `nowMs`.
51 * Let $t_{pen}$ be `lastPenTimestampMs` and $t_{touch}$ be `lastTouchTimestampMs`.
52 *
53 * Condition for Stylus Priority:
54 * $C_{stylus} = \text{pen.isDown} \lor \text{pen.inProximity} \lor \text{pen.isHovering} \lor (t - t_{pen} < T_{pen\_grace})$
55 *
56 * Condition for Touch Priority (if Stylus inactive):
57 * $C_{touch} = (\text{fingerCount} > 0) \lor (t - t_{touch} < T_{touch\_grace})$
58 *
59 * Otherwise:
60 * Fallback to Mouse.
61 *
62 * @param nowMs System tick in milliseconds from SDL_GetTicks().
63 */
64void InputStateMachine::UpdateHardwareState(uint64_t nowMs) {
65 size_t fingerCount = GetActiveFingerCount();
66 DeviceType oldDevice = ActiveDevice;
67
68 // -------------------------------------------------------------------------
69 // 1. DEVICE PRIORITY ARBITRATION
70 // -------------------------------------------------------------------------
71 if (pen.isDown || pen.inProximity || pen.isHovering || (nowMs - lastPenTimestampMs < config.penGracePeriodMs)) {
72 ActiveDevice = DeviceType::Stylus;
73 } else if (fingerCount > 0 || (nowMs - lastTouchTimestampMs < config.touchGracePeriodMs)) {
74 ActiveDevice = DeviceType::Touch;
75 } else {
76 ActiveDevice = DeviceType::Mouse;
77 }
78
79 // -------------------------------------------------------------------------
80 // 2. DEVICE SWITCH: RESTORE PER-DEVICE SAVED TOOL
81 // -------------------------------------------------------------------------
82 if (oldDevice != ActiveDevice) {
83 std::string deviceName = (ActiveDevice == DeviceType::Stylus) ? "Stylus" :
84 (ActiveDevice == DeviceType::Touch) ? "Touch" : "Mouse";
85 LOG_INFO(InputStateMachine, "Active input device changed to: " + deviceName);
86
87 // Restore saved tool for the incoming device
88 currentAction = GetActiveDeviceTool();
89 }
90
91 // -------------------------------------------------------------------------
92 // 3. STYLUS CONTACT STATE & TRANSIENT TOOL OVERRIDES
93 // -------------------------------------------------------------------------
94 InteractionState oldAction = currentAction;
95 StylusState oldStylus = currentStylusState;
96
97 if (ActiveDevice == DeviceType::Stylus) {
98 if (pen.isDown) {
99 currentStylusState = StylusState::Engaged;
100 } else if (pen.isHovering || pen.inProximity || (nowMs - lastPenTimestampMs < config.penGracePeriodMs)) {
101 currentStylusState = StylusState::Hovering;
102 } else {
103 currentStylusState = StylusState::OutOfRange;
104 }
105
106 // Barrel button 1 / barrel button 3 / physical eraser tail -> transient Eraser override
107 // Barrel button 2 -> transient Lasso/Select override
108 // No button held -> restore saved tool
109 if (pen.barrel1 || pen.barrel3 || pen.eraserTip) {
110 stylusButtons = StylusButtonState::BarrelPressed;
111 currentAction = InteractionState::Eraser;
112 } else if (pen.barrel2) {
113 stylusButtons = StylusButtonState::Barrel2Pressed;
114 currentAction = InteractionState::Selecting;
115 } else {
116 stylusButtons = StylusButtonState::None;
117 currentAction = stylusTool.savedTool;
118 }
119 } else {
120 currentStylusState = StylusState::OutOfRange;
121 stylusButtons = StylusButtonState::None;
122 }
123
124 if (oldStylus != currentStylusState) {
125 std::string stylusName = (currentStylusState == StylusState::OutOfRange) ? "OutOfRange" :
126 (currentStylusState == StylusState::Hovering) ? "Hovering" : "Engaged";
127 LOG_INFO(InputStateMachine, "Stylus state changed to: " + stylusName);
128 }
129
130 if (oldAction != currentAction) {
131 std::string actionName = (currentAction == InteractionState::Inking) ? "Inking" :
132 (currentAction == InteractionState::Eraser) ? "Eraser" :
133 (currentAction == InteractionState::Selecting) ? "Selecting" :
134 (currentAction == InteractionState::Panning) ? "Panning" :
135 (currentAction == InteractionState::DrawingShape) ? "DrawingShape" :
136 (currentAction == InteractionState::Text) ? "Text" :
137 (currentAction == InteractionState::Transforming) ? "Transforming" : "Idle";
138 LOG_INFO(InputStateMachine, "Interaction state changed to: " + actionName);
139 }
140}

142/**
143 * @brief High-level entry point called each frame to evaluate arbitration and route to dispatchers.
144 *
145 * @param canvas Reference to CanvasEngine for drawing, zooming, and viewport manipulation.
146 * @param session Reference to DocumentSession for stroke persistence and undo/redo.
147 * @param imguiWantsInput Boolean flag indicating whether ImGui UI chrome currently holds focus.
148 */
149void InputStateMachine::ProcessInputState(CanvasEngine& canvas, DocumentSession& session, bool imguiWantsInput) {
150 uint64_t now = SDL_GetTicks();
151 UpdateHardwareState(now);
152
153 auto activePg = session.GetActivePage();
154 isPdfModeActive = (activePg && activePg->isDedicatedPdf);
155 // Markdown editor is disabled for prerelease; dedicated PDF viewer manages its own pointer events.
156 bool isDedicatedDocActive = isPdfModeActive;
157
158 // If active page is a dedicated standalone PDF document, stroke projection and interaction
159 // are handled directly by its view to prevent writing to the background canvas.
160 if (!isDedicatedDocActive) {
161 switch (ActiveDevice) {
162 case DeviceType::Stylus: DispatchStylus(canvas, session, imguiWantsInput); break;
163 case DeviceType::Touch: DispatchTouch(canvas, session, imguiWantsInput); break;
164 case DeviceType::Mouse: DispatchMouse(canvas, session, imguiWantsInput); break;
165 default: break;
166 }
167 }
168
169 // Save previous frame states for edge transitions
170 oldStylusState = currentStylusState;
171 wasMouseDown = mouse.leftButton;
172 wasMiddleDown = mouse.middleButton;
173}

175/**
176 * @brief Computes velocity-scaled dynamic eraser radius when isDynamicEraser is active.
177 *
178 * Mathematical Model:
179 * Let dt = currentTimeSec - lastEraserTimeSec.
180 * Pointer displacement distPx = hypot(dx, dy).
181 * Instantaneous velocity v = distPx / dt in pixels per second.
182 *
183 * Normalized velocity parameter t = clamp((v - 100.0) / 1900.0, 0.0, 1.0).
184 * Response curve: s = sqrt(t) (concave curve providing fast initial scaling).
185 * Target radius: R_target = R_min + s * (R_max - R_min).
186 *
187 * Asymmetric smoothing:
188 * alpha = 0.35 when expanding (fast swipe opens broad wiping swath).
189 * alpha = 0.12 when contracting (gentle recovery prevents jitter).
190 * R_current += alpha * (R_target - R_current).
191 */
192float InputStateMachine::UpdateDynamicEraserRadius(float currentX, float currentY, double currentTimeSec, bool isMoving, bool isDown) {
193 if (!isDynamicEraser) {
194 return eraserRadiusMm;
195 }
196
197 if (currentTimeSec <= 0.0) {
198 currentTimeSec = SDL_GetTicks() * 0.001;
199 }
200
201 // Keep min/max radius bounds in sync with state machine parameters
202 auto params = eraserPhysics.GetParameters();
203 if (params.minRadiusMm != dynamicEraserMinRadiusMm || params.maxRadiusMm != dynamicEraserMaxRadiusMm) {
204 params.minRadiusMm = dynamicEraserMinRadiusMm;
205 params.maxRadiusMm = dynamicEraserMaxRadiusMm;
206 eraserPhysics.SetParameters(params);
207 }
208
209 // Handle initial state or temporal discontinuity (> 250ms gap)
210 double dt = currentTimeSec - lastEraserTimeSec;
211 if (lastEraserTimeSec <= 0.0 || dt > 0.25 || dt <= 0.0001) {
212 lastVelocityX = currentX;
213 lastVelocityY = currentY;
214 lastEraserTimeSec = currentTimeSec;
215 if (currentDynamicEraserRadiusMm < dynamicEraserMinRadiusMm) {
216 currentDynamicEraserRadiusMm = dynamicEraserMinRadiusMm;
217 eraserPhysics.Reset(dynamicEraserMinRadiusMm);
218 }
219 return currentDynamicEraserRadiusMm;
220 }
221
222 float dx = currentX - lastVelocityX;
223 float dy = currentY - lastVelocityY;
224 float distPx = std::hypot(dx, dy);
225
226 float speedPxPerSec = 0.0f;
227 if (distPx > 0.5f) {
228 speedPxPerSec = static_cast<float>(distPx / dt);
229 lastVelocityX = currentX;
230 lastVelocityY = currentY;
231 }
232
233 // Run physical mass-spring-damper simulation step
234 currentDynamicEraserRadiusMm = eraserPhysics.Update(speedPxPerSec, static_cast<float>(dt));
235 lastEraserTimeSec = currentTimeSec;
236
237 return currentDynamicEraserRadiusMm;
238}
