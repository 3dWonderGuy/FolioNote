46bool ImportManager::ImportLibraryFolder(
47 const std::string& folderPath,
48 LibraryManager& libManager,
49 const std::string& customName
50) {
51 std::error_code ec;
52 if (!std::filesystem::exists(folderPath, ec) || !std::filesystem::is_directory(folderPath, ec)) {
53 LOG_ERROR(FileManager, "ImportManager::ImportLibraryFolder: Invalid directory: " + folderPath);
54 return false;
55 }
56
57 std::string libName = customName.empty() ? std::filesystem::path(folderPath).filename().string() : customName;
58 libManager.AddLibrary(libName, folderPath);
59 LOG_INFO(FileManager, "ImportManager: Mounted external library directory: " + folderPath);
60 return true;
61}
