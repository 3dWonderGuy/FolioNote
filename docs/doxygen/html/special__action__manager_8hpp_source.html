1/**
2 * @file special_action_manager.hpp
3 * @brief Extensible shortcut and special gesture action system.
4 *
5 * ARCHITECTURAL ROLE:
6 * -------------------
7 * As FolioNote evolves with richer interaction paradigms (pen double-taps, long-press holds,
8 * multi-finger tap shortcuts like Undo/Redo, radial menu triggers), hardcoding these behaviors
9 * inside device dispatchers creates spaghetti code and breaks maintainability.
10 *
11 * SpecialActionManager decouples gesture/shortcut recognition from action execution:
12 * 1. Recognition: Analyzes timing and spatial windows for double-clicks, dwell holds, and multi-finger taps.
13 * 2. Semantic Action: Emits strongly typed `SpecialActionType` events (e.g. Undo, Redo, ToggleEraser).
14 * 3. Dispatch: Triggers registered action callbacks or executes built-in default canvas actions.
15 */
16

66 /**
67 * @brief Evaluates a click/tap event to determine if it completes a double-click gesture.
68 *
69 * MATHEMATICAL LOGIC:
70 * -------------------
71 * Let previous tap be at $(x_1, y_1)$ at time $t_1$.
72 * Let incoming tap be at $(x_2, y_2)$ at time $t_2$.
73 *
74 * The incoming tap is classified as a Double-Click if and only if:
75 * 1. Temporal constraint: $\Delta t = t_2 - t_1 \le T_{\text{max}} \quad (\text{default } 300\,\text{ms})$
76 * 2. Spatial constraint: $D = \sqrt{(x_2 - x_1)^2 + (y_2 - y_1)^2} \le R_{\text{max}} \quad (\text{default } 12\,\text{px})$
77 *
78 * @param x Pixel X position of the click in window coordinates.
79 * @param y Pixel Y position of the click in window coordinates.
80 * @param nowMs Current system timestamp in milliseconds.
81 * @param config Configuration parameters supplying temporal and spatial thresholds.
82 * @return true if a double-click was recognized, false otherwise.
83 */
84 bool EvaluateDoubleClick(float x, float y, uint64_t nowMs, const InputConfiguration& config);
85
86 /**
87 * @brief Resets the click history, discarding any pending double-click state.
88 */
89 void ResetClickHistory() noexcept;
90
91 /**
92 * @brief Registers an external listener callback to receive special action events.
93 *
94 * @param callback Function to invoke when a special action is triggered.
95 */
96 void RegisterActionCallback(ActionCallback callback);
97
98 /**
99 * @brief Dispatches a special action to all registered callbacks and executes default handling.
100 *
101 * @param action The semantic action to execute.
102 * @param x Local X coordinate where the action originated.
103 * @param y Local Y coordinate where the action originated.
104 * @param canvas Reference to CanvasEngine for viewport and object updates.
105 * @param session Reference to DocumentSession for undo/redo and object mutations.
106 */
107 void TriggerAction(SpecialActionType action, float x, float y, CanvasEngine& canvas, DocumentSession& session);

109 /**

111 */
112 [[nodiscard]] static std::string ActionToString(SpecialActionType action);

114private:

117};
