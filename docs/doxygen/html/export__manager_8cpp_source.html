40std::string ExportManager::SanitizeFilename(const std::string& name) {
41 std::string result;
42 for (char c : name) {
43 if (std::isalnum(static_cast<unsigned char>(c)) || c == '_' || c == '-' || c == ' ') {
44 result += (c == ' ' ? '_' : c);
45 }
46 }
47 return result.empty() ? "export" : result;
48}

50std::string ExportManager::GetExportFilename(
51 ExportScope scope,
52 ExportFormat format,
53 const std::shared_ptr<Notebook>& notebook,
54 const std::shared_ptr<Section>& section,
55 const std::shared_ptr<CanvasPage>& page,
56 const std::string& libraryPath
57) {
58 std::string base = "FolioNote_Export";
59
60 if (scope == ExportScope::CurrentPage && page) {
61 base = SanitizeFilename(page->title.empty() ? "Untitled_Page" : page->title);
62 } else if (scope == ExportScope::CurrentSection && section) {
63 base = SanitizeFilename(section->name.empty() ? "Untitled_Section" : section->name);
64 } else if (scope == ExportScope::EntireNotebook && notebook) {
65 base = SanitizeFilename(notebook->name.empty() ? "Untitled_Notebook" : notebook->name);
66 } else if (scope == ExportScope::EntireLibrary && !libraryPath.empty()) {
67 base = SanitizeFilename(FileManager::GetStem(libraryPath));
68 }
69
70 switch (format) {
71 case ExportFormat::PDF_Print: return base + "_Print.html";
72 case ExportFormat::HTML_Document: return base + ".html";
73 case ExportFormat::PDF_Vector: return base + ".pdf";
74 case ExportFormat::MarkdownText: return base + ".md";
75 case ExportFormat::FolioPackage: return base + ".folionb.7z";
76 case ExportFormat::LibraryPackage: return base + ".foliolib.7z";
77 default: return base + ".txt";
78 }
79}

81bool ExportManager::Export(
82 ExportScope scope,
83 ExportFormat format,
84 const std::shared_ptr<Notebook>& notebook,
85 const std::shared_ptr<Section>& section,
86 const std::shared_ptr<CanvasPage>& page,
87 const std::string& destinationPath,
88 const ExportOptions& options
89) {
90 std::string outPath = destinationPath;
91 if (outPath.empty() || FileManager::IsDirectory(outPath) || FileManager::GetExtension(outPath).empty()) {
92 std::string baseDir = outPath.empty() ? FileManager::GetExportsDirectory() : outPath;
93 FileManager::CreateDirectories(baseDir);
94 std::string defaultFilename = GetExportFilename(scope, format, notebook, section, page, options.libraryPath);
95 outPath = FileManager::JoinPath(baseDir, defaultFilename);
96 } else {
97 std::string parentDir = FileManager::GetParentPath(outPath);
98 if (!parentDir.empty()) {
99 FileManager::CreateDirectories(parentDir);
100 }
101 }
102
103 LOG_INFO(FileManager, "ExportManager: Starting export. Scope=" + std::to_string(static_cast<int>(scope)) +
104 ", Format=" + std::to_string(static_cast<int>(format)) + " -> " + outPath);
105
106 switch (format) {
107 case ExportFormat::HTML_Document:
108 case ExportFormat::PDF_Print: {
109 std::string html;
110 if (scope == ExportScope::CurrentPage && page) {
111 std::string secName = section ? section->name : "Section";
112 std::string nbName = notebook ? notebook->name : "Notebook";
113 html = HtmlSvgExporter::ExportPage(page, secName, nbName, options);
114 } else if (scope == ExportScope::CurrentSection && section) {
115 std::string nbName = notebook ? notebook->name : "Notebook";
116 html = HtmlSvgExporter::ExportSection(section, nbName, options);
117 } else if (notebook) {
118 html = HtmlSvgExporter::ExportNotebook(notebook, options);
119 } else {
120 return false;
121 }
122
123 if (!FileManager::WriteTextAtomic(outPath, html)) {
124 LOG_ERROR(FileManager, "ExportManager: Failed to write HTML to: " + outPath);
125 return false;
126 }
127
128 if (format == ExportFormat::PDF_Print && options.triggerPrintDialog) {
129#if defined(_WIN32)
130 ShellExecuteA(NULL, "print", outPath.c_str(), NULL, NULL, SW_SHOWNORMAL);
131#endif
132 }
133 return true;
134 }
135
136 case ExportFormat::PDF_Vector: {
137 if (scope == ExportScope::CurrentPage && page) {
138 return PdfVectorExporter::ExportPage(page, outPath, options);
139 } else if (scope == ExportScope::CurrentSection && section) {
140 return PdfVectorExporter::ExportSection(section, outPath, options);
141 } else if (notebook) {
142 return PdfVectorExporter::ExportNotebook(notebook, outPath, options);
143 }
144 return false;
145 }
146
147 case ExportFormat::MarkdownText: {
148 std::string outDir = FileManager::GetParentPath(outPath);
149 std::string md;
150
151 if (scope == ExportScope::CurrentPage && page) {
152 std::string secName = section ? section->name : "Section";
153 std::string nbName = notebook ? notebook->name : "Notebook";
154 md = MarkdownExporter::ExportPage(page, secName, nbName, outDir, options);
155 } else if (scope == ExportScope::CurrentSection && section) {
156 std::string nbName = notebook ? notebook->name : "Notebook";
157 md = MarkdownExporter::ExportSection(section, nbName, outDir, options);
158 } else if (notebook) {
159 md = MarkdownExporter::ExportNotebook(notebook, outDir, options);
160 } else {
161 return false;
162 }
163
164 return FileManager::WriteTextAtomic(outPath, md);
165 }
166
167 case ExportFormat::FolioPackage: {
168 if (!notebook || notebook->filePath.empty()) return false;
169 return PackageExporter::ExportNotebookPackage(notebook->filePath, outPath, options.compressionLevel);
170 }
171
172 case ExportFormat::LibraryPackage: {
173 std::string libPath = options.libraryPath;
174 if (libPath.empty() && notebook && !notebook->filePath.empty()) {
175 libPath = FileManager::GetParentPath(notebook->filePath);
176 }
177 if (libPath.empty()) return false;
178 return PackageExporter::ExportLibraryPackage(libPath, outPath, options.compressionLevel);
179 }
180
181 default:
182 return false;
183 }
184}
