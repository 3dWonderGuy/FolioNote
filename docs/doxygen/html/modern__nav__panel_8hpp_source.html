158 // -----------------------------------------------------------------------
159 // Smooth Collapse / Expand Animation
160 // animatedTotalWidth chases the logical target each frame via exponential
161 // smoothing so the sidebar glides in / out instead of snapping.
162 // ANIM_SPEED controls responsiveness: higher = faster (12 ≈ ~80 ms settle).
163 // -----------------------------------------------------------------------
164 float animatedTotalWidth = ModernNavConfig::DEFAULT_SECTION_W * 2.0f +
165 ModernNavConfig::SPLITTER_WIDTH * 2.0f;
166 static constexpr float ANIM_SPEED = 14.0f; // frames-per-second feel
167
168 // ------------------------------------------------------------------------
169 // Geometry Queries & State Transitions
170 // ------------------------------------------------------------------------
171 // Returns the *animated* width – used by app.hpp for canvas positioning so
172 // the canvas smoothly expands/contracts in lock-step with the sidebar.
173 // Pixel-snapped to avoid sub-pixel jitter at the sidebar/canvas seam.
174 [[nodiscard]] float GetTotalWidth() const noexcept {
175 return std::round(animatedTotalWidth);
176 }
177
178 // Returns the logical (target) width the sidebar is heading toward.
179 [[nodiscard]] float GetTargetWidth() const noexcept {
180 if (state == ModernNavState::Fully_Hidden) return 0.0f;
181 if (state == ModernNavState::PagesOnly) return pagesWidth + ModernNavConfig::SPLITTER_WIDTH;
182 return sectionWidth + ModernNavConfig::SPLITTER_WIDTH + pagesWidth + ModernNavConfig::SPLITTER_WIDTH;
183 }
184
185 [[nodiscard]] float GetCurrentMinWidth() const noexcept {
186 if (state == ModernNavState::PagesOnly) {
187 return ModernNavConfig::PAGES_MIN_WIDTH;
188 }
189 return ModernNavConfig::SECTION_MIN_WIDTH + ModernNavConfig::SPLITTER_WIDTH + ModernNavConfig::PAGES_MIN_WIDTH;
190 }
191
192 void ToggleState() noexcept {
193 state = (state == ModernNavState::Expanded) ? ModernNavState::PagesOnly : ModernNavState::Expanded;
194 }
195
196 // True while the sidebar is mid-transition (animated != target).
197 // Used externally to suppress expensive canvas resizes during the glide.
198 [[nodiscard]] bool IsAnimating() const noexcept {
199 return animatedTotalWidth != GetTargetWidth();
200 }
201
202 // Advance the animation one frame. Call once per frame BEFORE Render().
203 // Uses exponential smoothing: animatedWidth += (target - current) * (1 - e^(-k*dt))
204 // which gives a fast-start, ease-out curve independent of frame rate.
205 void TickAnimation() noexcept {
206 float dt = ImGui::GetIO().DeltaTime;
207 float target = GetTargetWidth();
208 // Clamp dt to avoid huge jump after a hitch / alt-tab
209 float k = 1.0f - std::exp(-ANIM_SPEED * std::min(dt, 0.1f));
210 animatedTotalWidth += (target - animatedTotalWidth) * k;
211 // Snap to exact target once within half a pixel to stop micro-drift
212 if (std::fabs(animatedTotalWidth - target) < 0.5f)
213 animatedTotalWidth = target;
214 }
