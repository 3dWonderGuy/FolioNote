14bool SpecialActionManager::EvaluateDoubleClick(float x, float y, uint64_t nowMs, const InputConfiguration& config) {
15 if (!lastTap.isValid) {
16 // First tap: initialize history
17 lastTap.x = x;
18 lastTap.y = y;
19 lastTap.timestampMs = nowMs;
20 lastTap.isValid = true;
21 return false;
22 }
23
24 // Compute temporal difference in milliseconds
25 uint64_t dt = (nowMs >= lastTap.timestampMs) ? (nowMs - lastTap.timestampMs) : 0;
26
27 // Compute spatial distance in screen pixels: D = sqrt((x2 - x1)^2 + (y2 - y1)^2)
28 float dx = x - lastTap.x;
29 float dy = y - lastTap.y;
30 float dist = std::hypot(dx, dy);
31
32 if (dt <= config.doubleTapMaxIntervalMs && dist <= config.doubleTapMaxDistancePx) {
33 // Double-tap criteria satisfied! Reset history to prevent a triple-tap from firing again
34 ResetClickHistory();
35 return true;
36 }
37
38 // Did not satisfy double-tap criteria; replace previous tap with this new baseline
39 lastTap.x = x;
40 lastTap.y = y;
41 lastTap.timestampMs = nowMs;
42 lastTap.isValid = true;
43 return false;
44}

56void SpecialActionManager::TriggerAction(SpecialActionType action, float x, float y, CanvasEngine& canvas, DocumentSession& session) {
57 if (action == SpecialActionType::None) return;
58
59 LOG_INFO(InputStateMachine, "Special Action Triggered: " + ActionToString(action) +
60 " at (" + std::to_string(x) + ", " + std::to_string(y) + ")");
61
62 // 1. Notify external subscribers (e.g. UI radial menu, ribbon bar, overlay)
63 for (auto& cb : callbacks) {
64 if (cb) {
65 cb(action, x, y);
66 }
67 }
68
69 // 2. Built-in default execution logic for core canvas actions
70 switch (action) {
71 case SpecialActionType::Undo: {
72 LOG_INFO(InputStateMachine, "Executing Undo via Special Action");
73 session.Undo(&canvas);
74 break;
75 }
76
77 case SpecialActionType::Redo: {
78 LOG_INFO(InputStateMachine, "Executing Redo via Special Action");
79 session.Redo(&canvas);
80 break;
81 }
82
83 case SpecialActionType::ClearSelection: {
84 canvas.ClearSelection(&session);
85 canvas.needsFullRebake = true;
86 canvas.isDirty = true;
87 break;
88 }
89
90 case SpecialActionType::SelectAtPoint: {
91 auto activePage = session.GetActivePage();
92 if (!activePage) {
93 LOG_WARN_CODE(InputStateMachine, FolioErrorCode::InputTargetPageNull,
94 "SelectAtPoint skipped: No active CanvasPage loaded.");
95 break;
96 }
97
98 // Hit test canvas objects at local position (x, y)
99 Point2D worldMm = canvas.transform.ScreenToWorld(x, y);
100 std::shared_ptr<CanvasObject> clickedObj = nullptr;
101 for (auto it = activePage->objects.rbegin(); it != activePage->objects.rend(); ++it) {
102 auto& obj = *it;
103 if (obj && obj->isVisible && obj->isSelectable &&
104 (obj->HitTest(worldMm.x, worldMm.y) || obj->HitTestCircle(worldMm.x, worldMm.y, 2.0))) {
105 clickedObj = obj;
106 break;
107 }
108 }
109
110 canvas.ClearSelection(&session);
111 if (clickedObj) {
112 clickedObj->isSelected = 1;
113 canvas.selectionGizmo.SetSelectedObjects(activePage->objects);
114 canvas.selectionGizmo.OnPointerDown(x, y, canvas.transform);
115 LOG_INFO(InputStateMachine, "Direct selection picked object UID " + std::to_string(clickedObj->uid) +
116 " at (" + std::to_string(x) + ", " + std::to_string(y) + ")");
117 } else {
118 LOG_INFO(InputStateMachine, "SelectAtPoint: No object hit at screen (" + std::to_string(x) +
119 ", " + std::to_string(y) + "). Selection cleared.");
120 }
121 canvas.needsFullRebake = true;
122 canvas.isDirty = true;
123 break;
124 }
125
126 case SpecialActionType::ToggleEraser:
127 case SpecialActionType::OpenContextMenu:
128 case SpecialActionType::ZoomToFit:
129 default:
130 // Handled by external callback or state machine caller
131 break;
132 }
133}
