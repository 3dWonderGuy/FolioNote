58/**
59 * @brief Commits finished stroke data (already containing pre-computed outline geometry).
60 * @param data Finished stroke outline path and live segment records.
61 * @param tool Active pen tool settings (color, base size, highlighter mode).
62 */
63void DocumentSession::CommitStroke(FinishedStrokeData&& data, const PenTool& tool) {
64 auto activePage = GetActivePage();
65 if (!activePage) {
66 LOG_WARN_CODE(DocumentSession, FolioErrorCode::InputTargetPageNull,
67 "CommitStroke rejected: No active CanvasPage available to receive stroke data.");
68 return;
69 }
70 if (data.outlinePath.is_empty() && data.liveSegments.empty()) return;
71
72 auto container = std::make_shared<InkContainer>();
73 container->uid = UIDGenerator::Next();
74 container->isHighlighter = (tool.penType == PenType::Highlighter);
75
76 Stroke stroke;
77 stroke.color = tool.color;
78 stroke.baseWidthMm = tool.baseSize;
79 stroke.strokePattern = tool.strokePattern;
80 stroke.capType = tool.capType;
81 stroke.blendMode = tool.blendMode;
82 stroke.penType = tool.penType;
83 stroke.opacity = tool.opacity;
84
85 if (!data.liveSegments.empty()) {
86 stroke.points.reserve(data.liveSegments.size() + 1);
87 stroke.points.push_back({ data.liveSegments[0].p0.x, data.liveSegments[0].p0.y, data.liveSegments[0].width * 0.5f });
88 for (const auto& s : data.liveSegments) {
89 stroke.points.push_back({ s.p1.x, s.p1.y, s.width * 0.5f });
90 }
91 }
92
93 stroke.outlinePath = std::move(data.outlinePath);
94 stroke.segments = std::move(data.liveSegments);
95 container->AddStroke(stroke);
96
97 activePage->AddObject(container);
98 RecordHistoryCommand(activePage, std::make_unique<Folio::AddObjectCommand>(container));
99 activePage->isModified = true;
100 NotifyPageModified(activePage);
101}

103/**
104 * @brief Commits raw 1D stroke segments by computing polygon outline hulls on the fly.
105 * @param segments Vector of 1D interpolated segments with pressure/width values.
106 * @param tool Active pen tool settings (color, base size, cap type).
107 */
108void DocumentSession::CommitStroke(std::vector<Segment1D>&& segments, const PenTool& tool) {
109 auto activePage = GetActivePage();
110 if (!activePage) {
111 LOG_WARN_CODE(DocumentSession, FolioErrorCode::InputTargetPageNull,
112 "CommitStroke (raw segments) rejected: No active CanvasPage available.");
113 return;
114 }
115 if (segments.empty()) return;
116
117 auto container = std::make_shared<InkContainer>();
118 container->uid = UIDGenerator::Next();
119 container->isHighlighter = (tool.penType == PenType::Highlighter);
120
121 Stroke stroke;
122 stroke.color = tool.color;
123 stroke.baseWidthMm = tool.baseSize;
124 stroke.strokePattern = tool.strokePattern;
125 stroke.capType = tool.capType;
126 stroke.blendMode = tool.blendMode;
127 stroke.penType = tool.penType;
128 stroke.opacity = tool.opacity;
129
130 std::vector<StrokeOutlineBuilder::InputPoint> pts;
131 pts.reserve(segments.size() + 1);
132 stroke.points.reserve(segments.size() + 1);
133
134 pts.push_back({ segments[0].p0.x, segments[0].p0.y, segments[0].width });
135 stroke.points.push_back({ segments[0].p0.x, segments[0].p0.y, segments[0].width * 0.5f });
136
137 for (const auto& s : segments) {
138 pts.push_back({ s.p1.x, s.p1.y, s.width });
139 stroke.points.push_back({ s.p1.x, s.p1.y, s.width * 0.5f });
140 }
141
142 stroke.outlinePath = StrokeOutlineBuilder::BuildOutline(pts, tool.capType, tool.strokePattern);
143 stroke.segments = std::move(segments);
144 container->AddStroke(stroke);
145
146 activePage->AddObject(container);
147 RecordHistoryCommand(activePage, std::make_unique<Folio::AddObjectCommand>(container));
148 activePage->isModified = true;
149 NotifyPageModified(activePage);

372/**
373 * @brief Pastes the contents of the session clipboard onto the active page, centered at (worldX, worldY).
374 *
375 * MATHEMATICAL PROCESS:
376 * 1. Evaluates collective bounding box B = Union(obj.bounds) for all clipboard items.
377 * 2. Computes centroid: C = ((B.minX + B.maxX)/2, (B.minY + B.maxY)/2).
378 * 3. Calculates world translation vector: ΔX = worldX - C.x, ΔY = worldY - C.y.
379 * 4. Clones each item, applies Δ translation, assigns fresh unique runtime UID via UIDGenerator::Next().
380 * 5. Commits a single atomic AddObjectsCommand to the page history.
381 *
382 * @param worldX Target world X coordinate (millimeters).
383 * @param worldY Target world Y coordinate (millimeters).
384 * @return Vector of newly created and inserted CanvasObjects.
385 */
386std::vector<std::shared_ptr<CanvasObject>> DocumentSession::PasteObjects(double worldX, double worldY) {
387 if (clipboardObjects.empty()) return {};
388 auto activePage = GetActivePage();
389 if (!activePage) return {};
390
391 AABB box;
392 for (const auto& obj : clipboardObjects) {
393 if (obj) box.Merge(obj->bounds);
394 }
395 Point2D center = { (box.minX + box.maxX) * 0.5, (box.minY + box.maxY) * 0.5 };
396 double dx = worldX - center.x;
397 double dy = worldY - center.y;
398 BLMatrix2D trans = BLMatrix2D::make_translation(dx, dy);
399
400 std::vector<std::shared_ptr<CanvasObject>> newObjects;
401 newObjects.reserve(clipboardObjects.size());
402
403 for (const auto& obj : clipboardObjects) {
404 if (!obj) continue;
405 auto clone = std::shared_ptr<CanvasObject>(obj->Clone().release());
406 clone->uid = UIDGenerator::Next();
407 clone->ApplyTransform(trans);
408 clone->UpdateBounds();
409 clone->isSelected = 1;
410 activePage->AddObject(clone);
411 newObjects.push_back(clone);
412 }
413
414 if (!newObjects.empty()) {
415 RecordHistoryCommand(activePage, std::make_unique<Folio::AddObjectsCommand>(newObjects));
416 activePage->isModified = true;
417 NotifyPageModified(activePage);
418 LOG_INFO(DocumentSession, "Pasted " + std::to_string(newObjects.size()) +
419 " objects at (" + std::to_string(worldX) + ", " + std::to_string(worldY) + ") mm");
420 }
421 return newObjects;
422}
