101FinishedStrokeData InkEngine::FinishStroke() {
102 if (!m_isStrokeActive) return {};
103
104 if (!m_activeStrokePoints.empty()) {
105 // Stage 1: Pen-Up event to flush lingering spring inertia in physics modeler
106 ink::stroke_model::Input input;
107 input.event_type = ink::stroke_model::Input::EventType::kUp;
108 input.position = {
109 static_cast<float>(m_activeStrokePoints.back().x),
110 static_cast<float>(m_activeStrokePoints.back().y)
111 };
112 input.time = ink::stroke_model::Time(m_activeStrokePoints.back().timeSeconds);
113 input.pressure = m_activeStrokePoints.back().pressure;
114
115 std::vector<ink::stroke_model::Result> results;
116 if (m_googleModeler->Update(input, results).ok()) {
117 ProcessGoogleResults(results);
118 }
119
120 // Stage 2: Single-tap dot handling (user tapped without lateral dragging)
121 if (m_liveModeledPoints.empty() && m_hasPreviousGooglePoint) {
122 float baseW = static_cast<float>(m_activeTool.baseSize);
123 m_liveModeledPoints.push_back({
124 static_cast<float>(m_previousGooglePos.x),
125 static_cast<float>(m_previousGooglePos.y),
126 baseW,
127 1.0f,
128 0.0f,
129 m_tiltX,
130 m_tiltY
131 });
132
133 Segment1D dot;
134 dot.p0 = Point2D{ m_previousGooglePos.x, m_previousGooglePos.y, 0, 0 };
135 dot.p1 = dot.p0;
136 dot.p1.x += 0.001; // Epsilon offset to establish valid vector direction
137 dot.width = baseW;
138 m_fullSmoothedSegments.push_back(dot);
139 }
140 }
141
142 // Stage 3: Package final vector stroke data
143 FinishedStrokeData data;
144 data.rawPoints = std::move(m_activeStrokePoints);
145 data.liveSegments = std::move(m_fullSmoothedSegments);
146 data.modeledPoints = m_liveModeledPoints;
147 data.outlinePath = StrokeOutlineBuilder::BuildOutline(
148 m_liveModeledPoints, m_activeTool.capType, m_activeTool.strokePattern);
149
150 // Stage 4: Reset live state
151 CancelStroke();
152
153 return data;
154}

174float InkEngine::CalculateDynamicWidth(const ink::stroke_model::Result& res) const {
175 float baseWidth = static_cast<float>(m_activeTool.baseSize);
176
177 // STAGE A: PRESSURE CEILING
178 float pressureWidth = baseWidth;
179 if (g_InkingConfig.google_use_pressure && res.pressure >= 0.0f) {
180 float pressureCurved = g_InkingConfig.EvaluatePressure(res.pressure);
181 float minFrac = g_InkingConfig.google_min_width_multiplier;
182 float maxFrac = g_InkingConfig.google_max_width_multiplier;
183 float widthMultiplier = minFrac + pressureCurved * (maxFrac - minFrac);
184 pressureWidth = baseWidth * std::clamp(widthMultiplier, minFrac, maxFrac);
185 }
186
187 // STAGE B: VELOCITY THINNING
188 float velocityThinFactor = 1.0f;
189 if (g_InkingConfig.google_use_velocity) {
190 float speed = std::hypot(res.velocity.x, res.velocity.y) * m_zoomScale;
191 float normalizedSpeed = std::clamp(
192 speed / g_InkingConfig.google_velocity_thinning_max_speed, 0.0f, 1.0f);
193
194 float minThin = g_InkingConfig.google_min_width_multiplier;
195 velocityThinFactor = 1.0f - normalizedSpeed * (1.0f - minThin);
196 }
197
198 return pressureWidth * velocityThinFactor;
199}

201void InkEngine::ProcessGoogleResults(const std::vector<ink::stroke_model::Result>& results) {
202 for (const auto& res : results) {
203 float targetWidth = CalculateDynamicWidth(res);
204
205 // Exponential Moving Average (EMA) width filter: removes single-frame pop
206 if (m_smoothedWidth < 0.0f) {
207 m_smoothedWidth = targetWidth;
208 } else {
209 m_smoothedWidth = std::lerp(
210 m_smoothedWidth, targetWidth, g_InkingConfig.google_dynamic_width_smoothing);
211 }
212
213 // Build 1D line segments for collision and spatial testing
214 if (m_hasPreviousGooglePoint) {
215 Segment1D seg;
216 seg.p0 = Point2D{ m_previousGooglePos.x, m_previousGooglePos.y, 0, 0 };
217 seg.p1 = Point2D{ res.position.x, res.position.y, 0, 0 };
218 seg.width = m_smoothedWidth;
219 m_fullSmoothedSegments.push_back(seg);
220 }
221
222 float speed = std::hypot(res.velocity.x, res.velocity.y);
223 m_liveModeledPoints.push_back({
224 res.position.x,
225 res.position.y,
226 m_smoothedWidth,
227 res.pressure,
228 speed,
229 m_tiltX,
230 m_tiltY
231 });
232
233 m_previousGooglePos = Point2D{ res.position.x, res.position.y, 0, 0 };
234 m_hasPreviousGooglePoint = true;
235 }
236
237 // Reconstruct the 2D polygon outline for live preview
238 if (!m_liveModeledPoints.empty()) {
239 m_liveStrokeOutline = StrokeOutlineBuilder::BuildOutline(
240 m_liveModeledPoints, m_activeTool.capType, m_activeTool.strokePattern);
241 }

244void InkEngine::UpdateGooglePrediction() {
245 m_googlePredictedSegments.clear();
246 m_predictedStrokeOutline.clear();
247 if (!m_hasPreviousGooglePoint) return;
248
249 std::vector<ink::stroke_model::Result> predictions;
250 if (m_googleModeler->Predict(predictions).ok() && !predictions.empty()) {
251 Point2D prev = m_previousGooglePos;
252 float predWidth = m_smoothedWidth;
253
254 std::vector<StrokeOutlineBuilder::InputPoint> predPoints;
255 if (!m_liveModeledPoints.empty()) {
256 predPoints.push_back(m_liveModeledPoints.back());
257 }
258
259 for (const auto& res : predictions) {
260 float targetWidth = CalculateDynamicWidth(res);
261 if (predWidth < 0.0f) {
262 predWidth = targetWidth;
263 } else {
264 predWidth = std::lerp(
265 predWidth, targetWidth, g_InkingConfig.google_dynamic_width_smoothing);
266 }
267
268 Segment1D seg;
269 seg.p0 = prev;
270 seg.p1 = Point2D{ res.position.x, res.position.y, 0, 0 };
271 seg.width = predWidth;
272 m_googlePredictedSegments.push_back(seg);
273 prev = seg.p1;
274
275 float speed = std::hypot(res.velocity.x, res.velocity.y);
276 predPoints.push_back({
277 res.position.x,
278 res.position.y,
279 predWidth,
280 res.pressure,
281 speed,
282 m_tiltX,
283 m_tiltY
284 });
285 }
286
287 if (predPoints.size() >= 2) {
288 m_predictedStrokeOutline = StrokeOutlineBuilder::BuildOutline(
289 predPoints, m_activeTool.capType, m_activeTool.strokePattern);
290 }
291 }

298LiveInkSnapshot InkEngine::GetLiveSnapshot() const {
299 LiveInkSnapshot snapshot;
300 snapshot.isStrokeActive = m_isStrokeActive;
301 snapshot.confirmedOutline = m_liveStrokeOutline;
302 snapshot.predictedOutline = m_predictedStrokeOutline;
303 snapshot.strokeColor = BLRgba32(m_activeTool.color.value);
304
305 if (m_isStrokeActive) {
306 BLBox confirmedBox;
307 if (m_liveStrokeOutline.get_bounding_box(&confirmedBox) == BL_SUCCESS) {
308 snapshot.dirtyBounds = AABB(confirmedBox.x0, confirmedBox.y0, confirmedBox.x1, confirmedBox.y1);
309 }
310 BLBox predictedBox;
311 if (m_predictedStrokeOutline.get_bounding_box(&predictedBox) == BL_SUCCESS) {
312 AABB predAABB(predictedBox.x0, predictedBox.y0, predictedBox.x1, predictedBox.y1);
313 snapshot.dirtyBounds.minX = std::min(snapshot.dirtyBounds.minX, predAABB.minX);
314 snapshot.dirtyBounds.minY = std::min(snapshot.dirtyBounds.minY, predAABB.minY);
315 snapshot.dirtyBounds.maxX = std::max(snapshot.dirtyBounds.maxX, predAABB.maxX);
316 snapshot.dirtyBounds.maxY = std::max(snapshot.dirtyBounds.maxY, predAABB.maxY);
317 }
318 }
319
320 return snapshot;
321}

352bool InkEngine::UpdateEphemeralStrokes(uint64_t currentTimeMs) {
353 if (m_ephemeralStrokes.empty()) return false;
354
355 size_t prevCount = m_ephemeralStrokes.size();
356 std::erase_if(m_ephemeralStrokes, [currentTimeMs](const EphemeralStroke& stroke) {
357 return (currentTimeMs - stroke.startTimeMs) >= stroke.durationMs;
358 });
359
360 // Redraw if any strokes are active (fading each frame) or any expired
361 return !m_ephemeralStrokes.empty() || (m_ephemeralStrokes.size() != prevCount);
362}

368void InkEngine::RenderEphemeralStrokes(BLContext& ctx, uint64_t currentTimeMs) const {
369 if (m_ephemeralStrokes.empty()) return;
370
371 for (const auto& stroke : m_ephemeralStrokes) {
372 if (stroke.outlinePath.is_empty()) continue;
373
374 uint64_t elapsed = (currentTimeMs >= stroke.startTimeMs) ? (currentTimeMs - stroke.startTimeMs) : 0;
375 if (elapsed >= stroke.durationMs) continue;
376
377 // Quadratic fade calculation: alpha = (1.0 - progress)^2
378 float progress = static_cast<float>(elapsed) / static_cast<float>(stroke.durationMs);
379 float alphaFactor = (1.0f - progress) * (1.0f - progress);
380
381 BLRgba32 decayedColor = stroke.baseColor;
382 uint32_t baseA = (decayedColor.value >> 24) & 0xFF;
383 uint32_t currentA = static_cast<uint32_t>(baseA * alphaFactor);
384 decayedColor.value = (decayedColor.value & 0x00FFFFFF) | (currentA << 24);
385
386 ctx.save();
387 ctx.set_fill_style(decayedColor);
388 ctx.fill_path(stroke.outlinePath);
389 ctx.restore();
390 }
391}
