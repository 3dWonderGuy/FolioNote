1/**
2 * =========================================================================================
3 * @file html_svg_exporter.cpp
4 * @brief Implementation of High-Fidelity Sheet-Tiled HTML & SVG Vector Exporter
5 * =========================================================================================
6 *
7 * ARCHITECTURAL IMPLEMENTATION DETAILS:
8 * 1. Sheet-Tiled Page Isolation:
9 * Each SheetTile is emitted as a discrete CSS-styled sheet (`.folio-sheet`) with physical
10 * millimeter dimensions matching standard paper (e.g. A4: 210 x 297 mm, Letter: 215.9 x 279.4 mm).
11 * In `@media print`, each sheet has `page-break-after: always;` ensuring crisp, 1:1 printing.
12 * 2. High-Resolution SVG Vectors:
13 * Renders vector ink strokes into inline SVG viewports with rounded line caps/joins and
14 * alpha transparency.
15 * 3. Searchable Text & Typography:
16 * TextBoxObjects are rendered as HTML text nodes positioned with millimeter offsets,
17 * preserving full selectability, clipboard copy, and search engine indexability.
18 * 4. Interactive Hyperlinks:
19 * Supports standard web URLs (`https://...`), in-document sheet anchors (`#sheet_...`),
20 * and external application deep links (`folionote://page/...`).
21 */

39std::string EscapeHtml(const std::string& input) {
40 std::string output;
41 for (char c : input) {
42 switch (c) {
43 case '&': output += "&amp;"; break;
44 case '<': output += "&lt;"; break;
45 case '>': output += "&gt;"; break;
46 case '"': output += "&quot;"; break;
47 case '\'': output += "&#39;"; break;
48 default: output += c; break;
49 }
50 }
51 return output;
52}

85std::string HtmlSvgExporter::FormatHtmlText(const std::string& rawText, const ExportOptions& options) {
86 if (!options.enableLinks) {
87 return EscapeHtml(rawText);
88 }
89
90 std::string escaped = EscapeHtml(rawText);
91
92 // Hyperlink regex for https://, http://, and folionote://
93 static const std::regex urlRegex(R"((https?://[^\s<>"]+|folionote://[^\s<>"]+))");
94 std::string result;
95 auto wordsBegin = std::sregex_iterator(escaped.begin(), escaped.end(), urlRegex);
96 auto wordsEnd = std::sregex_iterator();
97
98 size_t lastPos = 0;
99 for (auto it = wordsBegin; it != wordsEnd; ++it) {
100 std::smatch match = *it;
101 result += escaped.substr(lastPos, match.position() - lastPos);
102
103 std::string url = match.str();
104 if (url.rfind("folionote://", 0) == 0) {
105 if (options.enableAppDeepLinks) {
106 result += "<a href=\"" + url + "\" class=\"folio-app-link\" title=\"Open in FolioNote\">" + url + "</a>";
107 } else {
108 result += url;
109 }
110 } else if (options.enableWebLinks) {
111 result += "<a href=\"" + url + "\" target=\"_blank\" rel=\"noopener noreferrer\">" + url + "</a>";
112 } else {
113 result += url;
114 }
115
116 lastPos = match.position() + match.length();
117 }
118 result += escaped.substr(lastPos);
119 return result;
120}

122std::string HtmlSvgExporter::RenderSheetBlock(
123 const SheetTile& tile,
124 const std::shared_ptr<CanvasPage>& page,
125 const ExportOptions& options
126) {
127 if (!page) return "";
128 std::ostringstream ss;
129
130 ss << "<div class=\"folio-sheet\" id=\"" << tile.sheetAnchorId << "\">\n";
131
132 // Header on the first sheet tile of a page
133 if (tile.sequenceIndex == 0) {
134 ss << " <header class=\"sheet-header\">\n";
135 ss << " <div class=\"sheet-title\">" << EscapeHtml(page->title) << "</div>\n";
136 if (!page->createdDateStr.empty()) {
137 ss << " <div class=\"sheet-meta\">" << EscapeHtml(page->createdDateStr + " | " + page->createdTimeStr) << "</div>\n";
138 }
139 ss << " </header>\n";
140 }
141
142 double virtualW = tile.widthMm / tile.uniformScale;
143 double virtualH = tile.heightMm / tile.uniformScale;
144
145 // SVG Vector Ink Layer
146 std::ostringstream svgStrokes;
147 int strokeCount = 0;
148
149 for (const auto& obj : tile.intersectingObjects) {
150 if (auto ink = std::dynamic_pointer_cast<InkContainer>(obj)) {
151 for (const auto& stroke : ink->strokes) {
152 if (stroke.segments.empty()) continue;
153 strokeCount++;
154
155 uint8_t a = stroke.color.a();
156 uint8_t r = stroke.color.r();
157 uint8_t g = stroke.color.g();
158 uint8_t b = stroke.color.b();
159 float alpha = (a / 255.0f) * stroke.opacity;
160 float w = std::max(0.5f, static_cast<float>(stroke.baseWidthMm * tile.uniformScale));
161
162 svgStrokes << " <polyline fill=\"none\" stroke=\"rgba(" << (int)r << "," << (int)g << "," << (int)b << "," << alpha << ")\" "
163 << "stroke-width=\"" << w << "\" stroke-linecap=\"round\" stroke-linejoin=\"round\" points=\"";
164 svgStrokes << stroke.segments[0].p0.x << "," << stroke.segments[0].p0.y << " ";
165 for (const auto& seg : stroke.segments) {
166 svgStrokes << seg.p1.x << "," << seg.p1.y << " ";
167 }
168 svgStrokes << "\" />\n";
169 }
170 }
171 }
172
173 if (strokeCount > 0) {
174 ss << " <svg class=\"sheet-svg-layer\" viewBox=\""
175 << tile.worldX << " " << tile.worldY << " " << virtualW << " " << virtualH
176 << "\" preserveAspectRatio=\"none\">\n";
177 ss << svgStrokes.str();
178 ss << " </svg>\n";
179 }
180
181 // Searchable HTML Text Layer
182 ss << " <div class=\"sheet-text-layer\">\n";
183 for (const auto& obj : tile.intersectingObjects) {
184 if (auto tb = std::dynamic_pointer_cast<Folio::TextBoxObject>(obj)) {
185 // Transform world coordinates relative to this sheet tile
186 double relX = (tb->worldX - tile.worldX) * tile.uniformScale;
187 double relY = (tb->worldY - tile.worldY) * tile.uniformScale;
188 double widthMm = (tb->worldWidth > 0.0 ? tb->worldWidth : 100.0) * tile.uniformScale;
189 double fontSizePt = (tb->fontSize > 0.0f ? tb->fontSize : 12.0f) * static_cast<float>(tile.uniformScale);
190
191 ss << " <div class=\"canvas-textbox\" style=\"left: " << relX << "mm; top: " << relY
192 << "mm; width: " << widthMm << "mm; font-size: " << fontSizePt << "pt; font-family: '"
193 << EscapeHtml(tb->fontFamily.empty() ? "Segoe UI" : tb->fontFamily) << "'; color: #1f2937;\">\n";
194 ss << " <p>" << FormatHtmlText(tb->text, options) << "</p>\n";
195 ss << " </div>\n";
196 }
197 }
198 ss << " </div>\n";
199
200 ss << "</div>\n";
201 return ss.str();
202}

204std::string HtmlSvgExporter::ExportPage(
205 const std::shared_ptr<CanvasPage>& page,
206 const std::string& sectionName,
207 const std::string& notebookName,
208 const ExportOptions& options
209) {
210 if (!page) return "";
211 auto tiles = SheetTiler::TilePage(
212 *page, options.traversalOrder, options.scalingMode, options.skipEmptySheets
213 );
214
215 double sheetW = tiles.empty() ? 210.0 : tiles.front().widthMm;
216 double sheetH = tiles.empty() ? 297.0 : tiles.front().heightMm;
217
218 std::ostringstream ss;
219 ss << GetHtmlHeader(page->title, sheetW, sheetH);
220 for (const auto& tile : tiles) {
221 ss << RenderSheetBlock(tile, page, options);
222 }
223 ss << "</body>\n</html>\n";
224 return ss.str();
225}
