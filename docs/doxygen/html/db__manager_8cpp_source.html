61 // Attempt clean close

64 LOG_WARN(DBManager, "sqlite3_close reported busy handles; executing sqlite3_close_v2");

82/**
83 * @brief Executes performance, locking, and durability PRAGMAs.
84 *
85 * - journal_mode = WAL: Enables Write-Ahead Logging for high concurrency without blocking readers.
86 * - synchronous = NORMAL: In WAL mode, provides full ACID crash safety while skipping heavy fsync on each commit.
87 * - busy_timeout = 5000: Auto-retries busy locks for up to 5 seconds before returning SQLITE_BUSY.
88 * - foreign_keys = ON: Enforces relational cascade deletions (deleting section cascades to pages).
89 * - mmap_size = 256MB: Uses OS zero-copy memory mapping for fast sequential reads.
90 */
91void DBManager::ExecutePragmas() {
92 char* err = nullptr;
93 sqlite3_exec(db, "PRAGMA journal_mode = WAL;", nullptr, nullptr, &err);
94 if (err) sqlite3_free(err);
95
96 sqlite3_exec(db, "PRAGMA synchronous = NORMAL;", nullptr, nullptr, &err);
97 if (err) sqlite3_free(err);
98
99 sqlite3_exec(db, "PRAGMA busy_timeout = 5000;", nullptr, nullptr, &err);
100 if (err) sqlite3_free(err);
101
102 sqlite3_exec(db, "PRAGMA foreign_keys = ON;", nullptr, nullptr, &err);
103 if (err) sqlite3_free(err);
104
105 sqlite3_exec(db, "PRAGMA mmap_size = 268435456;", nullptr, nullptr, &err);
106 if (err) sqlite3_free(err);
