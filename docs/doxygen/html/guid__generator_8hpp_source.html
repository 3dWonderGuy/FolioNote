8/**
9 * @brief Utility class for generating RFC 4122 compliant UUID v4 strings.
10 *
11 * HOW IT WORKS:
12 * UUID v4 consists of 128 bits of random data with two specific bit fields:
13 * - Version field (bits 12-15 of the third group): fixed to 0b0100 (= 4)
14 * - Variant field (bits 6-7 of the fourth group): fixed to 0b10xx (RFC 4122)
15 * The remaining 122 bits are random, giving the output format:
16 * xxxxxxxx-xxxx-4xxx-[89ab]xxx-xxxxxxxxxxxx
17 *
18 * SEEDING STRATEGY (multi-source entropy):
19 * std::random_device alone can fall back to a deterministic PRNG on some platforms
20 * (e.g., MinGW on Windows). To prevent identical seeds across threads or runs:
21 * 1. Two calls to random_device are XOR-combined into a 64-bit seed.
22 * 2. A high-resolution timestamp is XOR'd in to ensure per-thread uniqueness
23 * even if random_device is non-entropic on the current platform.
24 * The resulting engine is stored as thread_local, so each thread has its own
25 * independent PRNG state with no locking or contention.
26 *
27 * PERFORMANCE:
28 * - thread_local: PRNG engine and distribution are initialized once per thread,
29 * not per call. Subsequent calls are pure PRNG generation with no heap allocation.
30 * - Stack buffer formatting via snprintf avoids heap allocation of std::ostringstream.
31 */
32class GUIDGenerator {
33public:
34 /**
35 * @brief Generates a standard RFC 4122 compliant UUID v4 string.
36 *
37 * Output format: "xxxxxxxx-xxxx-4xxx-[89ab]xxx-xxxxxxxxxxxx" (36 chars, lowercase hex)
38 * Example: "c4a760a8-dbcf-4e62-9783-26b6831e508c"
39 *
40 * @return std::string containing the UUID (always exactly 36 characters, no null terminator).
41 */
42 [[nodiscard]] static std::string GenerateV4() {
43 // Multi-source entropy seeding: XOR two random_device outputs with a timestamp
44 // to guard against deterministic fallback on platforms where random_device is weak.
45 // thread_local ensures each thread seeds independently with no contention.
46 thread_local std::mt19937_64 engine([]() {
47 std::random_device rd;
48 uint64_t seed = (static_cast<uint64_t>(rd()) << 32) ^ rd();
49 seed ^= static_cast<uint64_t>(std::chrono::high_resolution_clock::now().time_since_epoch().count());
50 return seed;
51 }());
52
53 // thread_local distribution: reused across calls with no re-construction cost.
54 thread_local std::uniform_int_distribution<uint64_t> dist;
55
56 // Generate 128 bits of random data as two 64-bit words.
57 uint64_t data0 = dist(engine);
58 uint64_t data1 = dist(engine);
59
60 // RFC 4122 Version 4: force bits 12-15 of the third UUID group to 0b0100.
61 // Mask clears those 4 bits, OR sets them to 4 (0x4000).
62 // data0 layout: [time_low(32)] [time_mid(16)] [time_hi_and_version(16)]
63 data0 = (data0 & 0xFFFFFFFFFFFF0FFFULL) | 0x0000000000004000ULL;
64
65 // RFC 4122 Variant (10xx): force the two most significant bits of the fourth group
66 // to 0b10. Mask clears the top 2 bits, OR sets them to 10xx (0x8000...).
67 // data1 layout: [clock_seq_hi_and_reserved(8)] [clock_seq_low(8)] [node(48)]
68 data1 = (data1 & 0x3FFFFFFFFFFFFFFFULL) | 0x8000000000000000ULL;
69
70 // Format into a 36-char UUID string on the stack (no heap allocation).
71 // Output groups: 8-4-4-4-12 hex digits separated by hyphens.
72 char buf[37]; // 36 UUID chars + null terminator
73 std::snprintf(buf, sizeof(buf),
74 "%08x-%04x-%04x-%04x-%012llx",
75 static_cast<uint32_t>(data0 >> 32), // 8 hex: time_low
76 static_cast<uint16_t>((data0 >> 16) & 0xFFFF), // 4 hex: time_mid
77 static_cast<uint16_t>(data0 & 0xFFFF), // 4 hex: time_hi_and_version (version=4)
78 static_cast<uint16_t>(data1 >> 48), // 4 hex: clock_seq (variant bits set)
79 static_cast<unsigned long long>(data1 & 0xFFFFFFFFFFFFULL) // 12 hex: node
80 );
81
82 // Construct string from exactly 36 chars, excluding the null terminator.
83 return std::string(buf, 36);
84 }
85};
