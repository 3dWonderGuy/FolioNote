3/**
4 * =========================================================================================
5 * @file printer_installer.hpp
6 * @brief Windows virtual printer management for FolioNote ("Print to FolioNote").
7 * =========================================================================================
8 *
9 * ARCHITECTURAL PURPOSE & DESIGN RATIONALE:
10 * 1. Virtual Printing Pipeline:
11 * Windows 10 and 11 provide the built-in "Microsoft Print to PDF" driver (ntprint.inf).
12 * FolioNote registers a specialized virtual printer named "Print to FolioNote" using this
13 * underlying driver. Any Windows application (browsers, office suites, CAD viewers) can
14 * print directly to FolioNote.
15 *
16 * 2. Automatic Import Routing & Event Dispatch:
17 * - To automatically route printed documents into FolioNote without manual file picking,
18 * we enable the Windows Print Service Operational event log (`wevtutil sl ...`).
19 * - When a document finishes spooling, the Windows Spooler raises Event 307.
20 * - In Event 307, `Param2` captures the source document title, while `Param5` identifies
21 * the printer name.
22 * - A Windows Scheduled Task (`FolioNotePrintDispatch`) subscribes to Event 307 for
23 * "Print to FolioNote", automatically launching FolioNote with `--import` and passing
24 * the document title.
25 *
26 * 3. UAC Administrator Privilege Escalation:
27 * - Configuring the Windows print spooler (adding printers, ports, or modifying event logs)
28 * requires administrative privileges (High Integrity Level).
29 * - FolioNote normally runs under standard user permissions (Medium Integrity Level).
30 * - `InstallPrinterElevated` and `UninstallPrinterElevated` use `ShellExecuteExW` with the
31 * verb `L"runas"` to prompt the user once via the standard Windows UAC dialog to execute
32 * the required PowerShell configuration task.
33 */

54 /**
55 * @brief Checks whether the "Print to FolioNote" virtual printer is registered in the spooler.
56 *
57 * Working Mechanism:
58 * - Queries the local print spooler subsystem via the Win32 Spooler API `OpenPrinterW`.
59 * - Returns true if the printer handle is acquired successfully.
60 *
61 * @param printerName The friendly name of the printer to verify (default: L"Print to FolioNote").
62 * @return true if the printer is installed and accessible in the Windows Spooler; false otherwise.
63 */
64 static bool IsPrinterInstalled(const std::wstring& printerName = L"Print to FolioNote") {
65#if defined(_WIN32)
66 PRINTER_DEFAULTSW defaults = { NULL, NULL, PRINTER_READ };
67 HANDLE hPrinter = NULL;
68 BOOL success = OpenPrinterW(const_cast<LPWSTR>(printerName.c_str()), &hPrinter, &defaults);
69 if (success && hPrinter != NULL) {
70 ClosePrinter(hPrinter);
71 return true;
72 }
73 return false;
74#else
75 (void)printerName;
76 return false;
77#endif
78 }

80 /**
81 * @brief Elevates privileges via UAC and installs the "Print to FolioNote" virtual printer
82 * alongside its automated event-dispatch scheduled task.
83 *
84 * Working Process:
85 * 1. Resolves the absolute path to the current running FolioNote binary.
86 * 2. Enables the `Microsoft-Windows-PrintService/Operational` channel.
87 * 3. Registers the printer targeting the 'Microsoft Print To PDF' driver.
88 * 4. Registers the `FolioNotePrintDispatch` scheduled task triggered by Event 307,
89 * routing `Param2` (the document title) into FolioNote's `--import` CLI pipeline.
90 *
91 * @param printerName Name of the virtual printer to register (default: L"Print to FolioNote").
92 * @return true if installation process exited successfully (exit code 0); false otherwise.
93 */
94 static bool InstallPrinterElevated(const std::wstring& printerName = L"Print to FolioNote") {
95#if defined(_WIN32)
96 // 1. Resolve absolute path of current executable to launch on print event
97 std::wstring exePathW(MAX_PATH, L'\0');
98 DWORD len = GetModuleFileNameW(NULL, &exePathW[0], static_cast<DWORD>(exePathW.size()));
99 while (len >= exePathW.size()) {
100 exePathW.resize(exePathW.size() * 2);
101 len = GetModuleFileNameW(NULL, &exePathW[0], static_cast<DWORD>(exePathW.size()));
102 }
103 exePathW.resize(len);
104
105 // Escape single quotes for safe PowerShell script block inclusion
106 std::wstring escapedExe = L"";
107 for (wchar_t c : exePathW) {
108 if (c == L'\'') escapedExe += L"''";
109 else escapedExe += c;
110 }
111
112 // PowerShell setup script:
113 // a) Enable Microsoft-Windows-PrintService/Operational log
114 // b) Register virtual printer with 'Microsoft Print To PDF' driver
115 // c) Configure scheduled task triggered by Event 307 (spool completion)
116 std::wstring psCommand =
117 L"-NoProfile -ExecutionPolicy Bypass -Command \""
118 L"$pName = '" + printerName + L"'; "
119 L"wevtutil sl 'Microsoft-Windows-PrintService/Operational' /e:true; "
120 L"if (-not (Get-Printer -Name $pName -ErrorAction SilentlyContinue)) { "
121 L" Add-Printer -Name $pName -DriverName 'Microsoft Print To PDF' -PortName 'PORTPROMPT:'; "
122 L" Set-Printer -Name $pName -Comment 'Virtual PDF printer for importing documents into FolioNote'; "
123 L" Restart-Service -Name Spooler -Force -ErrorAction SilentlyContinue "
124 L"}; "
125 L"$appExe = '" + escapedExe + L"'; "
126 L"$xml = @'`\r\n"
127 L"<?xml version=`\"1.0`\" encoding=`\"UTF-16`\"?>`r`\n"
128 L"<Task version=`\"1.2`\" xmlns=`\"http://schemas.microsoft.com/windows/2004/02/mit/task`\">`r`\n"
129 L" <RegistrationInfo><Description>Auto-dispatches printed PDFs into FolioNote</Description></RegistrationInfo>`r`\n"
130 L" <Triggers>`r`\n"
131 L" <EventTrigger>`r`\n"
132 L" <Enabled>true</Enabled>`r`\n"
133 L" <Subscription>&lt;QueryList&gt;&lt;Query Id=`\"0`\" Path=`\"Microsoft-Windows-PrintService/Operational`\"&gt;&lt;Select Path=`\"Microsoft-Windows-PrintService/Operational`\"&gt;*[System[EventID=307]] and *[UserData[DocumentPrinted[Param5='\" + printerName + \"']]]&lt;/Select&gt;&lt;/Query&gt;&lt;/QueryList&gt;</Subscription>`r`\n"
134 L" <ValueQueries>`r`\n"
135 L" <Value name=`\"DocTitle`\">Event/UserData/DocumentPrinted/Param2</Value>`r`\n"
136 L" </ValueQueries>`r`\n"
137 L" </EventTrigger>`r`\n"
138 L" </Triggers>`r`\n"
139 L" <Principals><Principal id=`\"Author`\"><LogonType>InteractiveToken</LogonType><RunLevel>LeastPrivilege</RunLevel></Principal></Principals>`r`\n"
140 L" <Settings>`r`\n"
141 L" <MultipleInstancesPolicy>Parallel</MultipleInstancesPolicy>`r`\n"
142 L" <DisallowStartIfOnBatteries>false</DisallowStartIfOnBatteries>`r`\n"
143 L" <StopIfGoingOnBatteries>false</StopIfGoingOnBatteries>`r`\n"
144 L" <ExecutionTimeLimit>PT1M</ExecutionTimeLimit>`r`\n"
145 L" <Priority>7</Priority>`r`\n"
146 L" </Settings>`r`\n"
147 L" <Actions Context=`\"Author`\">`r`\n"
148 L" <Exec>`r`\n"
149 L" <Command>$appExe</Command>`r`\n"
150 L" <Arguments>--import-print `\"`$(DocTitle)`\"</Arguments>`r`\n"
151 L" </Exec>`r`\n"
152 L" </Actions>`r`\n"
153 L"</Task>`r`\n"
154 L"'@; "
155 L"$xmlFile = [System.IO.Path]::GetTempFileName(); "
156 L"[System.IO.File]::WriteAllText($xmlFile,$xml, [System.Text.Encoding]::Unicode); "
157 L"schtasks.exe /create /tn 'FolioNotePrintDispatch' /xml $xmlFile /f; "
158 L"Remove-Item -Path $xmlFile -Force -ErrorAction SilentlyContinue"
159 L"\"";
160
161 SHELLEXECUTEINFOW sei = { sizeof(sei) };
162 sei.fMask = SEE_MASK_NOCLOSEPROCESS;
163 sei.hwnd = NULL;
164 sei.lpVerb = L"runas"; // Requests UAC administrator elevation
165 sei.lpFile = L"powershell.exe";
166 sei.lpParameters = psCommand.c_str();
167 sei.nShow = SW_HIDE;
168
169 if (ShellExecuteExW(&sei)) {
170 if (sei.hProcess != NULL) {
171 WaitForSingleObject(sei.hProcess, 30000);
172 DWORD exitCode = 1;
173 GetExitCodeProcess(sei.hProcess, &exitCode);
174 CloseHandle(sei.hProcess);
175 return (exitCode == 0);
176 }
177 return true;
178 }
179 return false;
180#else
181 (void)printerName;
182 return false;
183#endif
184 }
