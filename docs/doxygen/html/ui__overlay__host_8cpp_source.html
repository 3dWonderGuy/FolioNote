81void UIOverlayHost::Render(float dt) {
82 const auto& theme = ThemeManager::Instance();
83 bool isDark = (theme.colorBg.x < 0.5f);
84 ImVec2 displaySize = ImGui::GetIO().DisplaySize;
85
86 // =========================================================================
87 // 1. MODAL DIALOG PRESENTATION
88 // =========================================================================
89 if (currentModal_.isOpen && currentModal_.renderFunc) {
90 std::string scaleKey = "modal_scale_" + currentModal_.id;
91 std::string alphaKey = "modal_alpha_" + currentModal_.id;
92
93 float scale = UIAnimationManager::Instance().GetSpring(scaleKey, 1.0f, 260.0f, 22.0f);
94 float alpha = UIAnimationManager::Instance().GetFloat(alphaKey, 1.0f, 14.0f);
95
96 // Fullscreen dimmed backdrop
97 ImDrawList* fgDrawList = ImGui::GetForegroundDrawList();
98 ImU32 backdropCol = IM_COL32(0, 0, 0, static_cast<int>(140.0f * alpha));
99 fgDrawList->AddRectFilled(ImVec2(0, 0), displaySize, backdropCol);
100
101 // Centered modal window calculation
102 float modalW = currentModal_.width * scale;
103 float modalH = 340.0f * scale; // Approximate initial height
104 ImVec2 center(displaySize.x * 0.5f, displaySize.y * 0.5f);
105 ImVec2 modalPos(center.x - modalW * 0.5f, center.y - modalH * 0.5f);
106
107 ImGui::SetNextWindowPos(modalPos, ImGuiCond_Always);
108 ImGui::SetNextWindowSize(ImVec2(modalW, 0.0f), ImGuiCond_Always);
109
110 ImGuiWindowFlags flags = ImGuiWindowFlags_NoTitleBar |
111 ImGuiWindowFlags_NoResize |
112 ImGuiWindowFlags_NoMove |
113 ImGuiWindowFlags_AlwaysAutoResize |
114 ImGuiWindowFlags_NoSavedSettings;
115
116 ImU32 modalBg = isDark ? IM_COL32(28, 32, 44, 250) : IM_COL32(252, 254, 255, 250);
117 ImU32 modalBorder = isDark ? IM_COL32(65, 74, 95, 220) : IM_COL32(210, 218, 230, 240);
118
119 ImGui::PushStyleColor(ImGuiCol_WindowBg, modalBg);
120 ImGui::PushStyleColor(ImGuiCol_Border, modalBorder);
121 ImGui::PushStyleVar(ImGuiStyleVar_WindowRounding, Radii::lg);
122 ImGui::PushStyleVar(ImGuiStyleVar_WindowPadding, ImVec2(Spacing::xl, Spacing::xl));
123
124 std::string modalWinName = "##Modal_" + currentModal_.id;
125 if (ImGui::Begin(modalWinName.c_str(), nullptr, flags)) {
126 // Header: Title and Close button
127 ImGui::TextColored(ImVec4(0.95f, 0.45f, 0.15f, 1.0f), "%s", currentModal_.title.c_str());
128 ImGui::SameLine(ImGui::GetWindowWidth() - 36.0f);
129 if (UI::Button("✕").Size(UIButtonSize::Small).Variant(UIButtonVariant::Ghost).Width(28.0f).Render()) {
130 CloseModal();
131 }
132
133 ImGui::Separator();
134 ImGui::Dummy(ImVec2(0.0f, Spacing::sm));
135
136 // Body
137 currentModal_.renderFunc();
138
139 ImGui::End();
140 }
141
142 ImGui::PopStyleVar(2);
143 ImGui::PopStyleColor(2);
144 }
145
146 // =========================================================================
147 // 2. TOAST NOTIFICATION STACK (Top-Right Screen Corner)
148 // =========================================================================
149 {
150 std::lock_guard<std::mutex> lock(toastsMutex_);
151 if (!toasts_.empty()) {
152 float toastW = 320.0f;
153 float toastH = 68.0f;
154 float padY = 10.0f;
155 float startX = displaySize.x - toastW - Spacing::lg;
156 float startY = Spacing::xl + 32.0f; // Below titlebar
157
158 ImDrawList* fgDrawList = ImGui::GetForegroundDrawList();
159
160 for (auto it = toasts_.begin(); it != toasts_.end(); ) {
161 it->remainingSeconds -= dt;
162 if (it->remainingSeconds <= 0.0f) {
163 it = toasts_.erase(it);
164 continue;
165 }
166
167 ImVec2 tMin(startX, startY);
168 ImVec2 tMax(startX + toastW, startY + toastH);
169
170 // Acrylic surface & shadow
171 ImU32 bgCol = isDark ? IM_COL32(32, 36, 48, 245) : IM_COL32(250, 252, 255, 245);
172 ImU32 borderCol = isDark ? IM_COL32(65, 72, 90, 200) : IM_COL32(215, 222, 232, 220);
173 UIShapeManager::DrawAcrylicPanel(fgDrawList, tMin, tMax, bgCol, borderCol, Radii::md, Elevation::high);
174
175 // Accent status indicator bar on left edge
176 ImU32 accentCol = IM_COL32(50, 150, 255, 255); // Info blue
177 const char* iconSymbol = "ℹ";
178 if (it->type == UIToastType::Success) {
179 accentCol = IM_COL32(40, 195, 120, 255); // Emerald green
180 iconSymbol = "✔";
181 } else if (it->type == UIToastType::Warning) {
182 accentCol = IM_COL32(245, 175, 40, 255); // Amber yellow
183 iconSymbol = "⚠";
184 } else if (it->type == UIToastType::Error) {
185 accentCol = IM_COL32(235, 60, 60, 255); // Rose red
186 iconSymbol = "✖";
187 }
188
189 fgDrawList->AddRectFilled(tMin, ImVec2(tMin.x + 4.0f, tMax.y), accentCol, Radii::md);
190
191 // Icon symbol
192 fgDrawList->AddText(ImVec2(tMin.x + 14.0f, tMin.y + 14.0f), accentCol, iconSymbol);
193
194 // Title & message text
195 ImU32 titleCol = isDark ? IM_COL32(240, 243, 250, 255) : IM_COL32(20, 25, 35, 255);
196 ImU32 msgCol = isDark ? IM_COL32(160, 168, 185, 220) : IM_COL32(105, 112, 128, 220);
197
198 fgDrawList->AddText(ImVec2(tMin.x + 36.0f, tMin.y + 12.0f), titleCol, it->title.c_str());
199 fgDrawList->AddText(ImVec2(tMin.x + 36.0f, tMin.y + 32.0f), msgCol, it->message.c_str());
200
201 // Bottom countdown progress bar
202 float progressFraction = std::clamp(it->remainingSeconds / it->initialSeconds, 0.0f, 1.0f);
203 float progressW = (toastW - 8.0f) * progressFraction;
204 fgDrawList->AddRectFilled(
205 ImVec2(tMin.x + 4.0f, tMax.y - 2.5f),
206 ImVec2(tMin.x + 4.0f + progressW, tMax.y),
207 ColorUtils::WithAlpha(accentCol, 0.65f)
208 );
209
210 startY += toastH + padY;
211 ++it;
212 }
213 }
214 }
