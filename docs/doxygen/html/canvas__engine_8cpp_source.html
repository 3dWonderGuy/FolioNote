250/**
251 * @brief Thread-safe dispatch of tasks to be executed on the main UI/rendering thread.
252 *
253 * Architecture & Working Process:
254 * 1. Background threads (e.g. SDL3 dialog callbacks running on Android JNI thread) invoke this method.
255 * 2. Acquires m_mainThreadTasksMutex and pushes the task functor into m_mainThreadTasks.
256 * 3. Calls InvalidateLayer() to wake the render loop and schedule the next frame.
257 *
258 * @param task The lambda or std::function to execute on the main thread.
259 */
260void CanvasEngine::EnqueueMainThreadTask(std::function<void()> task) {
261 if (!task) return;
262 {
263 std::lock_guard<std::mutex> lock(m_mainThreadTasksMutex);
264 m_mainThreadTasks.push_back(std::move(task));
265 }
266 InvalidateLayer();
267}

269/**
270 * @brief Flushes and executes all pending main-thread tasks in FIFO order.
271 *
272 * Working Process:
273 * 1. Quickly swaps the queued task list under m_mainThreadTasksMutex to minimize lock contention.
274 * 2. Sequentially executes each task on the current calling thread (main render thread).
275 * 3. Catches and logs any unexpected exceptions to guarantee application resilience.
276 */
277void CanvasEngine::ProcessMainThreadTasks() {
278 std::vector<std::function<void()>> tasks;
279 {
280 std::lock_guard<std::mutex> lock(m_mainThreadTasksMutex);
281 if (m_mainThreadTasks.empty()) return;
282 tasks.swap(m_mainThreadTasks);
283 }
284
285 for (auto& task : tasks) {
286 if (task) {
287 try {
288 task();
289 } catch (const std::exception& e) {
290 LOG_ERROR(CanvasEngine, std::string("Exception in MainThreadTask: ") + e.what());
291 } catch (...) {
292 LOG_ERROR(CanvasEngine, "Unknown exception in MainThreadTask");
293 }
294 }
295 }
